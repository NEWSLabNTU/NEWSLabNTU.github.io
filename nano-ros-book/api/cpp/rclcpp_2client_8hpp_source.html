407 static void response_trampoline(const uint8_t* data, size_t len, void* ctx) {

463template <typename S, typename F, typename>
464Result Node::create_client(Client<S>& out, const char* service_name, F callback,
465 const ::rclcpp::QoS& qos, const ::rclcpp::ClientOptions& options) {
466 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
467 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
468
469 // The user handler becomes the ARENA's context (phase-456 W3); the
470 // conversion is still the compile error for a non-convertible `F`.
471 const typename Client<S>::TypedResponseFn user_fn =
472 typename Client<S>::TypedResponseFn(callback);
473
474 uint8_t sched = (options.sched_context == ::rclcpp::SCHED_CONTEXT_UNSET)
475 ? 0u
476 : static_cast<uint8_t>(options.sched_context);
477 size_t handle = static_cast<size_t>(-1);
478 nros_cpp_ret_t ret = nros_cpp_service_client_register(
479 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
480 reinterpret_cast<nros_cpp_service_response_callback_t>(&Client<S>::response_trampoline),
481 ::rclcpp::detail::fn_to_context(user_fn), sched, &handle);
482 if (ret == 0) {
483 out.executor_ = executor_handle_;
484 out.handle_id_ = handle;
485 // phase-444 — see `rclcpp::PollClient`'s overload, which does the same.
486 ::rclcpp::detail::assign_entity_name(out.service_name_, service_name);
487 out.initialized_ = true;
488 }
489 return Result(ret);
490}
