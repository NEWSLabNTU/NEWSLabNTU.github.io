312 static bool request_trampoline(const uint8_t* req, size_t req_len, uint8_t* resp,
313 size_t resp_cap, size_t* resp_len, void* ctx) {
314 const TypedServiceFn user_fn = ::rclcpp::detail::fn_from_context<TypedServiceFn>(ctx);
315 if (user_fn == nullptr) return false;
316 RequestType request;
317 if (RequestType::ffi_deserialize(req, req_len, &request) != 0) return false;
318 ResponseType response;
319 user_fn(request, response);
320 size_t len = 0;
321 if (ResponseType::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
322 *resp_len = len;
323 return true;
324 }

390template <typename S>
391Result Node::register_service_(Service<S>& out, const char* service_name,
392 bool (*trampoline)(const uint8_t*, size_t, uint8_t*, size_t, size_t*,
393 void*),
394 void* context, const ::rclcpp::QoS& qos,
395 const ::rclcpp::ServiceOptions& options, const void* capture,
396 size_t capture_len, void (*capture_drop)(void*)) {
397 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
398 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
399
400 uint8_t sched = (options.sched_context == ::rclcpp::SCHED_CONTEXT_UNSET)
401 ? 0u
402 : static_cast<uint8_t>(options.sched_context);
403 size_t handle = static_cast<size_t>(-1);
404 nros_cpp_ret_t ret =
405 (capture != nullptr && capture_len != 0)
406 ? nros_cpp_service_server_register_capturing(
407 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos, trampoline,
408 static_cast<const uint8_t*>(capture), capture_len, capture_drop, sched, &handle)
409 : nros_cpp_service_server_register(&handle_, service_name, S::TYPE_NAME,
410 S::Request::TYPE_HASH, ffi_qos, trampoline, context,
411 sched, &handle);
412 return this->template finish_register_service_<S>(out, service_name, ret, handle);
413}

478struct FrameBlock {
483 static const size_t CAPACITY = 96;
484 alignas(alignof(::std::max_align_t)) unsigned char bytes[CAPACITY];
485 bool used = false;
486};

488[[noreturn]] inline void abort_frame_block() {
489 RCLC_LOG_ERROR(log_handle(nullptr),
490 "rclcpp: a shared_ptr control block does not fit FrameBlock::CAPACITY");
491 ::std::abort();
492}

494template <typename T> struct FrameBlockAllocator {
495 using value_type = T;
496 FrameBlock* block;
497 explicit FrameBlockAllocator(FrameBlock* b) : block(b) {}
498 template <typename U> FrameBlockAllocator(const FrameBlockAllocator<U>& o) : block(o.block) {}
499 T* allocate(size_t n) {
500 if (block->used || n * sizeof(T) > FrameBlock::CAPACITY ||
501 alignof(T) > alignof(::std::max_align_t)) {
502 abort_frame_block();
503 }
504 block->used = true;
505 return reinterpret_cast<T*>(block->bytes);
506 }
507 void deallocate(T*, size_t) { block->used = false; }
508 template <typename U> bool operator==(const FrameBlockAllocator<U>& o) const {
509 return block == o.block;
510 }
511 template <typename U> bool operator!=(const FrameBlockAllocator<U>& o) const {
512 return block != o.block;
513 }
514};

546template <typename S, typename Invoke>
547inline bool serve_lent(const uint8_t* req, size_t req_len, const ::rmw_request_id_t* id,
548 uint8_t* resp, size_t resp_cap, size_t* resp_len, const Invoke& invoke) {
549 typename S::Request request;
550 if (S::Request::ffi_deserialize(req, req_len, &request) != 0) return false;
551 typename S::Response response;
552 // Issue 2081 — the id the backend reported for this request (all zero
553 // when it reported none).
554 ::rmw_request_id_t header = {};
555 if (id != nullptr) header = *id;
556 {
557 FrameBlock request_block;
558 FrameBlock response_block;
559 FrameBlock header_block;
560 ::std::shared_ptr<typename S::Request> request_ptr =
561 lend_from_frame(&request, &request_block);
562 ::std::shared_ptr<typename S::Response> response_ptr =
563 lend_from_frame(&response, &response_block);
564 ::std::shared_ptr<::rmw_request_id_t> header_ptr = lend_from_frame(&header, &header_block);
565 invoke(header_ptr, request_ptr, response_ptr);
566 if (request_ptr.use_count() != 1 || response_ptr.use_count() != 1 ||
567 header_ptr.use_count() != 1) {
568 NROS_RCLCPP_SAY_REFUSED(NROS_RCLCPP_REFUSE_RETAINED_SERVICE_PTR);
569 ::std::abort();
570 }
571 }
572 size_t len = 0;
573 if (S::Response::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
574 *resp_len = len;
575 return true;
576}
