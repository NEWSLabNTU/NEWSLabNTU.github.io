249 static bool request_trampoline(const uint8_t* req, size_t req_len, uint8_t* resp,
250 size_t resp_cap, size_t* resp_len, void* ctx) {
251 auto* self = static_cast<Service*>(ctx);
252 if (self == nullptr) return false;
253 RequestType request;
254 if (RequestType::ffi_deserialize(req, req_len, &request) != 0) return false;
255 ResponseType response;
256 if (self->user_fn_ != nullptr) {
257 self->user_fn_(request, response);
258 } else if (self->user_fn_ctx_ != nullptr) {
259 self->user_fn_ctx_(request, response, self->user_ctx_);
260 } else {
261 return false;
262 }
263 size_t len = 0;
264 if (ResponseType::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
265 *resp_len = len;
266 return true;
267 }

275 size_t handle_id_ = static_cast<size_t>(-1);

307// Phase 189.M3.3.e — callback-style (arena-registered) service. The arena owns
308// the server + dispatches `out`'s request handler during spin_once, so the
309// handle is real and `options.sched_context` is functional.
310template <typename S, typename F, typename>
311Result Node::create_service(Service<S>& out, const char* service_name, F callback, const QoS& qos,
312 const ServiceOptions& options) {
313 if (!initialized_) return Result(ErrorCode::NotInitialized);
314 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
315
316 // Store the user handler (compile error if F isn't convertible).
317 out.user_fn_ = typename Service<S>::TypedServiceFn(callback);
318 out.user_fn_ctx_ = nullptr;
319 out.user_ctx_ = nullptr;
320
321 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
322 ? 0u
323 : static_cast<uint8_t>(options.sched_context);
324 size_t handle = static_cast<size_t>(-1);
325 nros_cpp_ret_t ret = nros_cpp_service_server_register(
326 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
327 reinterpret_cast<nros_cpp_service_request_callback_t>(&Service<S>::request_trampoline),
328 &out, sched, &handle);
329 if (ret == 0) {
330 out.handle_id_ = handle;
331 out.callback_mode_ = true;
332 out.initialized_ = true;
333 }
334 return Result(ret);
335}
