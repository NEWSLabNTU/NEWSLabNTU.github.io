273 static bool request_trampoline(const uint8_t* req, size_t req_len, uint8_t* resp,
274 size_t resp_cap, size_t* resp_len, void* ctx) {
275 const TypedServiceFn user_fn = ::nros::detail::fn_from_context<TypedServiceFn>(ctx);
276 if (user_fn == nullptr) return false;
277 RequestType request;
278 if (RequestType::ffi_deserialize(req, req_len, &request) != 0) return false;
279 ResponseType response;
280 user_fn(request, response);
281 size_t len = 0;
282 if (ResponseType::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
283 *resp_len = len;
284 return true;
285 }

346template <typename S, typename F, typename>
347Result Node::create_service(Service<S>& out, const char* service_name, F callback,
348 const ::nros::QoS& qos, const ::nros::ServiceOptions& options) {
349 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
350 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
351
352 // The user handler becomes the ARENA's context (phase-456 W3). The
353 // conversion is still the compile error for a non-convertible `F`; what
354 // changed is where the resulting pointer is stored — in the registration,
355 // not in `out`, so `out` is no longer an object the arena knows by address.
356 const typename Service<S>::TypedServiceFn user_fn =
357 typename Service<S>::TypedServiceFn(callback);
358
359 uint8_t sched = (options.sched_context == ::nros::SCHED_CONTEXT_UNSET)
360 ? 0u
361 : static_cast<uint8_t>(options.sched_context);
362 size_t handle = static_cast<size_t>(-1);
363 nros_cpp_ret_t ret = nros_cpp_service_server_register(
364 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
365 reinterpret_cast<nros_cpp_service_request_callback_t>(&Service<S>::request_trampoline),
366 ::nros::detail::fn_to_context(user_fn), sched, &handle);
367 if (ret == 0) {
368 out.handle_id_ = handle;
369 // phase-444 — remember the name for `get_service_name()`; the runtime
370 // takes `service_name` and drops it.
371 ::nros::detail::assign_entity_name(out.service_name_, service_name);
372
373 // issue 1437 — the arena that owns the entity, beside its index.
374 out.executor_ = executor_handle_;
375 out.initialized_ = true;
376 }
377 return Result(ret);
378}
