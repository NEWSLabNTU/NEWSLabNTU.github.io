299 static bool request_trampoline(const uint8_t* req, size_t req_len, uint8_t* resp,
300 size_t resp_cap, size_t* resp_len, void* ctx) {
301 auto* self = static_cast<Service*>(ctx);
302 if (self == nullptr) return false;
303 RequestType request;
304 if (RequestType::ffi_deserialize(req, req_len, &request) != 0) return false;
305 ResponseType response;
306 if (self->user_fn_ != nullptr) {
307 self->user_fn_(request, response);
308 } else if (self->user_fn_ctx_ != nullptr) {
309 self->user_fn_ctx_(request, response, self->user_ctx_);
310 } else {
311 return false;
312 }
313 size_t len = 0;
314 if (ResponseType::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
315 *resp_len = len;
316 return true;
317 }

321 ::nros::QoS actual_qos_half(bool request) const {
322 nros_cpp_qos_t req{};
323 nros_cpp_qos_t resp{};
324 if (!initialized_) return ::nros::detail::qos_all_unknown();
325 const void* storage = callback_mode_ ? nullptr : static_cast<const void*>(storage_);
326 if (nros_cpp_service_server_get_actual_qos(storage, executor_, handle_id_, &req, &resp) !=
327 0) {
328 return ::nros::detail::qos_all_unknown();
329 }
330 return ::nros::detail::qos_from_ffi(request ? req : resp);

397
398 // Store the user handler (compile error if F isn't convertible).
399 out.user_fn_ = typename Service<S>::TypedServiceFn(callback);
400 out.user_fn_ctx_ = nullptr;
401 out.user_ctx_ = nullptr;
402
403 uint8_t sched = (options.sched_context == ::nros::SCHED_CONTEXT_UNSET)
404 ? 0u
405 : static_cast<uint8_t>(options.sched_context);
406 size_t handle = static_cast<size_t>(-1);
407 nros_cpp_ret_t ret = nros_cpp_service_server_register(
408 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
409 reinterpret_cast<nros_cpp_service_request_callback_t>(&Service<S>::request_trampoline),
410 &out, sched, &handle);
411 if (ret == 0) {
412 out.handle_id_ = handle;
413 out.callback_mode_ = true;
414 // phase-444 — see the poll-style overload above.
415 ::nros::detail::assign_entity_name(out.service_name_, service_name);
416
417 // issue 1437 — the arena that owns the entity, beside its index.
418 out.executor_ = executor_handle_;
419 out.initialized_ = true;
