271 static bool request_trampoline(const uint8_t* req, size_t req_len, uint8_t* resp,
272 size_t resp_cap, size_t* resp_len, void* ctx) {
273 const TypedServiceFn user_fn = ::rclcpp::detail::fn_from_context<TypedServiceFn>(ctx);
274 if (user_fn == nullptr) return false;
275 RequestType request;
276 if (RequestType::ffi_deserialize(req, req_len, &request) != 0) return false;
277 ResponseType response;
278 user_fn(request, response);
279 size_t len = 0;
280 if (ResponseType::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
281 *resp_len = len;
282 return true;
283 }

335template <typename S, typename F, typename>
336Result Node::create_service(Service<S>& out, const char* service_name, F callback,
337 const ::rclcpp::QoS& qos, const ::rclcpp::ServiceOptions& options) {
338 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
339 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
340
341 // The user handler becomes the ARENA's context (phase-456 W3). The
342 // conversion is still the compile error for a non-convertible `F`; what
343 // changed is where the resulting pointer is stored — in the registration,
344 // not in `out`, so `out` is no longer an object the arena knows by address.
345 const typename Service<S>::TypedServiceFn user_fn =
346 typename Service<S>::TypedServiceFn(callback);
347
348 uint8_t sched = (options.sched_context == ::rclcpp::SCHED_CONTEXT_UNSET)
349 ? 0u
350 : static_cast<uint8_t>(options.sched_context);
351 size_t handle = static_cast<size_t>(-1);
352 nros_cpp_ret_t ret = nros_cpp_service_server_register(
353 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
354 reinterpret_cast<nros_cpp_service_request_callback_t>(&Service<S>::request_trampoline),
355 ::rclcpp::detail::fn_to_context(user_fn), sched, &handle);
356 if (ret == 0) {
357 out.handle_id_ = handle;
358 // phase-444 — remember the name for `get_service_name()`; the runtime
359 // takes `service_name` and drops it.
360 ::rclcpp::detail::assign_entity_name(out.service_name_, service_name);
361
362 // issue 1437 — the arena that owns the entity, beside its index.
363 out.executor_ = executor_handle_;
364 out.initialized_ = true;
365 }
366 return Result(ret);
367}
