231 static bool request_trampoline(const uint8_t* req, size_t req_len, uint8_t* resp,
232 size_t resp_cap, size_t* resp_len, void* ctx) {
233 auto* self = static_cast<Service*>(ctx);
234 if (self == nullptr) return false;
235 RequestType request;
236 if (RequestType::ffi_deserialize(req, req_len, &request) != 0) return false;
237 ResponseType response;
238 if (self->user_fn_ != nullptr) {
239 self->user_fn_(request, response);
240 } else if (self->user_fn_ctx_ != nullptr) {
241 self->user_fn_ctx_(request, response, self->user_ctx_);
242 } else {
243 return false;
244 }
245 size_t len = 0;
246 if (ResponseType::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
247 *resp_len = len;
248 return true;
249 }

280// Phase 189.M3.3.e — callback-style (arena-registered) service. The arena owns
281// the server + dispatches `out`'s request handler during spin_once, so the
282// handle is real and `options.sched_context` is functional.
283template <typename S, typename F, typename>
284Result Node::create_service(Service<S>& out, const char* service_name, F callback, const QoS& qos,
285 const ServiceOptions& options) {
286 if (!initialized_) return Result(ErrorCode::NotInitialized);
287 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
288
289 // Store the user handler (compile error if F isn't convertible).
290 out.user_fn_ = typename Service<S>::TypedServiceFn(callback);
291 out.user_fn_ctx_ = nullptr;
292 out.user_ctx_ = nullptr;
293
294 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
295 ? 0u
296 : static_cast<uint8_t>(options.sched_context);
297 size_t handle = static_cast<size_t>(-1);
298 nros_cpp_ret_t ret = nros_cpp_service_server_register(
299 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
300 reinterpret_cast<nros_cpp_service_request_callback_t>(&Service<S>::request_trampoline),
301 &out, sched, &handle);
302 if (ret == 0) {
303 out.handle_id_ = handle;
304 out.callback_mode_ = true;
305 out.initialized_ = true;
306 }
307 return Result(ret);
308}
