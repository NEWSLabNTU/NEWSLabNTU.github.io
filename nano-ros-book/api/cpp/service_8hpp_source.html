271 static bool request_trampoline(const uint8_t* req, size_t req_len, uint8_t* resp,
272 size_t resp_cap, size_t* resp_len, void* ctx) {
273 const TypedServiceFn user_fn = ::rclcpp::detail::fn_from_context<TypedServiceFn>(ctx);
274 if (user_fn == nullptr) return false;
275 RequestType request;
276 if (RequestType::ffi_deserialize(req, req_len, &request) != 0) return false;
277 ResponseType response;
278 user_fn(request, response);
279 size_t len = 0;
280 if (ResponseType::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
281 *resp_len = len;
282 return true;
283 }

350Result Node::register_service_(Service<S>& out, const char* service_name,
351 bool (*trampoline)(const uint8_t*, size_t, uint8_t*, size_t, size_t*,
352 void*),
353 void* context, const ::rclcpp::QoS& qos,
354 const ::rclcpp::ServiceOptions& options) {

361 size_t handle = static_cast<size_t>(-1);
362 nros_cpp_ret_t ret = nros_cpp_service_server_register(&handle_, service_name, S::TYPE_NAME,

372 out.executor_ = executor_handle_;

412struct FrameBlock {

422[[noreturn]] inline void abort_frame_block() {
423 RCLC_LOG_ERROR(log_handle(nullptr),
424 "rclcpp: a shared_ptr control block does not fit FrameBlock::CAPACITY");

428template <typename T> struct FrameBlockAllocator {
429 using value_type = T;
430 FrameBlock* block;
431 explicit FrameBlockAllocator(FrameBlock* b) : block(b) {}
432 template <typename U> FrameBlockAllocator(const FrameBlockAllocator<U>& o) : block(o.block) {}
433 T* allocate(size_t n) {
434 if (block->used || n * sizeof(T) > FrameBlock::CAPACITY ||
435 alignof(T) > alignof(::std::max_align_t)) {
436 abort_frame_block();
437 }
438 block->used = true;
439 return reinterpret_cast<T*>(block->bytes);
440 }
441 void deallocate(T*, size_t) { block->used = false; }
442 template <typename U> bool operator==(const FrameBlockAllocator<U>& o) const {
443 return block == o.block;
444 }
445 template <typename U> bool operator!=(const FrameBlockAllocator<U>& o) const {
446 return block != o.block;
447 }
448};

475template <typename S, int Shape> struct shared_ptr_service_trampoline {
476 static bool call(const uint8_t* req, size_t req_len, uint8_t* resp, size_t resp_cap,
477 size_t* resp_len, void* ctx) {
478 using Fn = shared_ptr_service_fn<S>;
479 typename S::Request request;
480 if (S::Request::ffi_deserialize(req, req_len, &request) != 0) return false;
481 typename S::Response response;
482 ::rmw_request_id_t header = {};
483 {
484 FrameBlock request_block;
485 FrameBlock response_block;
486 FrameBlock header_block;
487 ::std::shared_ptr<typename S::Request> request_ptr =
488 lend_from_frame(&request, &request_block);
489 ::std::shared_ptr<typename S::Response> response_ptr =
490 lend_from_frame(&response, &response_block);
491 ::std::shared_ptr<::rmw_request_id_t> header_ptr =
492 lend_from_frame(&header, &header_block);
493 if (Shape == 3) {
494 fn_from_context<typename Fn::three>(ctx)(header_ptr, request_ptr, response_ptr);
495 } else {
496 fn_from_context<typename Fn::two>(ctx)(request_ptr, response_ptr);
497 }
498 if (request_ptr.use_count() != 1 || response_ptr.use_count() != 1 ||
499 header_ptr.use_count() != 1) {
500 NROS_RCLCPP_SAY_REFUSED(NROS_RCLCPP_REFUSE_RETAINED_SERVICE_PTR);
501 ::std::abort();
502 }
503 }
504 size_t len = 0;
505 if (S::Response::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
506 *resp_len = len;
507 return true;
508 }
509};
