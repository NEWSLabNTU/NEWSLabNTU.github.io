234 static bool request_trampoline(const uint8_t* req, size_t req_len, uint8_t* resp,
235 size_t resp_cap, size_t* resp_len, void* ctx) {
236 auto* self = static_cast<Service*>(ctx);
237 if (self == nullptr) return false;
238 RequestType request;
239 if (RequestType::ffi_deserialize(req, req_len, &request) != 0) return false;
240 ResponseType response;
241 if (self->user_fn_ != nullptr) {
242 self->user_fn_(request, response);
243 } else if (self->user_fn_ctx_ != nullptr) {
244 self->user_fn_ctx_(request, response, self->user_ctx_);
245 } else {
246 return false;
247 }
248 size_t len = 0;
249 if (ResponseType::ffi_serialize(&response, resp, resp_cap, &len) != 0) return false;
250 *resp_len = len;
251 return true;
252 }

254 alignas(8) uint8_t storage_[NROS_SERVICE_SERVER_SIZE];

258 TypedServiceFnWithCtx user_fn_ctx_ = nullptr;
259 void* user_ctx_ = nullptr;
260 size_t handle_id_ = static_cast<size_t>(-1);
261 bool callback_mode_ = false;

283// Phase 189.M3.3.e — callback-style (arena-registered) service. The arena owns
284// the server + dispatches `out`'s request handler during spin_once, so the
285// handle is real and `options.sched_context` is functional.
286template <typename S, typename F, typename>
287Result Node::create_service(Service<S>& out, const char* service_name, F callback, const QoS& qos,
288 const ServiceOptions& options) {
289 if (!initialized_) return Result(ErrorCode::NotInitialized);
290 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
291
292 // Store the user handler (compile error if F isn't convertible).
293 out.user_fn_ = typename Service<S>::TypedServiceFn(callback);
294 out.user_fn_ctx_ = nullptr;
295 out.user_ctx_ = nullptr;
296
297 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
298 ? 0u
299 : static_cast<uint8_t>(options.sched_context);
300 size_t handle = static_cast<size_t>(-1);
301 nros_cpp_ret_t ret = nros_cpp_service_server_register(
302 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
303 reinterpret_cast<nros_cpp_service_request_callback_t>(&Service<S>::request_trampoline),
304 &out, sched, &handle);
305 if (ret == 0) {
306 out.handle_id_ = handle;
307 out.callback_mode_ = true;
308 out.initialized_ = true;
309 }
310 return Result(ret);
311}
