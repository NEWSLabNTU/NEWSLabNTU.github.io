76 template <size_t Cap> Result try_next_sized(T& out) {
77 if (!take_fn_) return Result(ErrorCode::NotInitialized);
78 uint8_t buf[Cap];
79 size_t len = 0;
80 nros_cpp_ret_t ret = take_fn_(storage_, buf, sizeof(buf), &len);
81 if (ret != 0) return Result(ret);
82 if (len == 0) return Result(ErrorCode::TryAgain);
83 if (T::ffi_deserialize(buf, len, &out) != 0) return Result(ErrorCode::Error);
84 return Result::success();
85 }

104 template <size_t Cap>
105 Result wait_next_sized(void* executor_handle, uint32_t timeout_ms, T& out,
106 uint32_t poll_ms = 10) {
107 if (!take_fn_) return Result(ErrorCode::NotInitialized);
108 if (poll_ms == 0) poll_ms = 1;
109 // Phase 118.C: budget by wall-clock. Accumulating `step` per
110 // iteration breaks when `zpico_spin_once` returns early on a
111 // signaled condvar (keep-alives, discovery gossip) — the
112 // iteration loop collapses into milliseconds and returns
113 // Timeout before the message has a chance to land. Mirrors the
114 // Phase 89.2 fix on Future::wait().
115 const uint64_t start_ns = nros_cpp_time_ns();
116 const uint64_t budget_ns = static_cast<uint64_t>(timeout_ms) * 1000000ULL;
117 while (true) {
118 nros_cpp_ret_t ret = nros_cpp_spin_once(executor_handle, static_cast<int32_t>(poll_ms));
119 // Transient conditions: keep polling. Anything else propagates.
120 if (ret != 0 && ret != static_cast<nros_cpp_ret_t>(ErrorCode::Timeout) &&
121 ret != static_cast<nros_cpp_ret_t>(ErrorCode::TryAgain)) {
122 return Result(ret);
123 }
124 Result rn = try_next_sized<Cap>(out);
125 if (rn.ok()) return Result::success();
126 // TryAgain / NotInitialized / Error from try_next: keep polling
127 // unless we've hit a hard error that's not "no data yet".
128 if (rn.code() != ErrorCode::TryAgain && rn.code() != ErrorCode::NotInitialized) {
129 return rn;
130 }
131 const uint64_t now_ns = nros_cpp_time_ns();
132 if (now_ns - start_ns >= budget_ns) break;
133 }
134 return Result(ErrorCode::Timeout);
135 }
