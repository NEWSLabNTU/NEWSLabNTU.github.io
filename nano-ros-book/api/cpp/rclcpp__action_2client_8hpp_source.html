800 void adopt(Client& other) {
801 user_ = other.user_;
802 upstream_ = static_cast<detail::HostedBox&&>(other.upstream_);
803 feedback_len_ = other.feedback_len_;
804 feedback_ready_ = other.feedback_ready_;
805 feedback_truncated_ = other.feedback_truncated_;

819 // Issue 2011 -- the three callbacks every `Client<A>` keeps installed. The
820 // executor's dispatch (and `nros_cpp_action_client_poll`) hands each reply
821 // here, attributed to the goal its request named (issue 2009); it goes to
822 // the future waiting on THAT goal on THIS client, if any -- dropped
823 // otherwise, never handed to another goal -- and then to the user's
824 // callback (`set_callbacks`).
825 static void on_goal_response(bool accepted, const uint8_t goal_id[16], void* ctx) {
826 Client* self = static_cast<Client*>(ctx);
827 ::rclcpp::detail::PendingLink* owner = self->accepts_.resolve_goal(goal_id);
828 if (owner != nullptr) {
829 // The 17-byte layout `GoalAccept::ffi_deserialize` decodes
830 // (`nros_cpp_action_goal_accept_decode`).
831 uint8_t wire[17];
832 for (int i = 0; i < 16; ++i)
833 wire[i] = goal_id[i];
834 wire[16] = accepted ? 1 : 0;
835 self->accepts_.deliver(owner, wire, sizeof(wire));
836 }
837 self->upstream_goal_response(accepted, goal_id);
838 if (self->user_.goal_response != nullptr)
839 self->user_.goal_response(accepted, goal_id, self->user_.context);
840 }
841
842 static void on_feedback(const uint8_t goal_id[16], const uint8_t* data, size_t len, void* ctx) {
843 Client* self = static_cast<Client*>(ctx);
844 const size_t n = len < sizeof(self->feedback_buf_) ? len : sizeof(self->feedback_buf_);
845 for (size_t i = 0; i < n; ++i)
846 self->feedback_buf_[i] = data[i];
847 self->feedback_len_ = n;
848 self->feedback_truncated_ = n < len;
849 self->feedback_ready_ = true;
850 self->upstream_feedback(goal_id, data, len);
851 if (self->user_.feedback != nullptr)
852 self->user_.feedback(goal_id, data, len, self->user_.context);
853 }
854
855 static void on_result(const uint8_t goal_id[16], int32_t status, const uint8_t* data,
856 size_t len, void* ctx) {
857 Client* self = static_cast<Client*>(ctx);
858 ::rclcpp::detail::PendingLink* owner = self->results_.resolve_goal(goal_id);
859 if (owner != nullptr) self->results_.deliver(owner, data, len);
860 self->upstream_result(goal_id, status, data, len);
861 if (self->user_.result != nullptr)
862 self->user_.result(goal_id, status, data, len, self->user_.context);
863 }
