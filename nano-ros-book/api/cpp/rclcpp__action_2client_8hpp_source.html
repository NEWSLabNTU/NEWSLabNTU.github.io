1046 // Issue 2027 — the goal-slot leak. A request in the Rust core's bounded
1047 // per-sub-client table (`MAX_CONCURRENT_GOALS`) used to come back only
1048 // with its reply, so a future dropped first — with a reply that never
1049 // came — held its slot forever, and the fifth such goal refused every
1050 // later send. The future is the request's owner; when it is abandoned the
1051 // request is forgotten, UNLESS another owner still waits on that reply:
1052 // a client-wide callback (`set_callbacks`) or an upstream goal
1053 // (`async_send_goal` / `async_get_result`). A late reply for a forgotten
1054 // request is dropped by the correlator.

1056 static_cast<Client*>(ctx)->forget_request(NROS_CPP_ACTION_CHANNEL_SEND_GOAL, goal);

1059 static_cast<Client*>(ctx)->forget_request(NROS_CPP_ACTION_CHANNEL_GET_RESULT, goal);

1096 // Issue 2011 -- the three callbacks every `Client<A>` keeps installed. The
1097 // executor's dispatch (and `nros_cpp_action_client_poll`) hands each reply
1098 // here, attributed to the goal its request named (issue 2009); it goes to
1099 // the future waiting on THAT goal on THIS client, if any -- dropped
1100 // otherwise, never handed to another goal -- and then to the user's
1101 // callback (`set_callbacks`).
1102 static void on_goal_response(bool accepted, const uint8_t goal_id[16], void* ctx) {
1103 Client* self = static_cast<Client*>(ctx);
1104 ::rclcpp::detail::PendingLink* owner = self->accepts_.resolve_goal(goal_id);
1105 if (owner != nullptr) {
1106 // The 17-byte layout `GoalAccept::ffi_deserialize` decodes
1107 // (`nros_cpp_action_goal_accept_decode`).
1108 uint8_t wire[17];
1109 for (int i = 0; i < 16; ++i)
1110 wire[i] = goal_id[i];
1111 wire[16] = accepted ? 1 : 0;
1112 self->accepts_.deliver(owner, wire, sizeof(wire));
1113 }
1114 self->upstream_goal_response(accepted, goal_id);
1115 if (self->user_.goal_response != nullptr)
1116 self->user_.goal_response(accepted, goal_id, self->user_.context);
1117 }

1397 void upstream_goal_response(bool accepted, const uint8_t goal_id[16]) {
1398#ifdef NROS_CPP_HAS_SHARED_PTR // hosted-family: action-interop
1399 size_t at = 0;
1400 UpstreamState* st = upstream_goal(goal_id, at);
1401 if (st == nullptr) return;
1402 UpstreamGoal& g = st->goals[at];
1403 if (g.accepted) return; // a duplicate acceptance
1404 auto cb = g.options.goal_response_callback;
1405 if (!accepted) {
1406 st->goals.erase(st->goals.begin() + static_cast<long>(at));
1407 if (cb) cb(nullptr);
1408 return;
1409 }
1410 typename GoalHandle::SharedPtr handle = ::std::make_shared<GoalHandle>(g.id);
1411 handle->feedback_aware_ = static_cast<bool>(g.options.feedback_callback);
1412 handle->result_aware_ = static_cast<bool>(g.options.result_callback);
1413 g.accepted = true;
1414 if (handle->result_aware_)
1415 g.handle = handle; // the pending result request keeps it, as upstream's does
1416 else
1417 g.weak = handle; // feedback-only: the caller's copy keeps it
1418 if (!handle->feedback_aware_ && !handle->result_aware_)
1419 st->goals.erase(st->goals.begin() + static_cast<long>(at));
1420 // `g` may be gone from here on: the callback can send another goal.
1421 if (cb) cb(handle);
1422 if (handle->result_aware_ && !get_result_async(goal_id).ok()) {
1423 // Nobody could ever answer it: unbind, as upstream would have thrown.
1424 handle->result_aware_ = false;
1425 if (upstream_goal(goal_id, at) != nullptr) {
1426 st->goals[at].handle.reset();
1427 st->goals[at].weak = handle;
1428 }
1429 }
1430#else
1431 (void)accepted;
1432 (void)goal_id;
1433#endif
1434 }
