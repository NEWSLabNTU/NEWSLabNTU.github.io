1283 void pump_cancels() {
1284#ifdef NROS_CPP_HAS_STD_PROMISE // hosted-family: action-interop
1285 UpstreamState* st = upstream_.template as<UpstreamState>();
1286 if (st == nullptr || !initialized_) return;
1287 for (int i = 0; i < ::rclcpp::detail::MAX_STALE_DRAIN_PER_POLL && !st->cancels.empty();
1288 ++i) {
1289 uint8_t goal[16] = {};
1290 int8_t code = 0;
1291 size_t count = 0;
1292 int64_t seq = ::rclcpp::detail::SEQUENCE_ID_UNKNOWN;
1293 bool taken = false; // a malformed reply still names its request
1294 nros_cpp_action_goal_info_t infos[::rclcpp_action::MAX_CANCEL_INFOS];
1295 const nros_cpp_ret_t ret = nros_cpp_action_client_take_cancel_reply(
1296 storage_, reinterpret_cast<uint8_t(*)[16]>(goal), &code, infos,
1297 ::rclcpp_action::MAX_CANCEL_INFOS, &count, &seq);
1298 if (ret == NROS_CPP_RET_TRY_AGAIN) break;
1299 if (ret != 0 && seq == ::rclcpp::detail::SEQUENCE_ID_UNKNOWN) continue;
1300 taken = ret == 0;
1301 // The waiter whose REQUEST this answers: by sequence id, or — from
1302 // a backend that assigns none, which holds one cancel in flight —
1303 // by the goal the request named.
1304 const GoalUUID key(goal);
1305 size_t at = 0;
1306 while (at < st->cancels.size() &&
1307 !(seq != ::rclcpp::detail::SEQUENCE_ID_UNKNOWN ? st->cancels[at].seq == seq
1308 : st->cancels[at].key == key))
1309 ++at;
1310 if (at == st->cancels.size()) {
1311 if (taken) st->stray_cancels.push_back(code);
1312 continue;
1313 }
1314 CancelWaiter w = static_cast<CancelWaiter&&>(st->cancels[at]);
1315 st->cancels.erase(st->cancels.begin() + static_cast<long>(at));
1316 if (!taken) { // malformed: the request is over, with no response
1317 w.promise.set_value(typename CancelResponse::SharedPtr());
1318 if (w.callback) w.callback(typename CancelResponse::SharedPtr());
1319 continue;
1320 }
1321 typename CancelResponse::SharedPtr resp = ::std::make_shared<CancelResponse>();
1322 resp->return_code = code;
1323 const size_t n = count < ::rclcpp_action::MAX_CANCEL_INFOS
1324 ? count
1325 : ::rclcpp_action::MAX_CANCEL_INFOS;
1326 for (size_t k = 0; k < n; ++k) {
1327 detail::CancelGoalInfo info;
1328 for (int b = 0; b < 16; ++b)
1329 info.goal_id.uuid[b] = infos[k].uuid[b];
1330 info.stamp.sec = infos[k].stamp_sec;
1331 info.stamp.nanosec = infos[k].stamp_nanosec;
1332 resp->goals_canceling.push_back(info);
1333 }
1334 w.promise.set_value(resp);
1335 if (w.callback) w.callback(resp);
1336 }
1337#endif
1338 }
