344template <typename M>
345Result Node::create_publisher(Publisher<M>& out, const char* topic, const ::nros::QoS& qos) {
346 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
347 // message the linked backend cannot encode never reaches the wire.
348 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
349 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
350 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
351 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
352 ffi_qos, out.storage_);
353 if (ret == 0) {
354 // Topic name lives C++-side; copy + null-terminate into the
355 // fixed-size buffer. Truncation is silent.
356 size_t topic_len = 0;
357 while (topic[topic_len] != '\0' && topic_len + 1 < sizeof(out.topic_name_)) {
358 out.topic_name_[topic_len] = topic[topic_len];
359 ++topic_len;
360 }
361 out.topic_name_[topic_len] = '\0';
362 out.initialized_ = true;
363 }
364 return Result(ret);
365}
