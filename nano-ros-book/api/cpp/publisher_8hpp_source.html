537 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
538 // message the linked backend cannot encode never reaches the wire.
539 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
540 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
541 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
542 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
543 ffi_qos, out.storage_);
544 if (ret == 0) {
545 // Topic name lives C++-side; copy + null-terminate into the
546 // fixed-size buffer. Truncation is silent.
547 ::nros::detail::assign_entity_name(out.topic_name_, topic);
548 out.initialized_ = true;
549 }
