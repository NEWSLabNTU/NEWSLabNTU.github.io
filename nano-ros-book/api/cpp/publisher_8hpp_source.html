631Result Node::create_publisher(Publisher<M>& out, const char* topic, const ::rclcpp::QoS& qos) {
632 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
633 // message the linked backend cannot encode never reaches the wire.
634 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
635 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
636 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
637 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
638 ffi_qos, out.storage_);
639 if (ret == 0) {
640 // Topic name lives C++-side; copy + null-terminate into the
641 // fixed-size buffer. Truncation is silent.
642 ::rclcpp::detail::assign_entity_name(out.topic_name_, topic);
643 out.initialized_ = true;
644 }
645 return Result(ret);
646}
