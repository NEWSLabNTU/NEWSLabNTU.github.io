337template <typename M>
338Result Node::create_publisher(Publisher<M>& out, const char* topic, const QoS& qos) {
339 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
340 // message the linked backend cannot encode never reaches the wire.
341 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
342 if (!initialized_) return Result(ErrorCode::NotInitialized);
343 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
344 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
345 ffi_qos, out.storage_);
346 if (ret == 0) {
347 // Topic name lives C++-side; copy + null-terminate into the
348 // fixed-size buffer. Truncation is silent.
349 size_t topic_len = 0;
350 while (topic[topic_len] != '\0' && topic_len + 1 < sizeof(out.topic_name_)) {
351 out.topic_name_[topic_len] = topic[topic_len];
352 ++topic_len;
353 }
354 out.topic_name_[topic_len] = '\0';
355 out.initialized_ = true;
356 }
357 return Result(ret);
358}
