584 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
585 // message the linked backend cannot encode never reaches the wire.
586 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
587 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
588 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
589 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
590 ffi_qos, out.storage_);
591 if (ret == 0) {
592 // Topic name lives C++-side; copy + null-terminate into the
593 // fixed-size buffer. Truncation is silent.
594 ::rclcpp::detail::assign_entity_name(out.topic_name_, topic);
595 out.initialized_ = true;
596 }
597 return Result(ret);
