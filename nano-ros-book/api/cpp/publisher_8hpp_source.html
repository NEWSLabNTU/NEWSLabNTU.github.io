570 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
571 // message the linked backend cannot encode never reaches the wire.
572 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
573 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
574 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
575 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
576 ffi_qos, out.storage_);
577 if (ret == 0) {
578 // Topic name lives C++-side; copy + null-terminate into the
579 // fixed-size buffer. Truncation is silent.
580 ::rclcpp::detail::assign_entity_name(out.topic_name_, topic);
581 out.initialized_ = true;
582 }
