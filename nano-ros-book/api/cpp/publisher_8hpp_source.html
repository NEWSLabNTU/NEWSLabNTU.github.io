161 template <typename W> Result publish_streamed(size_t total_len, W&& writer) {
162 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
163 struct Ctx {
164 W writer;
165 size_t total;
166 } ctx{static_cast<W&&>(writer), total_len};
167 auto size_cb = [](size_t* out_total, void* user) {
168 auto* c = static_cast<Ctx*>(user);
169 *out_total = c->total;
170 };

173 *out_written = c->writer(out_buf, cap);
174 };
175 return Result(nros_cpp_publisher_publish_streamed(storage_, size_cb, chunk_cb,
176 static_cast<void*>(&ctx)));

217 Result commit(size_t actual_len) {
218 if (!token_) return Result(::nros::ErrorCode::NotInitialized);
219 nros_cpp_ret_t ret = nros_cpp_publisher_commit(pub_, token_, actual_len);
220 pub_ = nullptr;
221 token_ = nullptr;
222 buf_ = nullptr;

230 nros_cpp_ret_t ret = nros_cpp_publisher_discard(pub_, token_);
231 pub_ = nullptr;
232 token_ = nullptr;
233 buf_ = nullptr;
234 cap_ = 0;

436template <typename M>
437Result Node::create_publisher(Publisher<M>& out, const char* topic, const ::nros::QoS& qos) {
438 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
439 // message the linked backend cannot encode never reaches the wire.
440 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
441 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
442 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
443 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
444 ffi_qos, out.storage_);
445 if (ret == 0) {
446 // Topic name lives C++-side; copy + null-terminate into the
447 // fixed-size buffer. Truncation is silent.
448 ::nros::detail::assign_entity_name(out.topic_name_, topic);
449 out.initialized_ = true;
450 }
451 return Result(ret);
452}
