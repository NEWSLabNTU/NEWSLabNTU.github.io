648Result Node::create_publisher(Publisher<M>& out, const char* topic, const ::rclcpp::QoS& qos) {
649 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
650 // message the linked backend cannot encode never reaches the wire.
651 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
652 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
653 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
654 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
655 ffi_qos, out.storage_);
656 if (ret == 0) {
657 // Topic name lives C++-side; copy + null-terminate into the
658 // fixed-size buffer. Truncation is silent.
659 ::rclcpp::detail::assign_entity_name(out.topic_name_, topic);
660 out.initialized_ = true;
661 }
662 return Result(ret);
663}
