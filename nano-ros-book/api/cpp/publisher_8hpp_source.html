357template <typename M>
358Result Node::create_publisher(Publisher<M>& out, const char* topic, const ::nros::QoS& qos) {
359 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
360 // message the linked backend cannot encode never reaches the wire.
361 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
362 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
363 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
364 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
365 ffi_qos, out.storage_);
366 if (ret == 0) {
367 // Topic name lives C++-side; copy + null-terminate into the
368 // fixed-size buffer. Truncation is silent.
369 size_t topic_len = 0;
370 while (topic[topic_len] != '\0' && topic_len + 1 < sizeof(out.topic_name_)) {
371 out.topic_name_[topic_len] = topic[topic_len];
372 ++topic_len;
373 }
374 out.topic_name_[topic_len] = '\0';
375 out.initialized_ = true;
376 }
377 return Result(ret);
378}
