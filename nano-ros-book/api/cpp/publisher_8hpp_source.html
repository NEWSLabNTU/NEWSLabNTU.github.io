493Result Node::create_publisher(Publisher<M>& out, const char* topic, const ::nros::QoS& qos) {
494 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
495 // message the linked backend cannot encode never reaches the wire.
496 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
497 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
498 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
499 nros_cpp_ret_t ret = nros_cpp_publisher_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
500 ffi_qos, out.storage_);
501 if (ret == 0) {
502 // Topic name lives C++-side; copy + null-terminate into the
503 // fixed-size buffer. Truncation is silent.
504 ::nros::detail::assign_entity_name(out.topic_name_, topic);
505 out.initialized_ = true;
506 }
507 return Result(ret);
508}
