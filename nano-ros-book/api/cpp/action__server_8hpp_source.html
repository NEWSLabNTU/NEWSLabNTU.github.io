715 // ── C trampolines ───────────────────────────────────────────────

717 // `ctx` is a pointer to this `ActionServer<A>` instance, so the
718 // trampoline reads the user's stored function pointer via the
719 // instance's own fields — no shared mutable statics.

721 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
722 void* ctx) {
723 auto* self = static_cast<Server*>(ctx);
724 if (!self) return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
725 GoalType g;
726 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
727 return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
728 }
729 if (self->user_goal_fn_ctx_ != nullptr) {
730 return ::rclcpp_action::to_ffi(
731 self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
732 }
733 if (self->user_goal_fn_ != nullptr) {
734 return ::rclcpp_action::to_ffi(self->user_goal_fn_(goal_id, g));
735 }
736 return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
737 }

739 static int32_t cancel_trampoline(const uint8_t goal_id[16], void* ctx) {
740 auto* self = static_cast<Server*>(ctx);
741 if (!self) return ::rclcpp_action::to_ffi(::rclcpp_action::CancelResponse::ACCEPT);
742 if (self->user_cancel_fn_ctx_ != nullptr) {
743 return ::rclcpp_action::to_ffi(
744 self->user_cancel_fn_ctx_(goal_id, self->user_cancel_ctx_));
745 }
746 if (self->user_cancel_fn_ != nullptr) {
747 return ::rclcpp_action::to_ffi(self->user_cancel_fn_(goal_id));
748 }
749 return ::rclcpp_action::to_ffi(::rclcpp_action::CancelResponse::ACCEPT);
750 }

752 static void accepted_trampoline(const uint8_t goal_id[16], void* ctx) {
753 auto* self = static_cast<Server*>(ctx);
754 if (!self) return;
755 if (self->user_accepted_fn_ctx_ != nullptr) {
756 self->user_accepted_fn_ctx_(goal_id, self->user_accepted_ctx_);
757 return;
758 }
759 if (self->user_accepted_fn_ != nullptr) {
760 self->user_accepted_fn_(goal_id);
761 }

764 ::rclcpp::Result install_callbacks() {
765 bool goal_set = (user_goal_fn_ != nullptr) || (user_goal_fn_ctx_ != nullptr);
766 bool cancel_set = (user_cancel_fn_ != nullptr) || (user_cancel_fn_ctx_ != nullptr);
767 bool accepted_set = (user_accepted_fn_ != nullptr) || (user_accepted_fn_ctx_ != nullptr);
768 nros_cpp_goal_callback_t gcb = goal_set ? &goal_trampoline : nullptr;
769 nros_cpp_cancel_callback_t ccb = cancel_set ? &cancel_trampoline : nullptr;
770 nros_cpp_ret_t ret = nros_cpp_action_server_set_callbacks(storage_, gcb, ccb, this);
771 if (ret != 0) return ::rclcpp::Result(ret);
772 // Issue 0796 — a SECOND install call rather than a fourth parameter on
773 // the first: `nros_cpp_action_server_set_callbacks` is declared in
774 // three places and called directly by C components, so growing it
775 // would have broken every one of them. Both calls pass the same
776 // `this`, which is the shared context slot the runtime keeps.
777 nros_cpp_accepted_callback_t acb = accepted_set ? &accepted_trampoline : nullptr;
778 return ::rclcpp::Result(nros_cpp_action_server_set_accepted_callback(storage_, acb, this));
779 }
