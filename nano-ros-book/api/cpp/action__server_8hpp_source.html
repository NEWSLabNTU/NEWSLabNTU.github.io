372 // ── C trampolines ───────────────────────────────────────────────
373 //
374 // `ctx` is a pointer to this `ActionServer<A>` instance, so the
375 // trampoline reads the user's stored function pointer via the
376 // instance's own fields — no shared mutable statics.

378 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
379 void* ctx) {
380 auto* self = static_cast<ActionServer*>(ctx);
381 if (!self) return static_cast<int32_t>(GoalResponse::Reject);
382 GoalType g;
383 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
384 return static_cast<int32_t>(GoalResponse::Reject);
385 }
386 if (self->user_goal_fn_ctx_ != nullptr) {
387 return static_cast<int32_t>(self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
388 }
389 if (self->user_goal_fn_ != nullptr) {
390 return static_cast<int32_t>(self->user_goal_fn_(goal_id, g));
391 }
392 return static_cast<int32_t>(GoalResponse::Reject);
393 }
394
395 static int32_t cancel_trampoline(const uint8_t goal_id[16], void* ctx) {
396 auto* self = static_cast<ActionServer*>(ctx);
397 if (!self) return static_cast<int32_t>(CancelResponse::Accept);
398 if (self->user_cancel_fn_ctx_ != nullptr) {
399 return static_cast<int32_t>(self->user_cancel_fn_ctx_(goal_id, self->user_cancel_ctx_));
400 }
401 if (self->user_cancel_fn_ != nullptr) {
402 return static_cast<int32_t>(self->user_cancel_fn_(goal_id));
403 }
404 return static_cast<int32_t>(CancelResponse::Accept);
405 }
406
407 static void accepted_trampoline(const uint8_t goal_id[16], void* ctx) {
408 auto* self = static_cast<ActionServer*>(ctx);
409 if (!self) return;
410 if (self->user_accepted_fn_ctx_ != nullptr) {
411 self->user_accepted_fn_ctx_(goal_id, self->user_accepted_ctx_);
412 return;
413 }
414 if (self->user_accepted_fn_ != nullptr) {
415 self->user_accepted_fn_(goal_id);
416 }
417 }

419 Result install_callbacks() {
420 bool goal_set = (user_goal_fn_ != nullptr) || (user_goal_fn_ctx_ != nullptr);
421 bool cancel_set = (user_cancel_fn_ != nullptr) || (user_cancel_fn_ctx_ != nullptr);
422 bool accepted_set = (user_accepted_fn_ != nullptr) || (user_accepted_fn_ctx_ != nullptr);
423 nros_cpp_goal_callback_t gcb = goal_set ? &goal_trampoline : nullptr;
424 nros_cpp_cancel_callback_t ccb = cancel_set ? &cancel_trampoline : nullptr;
425 nros_cpp_ret_t ret = nros_cpp_action_server_set_callbacks(storage_, gcb, ccb, this);
426 if (ret != 0) return Result(ret);
427 // Issue 0796 — a SECOND install call rather than a fourth parameter on
428 // the first: `nros_cpp_action_server_set_callbacks` is declared in
429 // three places and called directly by C components, so growing it
430 // would have broken every one of them. Both calls pass the same
431 // `this`, which is the shared context slot the runtime keeps.
432 nros_cpp_accepted_callback_t acb = accepted_set ? &accepted_trampoline : nullptr;
433 return Result(nros_cpp_action_server_set_accepted_callback(storage_, acb, this));
434 }

461template <typename A>
462Result Node::create_action_server(ActionServer<A>& out, const char* action_name, const QoS& qos,
463 const ActionServerOptions& options) {
464 if (!initialized_) return Result(ErrorCode::NotInitialized);
465 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
466 nros_cpp_ret_t ret = nros_cpp_action_server_create(&handle_, action_name, A::TYPE_NAME,
467 A::Goal::TYPE_HASH, ffi_qos, out.storage_);
468 if (ret != 0) return Result(ret);
469 // Register with executor — creates transport handles (3 queryables + 2 publishers).
470 // Deferred from create to avoid FreeRTOS QEMU deadlocks. Phase 87.6:
471 // names are passed at register-time (buffers live on the C++
472 // `nros::ActionServer<A>` class, not in the runtime struct).
473 // Phase 189.M3.3.c — `sched_context` binds the action's (arena-registered)
474 // goal-service handle to a scheduling context. UNSET ⇒ 0 (inherit, no-op).
475 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
476 ? 0u
477 : static_cast<uint8_t>(options.sched_context);
478 ret = nros_cpp_action_server_register(out.storage_, executor_handle_, action_name, A::TYPE_NAME,
479 A::Goal::TYPE_HASH, sched);
480 if (ret == 0) {
481 // Phase 87.6: copy action_name into the C++-owned buffer for
482 // `get_action_name()` accessor.
483 size_t name_len = 0;
484 while (action_name[name_len] != '\0' && name_len + 1 < sizeof(out.action_name_)) {
485 out.action_name_[name_len] = action_name[name_len];
486 ++name_len;
487 }
488 out.action_name_[name_len] = '\0';
489 out.executor_ = executor_handle_;
490 out.initialized_ = true;
491 }
492 return Result(ret);
493}
