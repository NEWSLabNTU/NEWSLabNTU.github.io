94 explicit GoalUUID(const uint8_t* src) : uuid{} {
95 for (size_t i = 0; i < SIZE; ++i) {
96 uuid[i] = src[i];

99
101 uint8_t* data() { return uuid; }
102 const uint8_t* data() const { return uuid; }
103
105 bool is_zero() const {
106 for (size_t i = 0; i < SIZE; ++i) {
107 if (uuid[i] != 0) return false;
108 }
109 return true;
110 }

113 for (size_t i = 0; i < SIZE; ++i) {
114 if (uuid[i] != other.uuid[i]) return false;
115 }
116 return true;
117 }
118
119 bool operator!=(const GoalUUID& other) const { return !(*this == other); }

124 bool operator<(const GoalUUID& other) const {
125 for (size_t i = 0; i < SIZE; ++i) {

615 // ── C trampolines ───────────────────────────────────────────────
616 //
617 // `ctx` is a pointer to this `ActionServer<A>` instance, so the
618 // trampoline reads the user's stored function pointer via the
619 // instance's own fields — no shared mutable statics.

621 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
622 void* ctx) {
623 auto* self = static_cast<Server*>(ctx);
624 if (!self) return static_cast<int32_t>(::nros::GoalResponse::Reject);
625 GoalType g;
626 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
627 return static_cast<int32_t>(::nros::GoalResponse::Reject);
628 }
629 if (self->user_goal_fn_ctx_ != nullptr) {
630 return static_cast<int32_t>(self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
631 }
632 if (self->user_goal_fn_ != nullptr) {
633 return static_cast<int32_t>(self->user_goal_fn_(goal_id, g));
634 }
635 return static_cast<int32_t>(::nros::GoalResponse::Reject);
636 }
637
638 static int32_t cancel_trampoline(const uint8_t goal_id[16], void* ctx) {
639 auto* self = static_cast<Server*>(ctx);
640 if (!self) return static_cast<int32_t>(::nros::CancelResponse::Accept);
641 if (self->user_cancel_fn_ctx_ != nullptr) {
642 return static_cast<int32_t>(self->user_cancel_fn_ctx_(goal_id, self->user_cancel_ctx_));
643 }
644 if (self->user_cancel_fn_ != nullptr) {
645 return static_cast<int32_t>(self->user_cancel_fn_(goal_id));
646 }
647 return static_cast<int32_t>(::nros::CancelResponse::Accept);
648 }
649
650 static void accepted_trampoline(const uint8_t goal_id[16], void* ctx) {
651 auto* self = static_cast<Server*>(ctx);
652 if (!self) return;
653 if (self->user_accepted_fn_ctx_ != nullptr) {
654 self->user_accepted_fn_ctx_(goal_id, self->user_accepted_ctx_);
655 return;
656 }
657 if (self->user_accepted_fn_ != nullptr) {
658 self->user_accepted_fn_(goal_id);
659 }
660 }

662 ::nros::Result install_callbacks() {
663 bool goal_set = (user_goal_fn_ != nullptr) || (user_goal_fn_ctx_ != nullptr);
664 bool cancel_set = (user_cancel_fn_ != nullptr) || (user_cancel_fn_ctx_ != nullptr);
665 bool accepted_set = (user_accepted_fn_ != nullptr) || (user_accepted_fn_ctx_ != nullptr);
666 nros_cpp_goal_callback_t gcb = goal_set ? &goal_trampoline : nullptr;
667 nros_cpp_cancel_callback_t ccb = cancel_set ? &cancel_trampoline : nullptr;
668 nros_cpp_ret_t ret = nros_cpp_action_server_set_callbacks(storage_, gcb, ccb, this);
669 if (ret != 0) return ::nros::Result(ret);
670 // Issue 0796 — a SECOND install call rather than a fourth parameter on
671 // the first: `nros_cpp_action_server_set_callbacks` is declared in
672 // three places and called directly by C components, so growing it
673 // would have broken every one of them. Both calls pass the same
674 // `this`, which is the shared context slot the runtime keeps.
675 nros_cpp_accepted_callback_t acb = accepted_set ? &accepted_trampoline : nullptr;
676 return ::nros::Result(nros_cpp_action_server_set_accepted_callback(storage_, acb, this));
677 }

719Result Node::create_action_server(::nros::ActionServer<A>& out, const char* action_name,
720 const ::nros::QoS& qos,
721 const ::nros::ActionServerOptions& options) {
722 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
723 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
724 nros_cpp_ret_t ret = nros_cpp_action_server_create(&handle_, action_name, A::TYPE_NAME,
725 A::Goal::TYPE_HASH, ffi_qos, out.storage_);
726 if (ret != 0) return Result(ret);
727 // Register with executor — creates transport handles (3 queryables + 2 publishers).
728 // Deferred from create to avoid FreeRTOS QEMU deadlocks. Phase 87.6:
729 // names are passed at register-time (buffers live on the C++
730 // `nros::ActionServer<A>` class, not in the runtime struct).
731 // Phase 189.M3.3.c — `sched_context` binds the action's (arena-registered)
732 // goal-service handle to a scheduling context. UNSET ⇒ 0 (inherit, no-op).
733 uint8_t sched = (options.sched_context == ::nros::SCHED_CONTEXT_UNSET)
734 ? 0u
735 : static_cast<uint8_t>(options.sched_context);
736 ret = nros_cpp_action_server_register(out.storage_, executor_handle_, action_name, A::TYPE_NAME,
737 A::Goal::TYPE_HASH, sched);
738 if (ret == 0) {
739 out.executor_ = executor_handle_;
740 // phase-417 W4.b — remember the name for `get_action_name()`. Truncating
741 // copy, as `Subscription`'s and the two polling tiers' do: a name longer
742 // than `nros::ACTION_NAME_MAX` is not a reason to fail a create that the
743 // runtime already accepted.
744 ::nros::detail::assign_entity_name(out.action_name_, action_name);
745 out.initialized_ = true;
746 }
747 return Result(ret);
