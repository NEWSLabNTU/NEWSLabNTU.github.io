83struct GoalUUID {

88 uint8_t uuid[16];
89
91 GoalUUID() : uuid{} {}
92
94 explicit GoalUUID(const uint8_t* src) : uuid{} {
95 for (size_t i = 0; i < SIZE; ++i) {
96 uuid[i] = src[i];
97 }
98 }
99
101 uint8_t* data() { return uuid; }
102 const uint8_t* data() const { return uuid; }
103
105 bool is_zero() const {
106 for (size_t i = 0; i < SIZE; ++i) {
107 if (uuid[i] != 0) return false;
108 }
109 return true;
110 }
111
112 bool operator==(const GoalUUID& other) const {
113 for (size_t i = 0; i < SIZE; ++i) {
114 if (uuid[i] != other.uuid[i]) return false;
115 }
116 return true;
117 }
118
119 bool operator!=(const GoalUUID& other) const { return !(*this == other); }
120
124 bool operator<(const GoalUUID& other) const {
125 for (size_t i = 0; i < SIZE; ++i) {
126 if (uuid[i] != other.uuid[i]) return uuid[i] < other.uuid[i];
127 }
128 return false;
129 }

714 // ── C trampolines ───────────────────────────────────────────────
715 //
716 // `ctx` is a pointer to this `ActionServer<A>` instance, so the
717 // trampoline reads the user's stored function pointer via the
718 // instance's own fields — no shared mutable statics.
719
720 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
721 void* ctx) {
722 auto* self = static_cast<Server*>(ctx);
723 if (!self) return ::nros::to_ffi(::nros::GoalResponse::REJECT);
724 GoalType g;
725 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
726 return ::nros::to_ffi(::nros::GoalResponse::REJECT);
727 }
728 if (self->user_goal_fn_ctx_ != nullptr) {
729 return ::nros::to_ffi(self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
730 }
731 if (self->user_goal_fn_ != nullptr) {
732 return ::nros::to_ffi(self->user_goal_fn_(goal_id, g));
733 }
734 return ::nros::to_ffi(::nros::GoalResponse::REJECT);
735 }
736
737 static int32_t cancel_trampoline(const uint8_t goal_id[16], void* ctx) {
738 auto* self = static_cast<Server*>(ctx);
739 if (!self) return ::nros::to_ffi(::nros::CancelResponse::ACCEPT);
740 if (self->user_cancel_fn_ctx_ != nullptr) {
741 return ::nros::to_ffi(self->user_cancel_fn_ctx_(goal_id, self->user_cancel_ctx_));
742 }
743 if (self->user_cancel_fn_ != nullptr) {
744 return ::nros::to_ffi(self->user_cancel_fn_(goal_id));
745 }
746 return ::nros::to_ffi(::nros::CancelResponse::ACCEPT);
747 }
748
749 static void accepted_trampoline(const uint8_t goal_id[16], void* ctx) {
750 auto* self = static_cast<Server*>(ctx);
751 if (!self) return;
752 if (self->user_accepted_fn_ctx_ != nullptr) {
753 self->user_accepted_fn_ctx_(goal_id, self->user_accepted_ctx_);
754 return;
755 }
756 if (self->user_accepted_fn_ != nullptr) {
757 self->user_accepted_fn_(goal_id);
758 }
759 }

761 ::nros::Result install_callbacks() {
762 bool goal_set = (user_goal_fn_ != nullptr) || (user_goal_fn_ctx_ != nullptr);
763 bool cancel_set = (user_cancel_fn_ != nullptr) || (user_cancel_fn_ctx_ != nullptr);
764 bool accepted_set = (user_accepted_fn_ != nullptr) || (user_accepted_fn_ctx_ != nullptr);
765 nros_cpp_goal_callback_t gcb = goal_set ? &goal_trampoline : nullptr;
766 nros_cpp_cancel_callback_t ccb = cancel_set ? &cancel_trampoline : nullptr;
767 nros_cpp_ret_t ret = nros_cpp_action_server_set_callbacks(storage_, gcb, ccb, this);
768 if (ret != 0) return ::nros::Result(ret);
769 // Issue 0796 — a SECOND install call rather than a fourth parameter on
770 // the first: `nros_cpp_action_server_set_callbacks` is declared in
771 // three places and called directly by C components, so growing it
772 // would have broken every one of them. Both calls pass the same
773 // `this`, which is the shared context slot the runtime keeps.
774 nros_cpp_accepted_callback_t acb = accepted_set ? &accepted_trampoline : nullptr;
775 return ::nros::Result(nros_cpp_action_server_set_accepted_callback(storage_, acb, this));
776 }

817template <typename A>
818Result Node::create_action_server(::nros::ActionServer<A>& out, const char* action_name,
819 const ::nros::QoS& qos,
820 const ::nros::ActionServerOptions& options) {
821 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
822 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
823 nros_cpp_ret_t ret = nros_cpp_action_server_create(&handle_, action_name, A::TYPE_NAME,
824 A::Goal::TYPE_HASH, ffi_qos, out.storage_);
825 if (ret != 0) return Result(ret);
826 // Register with executor — creates transport handles (3 queryables + 2 publishers).
827 // Deferred from create to avoid FreeRTOS QEMU deadlocks. Phase 87.6:
828 // names are passed at register-time (buffers live on the C++
829 // `nros::ActionServer<A>` class, not in the runtime struct).
830 // Phase 189.M3.3.c — `sched_context` binds the action's (arena-registered)
831 // goal-service handle to a scheduling context. UNSET ⇒ 0 (inherit, no-op).
832 uint8_t sched = (options.sched_context == ::nros::SCHED_CONTEXT_UNSET)
833 ? 0u
834 : static_cast<uint8_t>(options.sched_context);
835 ret = nros_cpp_action_server_register(out.storage_, executor_handle_, action_name, A::TYPE_NAME,
836 A::Goal::TYPE_HASH, sched);
837 if (ret == 0) {
838 out.executor_ = executor_handle_;
839 // phase-417 W4.b — remember the name for `get_action_name()`. Truncating
840 // copy, as `Subscription`'s and the two polling tiers' do: a name longer
841 // than `nros::ACTION_NAME_MAX` is not a reason to fail a create that the
842 // runtime already accepted.
843 ::nros::detail::assign_entity_name(out.action_name_, action_name);
844 out.initialized_ = true;
845 }
846 return Result(ret);
