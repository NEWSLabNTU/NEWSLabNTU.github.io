515 // ── C trampolines ───────────────────────────────────────────────
516 //
517 // `ctx` is a pointer to this `ActionServer<A>` instance, so the
518 // trampoline reads the user's stored function pointer via the
519 // instance's own fields — no shared mutable statics.
520
521 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
522 void* ctx) {
523 auto* self = static_cast<ActionServer*>(ctx);
524 if (!self) return static_cast<int32_t>(GoalResponse::Reject);
525 GoalType g;
526 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
527 return static_cast<int32_t>(GoalResponse::Reject);
528 }
529 if (self->user_goal_fn_ctx_ != nullptr) {
530 return static_cast<int32_t>(self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
531 }
532 if (self->user_goal_fn_ != nullptr) {
533 return static_cast<int32_t>(self->user_goal_fn_(goal_id, g));
534 }
535 return static_cast<int32_t>(GoalResponse::Reject);
536 }
537
538 static int32_t cancel_trampoline(const uint8_t goal_id[16], void* ctx) {
539 auto* self = static_cast<ActionServer*>(ctx);
540 if (!self) return static_cast<int32_t>(CancelResponse::Accept);
541 if (self->user_cancel_fn_ctx_ != nullptr) {
542 return static_cast<int32_t>(self->user_cancel_fn_ctx_(goal_id, self->user_cancel_ctx_));
543 }
544 if (self->user_cancel_fn_ != nullptr) {
545 return static_cast<int32_t>(self->user_cancel_fn_(goal_id));
546 }
547 return static_cast<int32_t>(CancelResponse::Accept);
548 }
549
550 static void accepted_trampoline(const uint8_t goal_id[16], void* ctx) {
551 auto* self = static_cast<ActionServer*>(ctx);
552 if (!self) return;
553 if (self->user_accepted_fn_ctx_ != nullptr) {
554 self->user_accepted_fn_ctx_(goal_id, self->user_accepted_ctx_);
555 return;
556 }
557 if (self->user_accepted_fn_ != nullptr) {
558 self->user_accepted_fn_(goal_id);
559 }
560 }

562 Result install_callbacks() {
563 bool goal_set = (user_goal_fn_ != nullptr) || (user_goal_fn_ctx_ != nullptr);
564 bool cancel_set = (user_cancel_fn_ != nullptr) || (user_cancel_fn_ctx_ != nullptr);
565 bool accepted_set = (user_accepted_fn_ != nullptr) || (user_accepted_fn_ctx_ != nullptr);
566 nros_cpp_goal_callback_t gcb = goal_set ? &goal_trampoline : nullptr;
567 nros_cpp_cancel_callback_t ccb = cancel_set ? &cancel_trampoline : nullptr;
568 nros_cpp_ret_t ret = nros_cpp_action_server_set_callbacks(storage_, gcb, ccb, this);
569 if (ret != 0) return Result(ret);
570 // Issue 0796 — a SECOND install call rather than a fourth parameter on
571 // the first: `nros_cpp_action_server_set_callbacks` is declared in
572 // three places and called directly by C components, so growing it
573 // would have broken every one of them. Both calls pass the same
574 // `this`, which is the shared context slot the runtime keeps.
575 nros_cpp_accepted_callback_t acb = accepted_set ? &accepted_trampoline : nullptr;
576 return Result(nros_cpp_action_server_set_accepted_callback(storage_, acb, this));
577 }

610Result Node::create_action_server(ActionServer<A>& out, const char* action_name, const QoS& qos,
611 const ActionServerOptions& options) {
612 if (!initialized_) return Result(ErrorCode::NotInitialized);
613 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
614 nros_cpp_ret_t ret = nros_cpp_action_server_create(&handle_, action_name, A::TYPE_NAME,
615 A::Goal::TYPE_HASH, ffi_qos, out.storage_);
616 if (ret != 0) return Result(ret);
617 // Register with executor — creates transport handles (3 queryables + 2 publishers).
618 // Deferred from create to avoid FreeRTOS QEMU deadlocks. Phase 87.6:
619 // names are passed at register-time (buffers live on the C++
620 // `nros::ActionServer<A>` class, not in the runtime struct).
621 // Phase 189.M3.3.c — `sched_context` binds the action's (arena-registered)
622 // goal-service handle to a scheduling context. UNSET ⇒ 0 (inherit, no-op).
623 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
624 ? 0u
625 : static_cast<uint8_t>(options.sched_context);
626 ret = nros_cpp_action_server_register(out.storage_, executor_handle_, action_name, A::TYPE_NAME,
627 A::Goal::TYPE_HASH, sched);
628 if (ret == 0) {
629 out.executor_ = executor_handle_;
630 out.initialized_ = true;
631 }
632 return Result(ret);
633}
