92 explicit GoalUUID(const uint8_t* src) : uuid{} {
93 for (size_t i = 0; i < SIZE; ++i) {
94 uuid[i] = src[i];
95 }
96 }
97
99 uint8_t* data() { return uuid; }
100 const uint8_t* data() const { return uuid; }
101
103 bool is_zero() const {
104 for (size_t i = 0; i < SIZE; ++i) {
105 if (uuid[i] != 0) return false;
106 }
107 return true;
108 }
109

111 for (size_t i = 0; i < SIZE; ++i) {
112 if (uuid[i] != other.uuid[i]) return false;
113 }
114 return true;
115 }
116

118
122 bool operator<(const GoalUUID& other) const {
123 for (size_t i = 0; i < SIZE; ++i) {

548 // ── C trampolines ───────────────────────────────────────────────
549 //
550 // `ctx` is a pointer to this `ActionServer<A>` instance, so the
551 // trampoline reads the user's stored function pointer via the
552 // instance's own fields — no shared mutable statics.

554 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
555 void* ctx) {
556 auto* self = static_cast<Server*>(ctx);
557 if (!self) return static_cast<int32_t>(::nros::GoalResponse::Reject);
558 GoalType g;
559 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
560 return static_cast<int32_t>(::nros::GoalResponse::Reject);
561 }
562 if (self->user_goal_fn_ctx_ != nullptr) {
563 return static_cast<int32_t>(self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
564 }
565 if (self->user_goal_fn_ != nullptr) {
566 return static_cast<int32_t>(self->user_goal_fn_(goal_id, g));
567 }
568 return static_cast<int32_t>(::nros::GoalResponse::Reject);
569 }
570
571 static int32_t cancel_trampoline(const uint8_t goal_id[16], void* ctx) {
572 auto* self = static_cast<Server*>(ctx);
573 if (!self) return static_cast<int32_t>(::nros::CancelResponse::Accept);
574 if (self->user_cancel_fn_ctx_ != nullptr) {
575 return static_cast<int32_t>(self->user_cancel_fn_ctx_(goal_id, self->user_cancel_ctx_));
576 }
577 if (self->user_cancel_fn_ != nullptr) {
578 return static_cast<int32_t>(self->user_cancel_fn_(goal_id));
579 }
580 return static_cast<int32_t>(::nros::CancelResponse::Accept);
581 }

583 static void accepted_trampoline(const uint8_t goal_id[16], void* ctx) {
584 auto* self = static_cast<Server*>(ctx);
585 if (!self) return;
586 if (self->user_accepted_fn_ctx_ != nullptr) {
587 self->user_accepted_fn_ctx_(goal_id, self->user_accepted_ctx_);
588 return;

591 self->user_accepted_fn_(goal_id);

595 ::nros::Result install_callbacks() {
596 bool goal_set = (user_goal_fn_ != nullptr) || (user_goal_fn_ctx_ != nullptr);
597 bool cancel_set = (user_cancel_fn_ != nullptr) || (user_cancel_fn_ctx_ != nullptr);
598 bool accepted_set = (user_accepted_fn_ != nullptr) || (user_accepted_fn_ctx_ != nullptr);
599 nros_cpp_goal_callback_t gcb = goal_set ? &goal_trampoline : nullptr;
600 nros_cpp_cancel_callback_t ccb = cancel_set ? &cancel_trampoline : nullptr;
601 nros_cpp_ret_t ret = nros_cpp_action_server_set_callbacks(storage_, gcb, ccb, this);
602 if (ret != 0) return ::nros::Result(ret);
603 // Issue 0796 — a SECOND install call rather than a fourth parameter on
604 // the first: `nros_cpp_action_server_set_callbacks` is declared in
605 // three places and called directly by C components, so growing it
606 // would have broken every one of them. Both calls pass the same
607 // `this`, which is the shared context slot the runtime keeps.
608 nros_cpp_accepted_callback_t acb = accepted_set ? &accepted_trampoline : nullptr;
609 return ::nros::Result(nros_cpp_action_server_set_accepted_callback(storage_, acb, this));
610 }
