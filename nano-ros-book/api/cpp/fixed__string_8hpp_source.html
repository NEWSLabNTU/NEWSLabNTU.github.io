99 // phase-417 W1.c — `std::string` interop.
100 //
101 // Codegen emits a message's string field as `FixedString<N>` where
102 // upstream rosidl emits `std::string`, so the very first line of a ported
103 // node —
104 //
105 // message.data = "Hello, world! " + std::to_string(count_++);
106 //
107 // did not compile: the only assignment was from `const char*`. These
108 // members COPY and call through to the existing `const char*` paths; they
109 // add no data member and no second code path (RFC-0089 §"Who implements an
110 // adopted name" — container/string conversions that copy and delegate are
111 // wrapper-side ergonomics), so the `char[N]` layout the FFI relies on is
112 // unchanged.
113 //
114 // Truncation is the same bounded-capacity contract `operator=(const
115 // char*)` has always had: a longer source is cut at `N - 1` characters.
