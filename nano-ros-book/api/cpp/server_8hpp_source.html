61// these entry points used to pass matched none of them. Generated action types

124struct GoalUUID {
126 static const size_t SIZE = 16;
127
129 uint8_t uuid[16];
130
132 GoalUUID() : uuid{} {}
133
135 explicit GoalUUID(const uint8_t* src) : uuid{} {
136 for (size_t i = 0; i < SIZE; ++i) {
137 uuid[i] = src[i];
138 }
139 }
140
142 uint8_t* data() { return uuid; }
143 const uint8_t* data() const { return uuid; }
144
146 bool is_zero() const {
147 for (size_t i = 0; i < SIZE; ++i) {
148 if (uuid[i] != 0) return false;
149 }
150 return true;
151 }
152
153 bool operator==(const GoalUUID& other) const {
154 for (size_t i = 0; i < SIZE; ++i) {
155 if (uuid[i] != other.uuid[i]) return false;
156 }
157 return true;
158 }
159
160 bool operator!=(const GoalUUID& other) const { return !(*this == other); }
161
165 bool operator<(const GoalUUID& other) const {
166 for (size_t i = 0; i < SIZE; ++i) {
167 if (uuid[i] != other.uuid[i]) return uuid[i] < other.uuid[i];
168 }
169 return false;
170 }
171};

759 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
760 void* ctx) {
761 auto* self = static_cast<Server*>(ctx);
762 if (!self) return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
763 GoalType g;
764 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
765 return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
766 }
767 if (self->user_goal_fn_ctx_ != nullptr) {
768 return ::rclcpp_action::to_ffi(
769 self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
770 }
771 if (self->user_goal_fn_ != nullptr) {
772 return ::rclcpp_action::to_ffi(self->user_goal_fn_(goal_id, g));
773 }
774 return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
775 }

777 static int32_t cancel_trampoline(const uint8_t goal_id[16], void* ctx) {
778 auto* self = static_cast<Server*>(ctx);
779 if (!self) return ::rclcpp_action::to_ffi(::rclcpp_action::CancelResponse::ACCEPT);
780 if (self->user_cancel_fn_ctx_ != nullptr) {
781 return ::rclcpp_action::to_ffi(
782 self->user_cancel_fn_ctx_(goal_id, self->user_cancel_ctx_));
783 }
784 if (self->user_cancel_fn_ != nullptr) {
785 return ::rclcpp_action::to_ffi(self->user_cancel_fn_(goal_id));
786 }
787 return ::rclcpp_action::to_ffi(::rclcpp_action::CancelResponse::ACCEPT);
788 }

790 static void accepted_trampoline(const uint8_t goal_id[16], void* ctx) {
791 auto* self = static_cast<Server*>(ctx);
792 if (!self) return;
793 if (self->user_accepted_fn_ctx_ != nullptr) {
794 self->user_accepted_fn_ctx_(goal_id, self->user_accepted_ctx_);
795 return;
796 }
797 if (self->user_accepted_fn_ != nullptr) {

802 ::rclcpp::Result install_callbacks() {
803 bool goal_set = (user_goal_fn_ != nullptr) || (user_goal_fn_ctx_ != nullptr);
804 bool cancel_set = (user_cancel_fn_ != nullptr) || (user_cancel_fn_ctx_ != nullptr);
805 bool accepted_set = (user_accepted_fn_ != nullptr) || (user_accepted_fn_ctx_ != nullptr);
806 nros_cpp_goal_callback_t gcb = goal_set ? &goal_trampoline : nullptr;
807 nros_cpp_cancel_callback_t ccb = cancel_set ? &cancel_trampoline : nullptr;
808 nros_cpp_ret_t ret = nros_cpp_action_server_set_callbacks(storage_, gcb, ccb, this);
809 if (ret != 0) return ::rclcpp::Result(ret);
810 // Issue 0796 — a SECOND install call rather than a fourth parameter on
811 // the first: `nros_cpp_action_server_set_callbacks` is declared in
812 // three places and called directly by C components, so growing it
813 // would have broken every one of them. Both calls pass the same
814 // `this`, which is the shared context slot the runtime keeps.
815 nros_cpp_accepted_callback_t acb = accepted_set ? &accepted_trampoline : nullptr;
816 return ::rclcpp::Result(nros_cpp_action_server_set_accepted_callback(storage_, acb, this));
817 }

850Result Node::create_action_server(::rclcpp_action::Server<A>& out, const char* action_name,
851 const ::rclcpp::QoS& qos,
852 const ::rclcpp::ActionServerOptions& options) {
853 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
854 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
855 nros_cpp_ret_t ret = nros_cpp_action_server_create(&handle_, action_name, A::TYPE_NAME,
856 A::Goal::TYPE_HASH, ffi_qos, out.storage_);
857 if (ret != 0) return Result(ret);
858 // Register with executor — creates transport handles (3 queryables + 2 publishers).
859 // Deferred from create to avoid FreeRTOS QEMU deadlocks. Phase 87.6:
860 // names are passed at register-time (buffers live on the C++
861 // `rclcpp_action::Server<A>` class, not in the runtime struct).
862 // Phase 189.M3.3.c — `sched_context` binds the action's (arena-registered)
863 // goal-service handle to a scheduling context. UNSET ⇒ 0 (inherit, no-op).
864 uint8_t sched = (options.sched_context == ::rclcpp::SCHED_CONTEXT_UNSET)
865 ? 0u
866 : static_cast<uint8_t>(options.sched_context);
867 const nros_cpp_action_hashes_t hashes = ::rclcpp::detail::action_endpoint_hashes<A>();
868 ret = nros_cpp_action_server_register(out.storage_, executor_handle_, action_name, A::TYPE_NAME,
869 A::Goal::TYPE_HASH, &hashes, sched);
870 if (ret == 0) {
871 out.executor_ = executor_handle_;
872 // phase-417 W4.b — remember the name for `get_action_name()`. Truncating
873 // copy, as `Subscription`'s and the two polling tiers' do: a name longer
874 // than `rclcpp_action::ACTION_NAME_MAX` is not a reason to fail a create that the
875 // runtime already accepted.
876 ::rclcpp::detail::assign_entity_name(out.action_name_, action_name);
877 out.initialized_ = true;
878 }
879 return Result(ret);
880}
