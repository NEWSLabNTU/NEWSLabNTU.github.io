126struct GoalUUID {
128 static const size_t SIZE = 16;
129
131 uint8_t uuid[16];
132
134 GoalUUID() : uuid{} {}
135
137 explicit GoalUUID(const uint8_t* src) : uuid{} {
138 for (size_t i = 0; i < SIZE; ++i) {
139 uuid[i] = src[i];
140 }
141 }
142
144 uint8_t* data() { return uuid; }
145 const uint8_t* data() const { return uuid; }
146
148 bool is_zero() const {
149 for (size_t i = 0; i < SIZE; ++i) {
150 if (uuid[i] != 0) return false;
151 }
152 return true;
153 }
154
155 bool operator==(const GoalUUID& other) const {
156 for (size_t i = 0; i < SIZE; ++i) {
157 if (uuid[i] != other.uuid[i]) return false;
158 }
159 return true;
160 }
161
162 bool operator!=(const GoalUUID& other) const { return !(*this == other); }
163
167 bool operator<(const GoalUUID& other) const {
168 for (size_t i = 0; i < SIZE; ++i) {
169 if (uuid[i] != other.uuid[i]) return uuid[i] < other.uuid[i];
170 }
171 return false;
172 }
173};

812 bool goal_status(const uint8_t goal_id[16], int8_t& out) {
813 if (!initialized_) return false;
814 struct Query {
815 const uint8_t* id;
816 int8_t status;
817 bool found;
818 } q = {goal_id, 0, false};
819 auto visit = [](const uint8_t(*id)[16], int8_t status, void* ctx) {
820 Query* qq = static_cast<Query*>(ctx);
821 if (::memcmp(*id, qq->id, 16) == 0) {
822 qq->status = status;
823 qq->found = true;
824 }
825 };
826 if (nros_cpp_action_server_for_each_active_goal(storage_, executor_, +visit, &q) != 0)
827 return false;
828 out = q.status;
829 return q.found;

834 // `ctx` is a pointer to this `ActionServer<A>` instance, so the
835 // trampoline reads the user's stored function pointer via the
836 // instance's own fields — no shared mutable statics.

838 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
839 void* ctx) {
840 auto* self = static_cast<Server*>(ctx);
841 if (!self) return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
842 GoalType g;
843 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
844 return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
845 }
846 if (self->user_goal_fn_ctx_ != nullptr) {
847 return ::rclcpp_action::to_ffi(
848 self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
849 }
850 if (self->user_goal_fn_ != nullptr) {
851 return ::rclcpp_action::to_ffi(self->user_goal_fn_(goal_id, g));
852 }
853 return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
854 }

881 ::rclcpp::Result install_callbacks() {

1034 static void rehome(Server<A>& s) {
1035#ifdef NROS_CPP_HAS_SHARED_PTR // hosted-family: action-interop
1036 if (State* st = s.upstream_.template as<State>()) st->server = &s;
1037#else
1038 (void)s;

1042#ifdef NROS_CPP_HAS_SHARED_PTR // hosted-family: action-interop
1043 using GoalFn = ::std::function<GoalResponse(const GoalUUID&, ::std::shared_ptr<const Goal>)>;
1044 using CancelFn = ::std::function<CancelResponse(::std::shared_ptr<Handle>)>;
1045 using AcceptedFn = ::std::function<void(::std::shared_ptr<Handle>)>;
1046
1047 struct Entry {
1048 GoalUUID id;
1049 ::std::shared_ptr<const Goal> goal;
1052 bool execute;
1053 };
1054 struct State {
1055 GoalFn goal;
1056 CancelFn cancel;
1057 AcceptedFn accepted;
1058 Server<A>* server;
1059 ::std::vector<Entry> goals;
1060 };
1061
1062 static void destroy(void* p) { delete static_cast<State*>(p); }
1063
1064 template <typename G, typename C, typename Acc>
1065 static ::rclcpp::Result bind(Server<A>& s, G&& goal, C&& cancel, Acc&& accepted) {
1066 State* st = new State{GoalFn(static_cast<G&&>(goal)),
1067 CancelFn(static_cast<C&&>(cancel)),
1068 AcceptedFn(static_cast<Acc&&>(accepted)),
1069 &s,
1070 {}};
1071 s.upstream_.reset(st, &destroy);
1072 ::rclcpp::Result r = s.set_goal_callback_with_ctx(&on_goal, st);
1073 if (r.ok()) r = s.set_cancel_callback_with_ctx(&on_cancel, st);
1074 if (r.ok()) r = s.set_accepted_callback_with_ctx(&on_accepted, st);
1075 return r;
1076 }
1077
1079 static void forget(Server<A>& s, const GoalUUID& id) {
1080 State* st = s.upstream_.template as<State>();
1081 if (st == nullptr) return;
1082 for (size_t i = 0; i < st->goals.size(); ++i) {
1083 if (st->goals[i].id == id) {
1084 st->goals.erase(st->goals.begin() + static_cast<long>(i));
1085 return;
1086 }
1087 }
1088 }
1089
1090 private:
1091 static ::std::shared_ptr<Handle> make_handle(State* st, const GoalUUID& id) {
1092 ::std::shared_ptr<Handle> h = ::std::make_shared<Handle>(st->server, id);
1093 for (const Entry& e : st->goals) {
1094 if (e.id == id) {
1095 h->goal_.reset(new ::std::shared_ptr<const Goal>(e.goal), [](void* p) {
1096 delete static_cast<::std::shared_ptr<const Goal>*>(p);
1097 });
1098 break;
1099 }
1100 }
1101 return h;
1102 }
1103
1104 static GoalResponse on_goal(const uint8_t uuid[16], const Goal& goal, void* ctx) {
1105 State* st = static_cast<State*>(ctx);
1106 if (!st->goal) return GoalResponse::REJECT;
1107 const GoalUUID id(uuid);
1108 ::std::shared_ptr<const Goal> msg = ::std::make_shared<const Goal>(goal);
1109 const GoalResponse r = st->goal(id, msg);
1110 if (r != GoalResponse::REJECT) {
1111 forget(*st->server, id);
1112 st->goals.push_back(Entry{id, msg, r == GoalResponse::ACCEPT_AND_EXECUTE});
1113 }
1114 return r;
1115 }
1116
1117 static CancelResponse on_cancel(const uint8_t uuid[16], void* ctx) {
1118 State* st = static_cast<State*>(ctx);
1119 if (!st->cancel) return CancelResponse::REJECT;
1120 return st->cancel(make_handle(st, GoalUUID(uuid)));
1121 }
1122
1123 static void on_accepted(const uint8_t uuid[16], void* ctx) {
1124 State* st = static_cast<State*>(ctx);
1125 const GoalUUID id(uuid);
1126 // Issue 2027 — upstream's server calls `execute()` itself for
1127 // ACCEPT_AND_EXECUTE, so the goal is EXECUTING (and the status array
1128 // says so) by the time `handle_accepted` sees it; ACCEPT_AND_DEFER
1129 // stays ACCEPTED until the handle's `execute()`.
1130 for (const Entry& e : st->goals) {
1131 if (e.id == id) {
1132 if (e.execute) (void)st->server->execute(uuid);
1133 break;
1134 }
1135 }
1136 if (st->accepted) st->accepted(make_handle(st, id));
1137 }
1138#endif
