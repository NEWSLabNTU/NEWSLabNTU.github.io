126struct GoalUUID {
128 static const size_t SIZE = 16;
129
131 uint8_t uuid[16];
132
134 GoalUUID() : uuid{} {}
135
137 explicit GoalUUID(const uint8_t* src) : uuid{} {
138 for (size_t i = 0; i < SIZE; ++i) {
139 uuid[i] = src[i];
140 }
141 }
142
144 uint8_t* data() { return uuid; }
145 const uint8_t* data() const { return uuid; }
146
148 bool is_zero() const {
149 for (size_t i = 0; i < SIZE; ++i) {
150 if (uuid[i] != 0) return false;
151 }
152 return true;
153 }
154
155 bool operator==(const GoalUUID& other) const {
156 for (size_t i = 0; i < SIZE; ++i) {
157 if (uuid[i] != other.uuid[i]) return false;
158 }
159 return true;
160 }
161
162 bool operator!=(const GoalUUID& other) const { return !(*this == other); }
163
167 bool operator<(const GoalUUID& other) const {
168 for (size_t i = 0; i < SIZE; ++i) {
169 if (uuid[i] != other.uuid[i]) return uuid[i] < other.uuid[i];
170 }
171 return false;
172 }
173};

809 // `ctx` is a pointer to this `ActionServer<A>` instance, so the

813 static int32_t goal_trampoline(const uint8_t goal_id[16], const uint8_t* data, size_t len,
814 void* ctx) {
815 auto* self = static_cast<Server*>(ctx);
816 if (!self) return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
817 GoalType g;
818 if (GoalType::ffi_deserialize(data, len, &g) != 0) {
819 return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
820 }
821 if (self->user_goal_fn_ctx_ != nullptr) {
822 return ::rclcpp_action::to_ffi(
823 self->user_goal_fn_ctx_(goal_id, g, self->user_goal_ctx_));
824 }
825 if (self->user_goal_fn_ != nullptr) {
826 return ::rclcpp_action::to_ffi(self->user_goal_fn_(goal_id, g));
827 }
828 return ::rclcpp_action::to_ffi(::rclcpp_action::GoalResponse::REJECT);
829 }

857 bool goal_set = (user_goal_fn_ != nullptr) || (user_goal_fn_ctx_ != nullptr);
858 bool cancel_set = (user_cancel_fn_ != nullptr) || (user_cancel_fn_ctx_ != nullptr);
859 bool accepted_set = (user_accepted_fn_ != nullptr) || (user_accepted_fn_ctx_ != nullptr);
860 nros_cpp_goal_callback_t gcb = goal_set ? &goal_trampoline : nullptr;
861 nros_cpp_cancel_callback_t ccb = cancel_set ? &cancel_trampoline : nullptr;
862 nros_cpp_ret_t ret = nros_cpp_action_server_set_callbacks(storage_, gcb, ccb, this);

996 using Goal = typename A::Goal;
997 using Handle = ServerGoalHandle<A>;
998
1000 static void rehome(Server<A>& s) {
1001#ifdef NROS_CPP_HAS_SHARED_PTR // hosted-family: action-interop
1002 if (State* st = s.upstream_.template as<State>()) st->server = &s;
1003#else
1004 (void)s;
1005#endif
1006 }
1007
1008#ifdef NROS_CPP_HAS_SHARED_PTR // hosted-family: action-interop
1009 using GoalFn = ::std::function<GoalResponse(const GoalUUID&, ::std::shared_ptr<const Goal>)>;
1010 using CancelFn = ::std::function<CancelResponse(::std::shared_ptr<Handle>)>;
1011 using AcceptedFn = ::std::function<void(::std::shared_ptr<Handle>)>;
1012
1013 struct Entry {
1014 GoalUUID id;
1015 ::std::shared_ptr<const Goal> goal;
1016 };
1017 struct State {
1018 GoalFn goal;
1019 CancelFn cancel;
1020 AcceptedFn accepted;
1021 Server<A>* server;
1022 ::std::vector<Entry> goals;
1023 };
1024
1025 static void destroy(void* p) { delete static_cast<State*>(p); }
1026
1027 template <typename G, typename C, typename Acc>
1028 static ::rclcpp::Result bind(Server<A>& s, G&& goal, C&& cancel, Acc&& accepted) {
1029 State* st = new State{GoalFn(static_cast<G&&>(goal)),
1030 CancelFn(static_cast<C&&>(cancel)),
1031 AcceptedFn(static_cast<Acc&&>(accepted)),
1032 &s,
1033 {}};
1034 s.upstream_.reset(st, &destroy);
1035 ::rclcpp::Result r = s.set_goal_callback_with_ctx(&on_goal, st);
1036 if (r.ok()) r = s.set_cancel_callback_with_ctx(&on_cancel, st);
1037 if (r.ok()) r = s.set_accepted_callback_with_ctx(&on_accepted, st);
1038 return r;
1039 }
1040
1042 static void forget(Server<A>& s, const GoalUUID& id) {
1043 State* st = s.upstream_.template as<State>();
1044 if (st == nullptr) return;
1045 for (size_t i = 0; i < st->goals.size(); ++i) {
1046 if (st->goals[i].id == id) {
1047 st->goals.erase(st->goals.begin() + static_cast<long>(i));
1048 return;
1049 }
1050 }
1051 }
1052
1053 private:
1054 static ::std::shared_ptr<Handle> make_handle(State* st, const GoalUUID& id) {
1055 ::std::shared_ptr<Handle> h = ::std::make_shared<Handle>(st->server, id);
1056 for (const Entry& e : st->goals) {
1057 if (e.id == id) {
1058 h->goal_.reset(new ::std::shared_ptr<const Goal>(e.goal), [](void* p) {
1059 delete static_cast<::std::shared_ptr<const Goal>*>(p);
1060 });
1061 break;
1062 }
1063 }
1064 return h;
1065 }
1066
1067 static GoalResponse on_goal(const uint8_t uuid[16], const Goal& goal, void* ctx) {
1068 State* st = static_cast<State*>(ctx);
1069 if (!st->goal) return GoalResponse::REJECT;
1070 const GoalUUID id(uuid);
1071 ::std::shared_ptr<const Goal> msg = ::std::make_shared<const Goal>(goal);
1072 const GoalResponse r = st->goal(id, msg);
1073 if (r != GoalResponse::REJECT) {
1074 forget(*st->server, id);
1075 st->goals.push_back(Entry{id, msg});
1076 }
1077 return r;
1078 }
1079
1080 static CancelResponse on_cancel(const uint8_t uuid[16], void* ctx) {
1081 State* st = static_cast<State*>(ctx);
1082 if (!st->cancel) return CancelResponse::REJECT;
1083 return st->cancel(make_handle(st, GoalUUID(uuid)));
1084 }
1085
1086 static void on_accepted(const uint8_t uuid[16], void* ctx) {
1087 State* st = static_cast<State*>(ctx);
1088 if (st->accepted) st->accepted(make_handle(st, GoalUUID(uuid)));
1089 }
1090#endif
1091};
