113// ported node looks for them (`rclcpp::ReliabilityPolicy`, `qos.hpp:34-58`).
114// Three of these were PRIVATE members of `QoS` until phase-379 W5, which is why

116// type. Ledger: `cpp:ReliabilityPolicy` and its three siblings.

118// The two rclcpp enumerators we used to decline ARRIVED in phase-444 (issue
119// 1437), and the reason each was declined is the reason it is now needed:
120//
121// * `SystemDefault` — "defer to the middleware". Declined because nothing
122// DEFERS: the backend is linked at build time (RFC-0036). True of a

225constexpr uint32_t qos_window_ms(const Duration& d) {
226 return d.nanoseconds() <= 0 ? 0u
227 : d.nanoseconds() == INT64_MAX - 1
228 ? UINT32_MAX - 1u
229 : (d.nanoseconds() >= static_cast<int64_t>(UINT32_MAX - 1u) * 1000000 - 999999
230 ? UINT32_MAX
231 : static_cast<uint32_t>((d.nanoseconds() + 999999) / 1000000));
232}

241constexpr Duration qos_best_available_duration() {
242 return Duration::from_nanoseconds(INT64_MAX - 1);
243}

248constexpr Duration qos_window_duration(uint32_t ms) {
249 return ms == UINT32_MAX - 1u ? qos_best_available_duration()
250 : Duration::from_nanoseconds(static_cast<int64_t>(ms) * 1000000);
251}

490// phase-417 G6 — profile comparison. `rclcpp::operator==(const QoS&, const QoS&)`
491// ============================================================================
492//
493// FREE functions, as upstream's are, and written from the PUBLIC accessors —
494// every one of the ten members has one, so no `friend` is needed and the
495// comparison stays a thing a reader can check against the accessor list.
496//
497// TEN hand-written comparisons rather than `= default`. `operator==() =
498// default` is C++20 and this surface's floor is `cxx_std_17`
499// (`packages/api/nros-cpp/CMakeLists.txt` — `target_compile_features(
500// nros-cpp-headers INTERFACE cxx_std_17)`), so the defaulted form would not
501// compile for the targets this header exists to serve. Still `constexpr`,
502// still no allocator, still no exceptions.
503//
504// Rust's `QoSProfile` derives `PartialEq, Eq` and has had this for nothing
505// since it existed; the ledger row `cpp:operator==` recorded the asymmetry as
506// having no platform reason, and it did not.
507//
508// The three time windows are compared through `Duration::nanoseconds()`: the
509// storage is milliseconds and the conversion is exact in both directions
510// (`detail::qos_window_duration`).
511
517constexpr bool operator==(const QoS& left, const QoS& right) {
518 return left.reliability() == right.reliability() && left.durability() == right.durability() &&
519 left.history() == right.history() && left.liveliness() == right.liveliness() &&
520 left.depth() == right.depth() &&
521 left.deadline().nanoseconds() == right.deadline().nanoseconds() &&
522 left.lifespan().nanoseconds() == right.lifespan().nanoseconds() &&
523 left.liveliness_lease_duration().nanoseconds() ==
524 right.liveliness_lease_duration().nanoseconds() &&
525 left.avoid_ros_namespace_conventions() == right.avoid_ros_namespace_conventions() &&
526 left.tx_express() == right.tx_express();
527}

587// --- THE C++ QoS profile table (phase-428 W10) -------------------------------
588//
589// Every named profile the C++ surface offers is spelled HERE and nowhere else.
590// `rclcpp::QoS`'s three named statics and all six concrete `rclcpp::*QoS` classes
591// are defined by calling into this block; none of them restates a policy.
592//
593// Why a table and not a hand copy per class: W10 exists because upstream's 7
594// profiles were transcribed in four places across the repo — this header,
595// `nros-rmw/src/traits.rs`, `nros/src/lib.rs` and `nros-rmw-abi/.../
596// rmw_entity.h` — about 26 constants for a 7-row table, with no gate binding
597// them. `QOS_PROFILE_PARAMETER_EVENTS` is the proof: `KeepAll, 0` in Rust under
598// a comment claiming it matched `rmw_qos_profile_parameter_events`, which is
599// `KEEP_LAST, 1000`. This header had the same profile right, which is exactly
600// what an ungated mirror looks like until the day it is not.
601//
602// This block cannot CONSUME the cross-language SSoT. The C spellings in
603// `rmw_entity.h` are `((rmw_qos_profile_t){...})` compound literals, which are
604// C99 and not standard C++, over a struct whose field types differ from
605// `nros_cpp_qos_t`'s; and `qos.hpp` is deliberately self-contained and
606// freestanding (see issue 0112 at the top of this file), so it cannot take a
607// dependency on a build-generated header either. What the table DOES buy is
608// that the C++ side has ONE row per profile for a cross-language gate to read,
609// instead of nine constructor bodies to parse.
610//
611// phase-454 W1 collected on that. `scripts/check-qos-profile-ssot.py` now
612// EVALUATES these rows — the ctor supplies the base profile and each chained
613// call applies its one effect — and compares every field against
614// `QoSProfile::QOS_PROFILE_*`. So the table is bound after all; what it is not
615// is DERIVED, which is the weaker of the two and the strongest available to a
616// freestanding header. A `.setter()` the gate does not model is a hard failure
617// there, not a skipped field.
618//
619// Values read 2026-09-05 against ROS 2 Humble as installed:
620// /opt/ros/humble/include/rmw/rmw/qos_profiles.h :25,38,51,64,77,90
621// /opt/ros/humble/include/rcl/rcl/logging_rosout.h :36
622// /opt/ros/humble/include/rclcpp/rclcpp/qos.hpp :371-500
623//
624// Three fields are identical in EVERY upstream profile and are therefore folded
625// into `QoS`'s default constructor rather than spelled per row:
626// * `avoid_ros_namespace_conventions` is `false` upstream, `0` here.
627// * liveliness is `RMW_QOS_POLICY_LIVELINESS_SYSTEM_DEFAULT` upstream, and
628// `rclcpp::QoS` now defaults to `LivelinessPolicy::SystemDefault`, which is our spelling of
629// that sentinel. It defaulted to `LivelinessAutomatic` — "what every
630// reference RMW folds that sentinel to" — under a
631// `nros-qos-mirror-deviation` record whose own reason was that this
632// surface and `nros-c`'s "move together with the C ABI or not at all".
633// They moved together (issue 1329): this constructor, `nros-c`'s
634// `NROS_QOS_*` statics and (until phase-492 W4) `component.h`'s
635// `rclc_c_qos_default()`.
636//
637// The forcing argument is issue 1329's per-backend QoS mask. XRCE honours
