86// -- Policy enums (phase 379 W5) ------------------------------------------
87//
88// At NAMESPACE scope and public, under rclcpp's names, because that is where a
89// ported node looks for them (`rclcpp::ReliabilityPolicy`, `qos.hpp:34-58`).
90// Three of these were PRIVATE members of `QoS` until phase-379 W5, which is why
91// the getters had to hand back `int` — a public getter cannot name a private
92// type. Ledger: `cpp:ReliabilityPolicy` and its three siblings.
93//
94// Two deliberate differences from rclcpp survive the rename, both ledgered:
95// * no `SystemDefault` — it means "defer to the middleware" and there is
96// none to defer to; the backend is linked at build time (RFC-0036).
97// * no `Unknown` — it is a discovery artefact for a policy read off a remote
98// endpoint, and we do no dynamic discovery.
99//
100// These are UNSCOPED enums, not rclcpp's `enum class`. Both spellings work as a
101// result: `nros::Reliable` (ours, historical) and `nros::ReliabilityPolicy::
102// Reliable` (rclcpp's). Switching to `enum class` would break the first and is
103// its own decision, which no ledger row makes — see the qos.json rows.

157constexpr uint32_t qos_window_ms(const Duration& d) {
158 return d.nanoseconds() <= 0
159 ? 0u
160 : (d.nanoseconds() > static_cast<int64_t>(UINT32_MAX) * 1000000
161 ? UINT32_MAX
162 : static_cast<uint32_t>((d.nanoseconds() + 999999) / 1000000));
163}
164
167constexpr Duration qos_window_duration(uint32_t ms) {
168 return Duration::from_nanoseconds(static_cast<int64_t>(ms) * 1000000);
169}

451constexpr nros_cpp_qos_t qos_to_ffi(const QoS& qos) {
452 nros_cpp_qos_t f{};
453 f.reliability = static_cast<nros_cpp_qos_reliability_t>(qos.reliability());
454 f.durability = static_cast<nros_cpp_qos_durability_t>(qos.durability());
455 f.history = static_cast<nros_cpp_qos_history_t>(qos.history());
456 f.liveliness_kind = static_cast<nros_cpp_qos_liveliness_t>(qos.liveliness());
457 f.depth = qos.depth();
458 // The FFI FIELDS are milliseconds and keep their `_ms` names; only the
459 // accessors changed. `qos_window_ms` is the exact inverse of the
460 // `Duration` the getter built, so this round-trip loses nothing.
461 f.deadline_ms = qos_window_ms(qos.deadline());
462 f.lifespan_ms = qos_window_ms(qos.lifespan());
463 f.liveliness_lease_ms = qos_window_ms(qos.liveliness_lease_duration());
464 f.avoid_ros_namespace_conventions = qos.avoid_ros_namespace_conventions() ? 1 : 0;
465 f.tx_express = qos.tx_express() ? 1 : 0;
466 return f;
467}
