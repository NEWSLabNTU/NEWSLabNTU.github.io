251// `<chrono>` requires the EXPLICIT opt-in, not `__has_include`, and that is
252// measured rather than stylistic.
253//
254// `__has_include(<chrono>)` is TRUE on the Zephyr arm-none-eabi toolchain --
255// the file exists -- but under `-ffreestanding` libstdc++ ships it INCOMPLETE:
256// `std::chrono::duration_cast` is absent, so `create_wall_timer` and `Rate`
257// failed to compile on every Zephyr C++ image with
258// `'duration_cast' is not a member of 'std::chrono'`. Replayed from that build
259// dir's own recorded compile command, not inferred.
260//
261// So the idiom has a boundary worth stating: `__has_include` answers "does the
262// header exist", which is the right question for `<memory>` (present or absent
263// as a unit) and the WRONG one for `<chrono>` (present but hollowed out). Where
264// a freestanding toolchain ships a partial header, only the consumer's own
265// `NROS_CPP_STD` opt-in is a reliable signal. Issue 0112's class, one layer past
266// where step A caught it.
267// CORRECTED 2026-09-05. The `NROS_CPP_STD`-only gate above was measured on the
268// Zephyr lane and never on a plain hosted one, and NOTHING THAT SHIPS DEFINES
269// `NROS_CPP_STD`: it appears in no cmake module, no toolchain file, no build.rs
270// and no Kconfig -- only in docs and in `check.just`'s own probe TUs. So the
271// narrow gate did not merely tighten `<chrono>`; it removed `create_wall_timer`
272// and `Rate` from EVERY shipped configuration, including the hosted one, where
273// `examples/templates/cpp-port-minimal-publisher` calls `create_wall_timer` and
274// is phase-417's own acceptance criterion.
275//
276// The right predicate needs BOTH halves, because the two lanes fail
277// differently and neither test alone sees both:
278//
279// * Zephyr / ThreadX-RV64 build `-nostdinc++` against a minimal libcpp that
280// has no `<chrono>` at all, so `__has_include` answers FALSE and is exactly
281// right there.
282// * FreeRTOS armcm3 builds `-ffreestanding` WITHOUT `-nostdinc++`, so
283// `__has_include(<chrono>)` answers TRUE against the host's own libstdc++
284// -- which then refuses to be used, because GCC 13 gates
285// `bits/requires_hosted.h` on `__STDC_HOSTED__` and `-ffreestanding` clears
286// it. That is the "present but hollow" case, and `__STDC_HOSTED__` is
287// precisely the flag that distinguishes it.
288//
289// `__STDC_HOSTED__` alone is still not enough -- CLAUDE.md's pitfall entry says
290// so, and it is right: a hosted compiler can run `-nostdinc++` against Zephyr's
291// minimal libcpp, where the macro is 1 and the header is absent. Neither test
292// covers both lanes; the conjunction does.
293//
294// `NROS_CPP_STD` stays as an explicit override for a consumer who knows better
295// than the probes, which is what the parity extractor and the docs use it for.
296// CORRECTED AGAIN 2026-09-07, and this time the discriminator is the library's
297// own feature-test macro rather than a guess about which RTOS ships what.
298//
299// The conjunction above is still not enough. Measured under the safety island's
300// OWN recorded compile command (`-nostdinc++`, arm-zephyr-eabi, picolibc), not
301// inferred:
302//
303// __STDC_HOSTED__ 1
304// __has_include(<chrono>) TRUE
305// __cpp_lib_chrono ABSENT
306//
307// So Zephyr does NOT merely "have no <chrono> at all" -- this toolchain ships
308// one that is present and hollow, which is the case the note above attributes
309// to FreeRTOS alone. Both halves of the conjunction answer yes and
310// `duration_cast` is still missing, so `Rate`'s duration constructor failed to
311// compile on the island exactly as it did before that note was written.
312//
313// `<ratio>` is the third test, and it is a PREREQUISITE rather than a proxy.
314// `duration_cast<To>(duration<Rep, Period>)` is defined in terms of
315// `std::ratio_divide` -- a duration's `Period` IS a `std::ratio` -- so a
316// `<chrono>` shipped without `<ratio>` cannot provide `duration_cast`, and that
317// is exactly the shape this toolchain ships. Measured on both sides:
318//
319// island (Zephyr) hosted g++ -std=c++14
320// __has_include(<chrono>) TRUE TRUE
321// __has_include(<ratio>) FALSE TRUE
322//
323// `__cpp_lib_chrono` was tried first and is WRONG here: it is a C++17
324// feature-test macro, so it is absent under `-std=c++14` even where the library
325// is complete. `check-cpp` compiles the phase-417 probes at C++14 and caught
326// that immediately -- gating on it removed `create_wall_timer` from the hosted
327// build, which is the same over-tightening the note above records.
328// `_GLIBCXX_CHRONO` also separates the two but is libstdc++'s private spelling;
329// `__has_include` is the portable question.
330#if defined(NROS_CPP_STD) || (defined(__STDC_HOSTED__) && __STDC_HOSTED__ && __has_include(<chrono>) && __has_include(<ratio>))
331#include <chrono>
332#define NROS_CPP_HAS_STD_CHRONO 1
333#endif

358constexpr bool is_ros_args_flag(char const* s, int i = 0) {
359 return s == nullptr ? false
360 : "--ros-args"[i] == '\0' ? s[i] == '\0'
361 : s[i] != "--ros-args"[i] ? false
362 : is_ros_args_flag(s, i + 1);
363}
364
365constexpr bool argv_has_ros_args(int argc, char const* const* argv, int i = 0) {
366 return i >= argc ? false
367 : argv == nullptr ? false
368 : is_ros_args_flag(argv[i]) ? true
369 : argv_has_ros_args(argc, argv, i + 1);
370}

527// --- Node --------------------------------------------------------------------
528//
529// rclcpp users write `auto n = std::make_shared<rclcpp::Node>("name");` and
530// then `n->create_publisher<M>(topic, qos)` returning a shared_ptr. nros's
531// Node is created via an `Executor` and exposes out-ref `create_*` member
532// functions. This matches the rclcpp call shape onto it.
533//
534// A `rclcpp::Node` does NOT own an executor. It opens its `nros::Node` on the
535// GLOBAL one `rclcpp::init()` created (issue 0465 — one session per image), and
536// since phase-417 every entity it creates — publisher, subscription, wall
537// timer, service, client — is registered THERE. So this class holds no dispatch
538// state and runs no loop: it is a call-shape adapter over `nros::Node`, which
539// is all RFC-0019 permits a wrapper to be, and it is the property that lets a
540// later step merge the two node types safely.
541//
542// Threading: callbacks fire on whatever thread services a spin verb, mirroring
543// the rclcpp default. An `nros::Node` cannot move between executors (RFC-0002,
544// one executor per RTOS task), which is why the executor is decided at
545// construction and never afterwards.

615 void initialize(const std::string& name, const char* namespace_) {
616 // Bring up the underlying nros::Node on the GLOBAL executor — the one
617 // `rclcpp::init()` already created. Initialization failures leave the
618 // node marked uninitialized rather than throwing (nros-cpp is
619 // freestanding by default — no `<stdexcept>`), so a caller using
620 // `std::make_shared<rclcpp::Node>("n")` still mirrors rclcpp's
621 // "constructor never returns an error code" contract and subsequent
622 // `create_*` fail visibly.
623 //
624 // Issue 0465 — this used to call `Executor::create(executor_)`, giving
625 // every Node its OWN executor and therefore its own RMW session.
626 // A non-bridge application has exactly one session; two is the bridge
627 // shape. With `ZPICO_MAX_SESSIONS` at its default of 1 the second open
628 // exhausted the pool and returned -1, surfacing as
629 // `Transport(ConnectionFailed)` — the same text a real connection
630 // failure gives, which is why it read as one for two months. Raising
631 // the pool would have hidden the design error behind memory spent on
632 // every embedded target.
633 //
634 // `::nros::create_node` targets `Node::global_storage()`, so N
635 // `rclcpp::Node`s share the single session, which is also what rclcpp's
636 // own process-level Context model implies.
637 ::nros::Result r = ::nros::create_node(node_, name.c_str(), namespace_);
638 initialized_ = r.ok();

667 // `nros_cpp_subscription_register` call the native callback-style
668 // `nros::Node::create_subscription` makes, so the executor owns the
669 // subscriber and dispatches the callback during `spin_once` — whichever
670 // spin verb the caller drives. It used to create a POLL-mode subscription
671 // and drain it from a node-local `pump()`, which only `rclcpp::spin` /
672 // `spin_some` called: a file that spun `nros::spin_once()` instead got zero
673 // callbacks and no diagnostic (RFC-0089 §"There is also one mismatch the
674 // rename makes strictly worse").
675 //
676 // Why not `node_.create_subscription(*s, topic, cb, qos)`: that overload is
677 // SFINAE-restricted to `void(*)(const M&)` — a plain function pointer with
678 // NO context slot — and every ported rclcpp callback captures. The typed
679 // ctx-carrying delivery exists one layer down (`Subscription<M>::
680 // user_fn_ctx_` + `user_ctx_`, and the `message_trampoline` branch that
681 // reads them, `subscription.hpp:563`) but NO `create_*` sets those fields,
682 // so the raw register with a type-erased ctx is the reachable shape. A
683 // `create_subscription` overload taking `(void(*)(const M&, void*), void*
684 // ctx)` would let this call `nros::Node` instead, and is C++-side work.
685 //
686 // The receive-buffer hint is `rx_buffer_capacity<M>` — the same number the
687 // poll path's `take()` stacks — and NOT the strict `rx_size_bound<M>`,
688 // whose unbounded-type arm is a deliberate compile error (issue 0964). A
689 // ported file must not stop compiling because its message has an unbounded
690 // string.

692 // ONE THING THIS LOSES, stated rather than left silent:
693 // `create_subscription_raw` hardcodes an EMPTY type hash (it takes no such
694 // parameter), which `normalize_type_hash` turns into
695 // `"TypeHashNotSupported"`. The poll-mode create this replaced passed
696 // `M::TYPE_HASH`. It does not affect DELIVERY — a subscriber's data
697 // keyexpr puts `*` in the hash slot (`keyexpr.rs:51`), and on the default
698 // Humble edition the publisher's is the literal `TypeHashNotSupported`
699 // anyway (`keyexpr.rs:30`) — but under `ros-iron` / `ros-jazzy` the
700 // subscription's LIVELINESS token now advertises the placeholder instead of
701 // the real hash, so `ros2 topic info --verbose` reads differently. The fix
702 // is a `type_hash` parameter on `create_subscription_raw`, defaulted to
703 // `""` so no existing caller moves.
704 //
705 // OWNERSHIP: the arena stores the cell's address as its dispatch context
706 // and there is no unregister, so the cell must outlive the registration.
707 // `owned_entities_` holds it for the node's lifetime — the same rule
708 // `create_service` below states — and the returned `shared_ptr` is a
709 // co-owner via the aliasing constructor, so dropping it is safe.
710 //
711 // WHAT THE RETURNED POINTER IS: a keep-alive, which is all upstream source
712 // does with it (`rclcpp::Subscription<M>::SharedPtr sub_;`). The executor
713 // owns the real subscriber, so `sub->take(msg)` on it answers

1002// --- spin / spin_some --------------------------------------------------------
1003//
1004// phase-417 — FORWARDERS, and nothing else. `rclcpp::spin(node)` is
1005// `nros::spin()` (block until `ok()` goes false) and `rclcpp::spin_some(node)`
1006// is one 0-timeout `nros::spin_once`. Both used to call a node-local `pump()`
1007// first, which was the ONLY thing that ran a ported node's timers and
1008// subscriptions; now those live on the executor, so these two dispatch nothing
1009// of their own and a file that reaches for `nros::spin_once()` or drives an
1010// `nros::Executor` gets exactly the same callbacks. That equivalence is the
1011// structural prerequisite for step B: after it both node types share a name,
1012// and a mismatch would be invisible.
1013//
1014// One behaviour change for an existing caller: `nros::spin()` RETURNS on the
1015// first failing `spin_once`, where this loop used to discard the result and
1016// keep going. A dead session now ends the spin instead of looking alive
1017// forever, which is what `nros::spin()` and `Executor::spin` already promise.
1018//
1019// The `node` argument is still checked, and still otherwise unused: there is
1020// one session per image (issue 0465), so every `rclcpp::Node` is on the global
1021// executor these verbs drive. Upstream takes the node for the same reason and
1022// spins the executor it belongs to.
1023
1024inline void spin(const Node::SharedPtr& node) {
1025 if (!node || !node->initialized()) {
1026 return;
1027 }
1028 (void)::nros::spin();
1029}

1038// Future type is templated rather than `const auto& future` so the header
1039// stays parseable under `-std=c++14` (the C++20 abbreviated-function-template
1040// syntax breaks `just check cpp`'s freestanding probe).
1041//
1042// issue 0339 — the bounded branch used to call `Executor::spin(timeout_ms)`
1043// and never consult the future, so it BURNED THE WHOLE TIMEOUT even when the
1044// future completed on the first spin: a `wait_for_service` / `send_request`
1045// sequence ported from rclcpp paid the full timeout on every SUCCESSFUL call.
1046// The unbounded branch directly below already had the right shape; both now
1047// share it, differing only in whether a deadline exists.
1048template <typename Future>
1049inline FutureReturnCode spin_until_future_complete(const Node::SharedPtr& node,
1050 const Future& future, int32_t timeout_ms = -1) {
1051 if (!node || !node->initialized()) {
1052 return FutureReturnCode::INTERRUPTED;
1053 }
1054 // Poll slice: same 10 ms the unbounded loop always used.
1055 constexpr int32_t kPollMs = 10;
1056 const bool bounded = timeout_ms >= 0;
1057 const uint64_t start_ns = nros_cpp_time_ns();
1058 const uint64_t budget_ns = bounded ? static_cast<uint64_t>(timeout_ms) * 1000000ull : 0ull;
1059
1060 while (::nros::ok()) {
1061 if (future.is_ready()) {
1062 return FutureReturnCode::SUCCESS;
1063 }
1064 (void)::nros::spin_once(kPollMs);
1065 // Re-check before the deadline test: a future that became ready on the
1066 // spin just above must report SUCCESS even if the budget expired in
1067 // the same slice.
1068 if (future.is_ready()) {
1069 return FutureReturnCode::SUCCESS;
1070 }
1071 if (bounded && nros_cpp_time_ns() - start_ns >= budget_ns) {
1072 return FutureReturnCode::TIMEOUT;
1073 }
1074 }
1075 return FutureReturnCode::INTERRUPTED;
1076}

1082// --- Rate / WallRate (phase-417 W2.d) ----------------------------------------
1083//
1084// A FORWARDER onto `nros::spin(remaining_ms, poll_ms)`, which budgets by wall
1085// clock inside `nros_cpp_spin_for` — Rust-side, once.
1086//
1087// The obvious fifteen-line class with its own sleep loop is NOT admissible
1088// here, and the distinction is the whole point of this item: a loop in the
1089// wrapper that spins the executor is RFC-0020 violation class 2, and a wrapper
1090// that BLOCKS without spinning is what RFC-0021 forbids. Same capability, and
1091// only one of the two shapes is allowed. So `sleep()` computes a deadline and
1092// makes exactly one call into the entry point that already exists.
1093//
1094// ADOPT-BOUNDED, and the envelope is load-bearing enough that a ported loop
1095// behaves DIFFERENTLY here even though it compiles unchanged:
1096//
1097// * **Your callbacks RUN during the sleep.** `nros::spin` drives the
1098// executor, so subscriptions and timers registered on the arena fire while
1099// `rate.sleep()` is blocked. Under a single-threaded rclcpp executor they
1100// do not — `Rate::sleep()` there is a pure `sleep_for`, and callbacks only
1101// run when you call `spin_some`. The nano-ros behaviour is the one a
1102// `while (ok()) { work(); rate.sleep(); }` loop usually WANTS (it is why
1103// that loop needs a second thread upstream), but it is not the same
1104// contract, and a callback that races your loop body is the way it shows.
1105// Since phase-417 that includes a `rclcpp::Node`'s OWN timers and
1106// subscriptions: they are executor entities now, so `rate.sleep()` runs
1107// them.
1108// * `Rate` and `WallRate` are the SAME clock. Upstream's `Rate` measures ROS
1109// time and `WallRate` steady time; both here read the monotonic
1110// `nros_cpp_time_ns()`, so a `Rate` does not slow down under a sim clock.
1111// `WallRate` is faithful; `Rate` is `WallRate` under a second name.
1112// * Resolution is ONE MILLISECOND — the FFI budget is `uint32_t` ms — and the
1113// remaining time is rounded UP, so a period below 1 ms cannot be met and a
1114// rate faster than 1 kHz is not expressible.
1115// * Before `nros::init()` there is no executor to spin, so `sleep()` returns
1116// immediately rather than blocking. A rate loop that runs before init busy
1117// spins; upstream would have slept.
1118//
1119// Gated on `<chrono>` alone: the duration constructor and `period()` are spelled
1120// in `std::chrono`, and everything else it touches is an integer.
1121#ifdef NROS_CPP_HAS_STD_CHRONO
1122

1127class Rate {
1128 public:
1131 explicit Rate(double frequency_hz)
1132 : period_ns_(frequency_hz > 0.0 ? static_cast<int64_t>(1000000000.0 / frequency_hz) : 0) {
1133 reset();
1134 }

1137 template <typename Rep, typename Period>
1138 explicit Rate(std::chrono::duration<Rep, Period> period)
1139 : period_ns_(std::chrono::duration_cast<std::chrono::nanoseconds>(period).count()) {
1140 if (period_ns_ < 0) period_ns_ = 0;
1141 reset();
1142 }
1143
1149 bool sleep() {
1150 if (period_ns_ <= 0) {
1151 return false;
1152 }
1153 const uint64_t now_ns = nros_cpp_time_ns();
1154 if (now_ns >= next_tick_ns_) {
1155 // Overran. Re-anchor on now rather than firing a burst to catch up
1156 // — upstream's `Rate` contract, and the choice the retired
1157 // node-local `pump()` used to make for wall timers before they
1158 // became executor timers (which catch up instead; see the envelope
1159 // on `rclcpp::TimerBase`).
1160 next_tick_ns_ = now_ns + static_cast<uint64_t>(period_ns_);
1161 return false;
1162 }
1163 const uint64_t remaining_ns = next_tick_ns_ - now_ns;
1164 // Round UP: truncating a 0.4 ms remainder to 0 would turn `sleep()`
1165 // into a busy loop.
1166 uint32_t remaining_ms = static_cast<uint32_t>((remaining_ns + 999999ull) / 1000000ull);
1167 if (remaining_ms == 0) remaining_ms = 1;
1168 next_tick_ns_ += static_cast<uint64_t>(period_ns_);
1169 const int32_t poll_ms = remaining_ms < 10u ? static_cast<int32_t>(remaining_ms) : 10;
1170 (void)::nros::spin(remaining_ms, poll_ms);
1171 return true;
1172 }
1173
1175 void reset() { next_tick_ns_ = nros_cpp_time_ns() + static_cast<uint64_t>(period_ns_); }
1176
1178 std::chrono::nanoseconds period() const { return std::chrono::nanoseconds(period_ns_); }
1179
1180 private:
1181 int64_t period_ns_;
1182 uint64_t next_tick_ns_;
1183};
