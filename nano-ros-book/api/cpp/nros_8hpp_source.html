155inline Result spin() {
156 if (!Node::global_initialized()) {
157 return Result(ErrorCode::NotInitialized);
158 }
159 Result last = Result::success();
160 while (ok()) {
161 last = Result(nros_cpp_spin_once(Node::global_storage(), 10));

175inline Result spin(uint32_t duration_ms, int32_t poll_ms = 10) {
176 if (!Node::global_initialized()) {
177 return Result(ErrorCode::NotInitialized);
178 }
179 // Issue 0329 — forward to the single budgeted-spin CFFI entry point. This
180 // loop previously budgeted by ITERATION count (`elapsed += timeout`), which
181 // collapsed to milliseconds when `spin_once` returned early on a signaled
182 // wake — the exact bug `Executor::spin` fixed in Phase 118.C. The correct
183 // wall-clock budget now lives once, Rust-side, in `nros_cpp_spin_for`.
184 return Result(nros_cpp_spin_for(Node::global_storage(), duration_ms, poll_ms));
185}
