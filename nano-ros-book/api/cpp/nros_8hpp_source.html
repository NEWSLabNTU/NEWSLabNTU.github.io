174 if (!Node::global_initialized()) {
175 return Result(ErrorCode::NotInitialized);

214//
215// `rclcpp::init` / `shutdown` / `ok` / `Node` / `spin` / `spin_some` / `Rate` /
216// `spin_until_future_complete` used to live in `nros/rclcpp_compat.hpp`, a
217// separate header a ported file had to be force-included with. RFC-0089
218// §"Naming: replace, with alias as the migration step" makes the ROS 2 spelling
219// a FIRST-CLASS name declared by the API headers themselves; §"End state: no
220// compat layer survives" is what follows from it — with one spelling per
221// concept there is nothing left for a shim to bridge, so it dissolves by
222// construction rather than having to be argued away. `nros::` is untouched and
223// both spellings work; deprecating `nros::` and migrating the in-tree call
224// sites is step B.
225//
226// They land HERE, in the umbrella, and not in `node.hpp`, because that is where
227// their dependencies are. `rclcpp::Node` reaches `nros::create_subscription_raw`
228// (`component.hpp`), `nros::Timer`, `nros::ParameterServer`, `nros::Service` and
229// `nros::Client`, every one of which includes `node.hpp` itself; `Rate::sleep`
230// and `spin_until_future_complete` drive the `nros::spin` / `nros::spin_once`
231// free functions defined a few lines above. This IS the header they were
232// shimming: the process-level verbs `rclcpp::init` mirrors — `nros::init`,
233// `nros::spin`, `nros::spin_once`, `nros::on_shutdown` — are all reached
234// through `<nros/nros.hpp>`.
235//
236// FREESTANDING. Unlike the shim it replaces, this is not hosted-STL by
237// construction. `init` / `shutdown` / `ok` and the `--ros-args` predicate need
238// nothing but `<cstdlib>`, so they reach a `no_std` C++ build. `Node`, the spin
239// forwarders and `Rate` are gated on the standard-library pieces their
240// signatures are spelled in (`__has_include`, never `__STDC_HOSTED__` — issue

270constexpr bool is_ros_args_flag(char const* s, int i = 0) {
271 return s == nullptr ? false
272 : "--ros-args"[i] == '\0' ? s[i] == '\0'
273 : s[i] != "--ros-args"[i] ? false
274 : is_ros_args_flag(s, i + 1);

277constexpr bool argv_has_ros_args(int argc, char const* const* argv, int i = 0) {
278 return i >= argc ? false

280 : is_ros_args_flag(argv[i]) ? true
281 : argv_has_ros_args(argc, argv, i + 1);

466// --- Node ---------------------------------------------------------------------
467//
468// phase-427 W1-W3 — THE NODE TYPES ARE MERGED. `rclcpp::Node` is not a class
469// here any more; it is an ALIAS for `nros::Node`, and the hosted call shape
470// that used to be a separate adapter class in this file is now a set of
471// overloads on that one type (declared in `node.hpp`, defined below).
472//
473// What that closes, in order of how much it cost:
474//
475// * THE DUPLICATE PARAMETER FACADE. `nros.hpp` said so about itself ("KNOWN
476// DUPLICATION... There should be ONE helper"): the same rclcpp-shaped
477// `declare_parameter<T>` existed twice, once in C++14 here and once in
478// C++17 `if constexpr` on `ComponentNode`. One node type is what makes one
479// facade possible.
480// * THE `get_logger()` COLLISION. Two accessors with identical signatures
481// and different observable behaviour — a real node logger and the
482// hardcoded `"nros.compat"` sentinel. One type can hold only one, so the
483// merge forced the decision W5 records.
484// * THE TWO-VOCABULARY SPLIT. A ported file wrote `rclcpp::Node` and got a
485// type with no graph queries, no lifecycle, no callback groups, no action
486// entities and no out-ref creators; a native file wrote `nros::Node` and
487// got no `shared_ptr` creators and no parameters. Neither list was a
488// design; both were what the other file happened to have.
489//
490// The class holds no dispatch state and runs no loop — every entity it creates
491// is registered on the executor arena `rclcpp::init()` opened (issue 0465, one
492// session per image), which is all RFC-0019 permits a wrapper to be.
493//
494// Threading: callbacks fire on whatever thread services a spin verb, mirroring
495// the rclcpp default. A node cannot move between executors (RFC-0002, one
496// executor per RTOS task), which is why the executor is decided at construction
497// and never afterwards.

537// -- subscriptions ------------------------------------------------------------
538//
539// phase-417 — ONE DISPATCH PATH. This ARENA-REGISTERS the subscription through
540// `nros::create_subscription_raw` (`component.hpp`), the same
541// `nros_cpp_subscription_register` call the native callback-style
542// `create_subscription` makes, so the executor owns the subscriber and
543// dispatches the callback during `spin_once` — whichever spin verb the caller
544// drives. It used to create a POLL-mode subscription and drain it from a
545// node-local `pump()`, which only `rclcpp::spin` / `spin_some` called: a file
546// that spun `nros::spin_once()` instead got zero callbacks and no diagnostic.
547//
548// Why not the out-ref `create_subscription(sub, topic, cb, qos)` overload: that
549// one is SFINAE-restricted to `void(*)(const M&)` — a plain function pointer
550// with NO context slot — and every ported rclcpp callback captures.
551//
552// The receive-buffer hint is `rx_buffer_capacity<M>` — the same number the poll
553// path's `take()` stacks — and NOT the strict `rx_size_bound<M>`, whose
554// unbounded-type arm is a deliberate compile error (issue 0964). A ported file
555// must not stop compiling because its message has an unbounded string.
556//
557// ONE THING THIS LOSES, stated rather than left silent:
558// `create_subscription_raw` hardcodes an EMPTY type hash (it takes no such
559// parameter), which `normalize_type_hash` turns into `"TypeHashNotSupported"`.
560// It does not affect DELIVERY — a subscriber's data keyexpr puts `*` in the
561// hash slot — but under `ros-iron` / `ros-jazzy` the subscription's LIVELINESS
562// token advertises the placeholder instead of the real hash, so
563// `ros2 topic info --verbose` reads differently.
564//
565// WHAT THE RETURNED POINTER IS: a keep-alive, which is all upstream source does
566// with it (`rclcpp::Subscription<M>::SharedPtr sub_;`). The executor owns the
567// real subscriber, so `sub->take(msg)` on it answers `NotInitialized` — the
568// sample went to your callback.
569template <typename M, typename Cb>
570inline ::std::shared_ptr<Subscription<M>> Node::create_subscription(const ::std::string& topic,
571 const QoS& qos, Cb cb) {
572 auto cell = ::std::make_shared<::rclcpp::detail::SubscriptionCallback<M>>();
573 cell->fn = ::std::move(cb);
574 ::rclcpp::detail::require_created(
575 ::nros::create_subscription_raw(*this, topic.c_str(), M::TYPE_NAME,
576 &::rclcpp::detail::SubscriptionCallback<M>::trampoline,
577 cell.get(), qos, ::nros::rx_buffer_capacity<M>::value),
578 "create_subscription", topic.c_str());
579 this->hosted().owned_entities.push_back(cell);
580 return ::std::shared_ptr<Subscription<M>>(cell, &cell->handle);
581}

630// -- parameters ---------------------------------------------------------------
631//
632// Forwarders onto THE parameter store — the `nros_params::ParameterServer` the
633// EXECUTOR owns, reached through `nros/node_parameters.hpp`. There is no other
634// one any more.
635//
636// What changed in phase-426 W4, and why it is not a detail: these used to
637// forward to an inline `nros::ParameterServer<NROS_RCLCPP_MAX_PARAMS>` member
638// on the hosted block — a second store, node-local, which the six
639// `rcl_interfaces/srv/*` servers could not read. So a parameter declared here
640// was invisible to `ros2 param get`, a sibling node did not share it, and the
641// launch seed had to be copied across by a helper because the two stores could
642// not be the same object. The member is deleted; the seam is one FFI call;
643// `ros2 param get <node> <name>` sees what `declare_parameter` wrote. Issue
644// 0793 / RFC-0089 §"Parameters".
645//
646// ADOPT-BOUNDED still, and the envelope is now about TYPES rather than scope:
647// bool / int / int64_t / double reach the store, `std::string` and
648// `std::vector<T>` do under `NROS_CPP_STD`, and rclcpp's `ParameterDescriptor`
649// / `ignore_override` / callback arguments remain absent (the
650// compile-time-options rule). The WIRE half of the acceptance — that the six
651// services answer per node FQN — is phase-426 W3/W6.
652//
653// Where the image declares no `param_services` capability there is no store at
654// all, and every call answers `ErrorCode::Unsupported`. This facade wears
655// upstream's value-returning signature, which has nowhere to report that, so
656// `declare_parameter` returns the code default. The loud half USED to be
657// `nros::ComponentNode`, whose own facade recorded the failure on an `ok()`
658// flag and made it boot-fatal; phase-427 W4 deleted that type, so no C++ path
659// is boot-fatal on a missing store any more. `nros::Node` still carries the
660// flag (`set_error` / `ok()`); routing the parameter path back onto it is a
661// separate decision, because it changes what an upstream-shaped
662// `declare_parameter` does.

672template <typename T> inline T Node::declare_parameter(const char* name, T default_value) {
673 const ::nros_cpp_node_t* h = this->ffi_handle();
674 Result r = ::nros::detail::node_param_declare(h, name, default_value);
675 if (!r.ok() && r.raw() != NROS_RET_ALREADY_EXISTS) {
676 return default_value;
677 }
678 T out = T();
679 if (!::nros::detail::node_param_get(h, name, out).ok()) {
680 return default_value;
681 }
682 return out;
683}

797// --- rclcpp::create_timer (phase-430 W6) -------------------------------------
798//
799// Humble has NO `Node::create_timer` member — the clock-taking verb is a FREE
800// function, `rclcpp::create_timer(node, clock, period, callback[, group])`, and
801// that is the only form a Humble-era file can be written in. Ours takes the
802// same arguments in the same order and returns the same cell
803// `create_wall_timer` returns.
804//
805// The clock is `nros::Clock*`, which is exactly what `node->get_clock()` hands
806// back, so the ported line binds with no conversion. Upstream's parameter is
807// `rclcpp::Clock::SharedPtr`; there is no allocator here (RFC-0022) and the
808// clock is a member of the node, so the pointer is the counterpart — the same
809// ADOPT-BOUNDED trade `get_clock()` itself makes.
810//
811// WHAT THE CLOCK DOES, and it is the whole point of the verb: a
812// `NROS_CLOCK_ROS_TIME` clock follows `/clock`, so the timer stops when a bag
813// stops and re-times when the bag is replayed at another rate; a
814// `NROS_CLOCK_STEADY_TIME` one does not. `create_wall_timer` is the steady
815// verb and is unaffected by simulated time. Upstream distinguishes the two
816// cases by TYPE (`WallTimer` vs `GenericTimer<ClockT>`) *and* by VERB; the type
817// distinction is unportable here (the clock is a runtime field on one flat
818// `Timer` — RFC-0089 §"Timer, studied against RTOS semantics"), the verb
819// distinction is portable and is what ported code writes, so that is what we
820// take.
821//
822// NOT ADDED: a clock-less `Node::create_timer(period, callback)`. That member

851 auto t = ::std::make_shared<detail::WallTimer>();
852 t->callback = ::std::forward<CallbackT>(callback);
853 const int64_t ns = period.nanoseconds();

888// --- spin / spin_some --------------------------------------------------------
889//
890// phase-417 — FORWARDERS, and nothing else. `rclcpp::spin(node)` is
891// `nros::spin()` (block until `ok()` goes false) and `rclcpp::spin_some(node)`
892// is one 0-timeout `nros::spin_once`. Both used to call a node-local `pump()`
893// first, which was the ONLY thing that ran a ported node's timers and
894// subscriptions; now those live on the executor, so these two dispatch nothing
895// of their own and a file that reaches for `nros::spin_once()` or drives an
896// `nros::Executor` gets exactly the same callbacks. That equivalence is the
897// structural prerequisite for step B: after it both node types share a name,
898// and a mismatch would be invisible.
899//
900// One behaviour change for an existing caller: `nros::spin()` RETURNS on the
901// first failing `spin_once`, where this loop used to discard the result and
902// keep going. A dead session now ends the spin instead of looking alive
903// forever, which is what `nros::spin()` and `Executor::spin` already promise.
904//
905// The `node` argument is still checked, and still otherwise unused: there is
906// one session per image (issue 0465), so every `rclcpp::Node` is on the global
907// executor these verbs drive. Upstream takes the node for the same reason and
908// spins the executor it belongs to.
909
910inline void spin(const Node::SharedPtr& node) {
911 if (!node || !node->initialized()) {
912 return;
913 }
914 (void)::nros::spin();
915}
916
917inline void spin_some(const Node::SharedPtr& node) {
918 if (!node || !node->initialized()) {
919 return;
920 }
921 (void)::nros::spin_once(0);
922}

924// Future type is templated rather than `const auto& future` so the header
925// stays parseable under `-std=c++14` (the C++20 abbreviated-function-template
926// syntax breaks `just check cpp`'s freestanding probe).
927//
928// issue 0339 — the bounded branch used to call `Executor::spin(timeout_ms)`
929// and never consult the future, so it BURNED THE WHOLE TIMEOUT even when the
930// future completed on the first spin: a `wait_for_service` / `send_request`
931// sequence ported from rclcpp paid the full timeout on every SUCCESSFUL call.
932// The unbounded branch directly below already had the right shape; both now
933// share it, differing only in whether a deadline exists.
934template <typename Future>
935inline FutureReturnCode spin_until_future_complete(const Node::SharedPtr& node,
936 const Future& future, int32_t timeout_ms = -1) {
937 if (!node || !node->initialized()) {
938 return FutureReturnCode::INTERRUPTED;
939 }
940 // Poll slice: same 10 ms the unbounded loop always used.
941 constexpr int32_t kPollMs = 10;
942 const bool bounded = timeout_ms >= 0;
943 const uint64_t start_ns = nros_cpp_time_ns();
944 const uint64_t budget_ns = bounded ? static_cast<uint64_t>(timeout_ms) * 1000000ull : 0ull;
945
946 while (::nros::ok()) {
947 if (future.is_ready()) {
948 return FutureReturnCode::SUCCESS;
949 }
950 (void)::nros::spin_once(kPollMs);
951 // Re-check before the deadline test: a future that became ready on the
952 // spin just above must report SUCCESS even if the budget expired in
953 // the same slice.
954 if (future.is_ready()) {
955 return FutureReturnCode::SUCCESS;
956 }
957 if (bounded && nros_cpp_time_ns() - start_ns >= budget_ns) {
958 return FutureReturnCode::TIMEOUT;
959 }
960 }
961 return FutureReturnCode::INTERRUPTED;
962}

968// --- Rate / WallRate (phase-417 W2.d) ----------------------------------------
969//
970// A FORWARDER onto `nros::spin(remaining_ms, poll_ms)`, which budgets by wall
971// clock inside `nros_cpp_spin_for` — Rust-side, once.
972//
973// The obvious fifteen-line class with its own sleep loop is NOT admissible
974// here, and the distinction is the whole point of this item: a loop in the
975// wrapper that spins the executor is RFC-0020 violation class 2, and a wrapper
976// that BLOCKS without spinning is what RFC-0021 forbids. Same capability, and
977// only one of the two shapes is allowed. So `sleep()` computes a deadline and
978// makes exactly one call into the entry point that already exists.
979//
980// ADOPT-BOUNDED, and the envelope is load-bearing enough that a ported loop
981// behaves DIFFERENTLY here even though it compiles unchanged:
982//
983// * **Your callbacks RUN during the sleep.** `nros::spin` drives the
984// executor, so subscriptions and timers registered on the arena fire while
985// `rate.sleep()` is blocked. Under a single-threaded rclcpp executor they
986// do not — `Rate::sleep()` there is a pure `sleep_for`, and callbacks only
987// run when you call `spin_some`. The nano-ros behaviour is the one a
988// `while (ok()) { work(); rate.sleep(); }` loop usually WANTS (it is why
989// that loop needs a second thread upstream), but it is not the same
990// contract, and a callback that races your loop body is the way it shows.
991// Since phase-417 that includes a `rclcpp::Node`'s OWN timers and
992// subscriptions: they are executor entities now, so `rate.sleep()` runs
993// them.
994// * `Rate` and `WallRate` are the SAME clock. Upstream's `Rate` measures ROS
995// time and `WallRate` steady time; both here read the monotonic
996// `nros_cpp_time_ns()`, so a `Rate` does not slow down under a sim clock.
997// `WallRate` is faithful; `Rate` is `WallRate` under a second name.
998// * Resolution is ONE MILLISECOND — the FFI budget is `uint32_t` ms — and the
999// remaining time is rounded UP, so a period below 1 ms cannot be met and a
1000// rate faster than 1 kHz is not expressible.
1001// * Before `nros::init()` there is no executor to spin, so `sleep()` returns
1002// immediately rather than blocking. A rate loop that runs before init busy
1003// spins; upstream would have slept.
1004//
1005// Gated on `<chrono>` alone: the duration constructor and `period()` are spelled
1006// in `std::chrono`, and everything else it touches is an integer.
1007#ifdef NROS_CPP_HAS_STD_CHRONO
1008

1013class Rate {
1014 public:
1017 explicit Rate(double frequency_hz)
1018 : period_ns_(frequency_hz > 0.0 ? static_cast<int64_t>(1000000000.0 / frequency_hz) : 0) {
1019 reset();
1020 }

1023 template <typename Rep, typename Period>
1024 explicit Rate(std::chrono::duration<Rep, Period> period)
1025 : period_ns_(std::chrono::duration_cast<std::chrono::nanoseconds>(period).count()) {
1026 if (period_ns_ < 0) period_ns_ = 0;
1027 reset();
1028 }
1029
1035 bool sleep() {
1036 if (period_ns_ <= 0) {
1037 return false;
1038 }
1039 const uint64_t now_ns = nros_cpp_time_ns();
1040 if (now_ns >= next_tick_ns_) {
1041 // Overran. Re-anchor on now rather than firing a burst to catch up
1042 // — upstream's `Rate` contract, and the choice the retired
1043 // node-local `pump()` used to make for wall timers before they
1044 // became executor timers (which catch up instead; see the envelope
1045 // on `rclcpp::Timer`).
1046 next_tick_ns_ = now_ns + static_cast<uint64_t>(period_ns_);

1048 }
1049 const uint64_t remaining_ns = next_tick_ns_ - now_ns;
1050 // Round UP: truncating a 0.4 ms remainder to 0 would turn `sleep()`
1051 // into a busy loop.
1052 uint32_t remaining_ms = static_cast<uint32_t>((remaining_ns + 999999ull) / 1000000ull);
1053 if (remaining_ms == 0) remaining_ms = 1;
1054 next_tick_ns_ += static_cast<uint64_t>(period_ns_);
1055 const int32_t poll_ms = remaining_ms < 10u ? static_cast<int32_t>(remaining_ms) : 10;
1056 (void)::nros::spin(remaining_ms, poll_ms);
1057 return true;
1058 }
1059
1061 void reset() { next_tick_ns_ = nros_cpp_time_ns() + static_cast<uint64_t>(period_ns_); }
1062
1064 std::chrono::nanoseconds period() const { return std::chrono::nanoseconds(period_ns_); }
1065
1066 private:
1067 int64_t period_ns_;
1068 uint64_t next_tick_ns_;
