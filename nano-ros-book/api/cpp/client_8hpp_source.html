433 static void response_trampoline(const uint8_t* data, size_t len, void* ctx) {
434 auto* self = static_cast<Client*>(ctx);
435 if (self == nullptr) return;
436 ResponseType response;
437 if (ResponseType::ffi_deserialize(data, len, &response) != 0) return;
438 if (self->user_fn_ != nullptr) {
439 self->user_fn_(response);
440 } else if (self->user_fn_ctx_ != nullptr) {
441 self->user_fn_ctx_(response, self->user_ctx_);
442 }

447 ::nros::QoS actual_qos_half(bool request) const {
448 nros_cpp_qos_t req{};
449 nros_cpp_qos_t resp{};
450 if (!initialized_) return ::nros::detail::qos_all_unknown();
451 const void* storage = callback_mode_ ? nullptr : static_cast<const void*>(storage_);
452 if (nros_cpp_service_client_get_actual_qos(storage, executor_, handle_id_, &req, &resp) !=
453 0) {
454 return ::nros::detail::qos_all_unknown();
455 }
456 return ::nros::detail::qos_from_ffi(request ? req : resp);

517Result Node::create_client(Client<S>& out, const char* service_name, F callback,
518 const ::nros::QoS& qos, const ::nros::ClientOptions& options) {
519 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
520 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
521
522 out.user_fn_ = typename Client<S>::TypedResponseFn(callback);
523 out.user_fn_ctx_ = nullptr;
524 out.user_ctx_ = nullptr;
525
526 uint8_t sched = (options.sched_context == ::nros::SCHED_CONTEXT_UNSET)
527 ? 0u
528 : static_cast<uint8_t>(options.sched_context);
529 size_t handle = static_cast<size_t>(-1);
530 nros_cpp_ret_t ret = nros_cpp_service_client_register(
531 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
532 reinterpret_cast<nros_cpp_service_response_callback_t>(&Client<S>::response_trampoline),
533 &out, sched, &handle);
534 if (ret == 0) {
535 out.executor_ = executor_handle_;
536 out.handle_id_ = handle;
537 out.callback_mode_ = true;
538 // phase-444 — see the future-style overload above.
539 ::nros::detail::assign_entity_name(out.service_name_, service_name);
540 out.initialized_ = true;
541 }
542 return Result(ret);
543}
