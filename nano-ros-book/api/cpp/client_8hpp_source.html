335 // Move semantics (non-copyable). Bookkeeping only — there is no storage to

337 //
338 // phase-456 W3 — a callback-style client is MOVABLE, and the warning that

379 static void response_trampoline(const uint8_t* data, size_t len, void* ctx) {

381 if (user_fn == nullptr) return;

435template <typename S, typename F, typename>
436Result Node::create_client(Client<S>& out, const char* service_name, F callback,
437 const ::rclcpp::QoS& qos, const ::rclcpp::ClientOptions& options) {
438 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
439 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
440
441 // The user handler becomes the ARENA's context (phase-456 W3); the
442 // conversion is still the compile error for a non-convertible `F`.
443 const typename Client<S>::TypedResponseFn user_fn =
444 typename Client<S>::TypedResponseFn(callback);
445
446 uint8_t sched = (options.sched_context == ::rclcpp::SCHED_CONTEXT_UNSET)
447 ? 0u
448 : static_cast<uint8_t>(options.sched_context);
449 size_t handle = static_cast<size_t>(-1);
450 nros_cpp_ret_t ret = nros_cpp_service_client_register(
451 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
452 reinterpret_cast<nros_cpp_service_response_callback_t>(&Client<S>::response_trampoline),
453 ::rclcpp::detail::fn_to_context(user_fn), sched, &handle);
454 if (ret == 0) {
455 out.executor_ = executor_handle_;
456 out.handle_id_ = handle;
457 // phase-444 — see `rclcpp::PollClient`'s overload, which does the same.
458 ::rclcpp::detail::assign_entity_name(out.service_name_, service_name);
459 out.initialized_ = true;
460 }
461 return Result(ret);
462}
