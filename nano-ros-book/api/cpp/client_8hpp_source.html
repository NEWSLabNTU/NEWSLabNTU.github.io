251 // Move semantics (non-copyable). Bookkeeping only — there is no storage to
252 // relocate, so `nros_cpp_service_client_relocate` is not called here.
253 //
254 // phase-456 W3 — a callback-style client is MOVABLE, and the warning that
255 // used to stand here is gone with its subject. It said the arena holds
256 // `this` as the response trampoline context; the arena holds the user's

295 static void response_trampoline(const uint8_t* data, size_t len, void* ctx) {
296 const TypedResponseFn user_fn = ::nros::detail::fn_from_context<TypedResponseFn>(ctx);
297 if (user_fn == nullptr) return;
298 ResponseType response;
299 if (ResponseType::ffi_deserialize(data, len, &response) != 0) return;
300 user_fn(response);

311 ::nros::QoS actual_qos_half(bool request) const {
312 nros_cpp_qos_t req{};
313 nros_cpp_qos_t resp{};
314 if (!initialized_) return ::nros::detail::qos_all_unknown();

319 return ::nros::detail::qos_from_ffi(request ? req : resp);

360template <typename S, typename F, typename>
361Result Node::create_client(Client<S>& out, const char* service_name, F callback,
362 const ::nros::QoS& qos, const ::nros::ClientOptions& options) {
363 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
364 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
365
366 // The user handler becomes the ARENA's context (phase-456 W3); the
367 // conversion is still the compile error for a non-convertible `F`.
368 const typename Client<S>::TypedResponseFn user_fn =
369 typename Client<S>::TypedResponseFn(callback);
370
371 uint8_t sched = (options.sched_context == ::nros::SCHED_CONTEXT_UNSET)
372 ? 0u
373 : static_cast<uint8_t>(options.sched_context);
374 size_t handle = static_cast<size_t>(-1);
375 nros_cpp_ret_t ret = nros_cpp_service_client_register(
376 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
377 reinterpret_cast<nros_cpp_service_response_callback_t>(&Client<S>::response_trampoline),
378 ::nros::detail::fn_to_context(user_fn), sched, &handle);
379 if (ret == 0) {
380 out.executor_ = executor_handle_;
381 out.handle_id_ = handle;
382 // phase-444 — see `nros::PollClient`'s overload, which does the same.
383 ::nros::detail::assign_entity_name(out.service_name_, service_name);
384 out.initialized_ = true;
385 }
386 return Result(ret);
387}
