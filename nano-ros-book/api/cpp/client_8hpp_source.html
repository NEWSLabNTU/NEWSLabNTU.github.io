354 static void response_trampoline(const uint8_t* data, size_t len, void* ctx) {
355 auto* self = static_cast<Client*>(ctx);
356 if (self == nullptr) return;
357 ResponseType response;
358 if (ResponseType::ffi_deserialize(data, len, &response) != 0) return;
359 if (self->user_fn_ != nullptr) {
360 self->user_fn_(response);
361 } else if (self->user_fn_ctx_ != nullptr) {
362 self->user_fn_ctx_(response, self->user_ctx_);
363 }
364 }

406// Phase 189.M3.3.f — callback-style (arena-registered) client. The arena owns
407// the client + dispatches `out`'s response handler during spin_once; requests
408// go through `async_send_request`. `options.sched_context` is functional.
409template <typename S, typename F, typename>
410Result Node::create_client(Client<S>& out, const char* service_name, F callback, const QoS& qos,
411 const ClientOptions& options) {
412 if (!initialized_) return Result(ErrorCode::NotInitialized);
413 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
414
415 out.user_fn_ = typename Client<S>::TypedResponseFn(callback);
416 out.user_fn_ctx_ = nullptr;
417 out.user_ctx_ = nullptr;
418
419 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
420 ? 0u
421 : static_cast<uint8_t>(options.sched_context);
422 size_t handle = static_cast<size_t>(-1);
423 nros_cpp_ret_t ret = nros_cpp_service_client_register(
424 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
425 reinterpret_cast<nros_cpp_service_response_callback_t>(&Client<S>::response_trampoline),
426 &out, sched, &handle);
427 if (ret == 0) {
428 out.executor_ = executor_handle_;
429 out.handle_id_ = handle;
430 out.callback_mode_ = true;
431 out.initialized_ = true;
432 }
433 return Result(ret);
434}
