113 using Fut = Future<ResponseType, RespCap>;

178 Result call_polling_sized(const RequestType& req, ResponseType& resp,
179 uint32_t timeout_ms = 100) {
180 if (!initialized_) return Result(ErrorCode::NotInitialized);
181 uint8_t req_buf[::nros::detail::buffer_bounds<RequestType>::tx];
182 size_t req_len = 0;

184 return Result(ErrorCode::Error);

186 uint8_t resp_buf[RespCap];
187 size_t resp_len = 0;
188 nros_cpp_ret_t ret = nros_cpp_service_client_call_raw(
189 storage_, req_buf, req_len, resp_buf, sizeof(resp_buf), &resp_len, timeout_ms);
190 if (ret != 0) return Result(ret);
191 if (resp_len == 0) return Result(ErrorCode::Timeout);
192 if (ResponseType::ffi_deserialize(resp_buf, resp_len, &resp) != 0) {
193 return Result(ErrorCode::Error);

337 static void response_trampoline(const uint8_t* data, size_t len, void* ctx) {
338 auto* self = static_cast<Client*>(ctx);
339 if (self == nullptr) return;
340 ResponseType response;
341 if (ResponseType::ffi_deserialize(data, len, &response) != 0) return;
342 if (self->user_fn_ != nullptr) {
343 self->user_fn_(response);
344 } else if (self->user_fn_ctx_ != nullptr) {
345 self->user_fn_ctx_(response, self->user_ctx_);
346 }
347 }

380// Phase 189.M3.3.f — callback-style (arena-registered) client. The arena owns
381// the client + dispatches `out`'s response handler during spin_once; requests
382// go through `async_send_request`. `options.sched_context` is functional.
383template <typename S, typename F, typename>
384Result Node::create_client(Client<S>& out, const char* service_name, F callback, const QoS& qos,
385 const ClientOptions& options) {
386 if (!initialized_) return Result(ErrorCode::NotInitialized);
387 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
388
389 out.user_fn_ = typename Client<S>::TypedResponseFn(callback);
390 out.user_fn_ctx_ = nullptr;
391 out.user_ctx_ = nullptr;
392
393 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
394 ? 0u
395 : static_cast<uint8_t>(options.sched_context);
396 size_t handle = static_cast<size_t>(-1);
397 nros_cpp_ret_t ret = nros_cpp_service_client_register(
398 &handle_, service_name, S::TYPE_NAME, S::Request::TYPE_HASH, ffi_qos,
399 reinterpret_cast<nros_cpp_service_response_callback_t>(&Client<S>::response_trampoline),
400 &out, sched, &handle);
401 if (ret == 0) {
402 out.executor_ = executor_handle_;
403 out.handle_id_ = handle;
404 out.callback_mode_ = true;
405 out.initialized_ = true;
406 }
407 return Result(ret);
408}
