954 //
955 // OUTSIDE `NROS_CPP_NODE_HOSTED`, and that is the point rather than a
956 // placement detail. `create_publisher<M>("chatter", 10)` is the single most
957 // copied line in the porting corpus, and until W5 it was gated: the return

2901 Result build(::rclcpp::Node& out) const {

2903 out.executor_handle_ = executor_handle_;
2904 nros_cpp_ret_t ret =
2905 nros_cpp_node_create_ex(executor_handle_, name_, &options_, &out.handle_);
2906 if (ret == 0) {
2907 out.initialized_ = true;
2908 }
2909 return Result(ret);
2910 }

2912 private:
2913 static void copy_bounded(const char* src, uint8_t* dst, size_t* dst_len, size_t cap) {
2914 size_t n = 0;
2915 if (src != nullptr) {
2916 while (src[n] != '\0' && n < cap) {
2917 dst[n] = static_cast<uint8_t>(src[n]);
2918 ++n;
2919 }
2920 }
2921 // Zero out the tail so stale bytes don't leak across reuses.
2922 for (size_t i = n; i < cap; ++i) {
2923 dst[i] = 0;
2924 }
2925 *dst_len = n;
2926 }

2929 const char* name_;
2930 nros_cpp_node_options_t options_;
