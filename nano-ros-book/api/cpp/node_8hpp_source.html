2778//
2779// `ComponentNode` carried `Timer timers_[NROS_COMPONENT_MAX_TIMERS]` — 192
2780// unconditional bytes at the default depth of 8. Merging that onto `Node` would
2781// have charged every node in the tree for a pool most of them never touch, on
2782// targets chosen for having no memory to spare.
2783//
2784// So the depth is a template parameter and the DEFAULT IS ZERO: `Node` itself

3108inline int32_t apply_ros_args(void* executor, int argc, char const* const* argv) {
3109 // No `--ros-args` scope means nothing for the parser to read; skipping the
3110 // call keeps a plain `main(argc, argv)` off the FFI entirely.
3111 bool scoped = false;
3112 for (int i = 0; argv != nullptr && i < argc && !scoped; ++i) {
3113 const char* a = argv[i];
3114 const char* want = "--ros-args";
3115 while (a != nullptr && *a != '\0' && *a == *want) {
3116 ++a;
3117 ++want;
3118 }
3119 scoped = a != nullptr && *a == '\0' && *want == '\0';
3120 }
3121 if (!scoped) {
3122 return 0;
3123 }
3124 char why[256] = {0};
3125 const int32_t rc = nros_cpp_install_argv_remaps(executor, argc, argv, why, sizeof(why));
3126 if (rc != 0) {
3127 // issue 1303 — two records through `nros_log`, each short enough for
3128 // its format buffer: the refused token (bounded here), then the fixed
3129 // text, whose length `say_refused` checks at compile time. As one
3130 // `NROS_ERROR` record it was over 870 bytes, which a real sink cannot
3131 // carry: the body is dropped and only an ellipsis is printed.
3132 RCLC_LOG_ERROR(::rclcpp::detail::log_handle(nullptr), "rclcpp::init: refused: %.120s",
3133 why[0] != '\0' ? why : "invalid executor handle or argv");
3134 NROS_RCLCPP_SAY_REFUSED(NROS_RCLCPP_REFUSE_INIT_ARGV);

3305 }
3306 return Result(ret);

3310 static void copy_bounded(const char* src, uint8_t* dst, size_t* dst_len, size_t cap) {
3311 size_t n = 0;
3312 if (src != nullptr) {
3313 while (src[n] != '\0' && n < cap) {
3314 dst[n] = static_cast<uint8_t>(src[n]);
3315 ++n;
3316 }
3317 }
3318 // Zero out the tail so stale bytes don't leak across reuses.
3319 for (size_t i = n; i < cap; ++i) {
3320 dst[i] = 0;
3321 }
3322 *dst_len = n;
3323 }
