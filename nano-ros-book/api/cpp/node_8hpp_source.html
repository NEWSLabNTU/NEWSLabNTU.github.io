1009 // Global executor inline storage for init/shutdown free functions.
1010 //
1011 // Use a template-static-member trick instead of a function-local static.
1012 // Function-local statics need __cxa_guard_acquire/release on first-call
1013 // initialisation; on NuttX the resulting guard logic returns NULL for
1014 // the storage pointer (observed empirically with LTO on armv7a-nuttx-eabihf,
1015 // even with constant-initialisation `= {}`). A template static member is
1016 // emitted into .bss like a file-scope variable and gets COMDAT-folded by
1017 // the linker, sidestepping the guarded-init path entirely.
1018 template <int = 0> struct GlobalStorageHolder {
1019 alignas(8) static uint8_t storage[NROS_CPP_EXECUTOR_STORAGE_SIZE];
1020 static bool initialized;
1021 };
1022 static uint8_t* global_storage() { return GlobalStorageHolder<>::storage; }
1023 static bool& global_initialized() { return GlobalStorageHolder<>::initialized; }
1024};

1036 // Issue 0329 — forward RAW to the 3-arg overload, which resolves the whole

1038 // default, then the env overlay in the Rust resolver behind `nros_cpp_init`).
1039 // This overload previously re-resolved the locator rung itself — a second,
1040 // partial copy of the same ladder (it never applied `NROS_ENTRY_DOMAIN_ID`),

1046inline Result init(const char* locator, uint8_t domain_id, const char* session_name) {
1047 // NROS_CPP_RET_INVALID_ARGUMENT = -3 (defined in nros_cpp_ffi.h
1048 // which isn't included from this header — duplicate the value
1049 // inline; generated header is the source of truth).
1050 if (session_name == nullptr) {
1051 return Result(-3);
1052 }
1053#if defined(NROS_CPP_STD) || (__STDC_HOSTED__ + 0)
1054 // Issue #39 — apply the same `$NROS_LOCATOR` / `$ROS_DOMAIN_ID` env
1055 // fallback as the 2-arg `init()` when `locator` is null / `domain_id` is
1056 // 0. This makes `init_with_launch_auto()` (which delegates here with a
1057 // null locator) honor the env overlay instead of passing a null locator
1058 // to the backend → TransportError / degraded session.
1059 // Issue 0330 — the hard local default is GONE entirely (it was a zenoh
1060 // fact in an RMW-blind header); the backend now supplies it.
1061 // RFC-0045 / issue #206 — the env overlay (NROS_LOCATOR / ROS_DOMAIN_ID /
1062 // NROS_NODE_NAME) moved into the shared Rust resolver behind
1063 // nros_cpp_init (precedence model A: hosted env > this baked chain >
1064 // compiled default; malformed or >232 ROS_DOMAIN_ID is an init ERROR,
1065 // never a silent domain 0). This header only assembles the baked rung.
1066#endif
1067 // Phase-287 W6 — compile-time connect defaults, so ONE portable source
1068 // works native + embedded. `NROS_ENTRY_LOCATOR` / `NROS_ENTRY_DOMAIN_ID`
1069 // are target compile definitions the embedded board gate bakes
1070 // (NanoRosEntry.cmake; Kconfig on Zephyr via <nros/main.hpp>); on native
1071 // they are undefined. Precedence (model A, RFC-0045/#206): env (hosted,
1072 // applied in the Rust resolver) > explicit arg > baked macro > default.
1073#ifdef NROS_ENTRY_LOCATOR
1074 if (locator == nullptr) {
1075 locator = NROS_ENTRY_LOCATOR;
1076 }
1077#endif
1078#ifdef NROS_ENTRY_DOMAIN_ID

1080 // argument — including kDomainIdExplicitZero (255) for a literal
1081 // domain 0 (issue #227) — passes through untouched.
1082 if (domain_id == 0) {
1083 domain_id = static_cast<uint8_t>(NROS_ENTRY_DOMAIN_ID);
1084 }
1085#endif
1086 // Issue 0330 — there is deliberately NO hosted "tcp/127.0.0.1:7447"
1087 // fallback here. That value is a *zenoh* fact and this header is
1088 // RMW-blind; a cyclonedds or xrce build must not carry it. A null
1089 // locator flows through `nros_cpp_init` (→ `None` → the RFC-0045
1090 // resolver's empty bottom rung) to whichever backend is linked, and
1091 // that backend applies its own default (zenoh:
1092 // `nros_rmw_zenoh::DEFAULT_LOCATOR`; xrce: its agent default;
1093 // cyclonedds: ignores the locator). Precedence is otherwise unchanged:
1094 // hosted env > explicit arg > baked macro > backend default.
1095
1096 // Phase 128.C.1 / phase-241.D3-rev — RMW-blind init. The selected
1097 // backend registers itself before `main` via its `.init_array` ctor
1098 // (RFC-0042 §D3.3), and `nros_cpp_init` additionally calls the weak
1099 // `nros_app_register_backends()` board-override hook for RTOS targets
1100 // where `.init_array` ctors do not run. No `#ifdef NROS_RMW_*` chain
1101 // here, no CMake-driven fan-out — the user's
1102 // `target_link_libraries(... NanoRos::Rmw::<name>)` is the only
1103 // selector.
1104 // Issue 1050 defect (3) — the RMW selector's BAKED rung. `NROS_ENTRY_RMW`
1105 // is a target compile definition the entry gate bakes, exactly like
1106 // `NROS_ENTRY_LOCATOR` / `NROS_ENTRY_DOMAIN_ID` above; undefined means the
1107 // image names no backend and the registry must contain exactly one.
1108 //
1109 // This is what the paragraph above could not express. "The user's
1110 // `target_link_libraries(... NanoRos::Rmw::<name>)` is the only selector"
1111 // is true of the LINK and false of the REGISTRY: a hosted archive
1112 // registers whatever it carries, from `.init_array`, before `main`.
1113#ifdef NROS_ENTRY_RMW
1114 const char* rmw = NROS_ENTRY_RMW;
1115#else
1116 const char* rmw = nullptr;
1117#endif
1118 nros_cpp_ret_t ret =
1119 nros_cpp_init_rmw(rmw, locator, domain_id, session_name, nullptr, Node::global_storage());
1120 if (ret == 0) {
1121 Node::global_initialized() = true;
1122 }
1123 return Result(ret);
1124}

1265// -- Phase 104.C.9 — NodeBuilder ----------------------------------------
1266//
1267// Mirrors Rust's `Executor::node_builder(name).rmw(...).locator(...).
1268// domain_id(...).namespace(...).sched(...).build()` chain. The C++
1269// wrapper is value-typed and stack-allocated; it accumulates options
1270// into an inline `nros_cpp_node_options_t` and ships it to
1271// `nros_cpp_node_create_ex` on `.build()`.
1272//
1273// Usage:
1274// ```cpp
1275// nros::Node node;
1276// NROS_TRY(executor.node_builder("egress")
1277// .rmw("cyclonedds")
1278// .domain_id(0)
1279// .build(node));
1280// ```
1281
1282class NodeBuilder {
1283 public:
1284 NodeBuilder(void* executor_handle, const char* name)
1285 : executor_handle_(executor_handle), name_(name),
1286 options_(nros_cpp_node_get_default_options()) {}
1287
1292 NodeBuilder& rmw(const char* name) {
1293 copy_bounded(name, options_.rmw_name, &options_.rmw_name_len, NROS_CPP_RMW_NAME_LEN);
1294 return *this;
1295 }
1296
1299 NodeBuilder& locator(const char* loc) {
1300 copy_bounded(loc, options_.locator, &options_.locator_len, NROS_CPP_LOCATOR_LEN);
1301 return *this;
1302 }
1303
1306 NodeBuilder& domain_id(uint32_t id) {
1307 options_.domain_id_override = id;
1308 return *this;
1309 }
1310
1313 NodeBuilder& namespace_(const char* ns) {
1314 copy_bounded(ns, options_.namespace_, &options_.namespace_len, NROS_CPP_NAMESPACE_LEN);
1315 return *this;
1316 }
1317
1320 NodeBuilder& sched(uint8_t sc_id) {
1321 options_.sched_context_id = sc_id;
1322 return *this;
1323 }
1324
1326 Result build(Node& out) const {
1327 if (!executor_handle_) return Result(ErrorCode::NotInitialized);
1328 out.executor_handle_ = executor_handle_;
1329 nros_cpp_ret_t ret =
1330 nros_cpp_node_create_ex(executor_handle_, name_, &options_, &out.handle_);
1331 if (ret == 0) {
1332 out.initialized_ = true;
1333 }
1334 return Result(ret);
1335 }
1336
1337 private:
1338 static void copy_bounded(const char* src, uint8_t* dst, size_t* dst_len, size_t cap) {
1339 size_t n = 0;
1340 if (src != nullptr) {
1341 while (src[n] != '\0' && n < cap) {
1342 dst[n] = static_cast<uint8_t>(src[n]);
1343 ++n;
1344 }
1345 }
1346 // Zero out the tail so stale bytes don't leak across reuses.
1347 for (size_t i = n; i < cap; ++i) {
1348 dst[i] = 0;
1349 }
1350 *dst_len = n;
1351 }
1352
1353 void* executor_handle_;
1354 const char* name_;
1355 nros_cpp_node_options_t options_;
1356};
