2509 size_t n = 0;

2511 while (src[n] != '\0' && n < cap) {
2512 dst[n] = static_cast<uint8_t>(src[n]);
2513 ++n;
2514 }

2517 for (size_t i = n; i < cap; ++i) {

2519 }

2521 }
