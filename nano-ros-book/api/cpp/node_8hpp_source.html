61// `<chrono>` requires the EXPLICIT opt-in, not `__has_include`, and that is
62// measured rather than stylistic.
63//
64// `__has_include(<chrono>)` is TRUE on the Zephyr arm-none-eabi toolchain --
65// the file exists -- but under `-ffreestanding` libstdc++ ships it INCOMPLETE:
66// `std::chrono::duration_cast` is absent, so `create_wall_timer` and `Rate`
67// failed to compile on every Zephyr C++ image with
68// `'duration_cast' is not a member of 'std::chrono'`. Replayed from that build
69// dir's own recorded compile command, not inferred.
70//
71// So the idiom has a boundary worth stating: `__has_include` answers "does the
72// header exist", which is the right question for `<memory>` (present or absent
73// as a unit) and the WRONG one for `<chrono>` (present but hollowed out). Where
74// a freestanding toolchain ships a partial header, only the consumer's own
75// `NROS_CPP_STD` opt-in is a reliable signal. Issue 0112's class, one layer past
76// where step A caught it.
77// CORRECTED 2026-09-05. The `NROS_CPP_STD`-only gate above was measured on the
78// Zephyr lane and never on a plain hosted one, and NOTHING THAT SHIPS DEFINES
79// `NROS_CPP_STD`: it appears in no cmake module, no toolchain file, no build.rs
80// and no Kconfig -- only in docs and in `check.just`'s own probe TUs. So the
81// narrow gate did not merely tighten `<chrono>`; it removed `create_wall_timer`
82// and `Rate` from EVERY shipped configuration, including the hosted one, where
83// `examples/templates/cpp-port-minimal-publisher` calls `create_wall_timer` and
84// is phase-417's own acceptance criterion.
85//
86// The right predicate needs BOTH halves, because the two lanes fail
87// differently and neither test alone sees both:
88//
89// * Zephyr / ThreadX-RV64 build `-nostdinc++` against a minimal libcpp that
90// has no `<chrono>` at all, so `__has_include` answers FALSE and is exactly
91// right there.
92// * FreeRTOS armcm3 builds `-ffreestanding` WITHOUT `-nostdinc++`, so
93// `__has_include(<chrono>)` answers TRUE against the host's own libstdc++
94// -- which then refuses to be used, because GCC 13 gates
95// `bits/requires_hosted.h` on `__STDC_HOSTED__` and `-ffreestanding` clears
96// it. That is the "present but hollow" case, and `__STDC_HOSTED__` is
97// precisely the flag that distinguishes it.
98//
99// `__STDC_HOSTED__` alone is still not enough -- CLAUDE.md's pitfall entry says
100// so, and it is right: a hosted compiler can run `-nostdinc++` against Zephyr's
101// minimal libcpp, where the macro is 1 and the header is absent. Neither test
102// covers both lanes; the conjunction does.
103//
104// `NROS_CPP_STD` stays as an explicit override for a consumer who knows better
105// than the probes, which is what the parity extractor and the docs use it for.
106// CORRECTED AGAIN 2026-09-07, and this time the discriminator is the library's
107// own feature-test macro rather than a guess about which RTOS ships what.
108//
109// The conjunction above is still not enough. Measured under the safety island's
110// OWN recorded compile command (`-nostdinc++`, arm-zephyr-eabi, picolibc), not
111// inferred:
112//
113// __STDC_HOSTED__ 1
114// __has_include(<chrono>) TRUE
115// __cpp_lib_chrono ABSENT
116//
117// So Zephyr does NOT merely "have no <chrono> at all" -- this toolchain ships
118// one that is present and hollow, which is the case the note above attributes
119// to FreeRTOS alone. Both halves of the conjunction answer yes and
120// `duration_cast` is still missing, so `Rate`'s duration constructor failed to
121// compile on the island exactly as it did before that note was written.
122//
123// `<ratio>` is the third test, and it is a PREREQUISITE rather than a proxy.
124// `duration_cast<To>(duration<Rep, Period>)` is defined in terms of
125// `std::ratio_divide` -- a duration's `Period` IS a `std::ratio` -- so a
126// `<chrono>` shipped without `<ratio>` cannot provide `duration_cast`, and that
127// is exactly the shape this toolchain ships. Measured on both sides:
128//
129// island (Zephyr) hosted g++ -std=c++14
130// __has_include(<chrono>) TRUE TRUE
131// __has_include(<ratio>) FALSE TRUE
132//
133// `__cpp_lib_chrono` was tried first and is WRONG here: it is a C++17
134// feature-test macro, so it is absent under `-std=c++14` even where the library
135// is complete. `check-cpp` compiles the phase-417 probes at C++14 and caught
136// that immediately -- gating on it removed `create_wall_timer` from the hosted
137// build, which is the same over-tightening the note above records.
138// `_GLIBCXX_CHRONO` also separates the two but is libstdc++'s private spelling;
139// `__has_include` is the portable question.

1585 // Global executor inline storage for init/shutdown free functions.
1586 //
1587 // Use a template-static-member trick instead of a function-local static.
1588 // Function-local statics need __cxa_guard_acquire/release on first-call
1589 // initialisation; on NuttX the resulting guard logic returns NULL for
1590 // the storage pointer (observed empirically with LTO on armv7a-nuttx-eabihf,

1655 // Issue 0330 — the hard local default is GONE entirely (it was a zenoh

1683 // fallback here. That value is a *zenoh* fact and this header is
1684 // RMW-blind; a cyclonedds or xrce build must not carry it. A null

1687 // that backend applies its own default (zenoh:
1688 // `nros_rmw_zenoh::DEFAULT_LOCATOR`; xrce: its agent default;

1860//
1861// Mirrors Rust's `Executor::node_builder(name).rmw(...).locator(...).
1862// domain_id(...).namespace(...).sched(...).build()` chain. The C++
1863// wrapper is value-typed and stack-allocated; it accumulates options
1864// into an inline `nros_cpp_node_options_t` and ships it to
1865// `nros_cpp_node_create_ex` on `.build()`.
1866//
1867// Usage:
1868// ```cpp
1869// nros::Node node;
1870// NROS_TRY(executor.node_builder("egress")
1871// .rmw("cyclonedds")
1872// .domain_id(0)
1873// .build(node));
1874// ```

1876class NodeBuilder {
1877 public:

1879 : executor_handle_(executor_handle), name_(name),
1880 options_(nros_cpp_node_get_default_options()) {}

1888 return *this;
1889 }
1890

1894 copy_bounded(loc, options_.locator, &options_.locator_len, NROS_CPP_LOCATOR_LEN);
1895 return *this;
1896 }

1900 NodeBuilder& domain_id(uint32_t id) {

1902 return *this;
1903 }
1904

1910 }

1916 return *this;
1917 }

1920 Result build(Node& out) const {
1921 if (!executor_handle_) return Result(ErrorCode::NotInitialized);
1922 out.executor_handle_ = executor_handle_;
1923 nros_cpp_ret_t ret =
1924 nros_cpp_node_create_ex(executor_handle_, name_, &options_, &out.handle_);
1925 if (ret == 0) {
1926 out.initialized_ = true;
1927 }
1928 return Result(ret);
1929 }

1931 private:
1932 static void copy_bounded(const char* src, uint8_t* dst, size_t* dst_len, size_t cap) {
1933 size_t n = 0;
1934 if (src != nullptr) {
1935 while (src[n] != '\0' && n < cap) {
1936 dst[n] = static_cast<uint8_t>(src[n]);
1937 ++n;
1938 }
1939 }
1940 // Zero out the tail so stale bytes don't leak across reuses.
1941 for (size_t i = n; i < cap; ++i) {
1942 dst[i] = 0;
1943 }
1944 *dst_len = n;
1945 }

1947 void* executor_handle_;
1948 const char* name_;
1949 nros_cpp_node_options_t options_;
1950};
