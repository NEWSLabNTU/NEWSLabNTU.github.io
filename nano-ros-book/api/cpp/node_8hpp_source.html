3047inline int32_t apply_ros_args(void* executor, int argc, char const* const* argv) {
3048 // No `--ros-args` scope means nothing for the parser to read; skipping the
3049 // call keeps a plain `main(argc, argv)` off the FFI entirely.
3050 bool scoped = false;
3051 for (int i = 0; argv != nullptr && i < argc && !scoped; ++i) {
3052 const char* a = argv[i];
3053 const char* want = "--ros-args";
3054 while (a != nullptr && *a != '\0' && *a == *want) {
3055 ++a;
3056 ++want;
3057 }
3058 scoped = a != nullptr && *a == '\0' && *want == '\0';
3059 }
3060 if (!scoped) {
3061 return 0;
3062 }
3063 char why[256] = {0};
3064 const int32_t rc = nros_cpp_install_argv_remaps(executor, argc, argv, why, sizeof(why));
3065 if (rc != 0) {
3066 // issue 1303 — two records through `nros_log`, each short enough for
3067 // its format buffer: the refused token (bounded here), then the fixed
3068 // text, whose length `say_refused` checks at compile time. As one
3069 // `NROS_ERROR` record it was over 870 bytes, which a real sink cannot
3070 // carry: the body is dropped and only an ellipsis is printed.
3071 RCLC_LOG_ERROR(::rclcpp::detail::log_handle(nullptr), "rclcpp::init: refused: %.120s",
3072 why[0] != '\0' ? why : "invalid executor handle or argv");
3073 NROS_RCLCPP_SAY_REFUSED(NROS_RCLCPP_REFUSE_INIT_ARGV);
3074 }
3075 return rc;
3076}
