3495inline int32_t apply_ros_args(void* executor, int argc, char const* const* argv) {
3496 // No `--ros-args` scope means nothing for the parser to read; skipping the
3497 // call keeps a plain `main(argc, argv)` off the FFI entirely.
3498 bool scoped = false;
3499 for (int i = 0; argv != nullptr && i < argc && !scoped; ++i) {
3500 const char* a = argv[i];
3501 const char* want = "--ros-args";
3502 while (a != nullptr && *a != '\0' && *a == *want) {
3503 ++a;
3504 ++want;
3505 }
3506 scoped = a != nullptr && *a == '\0' && *want == '\0';
3507 }
3508 // Issue 2073 -- an image that baked arguments of its own has them to
3509 // apply whatever the command line holds (a board's is empty).
3510 if (!scoped && !nros_cpp_baked_ros_args_present()) {
3511 return 0;
3512 }
3513 char why[256] = {0};
3514 const int32_t rc = nros_cpp_install_argv_remaps(executor, argc, argv, why, sizeof(why));
3515 if (rc != 0) {
3516 // issue 1303 — two records through `nros_log`, each short enough for
3517 // its format buffer: the refused token (bounded here), then the fixed
3518 // text, whose length `say_refused` checks at compile time. As one
3519 // `NROS_ERROR` record it was over 870 bytes, which a real sink cannot
3520 // carry: the body is dropped and only an ellipsis is printed.
3521 RCLC_LOG_ERROR(::rclcpp::detail::log_handle(nullptr), "rclcpp::init: refused: %.120s",
3522 why[0] != '\0' ? why : "invalid executor handle or argv");
3523 NROS_RCLCPP_SAY_REFUSED(NROS_RCLCPP_REFUSE_INIT_ARGV);
3524 }
3525 return rc;
3526}
