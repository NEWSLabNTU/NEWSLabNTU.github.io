2518 // `nros_rmw_zenoh::DEFAULT_LOCATOR`; xrce: its agent default;
2519 // cyclonedds: ignores the locator). Precedence is otherwise unchanged:
2520 // hosted env > explicit arg > baked macro > backend default.

2522 // Phase 128.C.1 / phase-241.D3-rev — RMW-blind init. The selected
2523 // backend registers itself before `main` via its `.init_array` ctor
2524 // (RFC-0042 §D3.3), and `nros_cpp_init` additionally calls the weak
2525 // `nros_app_register_backends()` board-override hook for RTOS targets
2526 // where `.init_array` ctors do not run. No `#ifdef NROS_RMW_*` chain
2527 // here, no CMake-driven fan-out — the user's
2528 // `target_link_libraries(... NanoRos::Rmw::<name>)` is the only
2529 // selector.
2530 // Issue 1050 defect (3) — the RMW selector's BAKED rung. `NROS_ENTRY_RMW`
2531 // is a target compile definition the entry gate bakes, exactly like
2532 // `NROS_ENTRY_LOCATOR` / `NROS_ENTRY_DOMAIN_ID` above; undefined means the
2533 // image names no backend and the registry must contain exactly one.
2534 //
2535 // This is what the paragraph above could not express. "The user's
2536 // `target_link_libraries(... NanoRos::Rmw::<name>)` is the only selector"
2537 // is true of the LINK and false of the REGISTRY: a hosted archive
2538 // registers whatever it carries, from `.init_array`, before `main`.
2539#ifdef NROS_ENTRY_RMW
2540 const char* rmw = NROS_ENTRY_RMW;
2541#else
2542 const char* rmw = nullptr;
2543#endif
2544 // Issue 1434 — the namespace travels as the BAKED rung, like `rmw` above
2545 // and `session_name` beside it. `nullptr` reaches `nros_cpp_init_rmw`'s
2546 // own "unset" arm, which is the root; an EMPTY string is normalised here
2547 // rather than there, because a bake macro that did not resolve expands to
2548 // `""` and treating that as "the empty namespace" would be a configured
2549 // value where the image stated nothing (the same reasoning 1050 applied to
2550 // an empty `rmw` selector).
2551 if (node_namespace != nullptr && node_namespace[0] == '\0') {
2552 node_namespace = nullptr;
2553 }
2554 nros_cpp_ret_t ret = nros_cpp_init_rmw(rmw, locator, domain_id, session_name, node_namespace,

2699// -- Phase 104.C.9 — NodeBuilder ----------------------------------------
2700//
2701// Mirrors Rust's `Executor::node_builder(name).rmw(...).locator(...).

2703// wrapper is value-typed and stack-allocated; it accumulates options

2705// `nros_cpp_node_create_ex` on `.build()`.

2708// ```cpp

2710// NROS_TRY(executor.node_builder("egress")

2714// ```

2716class NodeBuilder {
2717 public:

2719 : executor_handle_(executor_handle), name_(name),
2720 options_(nros_cpp_node_get_default_options()) {}
2721

2728 return *this;
2729 }
2730

2734 copy_bounded(loc, options_.locator, &options_.locator_len, NROS_CPP_LOCATOR_LEN);
2735 return *this;
2736 }
2737
2740 NodeBuilder& domain_id(uint32_t id) {
2741 options_.domain_id_override = id;
2742 return *this;
2743 }
2744
2747 NodeBuilder& namespace_(const char* ns) {
2748 copy_bounded(ns, options_.namespace_, &options_.namespace_len, NROS_CPP_NAMESPACE_LEN);

2760 Result build(::rclcpp::Node& out) const {
2761 if (!executor_handle_) return Result(ErrorCode::NotInitialized);
2762 out.executor_handle_ = executor_handle_;
2763 nros_cpp_ret_t ret =
2764 nros_cpp_node_create_ex(executor_handle_, name_, &options_, &out.handle_);
2765 if (ret == 0) {
2766 out.initialized_ = true;
2767 }
2768 return Result(ret);
2769 }

2771 private:
2772 static void copy_bounded(const char* src, uint8_t* dst, size_t* dst_len, size_t cap) {
2773 size_t n = 0;
2774 if (src != nullptr) {
2775 while (src[n] != '\0' && n < cap) {
2776 dst[n] = static_cast<uint8_t>(src[n]);
2777 ++n;
2778 }
2779 }
2780 // Zero out the tail so stale bytes don't leak across reuses.
2781 for (size_t i = n; i < cap; ++i) {
2782 dst[i] = 0;
2783 }
2784 *dst_len = n;
2785 }

2788 const char* name_;
2789 nros_cpp_node_options_t options_;
