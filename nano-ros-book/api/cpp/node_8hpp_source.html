1509 // Global executor inline storage for init/shutdown free functions.
1510 //
1511 // Use a template-static-member trick instead of a function-local static.
1512 // Function-local statics need __cxa_guard_acquire/release on first-call
1513 // initialisation; on NuttX the resulting guard logic returns NULL for
1514 // the storage pointer (observed empirically with LTO on armv7a-nuttx-eabihf,

1519 alignas(8) static uint8_t storage[NROS_CPP_EXECUTOR_STORAGE_SIZE];
1520 };

1567 // NROS_CPP_RET_INVALID_ARGUMENT = -3 (defined in nros_cpp_ffi.h

1569 // inline; generated header is the source of truth).
1570 if (session_name == nullptr) {

1573#if defined(NROS_CPP_STD) || (__STDC_HOSTED__ + 0)

1578 // to the backend → TransportError / degraded session.
1579 // Issue 0330 — the hard local default is GONE entirely (it was a zenoh

1581 // RFC-0045 / issue #206 — the env overlay (NROS_LOCATOR / ROS_DOMAIN_ID /

1590 // (NanoRosEntry.cmake; Kconfig on Zephyr via <nros/main.hpp>); on native

1597#endif
1598#ifdef NROS_ENTRY_DOMAIN_ID

1605#endif

1612 // `nros_rmw_zenoh::DEFAULT_LOCATOR`; xrce: its agent default;
