766 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
767 // message the linked backend cannot encode never reaches the wire.

770 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
771 nros_cpp_ret_t ret = nros_cpp_subscription_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
772 ffi_qos, out.storage_);

837 // message the linked backend cannot encode never reaches the wire.

1054inline Result register_subscription_capturing(::rclcpp::Node& node, const char* topic,
1055 const QoS& qos, const Fn& fn, size_t* out_handle_id) {
1056 static_assert(sizeof(Fn) <= NROS_CPP_CALLBACK_CAPACITY + 2 * sizeof(void*),

1058 "NROS_CPP_CALLBACK_CAPACITY, and the arena's CALLBACK_CAPTURE_BYTES with it");

1063 void* ctx) {
1064 // `ctx` is the ARENA's copy of `fn`, not ours.
1065 const Fn* self = static_cast<const Fn*>(ctx);
1066 M msg;
1067 if (M::ffi_deserialize(data, len, &msg) != 0) return;
1068 (*self)(msg);
1069 };

1072 nros_cpp_subscription_options_t opts = {};
1073 opts.rx_buffer_hint = static_cast<uint32_t>(::nros::rx_buffer_capacity<M>::value);
1074 return Result(nros_cpp_subscription_register_capturing(
1075 h, topic, M::TYPE_NAME, M::TYPE_HASH, ffi_qos, invoke,
1076 reinterpret_cast<const uint8_t*>(&fn), sizeof(Fn), out_handle_id, &opts));
