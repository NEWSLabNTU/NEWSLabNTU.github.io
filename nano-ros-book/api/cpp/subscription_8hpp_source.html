631Result Node::create_subscription(Subscription<M>& out, const char* topic, const QoS& qos) {
632 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
633 // message the linked backend cannot encode never reaches the wire.
634 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
635 if (!initialized_) return Result(ErrorCode::NotInitialized);
636 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
637 nros_cpp_ret_t ret = nros_cpp_subscription_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
638 ffi_qos, out.storage_);
639 if (ret == 0) {
640 // Phase 87.6: topic name lives C++-side now.
641 size_t topic_len = 0;
642 while (topic[topic_len] != '\0' && topic_len + 1 < sizeof(out.topic_name_)) {
643 out.topic_name_[topic_len] = topic[topic_len];
644 ++topic_len;
645 }
646 out.topic_name_[topic_len] = '\0';
647 out.initialized_ = true;
648 }
649 return Result(ret);
650}

664template <typename M>
665Result Node::create_subscription(Subscription<M>& out, const char* topic, const QoS& qos,
666 const SubscriptionOptions& options) {
667 Result r = create_subscription<M>(out, topic, qos);
668 if (!r.ok()) return r;
669
670 // TODO(M3.4): honour options.message_info via the with-info arena path.
671
672 if (options.sched_context != SCHED_CONTEXT_UNSET && out.has_sched_handle()) {
673 nros_cpp_ret_t bind = nros_cpp_bind_handle_to_sched_context(
674 executor_handle_, out.sched_handle_id(), static_cast<uint8_t>(options.sched_context));
675 if (bind != 0) {
676 // Roll back so the caller doesn't observe a half-configured
677 // entity. Destructor-on-out would also fire, but explicit
678 // teardown keeps the returned error authoritative.
679 nros_cpp_subscription_destroy(out.storage_);
680 out.initialized_ = false;
681 return Result(bind);
682 }
683 }
684 return Result::success();
685}

687// Phase 189.M3.x — callback-style (arena-registered) subscription. The arena
688// owns the subscriber + dispatches `out`'s message handler during spin_once, so
689// the handle is real and `options.sched_context` is functional. Mirrors the
690// callback-style `create_service` one entity over.
691template <typename M, typename F, typename>
692Result Node::create_subscription(Subscription<M>& out, const char* topic, F callback,
693 const QoS& qos, const SubscriptionOptions& options) {
694 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
695 // message the linked backend cannot encode never reaches the wire.
696 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
697 if (!initialized_) return Result(ErrorCode::NotInitialized);
698 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
699
700 // Store the user handler (compile error if F isn't convertible to the
701 // plain-fn-ptr handler type).
702 out.user_fn_ = typename Subscription<M>::TypedSubscriptionFn(callback);
703 out.user_fn_ctx_ = nullptr;
704 out.user_ctx_ = nullptr;
705
706 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
707 ? 0u
708 : static_cast<uint8_t>(options.sched_context);
709 size_t handle = static_cast<size_t>(-1);
710 // phase-402: `sched_context` is a FIELD now. `callback_group` stays unset,
711 // i.e. the default group.
712 nros_cpp_subscription_options_t ffi_options = nros_cpp_subscription_default_options();
713 ffi_options.sched_context = sched;
714 nros_cpp_ret_t ret = nros_cpp_subscription_register(
715 &handle_, topic, M::TYPE_NAME, M::TYPE_HASH, ffi_qos, &Subscription<M>::message_trampoline,
716 &out, &handle, &ffi_options);
717 if (ret == 0) {
718 out.sched_handle_id_ = handle;
719 out.callback_mode_ = true;
720 out.initialized_ = true;
721 }
722 return Result(ret);
723}
