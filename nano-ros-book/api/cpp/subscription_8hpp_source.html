445template <typename M, typename F, typename>
446Result Node::create_subscription(Subscription<M>& out, const char* topic, F callback,
447 const ::nros::QoS& qos,
448 const ::nros::SubscriptionOptions& options) {
449 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
450 // message the linked backend cannot encode never reaches the wire.
451 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
452 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
453 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
454
455 // Store the user handler (compile error if F isn't convertible to the
456 // plain-fn-ptr handler type).
457 out.user_fn_ = typename Subscription<M>::TypedSubscriptionFn(callback);
458 out.user_fn_ctx_ = nullptr;
459 out.user_ctx_ = nullptr;
460
461 uint8_t sched = (options.sched_context == ::nros::SCHED_CONTEXT_UNSET)
462 ? 0u
463 : static_cast<uint8_t>(options.sched_context);
464 size_t handle = static_cast<size_t>(-1);
465 // phase-402: `sched_context` is a FIELD now. `callback_group` stays unset,
466 // i.e. the default group.
467 nros_cpp_subscription_options_t ffi_options = nros_cpp_subscription_default_options();
468 ffi_options.sched_context = sched;
469 // phase-456 W7 — state the bound. `M` is a template parameter of this
470 // function, so the number was always available here; leaving it at the
471 // option default put this registration on issue 1319's `c_raw_no_hint` row
472 // while the sizing descriptor credited the entry with a supplied hint.
473 ffi_options.rx_buffer_hint = static_cast<uint32_t>(::nros::rx_buffer_capacity<M>::value);
474 nros_cpp_ret_t ret = nros_cpp_subscription_register(
475 &handle_, topic, M::TYPE_NAME, M::TYPE_HASH, ffi_qos, &Subscription<M>::message_trampoline,
476 &out, &handle, &ffi_options);
477 if (ret == 0) {
478 out.sched_handle_id_ = handle;
479 // phase-456 W2b — `get_topic_name()` answered "" on every callback-style
480 // subscription until this line existed.
481 out.store_topic_name(topic);
482 out.initialized_ = true;
483 }
484 return Result(ret);
485}

651inline Result register_subscription_capturing(::rclcpp::Node& node, const char* topic,
652 const QoS& qos, const Fn& fn, size_t* out_handle_id) {
653 // phase-456 W8 — the per-entry capture budget this used to assert is GONE.
654 // It read `sizeof(Fn) <= NROS_CPP_CALLBACK_CAPACITY + 2 * sizeof(void*)` and
655 // named `CALLBACK_CAPTURE_BYTES`, an arena constant that no longer exists:
656 // the runtime allocates exactly `capture_len` bytes now, so there is no
657 // cross-language number for the two sides to agree about and nothing here
658 // to assert. `InplaceFn`'s own capacity `static_assert` is untouched and is
659 // still the knob a too-large CALLABLE hits (`inplace_fn.hpp`), which is a
660 // property of that type rather than of this registration.
661 const nros_cpp_node_t* h = node.ffi_handle();
662 if (h == nullptr) return Result(ErrorCode::NotInitialized);
663
664 nros_cpp_subscription_message_callback_t invoke = [](const uint8_t* data, size_t len,
