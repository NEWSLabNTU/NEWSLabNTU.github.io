440 // GONE, with the taking API that was their only reader. On this path the
441 // arena owns the subscriber, so those bytes were never filled, and
442 // `take()` reinterpreted them (`&mut *(storage as *mut RmwSubscriber)`)
443 // rather than refusing. `callback_mode_` went with them: it distinguished
444 // the two owners inside one class, and there is one owner here now.

447 // Phase 189.M3.1 — executor HandleId for sched-context binding, or
448 // SIZE_MAX (the default) when the registration has not happened yet.
449 size_t sched_handle_id_ = static_cast<size_t>(-1);
450 // The executor that owns the arena entry `sched_handle_id_` indexes —
451 // the phase-467 RMW gap-closure design study's Row 13.
452 //
453 // A `HandleId` is EXECUTOR-SCOPED: an index into one executor's arena, not
454 // a process-wide identity, and RFC-0002 puts one executor on one RTOS
455 // task, so a tiered image has several. That is why `get_actual_qos()`
456 // could not be answered from `sched_handle_id_` alone, and why resolving
457 // it "from the sched handle" would have needed a global executor
458 // registry. The pair is the same one `rclcpp::Client` already keeps
459 // (`{executor_, handle_id_}`), for the same reason.
460 void* executor_ = nullptr;

462 TypedSubscriptionFn user_fn_ = nullptr;

487template <typename M, typename F, typename>
488Result Node::create_subscription(Subscription<M>& out, const char* topic, const ::rclcpp::QoS& qos,
489 F callback, const ::rclcpp::SubscriptionOptions& options) {
490 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
491 // message the linked backend cannot encode never reaches the wire.
492 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
493 if (!initialized_) return Result(::rclcpp::ErrorCode::NotInitialized);
494 nros_cpp_qos_t ffi_qos = ::rclcpp::detail::qos_to_ffi(qos);
495
496 // Store the user handler (compile error if F isn't convertible to the
497 // plain-fn-ptr handler type).
498 out.user_fn_ = typename Subscription<M>::TypedSubscriptionFn(callback);
499 out.user_fn_ctx_ = nullptr;
500 out.user_ctx_ = nullptr;
501
502 uint8_t sched = (options.sched_context == ::rclcpp::SCHED_CONTEXT_UNSET)
503 ? 0u
504 : static_cast<uint8_t>(options.sched_context);
505 size_t handle = static_cast<size_t>(-1);
506 // phase-402: `sched_context` is a FIELD now. `callback_group` stays unset,
507 // i.e. the default group.
508 nros_cpp_subscription_options_t ffi_options = nros_cpp_subscription_default_options();
509 ffi_options.sched_context = sched;
510 // phase-456 W7 — state the bound. `M` is a template parameter of this
511 // function, so the number was always available here; leaving it at the
512 // option default put this registration on issue 1319's `c_raw_no_hint` row
513 // while the sizing descriptor credited the entry with a supplied hint.
514 ffi_options.rx_buffer_hint = static_cast<uint32_t>(::rclcpp::rx_buffer_capacity<M>::value);
515 nros_cpp_ret_t ret = nros_cpp_subscription_register(
516 &handle_, topic, M::TYPE_NAME, M::TYPE_HASH, ffi_qos, &Subscription<M>::message_trampoline,
517 &out, &handle, &ffi_options);
518 if (ret == 0) {
519 out.sched_handle_id_ = handle;
520 out.executor_ = executor_handle_;
521 // phase-456 W2b — `get_topic_name()` answered "" on every callback-style
522 // subscription until this line existed.
523 out.store_topic_name(topic);
524 out.initialized_ = true;
525 }
526 return Result(ret);
527}

685// ---- issue 1995: which callback shapes a subscription takes ------------------
686//
687// The arena delivers `const M&` from a message decoded in its dispatch frame,
688// and that stays the ONE delivery path. Upstream's other common shapes —
689// `M::UniquePtr`, `M::SharedPtr`, `M::ConstSharedPtr` (by value or `const&`),
690// `std::shared_ptr<const M>` — are taken on a HOSTED build by wrapping the
691// callable in an adapter that COPIES the message into a heap object per
692// delivery and hands the callback that pointer.
693//
694// Why a copy, not the service handler's frame loan (`detail::lend_from_frame`):
695// a subscription callback KEEPING its pointer (`last_msg_ = msg;`) is the
696// common case, not the rare one, and a loan turns it into an abort. A copy
697// makes keeping it legal, and the allocation is opted into by the signature —
698// a `const M&` callback still allocates nothing. Freestanding builds have no
699// allocator on this path, so there the pointer shapes are refused at compile
700// time (`NROS_RCLCPP_REFUSE_SUBSCRIPTION_CALLBACK_SHAPE`).

922template <typename M, typename Fn>
923inline Result register_subscription_capturing(::rclcpp::Node& node, const char* topic,
924 const QoS& qos, const Fn& fn, size_t* out_handle_id) {
925 // phase-456 W8 — the per-entry capture budget this used to assert is GONE.
926 // It read `sizeof(Fn) <= NROS_CPP_CALLBACK_CAPACITY + 2 * sizeof(void*)` and
927 // named `CALLBACK_CAPTURE_BYTES`, an arena constant that no longer exists:
928 // the runtime allocates exactly `capture_len` bytes now, so there is no
929 // cross-language number for the two sides to agree about and nothing here
930 // to assert. `InplaceFn`'s own capacity `static_assert` is untouched and is
931 // still the knob a too-large CALLABLE hits (`inplace_fn.hpp`), which is a
932 // property of that type rather than of this registration.
933 const nros_cpp_node_t* h = node.ffi_handle();
934 if (h == nullptr) return Result(ErrorCode::NotInitialized);
935
936 nros_cpp_subscription_message_callback_t invoke = [](const uint8_t* data, size_t len,
937 void* ctx) {
938 // `ctx` is the ARENA's copy of `fn`, not ours.
939 const Fn* self = static_cast<const Fn*>(ctx);
940 M msg;
941 if (M::ffi_deserialize(data, len, &msg) != 0) return;
942 (*self)(msg);
943 };
944
945 // phase-476 W2 — the arena's copy is destroyed when the subscription is
946 // released. `Fn` is an `rclcpp::InplaceFn` (the one caller passes one), so
947 // `clear()` runs the stored callable's own destructor.
948 void (*destroy)(void*) = [](void* ctx) { static_cast<Fn*>(ctx)->clear(); };
949
950 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
951 nros_cpp_subscription_options_t opts = {};
952 opts.rx_buffer_hint = static_cast<uint32_t>(::rclcpp::rx_buffer_capacity<M>::value);
953 return Result(nros_cpp_subscription_register_capturing(
954 h, topic, M::TYPE_NAME, M::TYPE_HASH, ffi_qos, invoke,
955 reinterpret_cast<const uint8_t*>(&fn), sizeof(Fn), destroy, out_handle_id, &opts));
956}
