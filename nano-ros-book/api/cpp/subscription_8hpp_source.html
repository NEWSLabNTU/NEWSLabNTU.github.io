669template <typename M>
670Result Node::create_subscription(Subscription<M>& out, const char* topic, const ::nros::QoS& qos) {
671 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
672 // message the linked backend cannot encode never reaches the wire.
673 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
674 if (!initialized_) return Result(::nros::ErrorCode::NotInitialized);
675 nros_cpp_qos_t ffi_qos = ::nros::detail::qos_to_ffi(qos);
676 nros_cpp_ret_t ret = nros_cpp_subscription_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
677 ffi_qos, out.storage_);
678 if (ret == 0) {
679 // Phase 87.6: topic name lives C++-side now.
680 size_t topic_len = 0;
681 while (topic[topic_len] != '\0' && topic_len + 1 < sizeof(out.topic_name_)) {
682 out.topic_name_[topic_len] = topic[topic_len];
683 ++topic_len;
684 }
685 out.topic_name_[topic_len] = '\0';
686 out.initialized_ = true;
687 }
688 return Result(ret);
689}

710Result Node::create_subscription(Subscription<M>& out, const char* topic, const ::nros::QoS& qos,
711 const ::nros::SubscriptionOptions& options) {
712 Result r = create_subscription<M>(out, topic, qos);
713 if (!r.ok()) return r;
714
715 // TODO(M3.4): honour options.message_info via the with-info arena path.
716
717 if (options.sched_context != ::nros::SCHED_CONTEXT_UNSET && out.has_sched_handle()) {
718 nros_cpp_ret_t bind = nros_cpp_bind_handle_to_sched_context(
719 executor_handle_, out.sched_handle_id(), static_cast<uint8_t>(options.sched_context));
720 if (bind != 0) {
721 // Roll back so the caller doesn't observe a half-configured
722 // entity. Destructor-on-out would also fire, but explicit
723 // teardown keeps the returned error authoritative.
724 nros_cpp_subscription_destroy(out.storage_);
725 out.initialized_ = false;
726 return Result(bind);
727 }
728 }
729 return Result::success();
