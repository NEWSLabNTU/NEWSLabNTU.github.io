663Result Node::create_subscription(Subscription<M>& out, const char* topic, const QoS& qos) {
664 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
665 // message the linked backend cannot encode never reaches the wire.
666 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
667 if (!initialized_) return Result(ErrorCode::NotInitialized);
668 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
669 nros_cpp_ret_t ret = nros_cpp_subscription_create(&handle_, topic, M::TYPE_NAME, M::TYPE_HASH,
670 ffi_qos, out.storage_);
671 if (ret == 0) {
672 // Phase 87.6: topic name lives C++-side now.
673 size_t topic_len = 0;
674 while (topic[topic_len] != '\0' && topic_len + 1 < sizeof(out.topic_name_)) {
675 out.topic_name_[topic_len] = topic[topic_len];
676 ++topic_len;
677 }
678 out.topic_name_[topic_len] = '\0';
679 out.initialized_ = true;
680 }
681 return Result(ret);

697Result Node::create_subscription(Subscription<M>& out, const char* topic, const QoS& qos,
698 const SubscriptionOptions& options) {

700 if (!r.ok()) return r;

702 // TODO(M3.4): honour options.message_info via the with-info arena path.

704 if (options.sched_context != SCHED_CONTEXT_UNSET && out.has_sched_handle()) {
705 nros_cpp_ret_t bind = nros_cpp_bind_handle_to_sched_context(

719// Phase 189.M3.x — callback-style (arena-registered) subscription. The arena
720// owns the subscriber + dispatches `out`'s message handler during spin_once, so
721// the handle is real and `options.sched_context` is functional. Mirrors the
722// callback-style `create_service` one entity over.
723template <typename M, typename F, typename>
724Result Node::create_subscription(Subscription<M>& out, const char* topic, F callback,
725 const QoS& qos, const SubscriptionOptions& options) {
726 // RFC-0088 D5 — one image, one backend, one encoding. Compile-time, so a
727 // message the linked backend cannot encode never reaches the wire.
728 NROS_CPP_ASSERT_MESSAGE_FORMAT(M);
729 if (!initialized_) return Result(ErrorCode::NotInitialized);
730 nros_cpp_qos_t ffi_qos = detail::qos_to_ffi(qos);
731
732 // Store the user handler (compile error if F isn't convertible to the
733 // plain-fn-ptr handler type).
734 out.user_fn_ = typename Subscription<M>::TypedSubscriptionFn(callback);
735 out.user_fn_ctx_ = nullptr;
736 out.user_ctx_ = nullptr;
737
738 uint8_t sched = (options.sched_context == SCHED_CONTEXT_UNSET)
