130 size_t count() const {

143 void link(PendingLink* l) {
144 l->next = head;

150 bool unlink(PendingLink* l, bool abandoned) {
151 for (PendingLink** pp = &head; *pp != nullptr; pp = &(*pp)->next) {
152 if (*pp == l) {
153 *pp = l->next;
154 l->next = nullptr;
155 // A goal-keyed (action) future names its reply exactly, so its
156 // removal costs no later reply (issue 2011).
157 if (abandoned && !l->by_goal && l->seq == SEQUENCE_ID_UNKNOWN && orphans < 255)
158 ++orphans;
159 return true;
160 }
161 }
162 return false;
163 }
164
166 void relink(PendingLink* from, PendingLink* to) {
167 for (PendingLink** pp = &head; *pp != nullptr; pp = &(*pp)->next) {
168 if (*pp == from) {
169 to->next = from->next;
170 *pp = to;
171 from->next = nullptr;
172 return;
173 }
174 }
175 }

179 PendingLink* resolve(int64_t seq) {
180 if (seq == SEQUENCE_ID_UNKNOWN) {
181 if (orphans > 0) {
182 --orphans;
183 return nullptr;
184 }
185 return (head != nullptr && head->next == nullptr) ? head : nullptr;
186 }
187 for (PendingLink* l = head; l != nullptr; l = l->next)
188 if (l->seq == seq) return l;
189 // The backend gave the REPLY an id but not the request: the sole
190 // pending request is the only candidate.
191 if (head != nullptr && head->next == nullptr && head->seq == SEQUENCE_ID_UNKNOWN)
192 return head;
193 return nullptr;
194 }
195
199 PendingLink* resolve_goal(const uint8_t goal[16]) {
200 for (PendingLink* l = head; l != nullptr; l = l->next) {
201 if (!l->by_goal) continue;
202 bool same = true;
203 for (int i = 0; i < 16; ++i) {
204 if (l->goal_id[i] != goal[i]) {
205 same = false;
206 break;
207 }
208 }
209 if (same) return l;
210 }
211 return nullptr;
212 }
213
216 void move_to(PendingList& to, void* storage) {
217 to.head = head;
218 to.orphans = orphans;
219 for (PendingLink* l = head; l != nullptr; l = l->next) {
220 l->list = &to;
221 l->storage = storage;
222 }
223 head = nullptr;
224 orphans = 0;
225 }

230 size_t prune() {
231 size_t n = 0;
232 while (head != nullptr) {
233 PendingLink* l = head;
234 (void)unlink(l, true);
235 l->list = nullptr;
236 l->orphaned = true;
237 ++n;
238 }
239 return n;
240 }
241
244 void orphan_all() {
245 PendingLink* l = head;
246 while (l != nullptr) {
247 PendingLink* next = l->next;
248 l->next = nullptr;
249 l->list = nullptr;
250 l->storage = nullptr;
251 l->orphaned = true;
252 l = next;
253 }
254 head = nullptr;
255 orphans = 0;
256 }

259 void deliver(PendingLink* owner, const uint8_t* data, size_t len) {
260 const size_t n = len < owner->cap ? len : owner->cap;
261 if (owner->buf != data) {
262 for (size_t i = 0; i < n; ++i)
263 owner->buf[i] = data[i];
264 }
265 owner->len = n;
266 owner->ready = true;
267 (void)unlink(owner, false);
268 }

357 Result wait(void* executor_handle, uint32_t timeout_ms, T& out, uint32_t poll_ms = 10) {
358 if (!live()) return Result(ErrorCode::Error);
359 if (poll_ms == 0) poll_ms = 1;
360 // Phase 89.2: budget by wall-clock. Accumulating `step` per iteration
361 // breaks when the underlying `zpico_spin_once` returns early on a
362 // signaled condvar (keep-alives, discovery gossip) — the 500-step
363 // default-timeout loop can then collapse into milliseconds and
364 // return ErrorCode::Timeout before the reply has a chance to land.
365 const uint64_t start_ns = nros_cpp_time_ns();
366 const uint64_t budget_ns = static_cast<uint64_t>(timeout_ms) * 1000000ULL;
367 while (true) {
368 nros_cpp_ret_t ret = nros_cpp_spin_once(executor_handle, static_cast<int32_t>(poll_ms));
369 // Transient conditions: keep polling. Anything else propagates.
370 // - Ok (0): nothing to dispatch this round.
371 // - Timeout (-2): spin_once returned after its timeout — normal.
372 // - TryAgain (-6): transport hint to retry.
373 if (ret != 0 && ret != static_cast<nros_cpp_ret_t>(ErrorCode::Timeout) &&
374 ret != static_cast<nros_cpp_ret_t>(ErrorCode::TryAgain)) {
375 return Result(ret);
376 }
377 if (is_ready()) return try_take(out);
378 const uint64_t now_ns = nros_cpp_time_ns();
379 if (now_ns - start_ns >= budget_ns) break;
380 }
381 return Result(ErrorCode::Timeout);
382 }

464 if (elapsed_ns >= budget_ns) break;

565 bool poll_keyed() {
566 ::rclcpp::detail::PendingList* list = link_.list;
567 if (list == nullptr) return false;
568 for (int i = 0; i < ::rclcpp::detail::MAX_STALE_DRAIN_PER_POLL; ++i) {
569 size_t len = 0;
570 int64_t seq = ::rclcpp::detail::SEQUENCE_ID_UNKNOWN;
571 nros_cpp_ret_t ret =
572 take_seq_fn_(link_.storage, cached_buf_, sizeof(cached_buf_), &len, &seq);
573 if (ret != 0) break; // TRY_AGAIN (nothing yet) or a transport error
574 ::rclcpp::detail::PendingLink* owner = list->resolve(seq);
575 if (owner != nullptr) list->deliver(owner, cached_buf_, len);
576 if (link_.ready) return true;
577 }
578 return link_.ready;
579 }

617 other.link_.next = nullptr;
618 other.slot_ = -1;
