75 public:
77 bool is_ready() {
78 if (slot_ < 0 || !take_fn_) return false;
79 if (ready_) return true;
80 uint8_t buf[Cap];
81 size_t len = 0;
82 nros_cpp_ret_t ret = take_fn_(client_storage_, buf, sizeof(buf), &len);
83 if (ret == 0 && len > 0) {
84 ready_ = true;
85 cached_len_ = len < sizeof(cached_buf_) ? len : sizeof(cached_buf_);
86 for (size_t i = 0; i < cached_len_; ++i)
87 cached_buf_[i] = buf[i];
88 return true;
89 }
90 return false;
91 }

97 Result try_take(T& out) {
98 if (slot_ < 0) return Result(ErrorCode::Error);
99 if (!ready_ && !is_ready()) return Result(ErrorCode::Error);
100 slot_ = -1; // consume
101 if (T::ffi_deserialize(cached_buf_, cached_len_, &out) != 0) {
102 return Result(ErrorCode::Error);
103 }
104 return Result::success();
105 }

118 Result wait(void* executor_handle, uint32_t timeout_ms, T& out, uint32_t poll_ms = 10) {
119 if (slot_ < 0) return Result(ErrorCode::Error);
120 if (poll_ms == 0) poll_ms = 1;
121 // Phase 89.2: budget by wall-clock. Accumulating `step` per iteration
122 // breaks when the underlying `zpico_spin_once` returns early on a
123 // signaled condvar (keep-alives, discovery gossip) — the 500-step
124 // default-timeout loop can then collapse into milliseconds and
125 // return ErrorCode::Timeout before the reply has a chance to land.
126 const uint64_t start_ns = nros_cpp_time_ns();
127 const uint64_t budget_ns = static_cast<uint64_t>(timeout_ms) * 1000000ULL;
128 while (true) {
129 nros_cpp_ret_t ret = nros_cpp_spin_once(executor_handle, static_cast<int32_t>(poll_ms));
130 // Transient conditions: keep polling. Anything else propagates.
131 // - Ok (0): nothing to dispatch this round.
132 // - Timeout (-2): spin_once returned after its timeout — normal.
133 // - TryAgain (-6): transport hint to retry.
134 if (ret != 0 && ret != static_cast<nros_cpp_ret_t>(ErrorCode::Timeout) &&
135 ret != static_cast<nros_cpp_ret_t>(ErrorCode::TryAgain)) {
136 return Result(ret);
137 }
138 if (is_ready()) return try_take(out);
139 const uint64_t now_ns = nros_cpp_time_ns();
140 if (now_ns - start_ns >= budget_ns) break;
141 }
142 return Result(ErrorCode::Timeout);
143 }
