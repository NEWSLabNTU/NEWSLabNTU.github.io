158 bool unlink(PendingLink* l, bool abandoned) {
159 for (PendingLink** pp = &head; *pp != nullptr; pp = &(*pp)->next) {
160 if (*pp == l) {
161 *pp = l->next;
162 l->next = nullptr;
163 // A goal-keyed (action) future names its reply exactly, so its
164 // removal costs no later reply (issue 2011).
165 if (abandoned && !l->by_goal && l->seq == SEQUENCE_ID_UNKNOWN && orphans < 255)
166 ++orphans;
167 // Issue 2027 — an abandoned goal-keyed future was the owner of
168 // one request in the Rust core's bounded table; tell the list's
169 // owner, so the slot does not outlive the last owner.
170 if (abandoned && l->by_goal && forget != nullptr) forget(forget_ctx, l->goal_id);
171 return true;
172 }
173 }
174 return false;
175 }

178 void relink(PendingLink* from, PendingLink* to) {
179 for (PendingLink** pp = &head; *pp != nullptr; pp = &(*pp)->next) {
180 if (*pp == from) {
181 to->next = from->next;
182 *pp = to;
183 from->next = nullptr;
184 return;
185 }
186 }

191 PendingLink* resolve(int64_t seq) {
192 if (seq == SEQUENCE_ID_UNKNOWN) {
193 if (orphans > 0) {
194 --orphans;
195 return nullptr;
196 }
197 return (head != nullptr && head->next == nullptr) ? head : nullptr;
198 }
199 for (PendingLink* l = head; l != nullptr; l = l->next)
200 if (l->seq == seq) return l;
201 // The backend gave the REPLY an id but not the request: the sole
202 // pending request is the only candidate.
203 if (head != nullptr && head->next == nullptr && head->seq == SEQUENCE_ID_UNKNOWN)
204 return head;
205 return nullptr;
206 }

211 PendingLink* resolve_goal(const uint8_t goal[16]) {
212 for (PendingLink* l = head; l != nullptr; l = l->next) {
213 if (!l->by_goal) continue;
214 bool same = true;
215 for (int i = 0; i < 16; ++i) {
216 if (l->goal_id[i] != goal[i]) {
217 same = false;
218 break;
219 }
220 }
221 if (same) return l;
222 }
223 return nullptr;
224 }

228 void move_to(PendingList& to, void* storage) {
229 to.head = head;
230 to.orphans = orphans;
231 for (PendingLink* l = head; l != nullptr; l = l->next) {
232 l->list = &to;
233 l->storage = storage;
234 }
235 head = nullptr;
236 orphans = 0;
237 }

242 size_t prune() {
243 size_t n = 0;
244 while (head != nullptr) {
245 PendingLink* l = head;
246 (void)unlink(l, true);
247 l->list = nullptr;
248 l->orphaned = true;
249 ++n;
250 }

259 bool remove_seq(int64_t seq) {
260 if (seq == SEQUENCE_ID_UNKNOWN) return false;
261 for (PendingLink* l = head; l != nullptr; l = l->next) {
262 if (l->by_goal || l->seq != seq) continue;
263 (void)unlink(l, true);
264 l->list = nullptr;
265 l->orphaned = true;
266 return true;
267 }
268 return false;
269 }
270
273 void orphan_all() {
274 PendingLink* l = head;
275 while (l != nullptr) {
276 PendingLink* next = l->next;
277 l->next = nullptr;
278 l->list = nullptr;
279 l->storage = nullptr;
280 l->orphaned = true;
281 l = next;
282 }
283 head = nullptr;
284 orphans = 0;
285 }

288 void deliver(PendingLink* owner, const uint8_t* data, size_t len) {
289 const size_t n = len < owner->cap ? len : owner->cap;
290 if (owner->buf != data) {
291 for (size_t i = 0; i < n; ++i)
292 owner->buf[i] = data[i];
293 }
294 owner->len = n;
295 owner->ready = true;
296 (void)unlink(owner, false);
297 }

386 Result wait(void* executor_handle, uint32_t timeout_ms, T& out, uint32_t poll_ms = 10) {
387 if (!live()) return Result(ErrorCode::Error);
388 if (poll_ms == 0) poll_ms = 1;
389 // Phase 89.2: budget by wall-clock. Accumulating `step` per iteration
390 // breaks when the underlying `zpico_spin_once` returns early on a
391 // signaled condvar (keep-alives, discovery gossip) — the 500-step
392 // default-timeout loop can then collapse into milliseconds and
393 // return ErrorCode::Timeout before the reply has a chance to land.
394 const uint64_t start_ns = nros_cpp_time_ns();
395 const uint64_t budget_ns = static_cast<uint64_t>(timeout_ms) * 1000000ULL;
396 while (true) {
397 nros_cpp_ret_t ret = nros_cpp_spin_once(executor_handle, static_cast<int32_t>(poll_ms));
398 // Transient conditions: keep polling. Anything else propagates.
399 // - Ok (0): nothing to dispatch this round.
400 // - Timeout (-2): spin_once returned after its timeout — normal.
401 // - TryAgain (-6): transport hint to retry.
402 if (ret != 0 && ret != static_cast<nros_cpp_ret_t>(ErrorCode::Timeout) &&
403 ret != static_cast<nros_cpp_ret_t>(ErrorCode::TryAgain)) {
404 return Result(ret);
405 }
406 if (is_ready()) return try_take(out);
407 const uint64_t now_ns = nros_cpp_time_ns();
408 if (now_ns - start_ns >= budget_ns) break;
409 }
410 return Result(ErrorCode::Timeout);
411 }

481 ::std::future_status wait_for(::std::chrono::duration<Rep, Period> timeout) {
482 if (!live()) return ::std::future_status::timeout;
483 if (link_.ready || is_ready()) return ::std::future_status::ready;
484 const int64_t ms = ::rclcpp::detail::chrono_wait_budget_ms(timeout);
485 if (ms <= 0 || executor_ == nullptr) {
486 if (executor_ != nullptr) (void)nros_cpp_spin_once(executor_, 0);

491 while (true) {
492 const uint64_t elapsed_ns = nros_cpp_time_ns() - start_ns;
493 if (elapsed_ns >= budget_ns) break;
494 const uint64_t left_ms = (budget_ns - elapsed_ns + 999999ULL) / 1000000ULL;
495 const int32_t slice = left_ms < 10ULL ? static_cast<int32_t>(left_ms) : 10;
496 (void)nros_cpp_spin_once(executor_, slice);
497 if (is_ready()) return ::std::future_status::ready;
498 }
499 return is_ready() ? ::std::future_status::ready : ::std::future_status::timeout;

594 bool poll_keyed() {
595 ::rclcpp::detail::PendingList* list = link_.list;
596 if (list == nullptr) return false;
597 for (int i = 0; i < ::rclcpp::detail::MAX_STALE_DRAIN_PER_POLL; ++i) {
598 size_t len = 0;
599 int64_t seq = ::rclcpp::detail::SEQUENCE_ID_UNKNOWN;
600 nros_cpp_ret_t ret =
601 take_seq_fn_(link_.storage, cached_buf_, sizeof(cached_buf_), &len, &seq);
602 if (ret != 0) break; // TRY_AGAIN (nothing yet) or a transport error
603 ::rclcpp::detail::PendingLink* owner = list->resolve(seq);
604 if (owner != nullptr) list->deliver(owner, cached_buf_, len);
605 if (link_.ready) return true;
606 }
607 return link_.ready;
608 }
