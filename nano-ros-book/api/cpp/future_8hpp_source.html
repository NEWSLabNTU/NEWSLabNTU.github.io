71 public:
73 bool is_ready() {
74 if (slot_ < 0 || !take_fn_) return false;
75 if (ready_) return true;
76 uint8_t buf[Cap];
77 size_t len = 0;
78 nros_cpp_ret_t ret = take_fn_(client_storage_, buf, sizeof(buf), &len);
79 if (ret == 0 && len > 0) {
80 ready_ = true;
81 cached_len_ = len < sizeof(cached_buf_) ? len : sizeof(cached_buf_);
82 for (size_t i = 0; i < cached_len_; ++i)
83 cached_buf_[i] = buf[i];
84 return true;
85 }
86 return false;

93 Result try_take(T& out) {
94 if (slot_ < 0) return Result(ErrorCode::Error);
95 if (!ready_ && !is_ready()) return Result(ErrorCode::Error);
96 slot_ = -1; // consume

114 Result wait(void* executor_handle, uint32_t timeout_ms, T& out, uint32_t poll_ms = 10) {
115 if (slot_ < 0) return Result(ErrorCode::Error);
116 if (poll_ms == 0) poll_ms = 1;
117 // Phase 89.2: budget by wall-clock. Accumulating `step` per iteration
118 // breaks when the underlying `zpico_spin_once` returns early on a
119 // signaled condvar (keep-alives, discovery gossip) — the 500-step
120 // default-timeout loop can then collapse into milliseconds and
121 // return ErrorCode::Timeout before the reply has a chance to land.
122 const uint64_t start_ns = nros_cpp_time_ns();
123 const uint64_t budget_ns = static_cast<uint64_t>(timeout_ms) * 1000000ULL;
124 while (true) {
125 nros_cpp_ret_t ret = nros_cpp_spin_once(executor_handle, static_cast<int32_t>(poll_ms));
126 // Transient conditions: keep polling. Anything else propagates.
127 // - Ok (0): nothing to dispatch this round.
128 // - Timeout (-2): spin_once returned after its timeout — normal.
129 // - TryAgain (-6): transport hint to retry.
130 if (ret != 0 && ret != static_cast<nros_cpp_ret_t>(ErrorCode::Timeout) &&
131 ret != static_cast<nros_cpp_ret_t>(ErrorCode::TryAgain)) {
132 return Result(ret);
133 }
134 if (is_ready()) return try_take(out);
135 const uint64_t now_ns = nros_cpp_time_ns();
136 if (now_ns - start_ns >= budget_ns) break;
137 }
138 return Result(ErrorCode::Timeout);
139 }
