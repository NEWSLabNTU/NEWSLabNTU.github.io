136 bool unlink(PendingLink* l, bool abandoned) {
137 for (PendingLink** pp = &head; *pp != nullptr; pp = &(*pp)->next) {
138 if (*pp == l) {
139 *pp = l->next;
140 l->next = nullptr;
141 if (abandoned && l->seq == SEQUENCE_ID_UNKNOWN && orphans < 255) ++orphans;
142 return true;
143 }
144 }
145 return false;
146 }
147
149 void relink(PendingLink* from, PendingLink* to) {
150 for (PendingLink** pp = &head; *pp != nullptr; pp = &(*pp)->next) {
151 if (*pp == from) {
152 to->next = from->next;
153 *pp = to;
154 from->next = nullptr;
155 return;
156 }
157 }
158 }

162 PendingLink* resolve(int64_t seq) {
163 if (seq == SEQUENCE_ID_UNKNOWN) {
164 if (orphans > 0) {
165 --orphans;
166 return nullptr;
167 }
168 return (head != nullptr && head->next == nullptr) ? head : nullptr;
169 }
170 for (PendingLink* l = head; l != nullptr; l = l->next)
171 if (l->seq == seq) return l;
172 // The backend gave the REPLY an id but not the request: the sole
173 // pending request is the only candidate.
174 if (head != nullptr && head->next == nullptr && head->seq == SEQUENCE_ID_UNKNOWN)
175 return head;
176 return nullptr;
177 }
178
181 void move_to(PendingList& to, void* storage) {
182 to.head = head;
183 to.orphans = orphans;
184 for (PendingLink* l = head; l != nullptr; l = l->next) {
185 l->list = &to;
186 l->storage = storage;
187 }
188 head = nullptr;
189 orphans = 0;
190 }
191
195 size_t prune() {
196 size_t n = 0;
197 while (head != nullptr) {
198 PendingLink* l = head;
199 (void)unlink(l, true);
200 l->list = nullptr;
201 l->orphaned = true;
202 ++n;
203 }
204 return n;
205 }
206
209 void orphan_all() {
210 PendingLink* l = head;
211 while (l != nullptr) {
212 PendingLink* next = l->next;
213 l->next = nullptr;
214 l->list = nullptr;
215 l->storage = nullptr;
216 l->orphaned = true;
217 l = next;
218 }
219 head = nullptr;
220 orphans = 0;
221 }

224 void deliver(PendingLink* owner, const uint8_t* data, size_t len) {
225 const size_t n = len < owner->cap ? len : owner->cap;
226 if (owner->buf != data) {
227 for (size_t i = 0; i < n; ++i)
228 owner->buf[i] = data[i];
229 }
230 owner->len = n;
231 owner->ready = true;
232 (void)unlink(owner, false);
233 }

314 Result wait(void* executor_handle, uint32_t timeout_ms, T& out, uint32_t poll_ms = 10) {
315 if (!live()) return Result(ErrorCode::Error);
316 if (poll_ms == 0) poll_ms = 1;
317 // Phase 89.2: budget by wall-clock. Accumulating `step` per iteration
318 // breaks when the underlying `zpico_spin_once` returns early on a
319 // signaled condvar (keep-alives, discovery gossip) — the 500-step
320 // default-timeout loop can then collapse into milliseconds and
321 // return ErrorCode::Timeout before the reply has a chance to land.
322 const uint64_t start_ns = nros_cpp_time_ns();
323 const uint64_t budget_ns = static_cast<uint64_t>(timeout_ms) * 1000000ULL;
324 while (true) {
325 nros_cpp_ret_t ret = nros_cpp_spin_once(executor_handle, static_cast<int32_t>(poll_ms));
326 // Transient conditions: keep polling. Anything else propagates.
327 // - Ok (0): nothing to dispatch this round.
328 // - Timeout (-2): spin_once returned after its timeout — normal.
329 // - TryAgain (-6): transport hint to retry.
330 if (ret != 0 && ret != static_cast<nros_cpp_ret_t>(ErrorCode::Timeout) &&
331 ret != static_cast<nros_cpp_ret_t>(ErrorCode::TryAgain)) {
332 return Result(ret);
333 }
334 if (is_ready()) return try_take(out);
335 const uint64_t now_ns = nros_cpp_time_ns();
336 if (now_ns - start_ns >= budget_ns) break;
337 }
338 return Result(ErrorCode::Timeout);
339 }

502 bool poll_keyed() {
503 ::rclcpp::detail::PendingList* list = link_.list;
504 if (list == nullptr) return false;
505 for (int i = 0; i < ::rclcpp::detail::MAX_STALE_DRAIN_PER_POLL; ++i) {
506 size_t len = 0;
507 int64_t seq = ::rclcpp::detail::SEQUENCE_ID_UNKNOWN;
508 nros_cpp_ret_t ret =
509 take_seq_fn_(link_.storage, cached_buf_, sizeof(cached_buf_), &len, &seq);
510 if (ret != 0) break; // TRY_AGAIN (nothing yet) or a transport error
511 ::rclcpp::detail::PendingLink* owner = list->resolve(seq);
512 if (owner != nullptr) list->deliver(owner, cached_buf_, len);
513 if (link_.ready) return true;
514 }
515 return link_.ready;
516 }
