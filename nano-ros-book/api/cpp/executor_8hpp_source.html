231 Result spin_once(int32_t timeout_ms) {
232 // FIRST, before the `initialized_` check: the defect is in the VALUE,
233 // so it is knowable without a session and a caller must not have to
234 // stand one up to be told (RFC-0089 §"Where the refusal fires"). It
235 // used to be `timeout_ms.max(0)` Rust-side, which turned upstream's
236 // "block forever" into a 0 ms poll with no diagnostic at all.
237 //
238 // `NROS_RCLCPP_SAY_REFUSED`, not the legacy `NROS_ERROR`: the legacy
239 // sink is a no-op on every freestanding target, and a refusal nobody
240 // can read on the targets nano-ros exists for is issue 1019 again. It
241 // is also not the `"%s"` printf path, whose 256-byte buffer would cut
242 // the message off before the alternative.
243 if (timeout_ms < 0) {
244 NROS_RCLCPP_SAY_REFUSED(NROS_RCLCPP_REFUSE_UNBOUNDED_SPIN_RUNTIME);
245 return Result(ErrorCode::Unsupported);
246 }
247 if (!initialized_) return Result(ErrorCode::NotInitialized);
248 return Result(nros_cpp_spin_once(storage_, timeout_ms));
249 }
