228 Result spin_once(int32_t timeout_ms) {
229 // FIRST, before the `initialized_` check: the defect is in the VALUE,
230 // so it is knowable without a session and a caller must not have to
231 // stand one up to be told (RFC-0089 §"Where the refusal fires"). It
232 // used to be `timeout_ms.max(0)` Rust-side, which turned upstream's
233 // "block forever" into a 0 ms poll with no diagnostic at all.
234 //
235 // `NROS_RCLCPP_SAY_REFUSED`, not the legacy `NROS_ERROR`: the legacy
236 // sink is a no-op on every freestanding target, and a refusal nobody
237 // can read on the targets nano-ros exists for is issue 1019 again. It
238 // is also not the `"%s"` printf path, whose 256-byte buffer would cut
239 // the message off before the alternative.
240 if (timeout_ms < 0) {
241 NROS_RCLCPP_SAY_REFUSED(NROS_RCLCPP_REFUSE_UNBOUNDED_SPIN_RUNTIME);
242 return Result(ErrorCode::Unsupported);
243 }
244 if (!initialized_) return Result(ErrorCode::NotInitialized);
245 return Result(nros_cpp_spin_once(storage_, timeout_ms));
246 }
