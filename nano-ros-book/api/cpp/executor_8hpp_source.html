230 Result spin_once(int32_t timeout_ms) {
231 // FIRST, before the `initialized_` check: the defect is in the VALUE,
232 // so it is knowable without a session and a caller must not have to
233 // stand one up to be told (RFC-0089 §"Where the refusal fires"). It
234 // used to be `timeout_ms.max(0)` Rust-side, which turned upstream's
235 // "block forever" into a 0 ms poll with no diagnostic at all.
236 //
237 // `NROS_RCLCPP_SAY_REFUSED`, not the legacy `NROS_ERROR`: the legacy
238 // sink is a no-op on every freestanding target, and a refusal nobody
239 // can read on the targets nano-ros exists for is issue 1019 again. It
240 // is also not the `"%s"` printf path, whose 256-byte buffer would cut
241 // the message off before the alternative.
242 if (timeout_ms < 0) {
243 NROS_RCLCPP_SAY_REFUSED(NROS_RCLCPP_REFUSE_UNBOUNDED_SPIN_RUNTIME);
244 return Result(ErrorCode::Unsupported);
245 }
246 if (!initialized_) return Result(ErrorCode::NotInitialized);
247 return Result(nros_cpp_spin_once(storage_, timeout_ms));
248 }
