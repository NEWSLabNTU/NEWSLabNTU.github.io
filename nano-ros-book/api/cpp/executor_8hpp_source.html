323 Result spin(int32_t poll_ms = 10) {
324 if (!initialized_) return Result(ErrorCode::NotInitialized);
325 Result last = Result::success();
326 while (initialized_) {
327 last = Result(nros_cpp_spin_once(storage_, poll_ms));
328 if (!last.ok()) return last;
329 }
330 return last;
331 }

341 Result spin_for(uint32_t duration_ms, int32_t poll_ms = 10) {
342 if (!initialized_) return Result(ErrorCode::NotInitialized);
343 // Issue 0329 — the wall-clock budgeted loop (Phase 118.C: iteration-count
344 // budgeting collapses when `spin_once` returns early on a signaled
345 // condvar) now lives once, Rust-side, in `nros_cpp_spin_for`; forward to
346 // it so this and `nros::spin()` share one implementation.
347 return Result(nros_cpp_spin_for(storage_, duration_ms, poll_ms));
348 }
