175static inline int8_t nros_platform_can_frame_check(const nros_platform_can_frame_t *frame, uint8_t fd_active) {
176 uint8_t flags;
177 uint8_t len;
178 if (frame == NULL) {
179 return NROS_PLATFORM_CAN_EINVAL;
180 }
181 flags = frame->flags;
182 len = frame->len;
183 if ((flags & (uint8_t) ~NROS_PLATFORM_CAN_FLAGS_KNOWN) != 0u) {
184 return NROS_PLATFORM_CAN_EINVAL;
185 }
186 if (frame->id > ((flags & NROS_PLATFORM_CAN_FLAG_EXT) ? NROS_PLATFORM_CAN_EXT_ID_MAX
187 : NROS_PLATFORM_CAN_STD_ID_MAX)) {
188 return NROS_PLATFORM_CAN_EINVAL;
189 }
190 if ((flags & NROS_PLATFORM_CAN_FLAG_FD) == 0u) {
191 if ((flags & NROS_PLATFORM_CAN_FLAG_BRS) != 0u || len > NROS_PLATFORM_CAN_CLASSIC_MAX_LEN) {
192 return NROS_PLATFORM_CAN_EINVAL;
193 }
194 return NROS_PLATFORM_CAN_OK;
195 }
196 if (fd_active == 0u) {
197 return NROS_PLATFORM_CAN_EINVAL;
198 }
199 if (len <= 8u || len == 12u || len == 16u || len == 20u || len == 24u || len == 32u ||
200 len == 48u || len == 64u) {
201 return NROS_PLATFORM_CAN_OK;
202 }
203 return NROS_PLATFORM_CAN_EINVAL;
204}
