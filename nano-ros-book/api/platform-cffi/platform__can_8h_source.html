183static inline int8_t nros_platform_can_frame_check(const nros_platform_can_frame_t *frame, uint8_t fd_active) {
184 uint8_t flags;
185 uint8_t len;
186 if (frame == NULL) {
187 return NROS_PLATFORM_CAN_EINVAL;
188 }
189 flags = frame->flags;
190 len = frame->len;
191 if ((flags & (uint8_t) ~NROS_PLATFORM_CAN_FLAGS_KNOWN) != 0u) {
192 return NROS_PLATFORM_CAN_EINVAL;
193 }
194 if (frame->id > ((flags & NROS_PLATFORM_CAN_FLAG_EXT) ? NROS_PLATFORM_CAN_EXT_ID_MAX
195 : NROS_PLATFORM_CAN_STD_ID_MAX)) {
196 return NROS_PLATFORM_CAN_EINVAL;
197 }
198 if ((flags & NROS_PLATFORM_CAN_FLAG_FD) == 0u) {
199 if ((flags & NROS_PLATFORM_CAN_FLAG_BRS) != 0u || len > NROS_PLATFORM_CAN_CLASSIC_MAX_LEN) {
200 return NROS_PLATFORM_CAN_EINVAL;
201 }
202 return NROS_PLATFORM_CAN_OK;
203 }
204 if (fd_active == 0u) {
205 return NROS_PLATFORM_CAN_ENOFD;
206 }
207 if (len <= 8u || len == 12u || len == 16u || len == 20u || len == 24u || len == 32u ||
208 len == 48u || len == 64u) {
209 return NROS_PLATFORM_CAN_OK;
210 }
211 return NROS_PLATFORM_CAN_EINVAL;
212}
