170/* ---- Socket accessor (Phase 154) ---- *
171 *
172 * Every `nros_platform_*` backend stores its socket struct with
173 * `int fd` (or equivalent) at offset 0:
174 *
175 * - POSIX: `typedef struct { int fd; } nros_posix_socket_t;`
176 * - ThreadX: `typedef struct { INT fd; } nros_threadx_socket_t;`
177 * - FreeRTOS+lwIP same shape.
178 *
179 * This accessor lets callers (e.g. zpico-sys's `get_session_fd`
180 * helper used by the read-task wakeup path on ThreadX-Linux) read
181 * the fd without depending on the per-RTOS struct typedef being
182 * visible at compile time. Static-inline so it has zero call
183 * overhead and ships with the header — no extra TU to link.
184 *
185 * Returns -1 if `sock` is NULL. Otherwise returns the leading
186 * `int` (which platforms treat as -1 for an unbound socket).
187 */
188static inline int nros_platform_socket_get_fd(const void *sock) {
189 if (sock == NULL) { return -1; }
190 /* SAFETY: every backend's socket struct is `{ int fd; ... }`,
191 * so the first sizeof(int) bytes of the opaque storage are
192 * the BSD-style fd. Cast-to-int read is the canonical way to
193 * extract it. */
194 return *(const int *)sock;
195}
