87typedef enum rcl_log_severity_t {
88 RCLC_LOG_SEVERITY_UNSET = 0,
89 RCLC_LOG_SEVERITY_TRACE = 5,
90 RCLC_LOG_SEVERITY_DEBUG = 10,
91 RCLC_LOG_SEVERITY_INFO = 20,
92 RCLC_LOG_SEVERITY_WARN = 30,
93 RCLC_LOG_SEVERITY_ERROR = 40,
94 RCLC_LOG_SEVERITY_FATAL = 50,
95 /* NOT a level — a WIDTH PIN (issue 1330). Every ARM EABI compiler defaults
96 * to `-fshort-enums` (it is the AAPCS default, not a flag this tree picks),
97 * so an enum whose largest value is 50 is packed to ONE byte there. That
98 * contradicts both the Rust mirror (`repr(transparent)` over `c_int`) and
99 * this enum's own promise above that any `int` a C caller can produce is
100 * representable — and it turned the `_Static_assert` below into a hard
101 * build error for every cross C/C++ image the day the guard landed.
102 * `-fno-short-enums` is not the fix: it would have to reach every consumer
103 * in and out of tree, and a TU that forgot would disagree about the width
104 * of a type in a shared header, silently. The pin belongs in the header.
105 *
106 * Trailing underscore: reserved, never emitted, never switched on.
107 * `to_facade` is TOTAL, so a caller that somehow passed it resolves to
108 * FATAL like any other out-of-range `int` rather than being rejected. */
109 RCLC_LOG_SEVERITY_FORCE_INT_WIDTH_ = 0x7fffffff,
110} rcl_log_severity_t;

112/* ── Width guard ──────────────────────────────────────────────────────────────
113 * This `enum` is HAND-MIRRORED in Rust (`packages/api/nros-c/src/log.rs`), which
114 * is the class `check-ffi-struct-mirrors` polices one layer over. The mirror is
115 * `#[repr(transparent)]` over `c_int`; it was `#[repr(u8)]` until phase-417
116 * stage 3, so one side passed a byte where the other passed four and the values
117 * survived only by how the ABIs in play happen to pass small integers. Nothing
118 * asserted the width anywhere in `include/` or `src/`. It does now. */

464/* ---- Throttled convenience macros ---- */
465/* One `static uint64_t` per call site, which is what makes the window
466 * per-SITE and not per-logger — the same shape as `rclcpp`'s
467 * `RCLCPP_INFO_THROTTLE`. Statements, not expressions (they declare storage),
468 * so they need their own `;` and cannot be used inside a comma expression.
469 *
470 * Severity is tested BEFORE the window: a record the logger's level already
471 * filters must not consume the window, or a raised-then-lowered log level
472 * loses its first record for a whole interval. */
473
474#define RCLC_LOG_THROTTLE_AT(logger, severity, interval_ms, ...) \
475 do { \
476 static uint64_t _nros_log_throttle_last = 0; \
477 if (rclc_logger_is_enabled((logger), (severity)) && \
478 rclc_log_throttle_admit(&_nros_log_throttle_last, (uint64_t)(interval_ms))) { \
479 rclc_log_emit_fmt_at((logger), (severity), __FILE__, (uint32_t)__LINE__, __VA_ARGS__); \
480 } \
481 } while (0)
482
483#define RCLC_LOG_TRACE_THROTTLE(logger, interval_ms, ...) \
484 RCLC_LOG_THROTTLE_AT((logger), RCLC_LOG_SEVERITY_TRACE, (interval_ms), __VA_ARGS__)
485#define RCLC_LOG_DEBUG_THROTTLE(logger, interval_ms, ...) \
486 RCLC_LOG_THROTTLE_AT((logger), RCLC_LOG_SEVERITY_DEBUG, (interval_ms), __VA_ARGS__)
487#define RCLC_LOG_INFO_THROTTLE(logger, interval_ms, ...) \
488 RCLC_LOG_THROTTLE_AT((logger), RCLC_LOG_SEVERITY_INFO, (interval_ms), __VA_ARGS__)
489#define RCLC_LOG_WARN_THROTTLE(logger, interval_ms, ...) \
490 RCLC_LOG_THROTTLE_AT((logger), RCLC_LOG_SEVERITY_WARN, (interval_ms), __VA_ARGS__)
491#define RCLC_LOG_ERROR_THROTTLE(logger, interval_ms, ...) \
492 RCLC_LOG_THROTTLE_AT((logger), RCLC_LOG_SEVERITY_ERROR, (interval_ms), __VA_ARGS__)
493#define RCLC_LOG_FATAL_THROTTLE(logger, interval_ms, ...) \
494 RCLC_LOG_THROTTLE_AT((logger), RCLC_LOG_SEVERITY_FATAL, (interval_ms), __VA_ARGS__)
