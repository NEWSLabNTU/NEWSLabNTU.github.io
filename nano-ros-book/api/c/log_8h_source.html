87typedef enum rcl_log_severity_t {
88 RCLC_LOG_SEVERITY_UNSET = 0,
89 RCLC_LOG_SEVERITY_TRACE = 5,
90 RCLC_LOG_SEVERITY_DEBUG = 10,
91 RCLC_LOG_SEVERITY_INFO = 20,
92 RCLC_LOG_SEVERITY_WARN = 30,
93 RCLC_LOG_SEVERITY_ERROR = 40,
94 RCLC_LOG_SEVERITY_FATAL = 50,
95 /* NOT a level — a WIDTH PIN. Every ARM EABI compiler defaults
96 * to `-fshort-enums` (it is the AAPCS default, not a flag this tree picks),
97 * so an enum whose largest value is 50 is packed to ONE byte there. That
98 * contradicts both the Rust mirror (`repr(transparent)` over `c_int`) and
99 * this enum's own promise above that any `int` a C caller can produce is
100 * representable — and it turned the `_Static_assert` below into a hard
101 * build error for every cross C/C++ image the day the guard landed.
102 * `-fno-short-enums` is not the fix: it would have to reach every consumer
103 * in and out of tree, and a TU that forgot would disagree about the width
104 * of a type in a shared header, silently. The pin belongs in the header.
105 *
106 * Trailing underscore: reserved, never emitted, never switched on.
107 * `to_facade` is TOTAL, so a caller that somehow passed it resolves to
108 * FATAL like any other out-of-range `int` rather than being rejected. */
109 RCLC_LOG_SEVERITY_FORCE_INT_WIDTH_ = 0x7fffffff,
110} rcl_log_severity_t;

112/* ── Width guard ──────────────────────────────────────────────────────────────
113 * This `enum` is HAND-MIRRORED in Rust (`packages/api/nros-c/src/log.rs`), which
114 * is the class `check-ffi-struct-mirrors` polices one layer over. The mirror is
115 * `#[repr(transparent)]` over `c_int`; it used to be `#[repr(u8)]`,
116 * so one side passed a byte where the other passed four and the values
117 * survived only by how the ABIs in play happen to pass small integers. Nothing
118 * asserted the width anywhere in `include/` or `src/`. It does now. */
119#if defined(__cplusplus) && __cplusplus >= 201103L
120static_assert(sizeof(rcl_log_severity_t) == sizeof(int),
121 "rcl_log_severity_t must be int-sized: the Rust mirror is repr(transparent) "
122 "over c_int");
123static_assert(RCLC_LOG_SEVERITY_INFO == 20, "rcl_log_severity_t must stay on rcutils's numbering");
124static_assert(RCLC_LOG_SEVERITY_FATAL == 50, "rcl_log_severity_t must stay on rcutils's numbering");
125#elif defined(__STDC_VERSION__) && __STDC_VERSION__ >= 201112L

207/* ---- Convenience macros ---- */
208/* The macros stage the printf args into a stack buffer inside
209 * `rclc_log_emit_fmt_at`, which attaches `__FILE__` / `__LINE__`.
210 * Filtering happens on the Rust side: the image's compile-time CEILING
211 * (`NROS_LOG_MAX_LEVEL`, `trace` unless stated) and then the
212 * per-logger threshold (`rclc_logger_set_level`), both inside the same check
213 * `rclc_logger_is_enabled` answers. The call site still evaluates its
214 * arguments; to keep a level's arguments out of the image, set the
215 * compile-time floor `RCLC_LOG_MIN_SEVERITY` below, or guard the call with
216 * `rclc_logger_is_enabled`. */
217
218/* ---- Compile-time floor ----
219 *
220 * `RCLC_LOG_MIN_SEVERITY` is the C twin of upstream's `RCUTILS_LOG_MIN_SEVERITY`
221 * (and of `RCLCPP_LOG_MIN_SEVERITY`, `<rclcpp/logging.hpp>`): define it, with
222 * `-D` or before the first include, to one of the seven values below and every
223 * level macro BELOW it -- `RCLC_LOG_<LEVEL>` and `RCLC_LOG_<LEVEL>_THROTTLE` --
224 * expands to nothing. Its arguments, the logger included, are then not
225 * evaluated at all, which is the difference from the run-time filters (the
226 * `log.max_level` ceiling and `rclc_logger_set_level`): those refuse a record
227 * after the call site has computed every argument.
228 *
229 * The numbers are `log.max_level`'s (`NROS_LOG_MAX_LEVEL`, 0 trace .. 6 off),
230 * so a Zephyr build can pass the image's ceiling straight through:
231 * `-DRCLC_LOG_MIN_SEVERITY=CONFIG_NROS_LOG_MAX_LEVEL`. Unset, the floor is
232 * TRACE: nothing is compiled out. `RCLC_LOG_THROTTLE_AT` takes its severity
233 * at run time and is not gated. */
234#define RCLC_LOG_MIN_SEVERITY_TRACE 0
235#define RCLC_LOG_MIN_SEVERITY_DEBUG 1
236#define RCLC_LOG_MIN_SEVERITY_INFO 2
237#define RCLC_LOG_MIN_SEVERITY_WARN 3
238#define RCLC_LOG_MIN_SEVERITY_ERROR 4
239#define RCLC_LOG_MIN_SEVERITY_FATAL 5
240#define RCLC_LOG_MIN_SEVERITY_NONE 6
241#ifndef RCLC_LOG_MIN_SEVERITY
242#define RCLC_LOG_MIN_SEVERITY RCLC_LOG_MIN_SEVERITY_TRACE
243#endif
244#if RCLC_LOG_MIN_SEVERITY < RCLC_LOG_MIN_SEVERITY_TRACE || \
245 RCLC_LOG_MIN_SEVERITY > RCLC_LOG_MIN_SEVERITY_NONE
246#error \
247 "RCLC_LOG_MIN_SEVERITY must be one of RCLC_LOG_MIN_SEVERITY_{TRACE,DEBUG,INFO,WARN,ERROR,FATAL,NONE} (0..6)"
248#endif

537/* ---- Throttled convenience macros ---- */
538/* One `static uint64_t` per call site, which is what makes the window
539 * per-SITE and not per-logger — the same shape as `rclcpp`'s
540 * `RCLCPP_INFO_THROTTLE`. Statements, not expressions (they declare storage),
541 * so they need their own `;` and cannot be used inside a comma expression.
542 *
543 * Severity is tested BEFORE the window: a record the logger's level already
544 * filters must not consume the window, or a raised-then-lowered log level
545 * loses its first record for a whole interval. */
546
547#define RCLC_LOG_THROTTLE_AT(logger, severity, interval_ms, ...) \
548 do { \
549 static uint64_t _nros_log_throttle_last = 0; \
550 if (rclc_logger_is_enabled((logger), (severity)) && \
551 rclc_log_throttle_admit(&_nros_log_throttle_last, (uint64_t)(interval_ms))) { \
552 rclc_log_emit_fmt_at((logger), (severity), __FILE__, (uint32_t)__LINE__, __VA_ARGS__); \
553 } \
554 } while (0)
