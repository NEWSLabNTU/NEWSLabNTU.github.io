1//! Phase 88 — portable leveled-logging facade for nano-ros.
2//!
3//! See [`docs/roadmap/archived/phase-88-nros-log.md`](../../../docs/roadmap/archived/phase-88-nros-log.md)
4//! for the design and acceptance criteria.
5//!
6//! ## Layering
7//!
8//! - This crate carries only the portable types + dispatcher +
9//!   macros + `PlatformSink`. No backend code.
10//! - Per-platform log delivery is the responsibility of each
11//!   `nros-platform-<rtos>` crate, exposing
12//!   `nros_platform_log_write` / `nros_platform_log_flush` via the
13//!   `nros_platform_*` ABI (header at
14//!   `packages/platform/nros-platform-api/include/nros/platform.h`).
15//! - `PlatformSink` is the bridge: a single `LogSink` impl that
16//!   forwards to the ABI. Apps that want fan-out (e.g.
17//!   `Platform + /rosout`) compose a `&'static [&dyn LogSink]`
18//!   manually and pass it to [`init`].
19//!
20//! ## Quick start
21//!
22//! ```ignore
23//! use nros_log::{Logger, Severity};
24//! use nros_log::{log_info, log_warn};
25//!
26//! static LOGGER: Logger = Logger::new("my_node");
27//!
28//! fn main() {
29//!     nros_log::register_logger(&LOGGER);
30//!     nros_log::init(nros_log::sinks::default());
31//!     log_info!(&LOGGER, "started; domain = {}", 42);
32//! }
33//! ```

74/// REP-2012 severity levels, mirroring `rcutils_log_severity_t`.
75///
76/// The integer representation is stable and part of the ABI for
77/// `nros_platform_log_write`. Lower value = more verbose.
78#[repr(u8)]
79#[derive(Copy, Clone, Debug, Eq, PartialEq, Ord, PartialOrd, Hash)]
80pub enum Severity {
81    /// Per-instruction granularity. Off unless `max-level-trace` is
82    /// the active ceiling.
83    Trace = 0,
84    /// Diagnostic information useful while developing.
85    Debug = 1,
86    /// Normal operation events worth surfacing once.
87    Info = 2,
88    /// Unexpected but recoverable conditions.
89    Warn = 3,
90    /// Errors the caller should surface; the system continues.
91    Error = 4,
92    /// Unrecoverable — the system is about to abort.
93    Fatal = 5,
94}
95
96impl Severity {
97    /// Short uppercase label suitable for log-line rendering.
98    #[must_use]
99    pub const fn as_str(self) -> &'static str {
100        match self {
101            Self::Trace => "TRACE",
102            Self::Debug => "DEBUG",
103            Self::Info => "INFO",
104            Self::Warn => "WARN",
105            Self::Error => "ERROR",
106            Self::Fatal => "FATAL",
107        }
108    }
109
110    /// Stable `u8` discriminant for cross-ABI use.
111    #[must_use]
112    pub const fn as_u8(self) -> u8 {
113        self as u8
114    }
115}

117/// Reconstruct a [`Severity`] from its `u8` discriminant.
118///
119/// Returns `None` for `> 5`.
120#[must_use]
121pub const fn severity_from_u8(value: u8) -> Option<Severity> {
122    match value {
123        0 => Some(Severity::Trace),
124        1 => Some(Severity::Debug),
125        2 => Some(Severity::Info),
126        3 => Some(Severity::Warn),
127        4 => Some(Severity::Error),
128        5 => Some(Severity::Fatal),
129        _ => None,
130    }
131}

133/// Compile-time ceiling check used by the `nros_*!` macros.
134///
135/// Returns `true` iff `severity` is allowed under the configured
136/// `max-level-*` feature.
137#[must_use]
138pub const fn severity_enabled_at_compile_time(severity: Severity) -> bool {
139    if cfg!(feature = "max-level-off") {
140        return false;
141    }
142    let ceiling = compile_time_ceiling();
143    (severity as u8) >= (ceiling as u8)
144}
145
146const fn compile_time_ceiling() -> Severity {
147    if cfg!(feature = "max-level-trace") {
148        Severity::Trace
149    } else if cfg!(feature = "max-level-debug") {
150        Severity::Debug
151    } else if cfg!(feature = "max-level-info") {
152        Severity::Info
153    } else if cfg!(feature = "max-level-warn") {
154        Severity::Warn
155    } else if cfg!(feature = "max-level-error") {
156        Severity::Error
157    } else {
158        // No ceiling feature = treat as `max-level-trace`.
159        Severity::Trace
160    }
161}

163/// One log entry, handed to each [`LogSink`].
164///
165/// `message` is already formatted — sinks must NOT re-format.
166#[derive(Debug)]
167pub struct Record<'a> {
168    /// Severity of the record.
169    pub severity: Severity,
170    /// Name of the originating [`Logger`].
171    pub logger_name: &'a str,
172    /// Formatted message text (no trailing newline).
173    pub message: &'a str,
174    /// File the macro invocation came from (`core::file!()`).
175    pub file: &'static str,
176    /// Line within `file` (`core::line!()`).
177    pub line: u32,
178    /// Monotonic timestamp in nanoseconds. `0` if unavailable.
179    pub timestamp_ns: u64,
180}

182/// Backend a log record is delivered to.
183///
184/// Implementations must be `Sync` so the dispatcher can hold them
185/// in `&'static [&dyn LogSink]`. ISR-safety is per-impl — see the
186/// table in `docs/roadmap/archived/phase-88-nros-log.md`.
187pub trait LogSink: Sync {
188    /// Render `record`. Called only when the record's severity passes
189    /// both the compile-time ceiling AND the [`Logger`]'s runtime
190    /// threshold.
191    fn log(&self, record: &Record<'_>);
192
193    /// Optional flush hook (default no-op).
194    fn flush(&self) {}
195}

197/// The stored byte meaning "this logger has NO level of its own", so its
198/// threshold is whatever [`Logger::default_level`] currently says.
199///
200/// phase-467 (ledger rows `c:log_severity_t` + `rust:Logger::set_default_level`).
201/// It cannot be `0`: that is `Severity::Trace`, the most verbose level, which
202/// is why storing rcutils's `UNSET` in a level byte used to select TRACE here
203/// where upstream restores a default. `u8::MAX` is outside the enum in the
204/// other direction, and being ABOVE `Fatal` is load-bearing in
205/// [`Logger::is_enabled`] — see the note there.
206const LEVEL_UNSET: u8 = u8::MAX;
207
208/// The threshold a logger with no level of its own resolves to.
209///
210/// ONE process-wide byte, which is the whole of our level hierarchy. rcutils
211/// resolves an unset logger by walking a DOTTED ANCESTRY (`x.y.z` → `x.y` →
212/// `x`) to `g_rcutils_logging_default_logger_level`; `nros_log` has no
213/// ancestry — [`get_logger`] is exact string equality over [`MAX_LOGGERS`]
214/// slots — so that walk degenerates here to exactly one step, and one step is
215/// what this is.
216///
217/// Deliberately not built further, and phase-467 recorded that as PERMANENT on
218/// ledger row `c:log_severity_t` rather than as work. A dotted name is accepted
219/// and is just a name with a dot in it; what is missing is the PARENT, and the
220/// two things that would give it one are both absent for the same reason.
221/// Upstream keys a level by NAME (`rcutils_logging_set_logger_level(const char
222/// *, int)` writes an allocator-backed severity map), so an ancestor there can
223/// carry a level without being a logger; here the level is an `AtomicU8` INSIDE
224/// the [`Logger`], so an ancestor would have to occupy one of the 32 slots to
225/// hold one. And the producer of dotted names upstream, `Logger::get_child`,
226/// has no allocator here to concatenate `parent.child` with. A resolver for a
227/// hierarchy nothing can construct is not a missing feature.
228static DEFAULT_LEVEL: AtomicU8 = AtomicU8::new(Severity::Info as u8);
229
230/// A named logger with a runtime severity threshold.
231///
232/// A logger constructed with [`Logger::new`] has NO level of its own and
233/// follows [`Logger::default_level`] (itself [`Severity::Info`] until someone
234/// calls [`Logger::set_default_level`]). [`Logger::with_level`] and
235/// [`Logger::set_level`] give it one; [`Logger::unset_level`] takes it away
236/// again. Use [`register_logger`] to publish a `'static Logger` so multiple
237/// call sites with the same name share the same threshold.
238pub struct Logger {
239    name: &'static str,
240    /// Either a [`Severity`] discriminant or the private `LEVEL_UNSET`.
241    level: AtomicU8,
242}
243
244impl Logger {
245    /// `const`-construct with NO level of its own — the logger follows
246    /// [`Logger::default_level`], which is [`Severity::Info`] until something
247    /// moves it.
248    ///
249    /// phase-467: this stored `Severity::Info` outright until the process
250    /// default existed, and "the author wrote `new`" then looked exactly like
251    /// "the author wrote `with_level(.., Info)`". They are different
252    /// statements — one is a choice and the other is the absence of one — and
253    /// telling them apart is what lets [`Logger::set_default_level`] move the
254    /// first without overriding the second.
255    #[must_use]
256    pub const fn new(name: &'static str) -> Self {
257        Self {
258            name,
259            level: AtomicU8::new(LEVEL_UNSET),
260        }
261    }
262
263    /// `const`-construct with an explicit threshold.
264    #[must_use]
265    pub const fn with_level(name: &'static str, level: Severity) -> Self {
266        Self {
267            name,
268            level: AtomicU8::new(level as u8),
269        }
270    }
271
272    /// Logger name (used as `Record::logger_name`).
273    #[must_use]
274    pub const fn name(&self) -> &'static str {
275        self.name
276    }
277
278    /// The threshold this logger actually filters on — its own level if it has
279    /// one, otherwise [`Logger::default_level`].
280    ///
281    /// This is `rcutils_logging_get_logger_effective_level`'s answer, not
282    /// `rcutils_logging_get_logger_level`'s: it is the number that decides
283    /// whether a record is emitted, which is the question a call site asks.
284    /// There is deliberately no accessor for "is a level set on this logger",
285    /// because nothing in the tree has a use for the difference and an
286    /// accessor per language is three more names for one fact.
287    #[must_use]
288    pub fn level(&self) -> Severity {
289        let own = self.level.load(Ordering::Relaxed);
290        if own == LEVEL_UNSET {
291            return Self::default_level();
292        }
293        severity_from_u8(own).unwrap_or(Severity::Info)
294    }
295
296    /// Give this logger a level of its own. It stops following
297    /// [`Logger::default_level`] until [`Logger::unset_level`].
298    pub fn set_level(&self, level: Severity) {
299        self.level.store(level as u8, Ordering::Relaxed);
300    }
301
302    /// Take this logger's own level away, so it follows
303    /// [`Logger::default_level`] again.
304    ///
305    /// `rcutils_logging_set_logger_level(name, RCUTILS_LOG_SEVERITY_UNSET)`'s
306    /// effect, and what `nros_logger_set_level(logger, NROS_LOG_SEVERITY_UNSET)`
307    /// reaches. phase-467.
308    pub fn unset_level(&self) {
309        self.level.store(LEVEL_UNSET, Ordering::Relaxed);
310    }
311
312    /// The threshold every logger with no level of its own filters on.
313    ///
314    /// [`Severity::Info`] until [`Logger::set_default_level`] moves it, which
315    /// is `RCUTILS_DEFAULT_LOGGER_DEFAULT_LEVEL`'s value and upstream's
316    /// starting point too.
317    #[must_use]
318    pub fn default_level() -> Severity {
319        severity_from_u8(DEFAULT_LEVEL.load(Ordering::Relaxed)).unwrap_or(Severity::Info)
320    }
321
322    /// Move the threshold that every logger WITHOUT a level of its own filters
323    /// on — `rclrs::Logger::set_default_level`.
324    ///
325    /// Takes effect immediately and retroactively: a `static LOGGER:
326    /// Logger = Logger::new("x")` that was compiled into the image long before
327    /// this call follows it, because "no level of its own" is a stored state
328    /// and not a value copied in at construction. That is the whole reason
329    /// this is a capability rather than a re-export of
330    /// `DEFAULT_LOGGER.set_level`, which moves the threshold of the ONE logger
331    /// named `"nros"` (ledger row `rust:Logger::set_default_level`).
332    ///
333    /// It does NOT override a logger that states its own level: after
334    /// `with_level(.., Warn)` or `set_level(Warn)`, `Warn` wins.
335    pub fn set_default_level(level: Severity) {
336        DEFAULT_LEVEL.store(level as u8, Ordering::Relaxed);
337    }
338
339    /// Whether a record at `severity` would be emitted by this
340    /// logger AT RUNTIME.
341    ///
342    /// THE HOT PATH of every log macro, on every target, so the shape here is
343    /// measured rather than chosen (phase-467). Resolving the unset level
344    /// SELECTS the byte to compare against and compares once, instead of
345    /// comparing twice and short-circuiting: measured at an inlined call site
346    /// with `rustc -O`, the select form is 13 instructions on
347    /// `thumbv7m-none-eabi` and **branch-free** (LLVM predicates the whole
348    /// fallback into one `ittt eq` block) against 16 with a branch for the
349    /// short-circuit form, and 10 against 12 on `x86_64`. The cost over the
350    /// single-load original (8 thumb / 4 x86_64 instructions) is one compare
351    /// against an immediate, plus a byte load of one process-global that a
352    /// logger with its own level never performs.
353    ///
354    /// `LEVEL_UNSET` being ABOVE `Fatal` rather than below `Trace` is what
355    /// keeps this correct if the select is ever reordered: `severity >=
356    /// u8::MAX` is false for every `Severity`, so an unresolved sentinel can
357    /// only ever suppress a record, never emit one at the most verbose level —
358    /// which is the failure `0` had.
359    #[must_use]
360    pub fn is_enabled(&self, severity: Severity) -> bool {
361        let own = self.level.load(Ordering::Relaxed);
362        let effective = if own == LEVEL_UNSET {
363            DEFAULT_LEVEL.load(Ordering::Relaxed)
364        } else {
365            own
366        };
367        (severity as u8) >= effective
368    }
369
370    /// Whether a record at `severity` would be emitted, given both this
371    /// logger's runtime threshold AND a per-call-site throttle window.
372    ///
373    /// phase-417 W4.d. The order is load-bearing and is the same order
374    /// `rclcpp` uses: the SEVERITY test runs first, so a record the level
375    /// already filters does not consume the throttle window. Reversing them
376    /// makes a raised-then-lowered log level lose its first record for a whole
377    /// interval, which is the shape of bug nobody reports because it looks
378    /// like the throttle working.
379    ///
380    /// Consumes the window when it returns `true`, so call it exactly once per
381    /// candidate record — the `nros_*_throttle!` macros do.
382    ///
383    /// `now_ns` is the caller's monotonic clock; the macros pass
384    /// [`__timestamp_ns`]. Taking it as an argument rather than reading the
385    /// clock here is what lets the whole rule be tested without a platform
386    /// port, and is the same shape `rclcpp::RCLCPP_*_THROTTLE` uses when it
387    /// asks for a `Clock`.
388    #[must_use]
389    pub fn is_enabled_throttled(
390        &self,
391        severity: Severity,
392        state: &ThrottleState,
393        now_ns: u64,
394        interval_ns: u64,
395    ) -> bool {
396        self.is_enabled(severity) && state.should_log(now_ns, interval_ns)
397    }
398
399    /// Hand `record` to every registered sink, after the runtime
400    /// threshold check.
401    ///
402    /// Macros call this; user code should not.
403    pub fn dispatch(&self, record: &Record<'_>) {
404        if !self.is_enabled(record.severity) {
405            return;
406        }
407        dispatch_to_sinks(record);
408    }
409}

415/// Maximum number of named loggers that can be registered via
416/// [`register_logger`]. Beyond this, [`get_logger`] returns
417/// [`DEFAULT_LOGGER`].
418pub const MAX_LOGGERS: usize = 32;
419
420/// Catch-all logger returned when the requested name is not
421/// registered (or the intern table is full).
422pub static DEFAULT_LOGGER: Logger = Logger::new("nros");
423
424mod intern {
425    use super::{AtomicPtr, Logger, MAX_LOGGERS, Ordering};
426
427    pub(super) struct InternTable {
428        slots: [AtomicPtr<Logger>; MAX_LOGGERS],
429    }
430
431    impl InternTable {
432        pub(super) const fn new() -> Self {
433            // `AtomicPtr::new` is `const` on both `core::sync::atomic`
434            // and `portable_atomic`, so we can initialise the array
435            // by repeating the call rather than naming a `const` —
436            // which clippy flags as interior-mutable.
437            #[allow(clippy::declare_interior_mutable_const)]
438            const NULL: AtomicPtr<Logger> = AtomicPtr::new(core::ptr::null_mut());
439            Self {
440                slots: [NULL; MAX_LOGGERS],
441            }
442        }
443
444        pub(super) fn lookup(&self, name: &str) -> Option<&'static Logger> {
445            for slot in &self.slots {
446                let ptr = slot.load(Ordering::Acquire);
447                if ptr.is_null() {
448                    return None;
449                }
450                // SAFETY: pointer published via Release after the
451                // owner constructed a `'static Logger`. The Acquire
452                // load synchronizes.
453                let logger: &'static Logger = unsafe { &*ptr };
454                if logger.name() == name {
455                    return Some(logger);
456                }
457            }
458            None
459        }
460
461        pub(super) fn insert(&self, logger: &'static Logger) -> Option<&'static Logger> {
462            if let Some(existing) = self.lookup(logger.name()) {
463                return Some(existing);
464            }
465            let ptr = logger as *const _ as *mut Logger;
466            for slot in &self.slots {
467                if slot
468                    .compare_exchange(
469                        core::ptr::null_mut(),
470                        ptr,
471                        Ordering::AcqRel,
472                        Ordering::Acquire,
473                    )
474                    .is_ok()
475                {
476                    return Some(logger);
477                }
478                let existing_ptr = slot.load(Ordering::Acquire);
479                if !existing_ptr.is_null() {
480                    // SAFETY: same publication invariant as `lookup`.
481                    let existing: &'static Logger = unsafe { &*existing_ptr };
482                    if existing.name() == logger.name() {
483                        return Some(existing);
484                    }
485                }
486            }
487            None
488        }
489    }
490}

494/// Publish `logger` under its name so subsequent `get_logger`
495/// calls with that name return THIS reference.
496///
497/// On name collision returns the pre-existing entry (the input
498/// `logger` is NOT inserted). On a full table returns
499/// [`DEFAULT_LOGGER`].
500pub fn register_logger(logger: &'static Logger) -> &'static Logger {
501    INTERN.insert(logger).unwrap_or(&DEFAULT_LOGGER)
502}

504/// Look up a registered logger by name. Returns [`DEFAULT_LOGGER`]
505/// if none is registered (call [`register_logger`] for a `'static
506/// Logger` to publish one).
507///
508/// Total — every call returns a usable handle the macros can
509/// dispatch through.
510#[must_use]
511pub fn get_logger(name: &str) -> &'static Logger {
512    INTERN.lookup(name).unwrap_or(&DEFAULT_LOGGER)
513}

515/// Look up a logger by name, CREATING one if the name is new.
516///
517/// phase-417 W4.d. This is the shape `rclcpp::get_logger` / `rcutils`' logger
518/// lookup have, and the one a wrapper needs: [`get_logger`] answers
519/// [`DEFAULT_LOGGER`] for an unregistered name, so a C or C++ caller doing
520/// `set_level(get_logger("nav"), Debug)` over the lookup-only form would have
521/// moved the threshold of the catch-all logger that EVERY other unregistered
522/// name also resolves to. Same call, same types, different effect — precisely
523/// the compile-and-differ RFC-0089 forbids.
524///
525/// Storage is the bounded static arena in [`pool`], sized by the
526/// `dynamic-loggers-<N>` feature. Returns `None` — never a logger under the
527/// wrong name — when:
528///
529/// * `name` is empty or longer than [`MAX_LOGGER_NAME_LEN`],
530/// * the name arena or the logger arena is full,
531/// * the [`MAX_LOGGERS`] intern table is full.
532///
533/// Callers decide what an exhausted arena means for them; `nros-c` reports it
534/// and falls back to [`DEFAULT_LOGGER`] loudly rather than silently.
535///
536/// Loggers created this way are never destroyed — the arena has no free list,
537/// which is what makes the returned `&'static Logger` honest on `no_std`.
538#[must_use]
539pub fn get_or_create_logger(name: &str) -> Option<&'static Logger> {
540    if let Some(existing) = INTERN.lookup(name) {
541        return Some(existing);
542    }
543    let interned = pool::intern_name(name)?;
544    let placed = pool::place(Logger::new(interned))?;
545    // Not `register_logger`: that answers `DEFAULT_LOGGER` on a full table,
546    // which would hand back a logger under the wrong name — the exact aliasing
547    // this function exists to avoid. On a lost race `insert` returns the
548    // winner and our slot is spent, which `pool::dynamic_loggers_in_use`
549    // reports.
550    INTERN.insert(placed)
551}

553/// Resolve `name` to a logger — the TOTAL form of [`get_or_create_logger`],
554/// and the one spelling every node-shaped accessor uses.
555///
556/// phase-427 W5. Four accessors answer "which logger is this node's?" —
557/// `nros_node::executor::Node::logger`, `nros_node_get_logger`,
558/// `nros_cpp_node_get_logger` and `nros_log_get_logger`. All four had the same
559/// job and, before this, three of them called [`get_logger`], which is a
560/// LOOKUP: a node whose name nobody `register_logger`ed resolved to
561/// [`DEFAULT_LOGGER`]. Two nodes in one image therefore emitted every record
562/// under the same name, `"nros"` — the sentinel W5 replaced, arrived at by a
563/// different route.
564///
565/// This creates instead, and falls back to [`DEFAULT_LOGGER`] only when it
566/// cannot — an empty or over-long name, or an exhausted arena — and says so at
567/// WARN when it does, because the catch-all's threshold is SHARED and a silent
568/// alias makes `set_level` on the result move every other unnamed logger's bar
569/// with it. The report is ONE per process: a C++ `NROS_LOG_*` call site asks
570/// `node.get_logger()` on every line, so a per-call report would be a flood.
571///
572/// Lookup comes first, so an image that pre-registered a `'static Logger` under
573/// the node's name still gets that exact reference and its threshold.
574#[must_use]
575pub fn resolve_logger(name: &str) -> &'static Logger {
576    if let Some(logger) = get_or_create_logger(name) {
577        return logger;
578    }
579    // ONE report per process. This is reached from `Node::get_logger()`, which
580    // a C++ call site invokes inside every `NROS_LOG_*` macro — so a node whose
581    // name is longer than `MAX_LOGGER_NAME_LEN` (node names may be 64 bytes,
582    // logger names 48) would otherwise emit a warning per log line, and the
583    // diagnostic would be the flood. The first failure is the informative one:
584    // the arena only ever gets fuller.
585    static REPORTED: AtomicU8 = AtomicU8::new(0);
586    if REPORTED.swap(1, Ordering::Relaxed) != 0 {
587        return &DEFAULT_LOGGER;
588    }
589    let (used, total) = dynamic_logger_name_arena();
590    crate::log_warn!(
591        &DEFAULT_LOGGER,
592        "resolve_logger(\"{name}\"): no logger created — {} of {} slots and {used} of {total} \
593         name bytes are spent, or the name is over {} bytes. Returning the catch-all logger, \
594         whose threshold is SHARED: raise `dynamic-loggers-<N>` on nros-log rather than calling \
595         set_level on this handle. Reported once per process.",
596        dynamic_loggers_in_use(),
597        dynamic_logger_capacity(),
598        MAX_LOGGER_NAME_LEN,
599    );
600    &DEFAULT_LOGGER
601}

603// -----------------------------------------------------------------------------
604// Sink list. Set once at `init`; read every dispatch.
605// -----------------------------------------------------------------------------
606
607static SINKS_PTR: AtomicPtr<&'static [&'static dyn LogSink]> =
608    AtomicPtr::new(core::ptr::null_mut());
609
610// issue 0710 — `init_default()` was here. It named a default this crate can no
611// longer name: the platform sink moved to `nros_platform_cffi::log`, where the
612// ABI it speaks is a dependency rather than a feature. Call
613// `nros_platform_cffi::log::init_default()`, or `init()` with your own sinks.
614
615/// Install the global sink list.
616///
617/// MUST be called at app startup BEFORE any record-emitting macro
618/// runs (otherwise the dispatch is a no-op — records are silently
619/// dropped). Calling `init` more than once swaps the list
620/// atomically; the previous pointer is leaked (intentional: the
621/// read path is lock-free so we can't safely free).
622///
623/// The sinks themselves must outlive the program (`'static`).
624pub fn init(sinks: &'static [&'static dyn LogSink]) {
625    // Indirect through a small `'static` cell so the read path
626    // dereferences a fat-pointer-sized slot rather than reading
627    // a wide pointer atomically.
628    #[cfg(feature = "alloc")]
629    {
630        let boxed: alloc::boxed::Box<&'static [&'static dyn LogSink]> =
631            alloc::boxed::Box::new(sinks);
632        let ptr = alloc::boxed::Box::into_raw(boxed);
633        SINKS_PTR.store(ptr, Ordering::Release);
634    }
635    #[cfg(not(feature = "alloc"))]
636    {
637        static CELL: SinkSlot = SinkSlot::new();
638        CELL.store(sinks);
639        SINKS_PTR.store(CELL.as_ptr(), Ordering::Release);
640    }
641    // AFTER publishing, so a record raised during the drain reaches `sinks`
642    // directly rather than joining a ring nobody will drain again.
643    early::drain(sinks);
644    let lost = early::overflowed();
645    if lost > 0 {
646        // Reported through the sinks just installed, because the alternative is
647        // a silent hole exactly where the boot story is (`nros-log` cannot know
648        // what those records said, but it does know how many there were).
649        let logger = Logger::new("nros_log");
650        crate::log_warn!(
651            &logger,
652            "{lost} record(s) raised before `init` did not fit the early ring \
653             (see `nros_log::early`; raise `early-records-<N>`)"
654        );
655    }
656}

658#[cfg(not(feature = "alloc"))]
659struct SinkSlot {
660    inner: core::cell::UnsafeCell<Option<&'static [&'static dyn LogSink]>>,
661}
662
663#[cfg(not(feature = "alloc"))]
664// SAFETY: only written from `init`, which the user contracts to call
665// once at startup before any concurrent reader exists.
666unsafe impl Sync for SinkSlot {}
667
668#[cfg(not(feature = "alloc"))]
669impl SinkSlot {
670    const fn new() -> Self {
671        Self {
672            inner: core::cell::UnsafeCell::new(None),
673        }
674    }
675    fn store(&self, sinks: &'static [&'static dyn LogSink]) {
676        // SAFETY: see Sync note above.
677        unsafe {
678            *self.inner.get() = Some(sinks);
679        }
680    }
681    fn as_ptr(&self) -> *mut &'static [&'static dyn LogSink] {
682        self.inner.get().cast()
683    }
684}

686// -----------------------------------------------------------------------------
687// Appendable sink registry (phase-417 W4.d).
688//
689// `init` REPLACES the list, which is right for the board — it names, once, the
690// whole delivery it chose. It is the wrong verb for a consumer: C and C++ had
691// no way to install a sink at all (`nros_log_init(void)` takes no arguments
692// and hardcodes the platform default), and a library that wants to tee records
693// to /rosout cannot call `init` without discarding whatever the board
694// installed.
695//
696// So `add_sink` APPENDS, and dispatch walks both lists. The two are kept
697// separate rather than merged into one growable list because they have
698// different owners and different lifetimes-of-decision: `init` may be called
699// again and swap the board's choice wholesale, and doing so must not silently
700// unregister a consumer's sink.
701// -----------------------------------------------------------------------------
702
703/// How many sinks [`add_sink`] can append on top of the [`init`] list.
704///
705/// Small on purpose: the appendable list is for consumers that TEE (a /rosout
706/// bridge, a test collector), not for composing a delivery — a board composing
707/// its own delivery passes the whole slice to [`init`].
708pub const MAX_ADDED_SINKS: usize = 4;
709
710struct AddedSinks {
711    slots: core::cell::UnsafeCell<[Option<&'static dyn LogSink>; MAX_ADDED_SINKS]>,
712    /// Number of slots fully written. `Release` on publish, `Acquire` on read.
713    len: AtomicUsize,
714}
715
716// SAFETY: writes happen only under `ADD_LOCK` and only to the slot at index
717// `len`, which no reader looks at until the `Release` store to `len` that
718// follows the write is observed by an `Acquire` load. A slot is written once
719// and never rewritten.
720unsafe impl Sync for AddedSinks {}
721
722static ADDED: AddedSinks = AddedSinks {
723    slots: core::cell::UnsafeCell::new([None; MAX_ADDED_SINKS]),
724    len: AtomicUsize::new(0),
725};
726
727/// Serialises `add_sink` against itself.
728///
729/// A spin lock, held for two stores with no call inside it, is admissible here
730/// where the in-order-publish alternative is not: that one has a thread
731/// waiting on ANOTHER thread's progress, which on a single-core RTOS with
732/// priorities is a deadlock rather than a delay. Do not call `add_sink` from
733/// an ISR.
734static ADD_LOCK: AtomicBool = AtomicBool::new(false);
735
736/// Append `sink` to the sinks every record is delivered to.
737///
738/// Returns `false` when [`MAX_ADDED_SINKS`] are already registered — the sink
739/// is NOT installed, and the caller is expected to say so rather than carry on
740/// as if it were.
741///
742/// Records raised before any sink existed are replayed into `sink` here, the
743/// same way [`init`] replays them, so the first sink to arrive still sees the
744/// boot story regardless of which of the two calls installed it.
745pub fn add_sink(sink: &'static dyn LogSink) -> bool {
746    while ADD_LOCK
747        .compare_exchange(false, true, Ordering::Acquire, Ordering::Acquire)
748        .is_err()
749    {
750        core::hint::spin_loop();
751    }
752    let idx = ADDED.len.load(Ordering::Relaxed);
753    let installed = if idx < MAX_ADDED_SINKS {
754        // SAFETY: `ADD_LOCK` makes this the only writer, `idx` is beyond what
755        // any reader may look at until the `Release` store below publishes it,
756        // and the slot is never written again.
757        unsafe {
758            ADDED
759                .slots
760                .get()
761                .cast::<Option<&'static dyn LogSink>>()
762                .add(idx)
763                .write(Some(sink));
764        }
765        ADDED.len.store(idx + 1, Ordering::Release);
766        true
767    } else {
768        false
769    };
770    ADD_LOCK.store(false, Ordering::Release);
771
772    if installed && SINKS_PTR.load(Ordering::Acquire).is_null() && idx == 0 {
773        // First delivery target in the program, and `init` has not run: the
774        // held records are ours to replay. AFTER publishing, for the reason
775        // `init` states — a record raised during the drain reaches the sink
776        // directly rather than joining a ring nobody will drain again.
777        early::drain_with(&mut |record| sink.log(record));
778    }
779    installed
780}
781
782/// How many sinks [`add_sink`] has installed.
783#[must_use]
784pub fn added_sink_count() -> usize {
785    ADDED.len.load(Ordering::Acquire)
786}
787
788/// Hand `record` to every sink installed by [`add_sink`]. Returns how many
789/// saw it.
790fn deliver_to_added(record: &Record<'_>) -> usize {
791    let len = ADDED.len.load(Ordering::Acquire);
792    let base = ADDED.slots.get().cast::<Option<&'static dyn LogSink>>();
793    for idx in 0..len {
794        // SAFETY: `idx < len`, and the `Acquire` load above pairs with the
795        // `Release` store in `add_sink` that published slot `idx`. Slots below
796        // `len` are immutable for the rest of the program.
797        if let Some(sink) = unsafe { base.add(idx).read() } {
798            sink.log(record);
799        }
800    }
801    len
802}

804/// Whether [`__timestamp_ns`] can answer with a real clock.
805///
806/// `false` means every `Record::timestamp_ns` is `0` and anything DERIVED from
807/// the timestamp — the throttle windows above all of them — has no time base:
808/// a throttled site then emits its FIRST record and nothing after it (a clock
809/// that never moves is a window that never elapses — issue 1152). Exposed so a
810/// wrapper can say so out loud instead of degrading quietly: without this, a
811/// throttled C call site looks identical whether it is rate-limiting or
812/// permanently shut.
813#[must_use]
814pub const fn timestamp_available() -> bool {
815    cfg!(feature = "platform-clock")
816}
817
818/// Current monotonic time for `Record::timestamp_ns` (issue #503).
819///
820/// With the `platform-clock` feature this reads
821/// `nros_platform_clock_ns` — the universal per-platform export the
822/// executor's timer accounting already links — scaled to nanoseconds.
823/// Without the feature it returns `0` ("unavailable"), the historical
824/// behavior, and imposes no link-time requirement.
825///
826/// Public because the emission macros expand it in user crates; not
827/// part of the supported API surface.
828#[doc(hidden)]
829#[must_use]
830pub fn __timestamp_ns() -> u64 {
831    #[cfg(feature = "platform-clock")]
832    {
833        unsafe extern "C" {
834            fn nros_platform_clock_ns() -> u64;
835        }
836        // SAFETY: bare query of the platform's monotonic us counter;
837        // the symbol comes from whichever `nros-platform-<rtos>` port
838        // linked the binary (the contract `PlatformSink ->
839        // nros_platform_log_write` already relies on).
840        unsafe { nros_platform_clock_ns() }
841    }
842    #[cfg(not(feature = "platform-clock"))]
843    {
844        0
845    }
846}

848fn dispatch_to_sinks(record: &Record<'_>) {
849    if recursion_guard_check_and_set() {
850        return;
851    }
852    // phase-417 W4.d — the appendable list is walked whether or not `init`
853    // ran, and a record with at least one destination is NOT held.
854    let added = deliver_to_added(record);
855    let ptr = SINKS_PTR.load(Ordering::Acquire);
856    if ptr.is_null() {
857        if added > 0 {
858            recursion_guard_clear();
859            return;
860        }
861        // Nothing installed yet — HOLD the record; `init` replays it into
862        // whatever sinks the board chooses. See `early` for why this crate
863        // does not reach for the platform sink itself: doing that (issue 0710)
864        // put `nros_platform_log_write` on a path every binary executes, which
865        // turned a pluggable delivery into a link-time requirement that no
866        // Cargo feature can undo under workspace feature unification.
867        //
868        // It is also a better answer to issue 0708 than installing a default
869        // was: the early records land in the sink the board picked, not in the
870        // one dispatch guessed before the board had spoken.
871        early::hold(record);
872        recursion_guard_clear();
873        return;
874    }
875    {
876        // SAFETY: `init` published a valid `'static` slice reference.
877        let sinks: &'static [&'static dyn LogSink] = unsafe { *ptr };
878        for sink in sinks {
879            sink.log(record);
880        }
881    }
882    recursion_guard_clear();
883}
884
885/// Flush every registered sink, appended ones included.
886pub fn flush() {
887    let len = ADDED.len.load(Ordering::Acquire);
888    let base = ADDED.slots.get().cast::<Option<&'static dyn LogSink>>();
889    for idx in 0..len {
890        // SAFETY: same publication invariant as `deliver_to_added`.
891        if let Some(sink) = unsafe { base.add(idx).read() } {
892            sink.flush();
893        }
894    }
895    let ptr = SINKS_PTR.load(Ordering::Acquire);
896    if ptr.is_null() {
897        return;
898    }
899    // SAFETY: same invariant as `dispatch_to_sinks`.
900    let sinks: &'static [&'static dyn LogSink] = unsafe { *ptr };
901    for sink in sinks {
902        sink.flush();
903    }
904}

906// -----------------------------------------------------------------------------
907// Recursion guard — process-global single AtomicBool.
908//
909// Granularity is intentionally coarse (process-wide, not per-thread).
910// The guard exists to break a sink that triggers log() during write
911// — not to serialize concurrent loggers across threads. A thread
912// re-entering through its own sink loses its other in-flight
913// sinks for that call; a different thread logging concurrently is
914// also short-circuited momentarily. This is acceptable: the alt is
915// per-thread storage which doesn't exist uniformly across our
916// `no_std` targets (`thread_local!` requires `std`).
917// -----------------------------------------------------------------------------
918
919use portable_atomic::AtomicBool;
920static RECURSION_GUARD: AtomicBool = AtomicBool::new(false);
921
922fn recursion_guard_check_and_set() -> bool {
923    RECURSION_GUARD
924        .compare_exchange(false, true, Ordering::Acquire, Ordering::Acquire)
925        .is_err()
926}
927
928fn recursion_guard_clear() {
929    RECURSION_GUARD.store(false, Ordering::Release);
930}
