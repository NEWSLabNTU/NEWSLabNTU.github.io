1//! Phase 88 — portable leveled-logging facade for nano-ros.
2//!
3//! See [`docs/roadmap/archived/phase-88-nros-log.md`](../../../docs/roadmap/archived/phase-88-nros-log.md)
4//! for the design and acceptance criteria.
5//!
6//! ## Layering
7//!
8//! - This crate carries only the portable types + dispatcher +
9//!   macros + `PlatformSink`. No backend code.
10//! - Per-platform log delivery is the responsibility of each
11//!   `nros-platform-<rtos>` crate, exposing
12//!   `nros_platform_log_write` / `nros_platform_log_flush` via the
13//!   `nros_platform_*` ABI (header at
14//!   `packages/platform/nros-platform-api/include/nros/platform.h`).
15//! - `PlatformSink` is the bridge: a single `LogSink` impl that
16//!   forwards to the ABI. Apps that want fan-out (e.g.
17//!   `Platform + /rosout`) compose a `&'static [&dyn LogSink]`
18//!   manually and pass it to [`init`].
19//!
20//! ## Quick start
21//!
22//! ```ignore
23//! use nros_log::{Logger, Severity};
24//! use nros_log::{log_info, log_warn};
25//!
26//! static LOGGER: Logger = Logger::new("my_node");
27//!
28//! fn main() {
29//!     nros_log::register_logger(&LOGGER);
30//!     nros_log::init(nros_log::sinks::default());
31//!     log_info!(&LOGGER, "started; domain = {}", 42);
32//! }
33//! ```

65/// REP-2012 severity levels, mirroring `rcutils_log_severity_t`.
66///
67/// The integer representation is stable and part of the ABI for
68/// `nros_platform_log_write`. Lower value = more verbose.
69#[repr(u8)]
70#[derive(Copy, Clone, Debug, Eq, PartialEq, Ord, PartialOrd, Hash)]
71pub enum Severity {
72    /// Per-instruction granularity. Off unless `max-level-trace` is
73    /// the active ceiling.
74    Trace = 0,
75    /// Diagnostic information useful while developing.
76    Debug = 1,
77    /// Normal operation events worth surfacing once.
78    Info = 2,
79    /// Unexpected but recoverable conditions.
80    Warn = 3,
81    /// Errors the caller should surface; the system continues.
82    Error = 4,
83    /// Unrecoverable — the system is about to abort.
84    Fatal = 5,
85}
86
87impl Severity {
88    /// Short uppercase label suitable for log-line rendering.
89    #[must_use]
90    pub const fn as_str(self) -> &'static str {
91        match self {
92            Self::Trace => "TRACE",
93            Self::Debug => "DEBUG",
94            Self::Info => "INFO",
95            Self::Warn => "WARN",
96            Self::Error => "ERROR",
97            Self::Fatal => "FATAL",
98        }
99    }
100
101    /// Stable `u8` discriminant for cross-ABI use.
102    #[must_use]
103    pub const fn as_u8(self) -> u8 {
104        self as u8
105    }
106}

108/// Reconstruct a [`Severity`] from its `u8` discriminant.
109///
110/// Returns `None` for `> 5`.
111#[must_use]
112pub const fn severity_from_u8(value: u8) -> Option<Severity> {
113    match value {
114        0 => Some(Severity::Trace),
115        1 => Some(Severity::Debug),
116        2 => Some(Severity::Info),
117        3 => Some(Severity::Warn),
118        4 => Some(Severity::Error),
119        5 => Some(Severity::Fatal),
120        _ => None,
121    }
122}

124/// Compile-time ceiling check used by the `nros_*!` macros.
125///
126/// Returns `true` iff `severity` is allowed under the configured
127/// `max-level-*` feature.
128#[must_use]
129pub const fn severity_enabled_at_compile_time(severity: Severity) -> bool {
130    if cfg!(feature = "max-level-off") {
131        return false;
132    }
133    let ceiling = compile_time_ceiling();
134    (severity as u8) >= (ceiling as u8)
135}
136
137const fn compile_time_ceiling() -> Severity {
138    if cfg!(feature = "max-level-trace") {
139        Severity::Trace
140    } else if cfg!(feature = "max-level-debug") {
141        Severity::Debug
142    } else if cfg!(feature = "max-level-info") {
143        Severity::Info
144    } else if cfg!(feature = "max-level-warn") {
145        Severity::Warn
146    } else if cfg!(feature = "max-level-error") {
147        Severity::Error
148    } else {
149        // No ceiling feature = treat as `max-level-trace`.
150        Severity::Trace
151    }
152}

154/// One log entry, handed to each [`LogSink`].
155///
156/// `message` is already formatted — sinks must NOT re-format.
157#[derive(Debug)]
158pub struct Record<'a> {
159    /// Severity of the record.
160    pub severity: Severity,
161    /// Name of the originating [`Logger`].
162    pub logger_name: &'a str,
163    /// Formatted message text (no trailing newline).
164    pub message: &'a str,
165    /// File the macro invocation came from (`core::file!()`).
166    pub file: &'static str,
167    /// Line within `file` (`core::line!()`).
168    pub line: u32,
169    /// Monotonic timestamp in nanoseconds. `0` if unavailable.
170    pub timestamp_ns: u64,
171}

173/// Backend a log record is delivered to.
174///
175/// Implementations must be `Sync` so the dispatcher can hold them
176/// in `&'static [&dyn LogSink]`. ISR-safety is per-impl — see the
177/// table in `docs/roadmap/archived/phase-88-nros-log.md`.
178pub trait LogSink: Sync {
179    /// Render `record`. Called only when the record's severity passes
180    /// both the compile-time ceiling AND the [`Logger`]'s runtime
181    /// threshold.
182    fn log(&self, record: &Record<'_>);
183
184    /// Optional flush hook (default no-op).
185    fn flush(&self) {}
186}
187
188/// A named logger with a runtime severity threshold.
189///
190/// Threshold defaults to [`Severity::Info`]. Use [`register_logger`]
191/// to publish a `'static Logger` so multiple call sites with the
192/// same name share the same threshold.
193pub struct Logger {
194    name: &'static str,
195    level: AtomicU8,
196}
197
198impl Logger {
199    /// `const`-construct with the default threshold ([`Severity::Info`]).
200    #[must_use]
201    pub const fn new(name: &'static str) -> Self {
202        Self {
203            name,
204            level: AtomicU8::new(Severity::Info as u8),
205        }
206    }
207
208    /// `const`-construct with an explicit threshold.
209    #[must_use]
210    pub const fn with_level(name: &'static str, level: Severity) -> Self {
211        Self {
212            name,
213            level: AtomicU8::new(level as u8),
214        }
215    }
216
217    /// Logger name (used as `Record::logger_name`).
218    #[must_use]
219    pub const fn name(&self) -> &'static str {
220        self.name
221    }
222
223    /// Current runtime threshold.
224    #[must_use]
225    pub fn level(&self) -> Severity {
226        severity_from_u8(self.level.load(Ordering::Relaxed)).unwrap_or(Severity::Info)
227    }
228
229    /// Update the runtime threshold.
230    pub fn set_level(&self, level: Severity) {
231        self.level.store(level as u8, Ordering::Relaxed);
232    }
233
234    /// Whether a record at `severity` would be emitted by this
235    /// logger AT RUNTIME.
236    #[must_use]
237    pub fn is_enabled(&self, severity: Severity) -> bool {
238        (severity as u8) >= self.level.load(Ordering::Relaxed)
239    }
240
241    /// Whether a record at `severity` would be emitted, given both this
242    /// logger's runtime threshold AND a per-call-site throttle window.
243    ///
244    /// phase-417 W4.d. The order is load-bearing and is the same order
245    /// `rclcpp` uses: the SEVERITY test runs first, so a record the level
246    /// already filters does not consume the throttle window. Reversing them
247    /// makes a raised-then-lowered log level lose its first record for a whole
248    /// interval, which is the shape of bug nobody reports because it looks
249    /// like the throttle working.
250    ///
251    /// Consumes the window when it returns `true`, so call it exactly once per
252    /// candidate record — the `nros_*_throttle!` macros do.
253    ///
254    /// `now_ns` is the caller's monotonic clock; the macros pass
255    /// [`__timestamp_ns`]. Taking it as an argument rather than reading the
256    /// clock here is what lets the whole rule be tested without a platform
257    /// port, and is the same shape `rclcpp::RCLCPP_*_THROTTLE` uses when it
258    /// asks for a `Clock`.
259    #[must_use]
260    pub fn is_enabled_throttled(
261        &self,
262        severity: Severity,
263        state: &ThrottleState,
264        now_ns: u64,
265        interval_ns: u64,
266    ) -> bool {
267        self.is_enabled(severity) && state.should_log(now_ns, interval_ns)
268    }
269
270    /// Hand `record` to every registered sink, after the runtime
271    /// threshold check.
272    ///
273    /// Macros call this; user code should not.
274    pub fn dispatch(&self, record: &Record<'_>) {
275        if !self.is_enabled(record.severity) {
276            return;
277        }
278        dispatch_to_sinks(record);
279    }
280}

286/// Maximum number of named loggers that can be registered via
287/// [`register_logger`]. Beyond this, [`get_logger`] returns
288/// [`DEFAULT_LOGGER`].
289pub const MAX_LOGGERS: usize = 32;
290
291/// Catch-all logger returned when the requested name is not
292/// registered (or the intern table is full).
293pub static DEFAULT_LOGGER: Logger = Logger::new("nros");
294
295mod intern {
296    use super::{AtomicPtr, Logger, MAX_LOGGERS, Ordering};
297
298    pub(super) struct InternTable {
299        slots: [AtomicPtr<Logger>; MAX_LOGGERS],
300    }
301
302    impl InternTable {
303        pub(super) const fn new() -> Self {
304            // `AtomicPtr::new` is `const` on both `core::sync::atomic`
305            // and `portable_atomic`, so we can initialise the array
306            // by repeating the call rather than naming a `const` —
307            // which clippy flags as interior-mutable.
308            #[allow(clippy::declare_interior_mutable_const)]
309            const NULL: AtomicPtr<Logger> = AtomicPtr::new(core::ptr::null_mut());
310            Self {
311                slots: [NULL; MAX_LOGGERS],
312            }
313        }
314
315        pub(super) fn lookup(&self, name: &str) -> Option<&'static Logger> {
316            for slot in &self.slots {
317                let ptr = slot.load(Ordering::Acquire);
318                if ptr.is_null() {
319                    return None;
320                }
321                // SAFETY: pointer published via Release after the
322                // owner constructed a `'static Logger`. The Acquire
323                // load synchronizes.
324                let logger: &'static Logger = unsafe { &*ptr };
325                if logger.name() == name {
326                    return Some(logger);
327                }
328            }
329            None
330        }
331
332        pub(super) fn insert(&self, logger: &'static Logger) -> Option<&'static Logger> {
333            if let Some(existing) = self.lookup(logger.name()) {
334                return Some(existing);
335            }
336            let ptr = logger as *const _ as *mut Logger;
337            for slot in &self.slots {
338                if slot
339                    .compare_exchange(
340                        core::ptr::null_mut(),
341                        ptr,
342                        Ordering::AcqRel,
343                        Ordering::Acquire,
344                    )
345                    .is_ok()
346                {
347                    return Some(logger);
348                }
349                let existing_ptr = slot.load(Ordering::Acquire);
350                if !existing_ptr.is_null() {
351                    // SAFETY: same publication invariant as `lookup`.
352                    let existing: &'static Logger = unsafe { &*existing_ptr };
353                    if existing.name() == logger.name() {
354                        return Some(existing);
355                    }
356                }
357            }
358            None
359        }
360    }
361}

365/// Publish `logger` under its name so subsequent `get_logger`
366/// calls with that name return THIS reference.
367///
368/// On name collision returns the pre-existing entry (the input
369/// `logger` is NOT inserted). On a full table returns
370/// [`DEFAULT_LOGGER`].
371pub fn register_logger(logger: &'static Logger) -> &'static Logger {
372    INTERN.insert(logger).unwrap_or(&DEFAULT_LOGGER)
373}

375/// Look up a registered logger by name. Returns [`DEFAULT_LOGGER`]
376/// if none is registered (call [`register_logger`] for a `'static
377/// Logger` to publish one).
378///
379/// Total — every call returns a usable handle the macros can
380/// dispatch through.
381#[must_use]
382pub fn get_logger(name: &str) -> &'static Logger {
383    INTERN.lookup(name).unwrap_or(&DEFAULT_LOGGER)
384}

386/// Look up a logger by name, CREATING one if the name is new.
387///
388/// phase-417 W4.d. This is the shape `rclcpp::get_logger` / `rcutils`' logger
389/// lookup have, and the one a wrapper needs: [`get_logger`] answers
390/// [`DEFAULT_LOGGER`] for an unregistered name, so a C or C++ caller doing
391/// `set_level(get_logger("nav"), Debug)` over the lookup-only form would have
392/// moved the threshold of the catch-all logger that EVERY other unregistered
393/// name also resolves to. Same call, same types, different effect — precisely
394/// the compile-and-differ RFC-0089 forbids.
395///
396/// Storage is the bounded static arena in [`pool`], sized by the
397/// `dynamic-loggers-<N>` feature. Returns `None` — never a logger under the
398/// wrong name — when:
399///
400/// * `name` is empty or longer than [`MAX_LOGGER_NAME_LEN`],
401/// * the name arena or the logger arena is full,
402/// * the [`MAX_LOGGERS`] intern table is full.
403///
404/// Callers decide what an exhausted arena means for them; `nros-c` reports it
405/// and falls back to [`DEFAULT_LOGGER`] loudly rather than silently.
406///
407/// Loggers created this way are never destroyed — the arena has no free list,
408/// which is what makes the returned `&'static Logger` honest on `no_std`.
409#[must_use]
410pub fn get_or_create_logger(name: &str) -> Option<&'static Logger> {
411    if let Some(existing) = INTERN.lookup(name) {
412        return Some(existing);
413    }
414    let interned = pool::intern_name(name)?;
415    let placed = pool::place(Logger::new(interned))?;
416    // Not `register_logger`: that answers `DEFAULT_LOGGER` on a full table,
417    // which would hand back a logger under the wrong name — the exact aliasing
418    // this function exists to avoid. On a lost race `insert` returns the
419    // winner and our slot is spent, which `pool::dynamic_loggers_in_use`
420    // reports.
421    INTERN.insert(placed)
422}

424// -----------------------------------------------------------------------------

428static SINKS_PTR: AtomicPtr<&'static [&'static dyn LogSink]> =
429    AtomicPtr::new(core::ptr::null_mut());
430
431// issue 0710 — `init_default()` was here. It named a default this crate can no
432// longer name: the platform sink moved to `nros_platform_cffi::log`, where the
433// ABI it speaks is a dependency rather than a feature. Call
434// `nros_platform_cffi::log::init_default()`, or `init()` with your own sinks.
435
436/// Install the global sink list.
437///
438/// MUST be called at app startup BEFORE any record-emitting macro
439/// runs (otherwise the dispatch is a no-op — records are silently
440/// dropped). Calling `init` more than once swaps the list
441/// atomically; the previous pointer is leaked (intentional: the
442/// read path is lock-free so we can't safely free).
443///
444/// The sinks themselves must outlive the program (`'static`).
445pub fn init(sinks: &'static [&'static dyn LogSink]) {
446    // Indirect through a small `'static` cell so the read path
447    // dereferences a fat-pointer-sized slot rather than reading
448    // a wide pointer atomically.
449    #[cfg(feature = "alloc")]
450    {
451        let boxed: alloc::boxed::Box<&'static [&'static dyn LogSink]> =
452            alloc::boxed::Box::new(sinks);
453        let ptr = alloc::boxed::Box::into_raw(boxed);
454        SINKS_PTR.store(ptr, Ordering::Release);
455    }
456    #[cfg(not(feature = "alloc"))]
457    {
458        static CELL: SinkSlot = SinkSlot::new();
459        CELL.store(sinks);
460        SINKS_PTR.store(CELL.as_ptr(), Ordering::Release);
461    }
462    // AFTER publishing, so a record raised during the drain reaches `sinks`
463    // directly rather than joining a ring nobody will drain again.
464    early::drain(sinks);
465    let lost = early::overflowed();
466    if lost > 0 {
467        // Reported through the sinks just installed, because the alternative is
468        // a silent hole exactly where the boot story is (`nros-log` cannot know
469        // what those records said, but it does know how many there were).
470        let logger = Logger::new("nros_log");
471        crate::log_warn!(
472            &logger,
473            "{lost} record(s) raised before `init` did not fit the early ring \
474             (see `nros_log::early`; raise `early-records-<N>`)"
475        );
476    }
477}

479#[cfg(not(feature = "alloc"))]
480struct SinkSlot {
481    inner: core::cell::UnsafeCell<Option<&'static [&'static dyn LogSink]>>,
482}
483
484#[cfg(not(feature = "alloc"))]
485// SAFETY: only written from `init`, which the user contracts to call
486// once at startup before any concurrent reader exists.
487unsafe impl Sync for SinkSlot {}
488
489#[cfg(not(feature = "alloc"))]
490impl SinkSlot {
491    const fn new() -> Self {
492        Self {
493            inner: core::cell::UnsafeCell::new(None),
494        }
495    }
496    fn store(&self, sinks: &'static [&'static dyn LogSink]) {
497        // SAFETY: see Sync note above.
498        unsafe {
499            *self.inner.get() = Some(sinks);
500        }
501    }
502    fn as_ptr(&self) -> *mut &'static [&'static dyn LogSink] {
503        self.inner.get().cast()
504    }
505}

507// -----------------------------------------------------------------------------
508// Appendable sink registry (phase-417 W4.d).
509//
510// `init` REPLACES the list, which is right for the board — it names, once, the
511// whole delivery it chose. It is the wrong verb for a consumer: C and C++ had
512// no way to install a sink at all (`nros_log_init(void)` takes no arguments
513// and hardcodes the platform default), and a library that wants to tee records
514// to /rosout cannot call `init` without discarding whatever the board
515// installed.
516//
517// So `add_sink` APPENDS, and dispatch walks both lists. The two are kept
518// separate rather than merged into one growable list because they have
519// different owners and different lifetimes-of-decision: `init` may be called
520// again and swap the board's choice wholesale, and doing so must not silently
521// unregister a consumer's sink.
522// -----------------------------------------------------------------------------
523
524/// How many sinks [`add_sink`] can append on top of the [`init`] list.
525///
526/// Small on purpose: the appendable list is for consumers that TEE (a /rosout
527/// bridge, a test collector), not for composing a delivery — a board composing
528/// its own delivery passes the whole slice to [`init`].
529pub const MAX_ADDED_SINKS: usize = 4;
530
531struct AddedSinks {
532    slots: core::cell::UnsafeCell<[Option<&'static dyn LogSink>; MAX_ADDED_SINKS]>,
533    /// Number of slots fully written. `Release` on publish, `Acquire` on read.
534    len: AtomicUsize,
535}
536
537// SAFETY: writes happen only under `ADD_LOCK` and only to the slot at index
538// `len`, which no reader looks at until the `Release` store to `len` that
539// follows the write is observed by an `Acquire` load. A slot is written once
540// and never rewritten.
541unsafe impl Sync for AddedSinks {}
542
543static ADDED: AddedSinks = AddedSinks {
544    slots: core::cell::UnsafeCell::new([None; MAX_ADDED_SINKS]),
545    len: AtomicUsize::new(0),
546};
547
548/// Serialises `add_sink` against itself.
549///
550/// A spin lock, held for two stores with no call inside it, is admissible here
551/// where the in-order-publish alternative is not: that one has a thread
552/// waiting on ANOTHER thread's progress, which on a single-core RTOS with
553/// priorities is a deadlock rather than a delay. Do not call `add_sink` from
554/// an ISR.
555static ADD_LOCK: AtomicBool = AtomicBool::new(false);
556
557/// Append `sink` to the sinks every record is delivered to.
558///
559/// Returns `false` when [`MAX_ADDED_SINKS`] are already registered — the sink
560/// is NOT installed, and the caller is expected to say so rather than carry on
561/// as if it were.
562///
563/// Records raised before any sink existed are replayed into `sink` here, the
564/// same way [`init`] replays them, so the first sink to arrive still sees the
565/// boot story regardless of which of the two calls installed it.
566pub fn add_sink(sink: &'static dyn LogSink) -> bool {
567    while ADD_LOCK
568        .compare_exchange(false, true, Ordering::Acquire, Ordering::Acquire)
569        .is_err()
570    {
571        core::hint::spin_loop();
572    }
573    let idx = ADDED.len.load(Ordering::Relaxed);
574    let installed = if idx < MAX_ADDED_SINKS {
575        // SAFETY: `ADD_LOCK` makes this the only writer, `idx` is beyond what
576        // any reader may look at until the `Release` store below publishes it,
577        // and the slot is never written again.
578        unsafe {
579            ADDED
580                .slots
581                .get()
582                .cast::<Option<&'static dyn LogSink>>()
583                .add(idx)
584                .write(Some(sink));
585        }
586        ADDED.len.store(idx + 1, Ordering::Release);
587        true
588    } else {
589        false
590    };
591    ADD_LOCK.store(false, Ordering::Release);
592
593    if installed && SINKS_PTR.load(Ordering::Acquire).is_null() && idx == 0 {
594        // First delivery target in the program, and `init` has not run: the
595        // held records are ours to replay. AFTER publishing, for the reason
596        // `init` states — a record raised during the drain reaches the sink
597        // directly rather than joining a ring nobody will drain again.
598        early::drain_with(&mut |record| sink.log(record));
599    }
600    installed
601}

603/// How many sinks [`add_sink`] has installed.
604#[must_use]
605pub fn added_sink_count() -> usize {
606    ADDED.len.load(Ordering::Acquire)
607}

609/// Hand `record` to every sink installed by [`add_sink`]. Returns how many
610/// saw it.
611fn deliver_to_added(record: &Record<'_>) -> usize {
612    let len = ADDED.len.load(Ordering::Acquire);
613    let base = ADDED.slots.get().cast::<Option<&'static dyn LogSink>>();
614    for idx in 0..len {
615        // SAFETY: `idx < len`, and the `Acquire` load above pairs with the
616        // `Release` store in `add_sink` that published slot `idx`. Slots below
617        // `len` are immutable for the rest of the program.
618        if let Some(sink) = unsafe { base.add(idx).read() } {
619            sink.log(record);
620        }
621    }
622    len
623}

625/// Whether [`__timestamp_ns`] can answer with a real clock.
626///
627/// `false` means every `Record::timestamp_ns` is `0` and anything DERIVED from
628/// the timestamp — the throttle windows above all of them — has no time base:
629/// a throttled site then emits its FIRST record and nothing after it (a clock
630/// that never moves is a window that never elapses — issue 1152). Exposed so a
631/// wrapper can say so out loud instead of degrading quietly: without this, a
632/// throttled C call site looks identical whether it is rate-limiting or
633/// permanently shut.
634#[must_use]
635pub const fn timestamp_available() -> bool {
636    cfg!(feature = "platform-clock")
637}

639/// Current monotonic time for `Record::timestamp_ns` (issue #503).
640///
641/// With the `platform-clock` feature this reads
642/// `nros_platform_clock_ns` — the universal per-platform export the
643/// executor's timer accounting already links — scaled to nanoseconds.
644/// Without the feature it returns `0` ("unavailable"), the historical
645/// behavior, and imposes no link-time requirement.
646///
647/// Public because the emission macros expand it in user crates; not
648/// part of the supported API surface.
649#[doc(hidden)]
650#[must_use]
651pub fn __timestamp_ns() -> u64 {
652    #[cfg(feature = "platform-clock")]
653    {
654        unsafe extern "C" {
655            fn nros_platform_clock_ns() -> u64;
656        }
657        // SAFETY: bare query of the platform's monotonic us counter;
658        // the symbol comes from whichever `nros-platform-<rtos>` port
659        // linked the binary (the contract `PlatformSink ->
660        // nros_platform_log_write` already relies on).
661        unsafe { nros_platform_clock_ns() }
662    }
663    #[cfg(not(feature = "platform-clock"))]
664    {
665        0
666    }
667}

669fn dispatch_to_sinks(record: &Record<'_>) {
670    if recursion_guard_check_and_set() {
671        return;
672    }
673    // phase-417 W4.d — the appendable list is walked whether or not `init`
674    // ran, and a record with at least one destination is NOT held.
675    let added = deliver_to_added(record);
676    let ptr = SINKS_PTR.load(Ordering::Acquire);
677    if ptr.is_null() {
678        if added > 0 {
679            recursion_guard_clear();
680            return;
681        }
682        // Nothing installed yet — HOLD the record; `init` replays it into
683        // whatever sinks the board chooses. See `early` for why this crate
684        // does not reach for the platform sink itself: doing that (issue 0710)
685        // put `nros_platform_log_write` on a path every binary executes, which
686        // turned a pluggable delivery into a link-time requirement that no
687        // Cargo feature can undo under workspace feature unification.
688        //
689        // It is also a better answer to issue 0708 than installing a default
690        // was: the early records land in the sink the board picked, not in the
691        // one dispatch guessed before the board had spoken.
692        early::hold(record);
693        recursion_guard_clear();
694        return;
695    }
696    {
697        // SAFETY: `init` published a valid `'static` slice reference.
698        let sinks: &'static [&'static dyn LogSink] = unsafe { *ptr };
699        for sink in sinks {
700            sink.log(record);
701        }
702    }
703    recursion_guard_clear();
704}
705
706/// Flush every registered sink, appended ones included.
707pub fn flush() {
708    let len = ADDED.len.load(Ordering::Acquire);
709    let base = ADDED.slots.get().cast::<Option<&'static dyn LogSink>>();
710    for idx in 0..len {
711        // SAFETY: same publication invariant as `deliver_to_added`.
712        if let Some(sink) = unsafe { base.add(idx).read() } {
713            sink.flush();
714        }
715    }
716    let ptr = SINKS_PTR.load(Ordering::Acquire);
717    if ptr.is_null() {
718        return;
719    }
720    // SAFETY: same invariant as `dispatch_to_sinks`.
721    let sinks: &'static [&'static dyn LogSink] = unsafe { *ptr };
722    for sink in sinks {
723        sink.flush();
724    }
725}

727// -----------------------------------------------------------------------------
728// Recursion guard — process-global single AtomicBool.
729//
730// Granularity is intentionally coarse (process-wide, not per-thread).
731// The guard exists to break a sink that triggers log() during write
732// — not to serialize concurrent loggers across threads. A thread
733// re-entering through its own sink loses its other in-flight
734// sinks for that call; a different thread logging concurrently is
735// also short-circuited momentarily. This is acceptable: the alt is
736// per-thread storage which doesn't exist uniformly across our
737// `no_std` targets (`thread_local!` requires `std`).
738// -----------------------------------------------------------------------------
739
740use portable_atomic::AtomicBool;
741static RECURSION_GUARD: AtomicBool = AtomicBool::new(false);
742
743fn recursion_guard_check_and_set() -> bool {
744    RECURSION_GUARD
745        .compare_exchange(false, true, Ordering::Acquire, Ordering::Acquire)
746        .is_err()
747}
748
749fn recursion_guard_clear() {
750    RECURSION_GUARD.store(false, Ordering::Release);
751}
