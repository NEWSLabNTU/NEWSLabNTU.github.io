1//! phase-417 W4.d — bounded storage so `get_or_create_logger` can hand out a
2//! logger for a name nobody registered at compile time.
3//!
4//! ## The gap this closes
5//!
6//! [`crate::get_logger`] is a LOOKUP: it answers with
7//! [`crate::DEFAULT_LOGGER`] for any name no `'static` [`crate::Logger`] was
8//! [`crate::register_logger`]ed under. That is correct for the Rust call sites
9//! it was written for, which declare their logger as a `static`. It is the
10//! wrong answer for a wrapper: `rclcpp::get_logger("x")` and
11//! `rcutils_get_logger("x")` CREATE, so a C caller doing
12//!
13//! ```c
14//! nros_logger_set_level(nros_log_get_logger("nav"), NROS_LOG_SEVERITY_DEBUG);
15//! ```
16//!
17//! over a lookup-only implementation would silently have moved the threshold
18//! of the catch-all logger every other unregistered name also resolves to —
19//! compile, and differ. RFC-0089 forbids exactly that, so the creation has to
20//! be real, and by RFC-0019 it has to be real HERE rather than in `nros-c`.
21//!
22//! ## Bounded, because `no_std`
23//!
24//! Two static arenas, both sized by the `dynamic-loggers-<N>` feature family:
25//! one for the [`crate::Logger`] values, one for their names (a `&'static str`
26//! is what `Logger` holds, and a `const char *` from C is not one). Exhaustion
27//! is REPORTED, never silently aliased onto the default logger — see
28//! [`crate::get_or_create_logger`].
29
30use core::{cell::UnsafeCell, mem::MaybeUninit};
31
32use portable_atomic::{AtomicUsize, Ordering};
33
34use crate::Logger;
35
36/// How many loggers [`crate::get_or_create_logger`] may create.
37///
38/// Picked by the `dynamic-loggers-<N>` feature family; `dynamic-loggers-0`
39/// declines the arena entirely (creation always fails, lookup still works),
40/// which is the right build for an image whose loggers are all `static`.
41#[must_use]
42pub const fn dynamic_logger_capacity() -> usize {
43    if cfg!(feature = "dynamic-loggers-0") {
44        0
45    } else if cfg!(feature = "dynamic-loggers-8") {
46        8
47    } else if cfg!(feature = "dynamic-loggers-32") {
48        32
49    } else {
50        16
51    }
52}
53
54/// Longest logger name the arena will accept, in bytes.
55///
56/// A cap per NAME as well as in total, so one absurd name cannot consume the
57/// budget every other logger in the image needs.
58pub const MAX_LOGGER_NAME_LEN: usize = 48;
59
60const CAPACITY: usize = dynamic_logger_capacity();
61const NAME_ARENA_BYTES: usize = CAPACITY * 24;
62
63struct Pool {
64    loggers: UnsafeCell<[MaybeUninit<Logger>; CAPACITY]>,
65    loggers_used: AtomicUsize,
66    names: UnsafeCell<[u8; NAME_ARENA_BYTES]>,
67    names_used: AtomicUsize,
68}
69
70// SAFETY: every mutation reserves its region first with a `compare_exchange`
71// on the matching `*_used` counter, so no two writers ever touch the same
72// bytes; a region is written once and never rewritten or freed, so a reader
73// holding a `&'static` into it can never observe a change. Publication to
74// OTHER threads happens through the intern table's `AcqRel` CAS in
75// `crate::get_or_create_logger`, which is the only way another thread can
76// reach one of these references.
77unsafe impl Sync for Pool {}
78
79static POOL: Pool = Pool {
80    loggers: UnsafeCell::new([const { MaybeUninit::uninit() }; CAPACITY]),
81    loggers_used: AtomicUsize::new(0),
82    names: UnsafeCell::new([0_u8; NAME_ARENA_BYTES]),
83    names_used: AtomicUsize::new(0),
84};
85
86/// Copy `name` into the static name arena and return a `&'static str` over the
87/// copy. `None` if the name is too long or the arena is exhausted.
88pub(crate) fn intern_name(name: &str) -> Option<&'static str> {
89    let len = name.len();
90    if len == 0 || len > MAX_LOGGER_NAME_LEN || len > NAME_ARENA_BYTES {
91        return None;
92    }
93    loop {
94        let used = POOL.names_used.load(Ordering::Acquire);
95        let end = used.checked_add(len)?;
96        if end > NAME_ARENA_BYTES {
97            return None;
98        }
99        if POOL
100            .names_used
101            .compare_exchange(used, end, Ordering::AcqRel, Ordering::Acquire)
102            .is_err()
103        {
104            continue;
105        }
106        // SAFETY: `[used, end)` is reserved to this call by the CAS above and
107        // is never written again. The arena is a `'static`, so the slice we
108        // build over it lives for the program.
109        let interned: &'static str = unsafe {
110            let base = POOL.names.get().cast::<u8>();
111            core::ptr::copy_nonoverlapping(name.as_ptr(), base.add(used), len);
112            core::str::from_utf8_unchecked(core::slice::from_raw_parts(base.add(used), len))
113        };
114        return Some(interned);
115    }
116}

118/// Move `logger` into the static logger arena and return a `&'static` to it.
119/// `None` when the arena is exhausted.
120pub(crate) fn place(logger: Logger) -> Option<&'static Logger> {
121    loop {
122        let used = POOL.loggers_used.load(Ordering::Acquire);
123        if used >= CAPACITY {
124            return None;
125        }
126        if POOL
127            .loggers_used
128            .compare_exchange(used, used + 1, Ordering::AcqRel, Ordering::Acquire)
129            .is_err()
130        {
131            continue;
132        }
133        // SAFETY: slot `used` is reserved to this call by the CAS above, is
134        // uninitialised until this write, and is never written again.
135        let placed: &'static Logger = unsafe {
136            let base = POOL.loggers.get().cast::<MaybeUninit<Logger>>();
137            let slot = base.add(used);
138            slot.write(MaybeUninit::new(logger));
139            &*slot.cast::<Logger>()
140        };
141        return Some(placed);
142    }
143}

145/// How many of the [`dynamic_logger_capacity`] slots are spent.
146///
147/// Counts slots CLAIMED, which is one more than the loggers reachable by name
148/// whenever two threads raced the same new name: the loser's slot is spent and
149/// unreferenced. Bounded and reported rather than reclaimed — this arena has
150/// no free list by construction.
151#[must_use]
152pub fn dynamic_loggers_in_use() -> usize {
153    POOL.loggers_used.load(Ordering::Acquire)
154}
155
156/// Bytes of the name arena spent, and its total size.
157#[must_use]
158pub fn dynamic_logger_name_arena() -> (usize, usize) {
159    (POOL.names_used.load(Ordering::Acquire), NAME_ARENA_BYTES)
160}
