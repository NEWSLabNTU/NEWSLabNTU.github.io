1//! Phase 212.M.5.a.2 — the executor-backed component runtime for nano-ros.
2//!
3//! [`MetadataRecorder`] (the planner sink) records what a [`Component`] /
4//! [`ExecutableNode`] declares. This module is the
5//! missing twin: it binds the same traits to a live [`Executor`] so
6//! a Node pkg can actually run — nodes, publishers,
7//! subscriptions, timers materialise as real executor handles, and
8//! every fired callback dispatches into
9//! [`ExecutableNode::on_callback`] with the right
10//! [`CallbackId`].
11//!
12//! Shape:
13//!
14//! ```ignore
15//! use nros::{Executor, ExecutorConfig};
16//! use nros::node_runtime::ExecutorNodeRuntime;
17//!
18//! let cfg = ExecutorConfig::from_env().node_name("talker_main");
19//! let executor = Executor::open(&cfg).unwrap();
20//! let mut runtime = ExecutorNodeRuntime::from_executor(executor);
21//! let _handle = runtime.register_node::<Talker>().unwrap();
22//! runtime.spin().unwrap();
23//! ```
24//!
25//! Owned-spin / board consumer: the macro-emitted `<pkg>::register(runtime)`
26//! wrapper (Phase 258, Track 2) installs each Node onto the runtime's executor
27//! through the uniform `install_node_typed` seam — the same seam the C/C++
28//! typed entries use. The codegen `run_plan(runtime)` body + `nros::main!`
29//! owned-spin loop drive one `register(runtime)?` per launch-XML `<node>`,
30//! then `runtime.spin()`. (The retired Phase 212.M.5.a four-fn-ptr BSP-baker
31//! ABI — ~~`register_dispatch_slot`~~ / ~~`nros_run_components`~~ — is gone.)
32//!
33//! ## Coverage today (Phase 212.M.5.a.2)
34//!
35//! Publishers, subscriptions, and repeating timers wire end-to-end:
36//! the live executor delivers callbacks; the bound
37//! [`ExecutableNode::on_callback`] body runs with a
38//! [`CallbackCtx`] backed by the per-component publisher resolver.
39//! Service servers / clients and action servers / clients wire
40//! end-to-end too (Phase 212.M-F.23): `create_entity` registers them on
41//! the executor with C-ABI trampolines that route inbound requests /
42//! goals into the component's `on_callback`, and the tick-time client /
43//! action surface ([`TickCtx`]) is backed by `RuntimeClientDispatch` /
44//! `RuntimeActions` over the live executor. Parameters are still a
45//! follow-up (registration succeeds; param callbacks don't fire yet).

212/// phase-391 W5.3b — per-class slot storage the `nros::node!` macro emits.
213///
214/// One `static` of this type per macro expansion, so `C` is CONCRETE at the
215/// emission site and the storage is sized `size_of::<TypedSlot<C>>()` exactly —
216/// no byte-budget guessing on the FFI install path. The type is public API for
217/// the MACRO EMIT only; it never crosses the C ABI (the trampoline keeps its
218/// `(ptr, ptr, ptr) -> i32` shape), which is what keeps the const-generic
219/// parameter legal under the "no const generics on FFI-visible types" rule.
220///
221/// `N` is the per-class INSTANCE cap: the launch path bakes one identity per
222/// plan node and can name the same class twice, so this is an array, not one
223/// slot. Taking past `N` is a registration error (`take` returns `None`), the
224/// same Full shape as the executor's node table. Slots are one-shot — `next`
225/// never rewinds — so storage is never re-initialised under a stale reference.
226pub struct ComponentSlotStorage<
227    C: ExecutableNode,
228    const N: usize = { crate::config::MAX_CLASS_INSTANCES },
229    const PUBS: usize = { crate::config::MAX_CELL_ENTITIES },
230    const SVCS: usize = { crate::config::MAX_CELL_ENTITIES },
231    const ACTC: usize = { crate::config::MAX_CELL_ENTITIES },
232    const ACTS: usize = { crate::config::MAX_CELL_ENTITIES },
233    const SSRV: usize = { crate::config::MAX_CELL_ENTITIES },
234    const TIMS: usize = { crate::config::MAX_CELL_ENTITIES },
235> {
236    slots: [UnsafeCell<MaybeUninit<TypedSlot<C>>>; N],
237    /// W5-endgame step 2b (issue 0857) — the per-instance CELL lives here too,
238    /// so the whole component (state + registries) is `.bss`, not heap. Handed
239    /// out pairwise with its slot by `take`; the cell's `slot` field then
240    /// borrows the sibling region, which is sound because the two arrays are
241    /// distinct `UnsafeCell`s claimed by the same monotonic index.
242    cells: [UnsafeCell<MaybeUninit<ComponentCell<PUBS, SVCS, ACTC, ACTS, SSRV, TIMS>>>; N],
243    next: AtomicUsize,
244}

261impl<
262    C: ExecutableNode,
263    const N: usize,
264    const PUBS: usize,
265    const SVCS: usize,
266    const ACTC: usize,
267    const ACTS: usize,
268    const SSRV: usize,
269    const TIMS: usize,
270> ComponentSlotStorage<C, N, PUBS, SVCS, ACTC, ACTS, SSRV, TIMS>
271{
272    /// Const constructor — the macro emits `static STORE: ... = ...::new();`.
273    #[allow(clippy::new_without_default)]
274    pub const fn new() -> Self {
275        Self {
276            slots: [const { UnsafeCell::new(MaybeUninit::uninit()) }; N],
277            cells: [const { UnsafeCell::new(MaybeUninit::uninit()) }; N],
278            next: AtomicUsize::new(0),
279        }
280    }
281
282    /// Hand out the next uninitialised slot, or `None` past the instance cap.
283    ///
284    /// `clippy::mut_from_ref` fires because a `&mut` derived from a `&self` is
285    /// usually unsound. It is not here, and the reason is already written down
286    /// on the `unsafe impl Sync` above: the monotonic `fetch_add` hands each
287    /// index out at most once, so no two `&mut` to one `UnsafeCell` can coexist,
288    /// and `i >= N` bounds it. Allowed rather than restructured — the lint
289    /// cannot see the atomic, and the invariant is the point of the type.
290    #[allow(clippy::mut_from_ref)]
291    fn take(
292        &'static self,
293    ) -> Option<(
294        &'static mut MaybeUninit<TypedSlot<C>>,
295        &'static mut MaybeUninit<ComponentCell<PUBS, SVCS, ACTC, ACTS, SSRV, TIMS>>,
296    )> {
297        let i = self.next.fetch_add(1, Ordering::Relaxed);
298        if i >= N {
299            return None;
300        }
301        // SAFETY: `i` was claimed exactly once by the fetch_add above, so these
302        // are the only references that will ever exist to `slots[i]`'s /
303        // `cells[i]`'s contents; the storage is a `static`, so `'static` is
304        // honest.
305        Some(unsafe { (&mut *self.slots[i].get(), &mut *self.cells[i].get()) })
306    }
307}

348/// phase-391 W5-endgame step 2b (issue 0857) — the NON-GENERIC head of every
349/// `ComponentCell<..>`, guaranteed FIRST FIELD by the cell's `#[repr(C)]`.
350///
351/// The executor's enrolled-component state is a thin pointer to a cell whose
352/// const parameters only the enrolling monomorphization knows. The tick/drop
353/// trampolines are monomorphized alongside it and cast back to the concrete
354/// type; the one consumer that iterates ALL enrolled states regardless of
355/// class — the dispatch-stats fold — casts to this header instead. Both casts
356/// are sound because the header is the first field of a `repr(C)` struct, so
357/// a cell pointer IS a header pointer.
358struct CellHeader {
359    callback_dispatches: AtomicUsize,
360    message_dispatches: AtomicUsize,
361}
362
363/// Issue 1865 — the NON-GENERIC prefix of every `ComponentCell<..>`: its
364/// header and its slot, the two fields that precede every const-generic one.
365///
366/// [`ExecutorNodeRuntime::dispatch_callback`] must reach the component state
367/// of an ENROLLED cell (the install seam's) to hand it a deferred callback,
368/// and like the stats fold it cannot name the cell's const parameters. Under
369/// `#[repr(C)]` a field's offset depends only on the fields before it, so a
370/// `repr(C)` struct that repeats the first two fields has their offsets —
371/// asserted below rather than trusted, because reordering the cell's fields
372/// would otherwise turn this cast into a read of the wrong bytes.
373#[repr(C)]
374struct CellPrefix {
375    header: CellHeader,
376    slot: RefCell<&'static mut dyn ComponentSlot>,
377}
378

380    assert!(
381        core::mem::offset_of!(ComponentCell<0, 0, 0, 0, 0, 0>, header)
382            == core::mem::offset_of!(CellPrefix, header)

384    assert!(
385        core::mem::offset_of!(ComponentCell<0, 0, 0, 0, 0, 0>, slot)
386            == core::mem::offset_of!(CellPrefix, slot)

388    assert!(
389        core::mem::offset_of!(ComponentCell<1, 1, 1, 1, 1, 1>, slot)

415    /// phase-391 W5.3b — BORROWED from per-class or pool storage, not boxed.

419    publishers: RefCell<heapless::Vec<(IdStr, EmbeddedRawPublisher), PUBS>>,

472impl<
473    const PUBS: usize,
474    const SVCS: usize,
475    const ACTC: usize,
476    const ACTS: usize,
477    const SSRV: usize,
478    const TIMS: usize,
479> Drop for ComponentCell<PUBS, SVCS, ACTC, ACTS, SSRV, TIMS>
480{
481    fn drop(&mut self) {
482        // phase-391 W5.3b — the slot is BORROWED from one-shot storage, so the
483        // component state's destructor no longer rides a `Box` drop. Run it
484        // here instead: the cell drops exactly once (the executor's drop
485        // trampoline runs it in place on the enrolled path; the dynamic
486        // path's `Arc` drops it on runtime drop), and the
487        // storage slot is never handed out again (`take`/`next_slot` counters
488        // are monotonic), so nothing can observe the dropped bytes.
489        // W5-endgame ctx slabs — run the placed ctxs' destructors (their
490        // `CellHandle` may hold an `Arc` on the pooled arm). Monotonic
491        // counters bound the initialized prefix exactly.
492        for i in 0..self.svc_ctxs_used.get() {
493            // SAFETY: entries [0, used) were initialized by `place_service_ctx`
494            // and are dropped exactly once, here.
495            unsafe { (*self.svc_ctxs[i].get()).assume_init_drop() };
496        }
497        for i in 0..self.act_srv_ctxs_used.get() {
498            // SAFETY: as above, for `place_action_server_ctx`.
499            unsafe { (*self.act_srv_ctxs[i].get()).assume_init_drop() };
500        }
501        for i in 0..self.act_cli_ctxs_used.get() {
502            // SAFETY: as above, for `place_action_client_ctx`.
503            unsafe { (*self.act_cli_ctxs[i].get()).assume_init_drop() };
504        }
505        let slot: &mut &'static mut dyn ComponentSlot = self.slot.get_mut();
506        let p: *mut dyn ComponentSlot = &raw mut **slot;
507        // SAFETY: `p` targets storage uniquely owned by this cell (handed out
508        // at most once), reachable only through the reference being dropped
509        // with us; it is dropped at most once because the cell is.
510        unsafe { core::ptr::drop_in_place(p) };
511    }
512}

540/// phase-391 W5-endgame step 1 (issue 0857) — the NON-GENERIC view every
541/// dispatch/tick path consumes instead of the concrete [`ComponentCell`].
542///
543/// Why a trait and not the cell: the endgame emits a per-class cell whose
544/// registries are sized to the class's DECLARED entity counts (const generics,
545/// which the "never public to other languages" rule confines to the macro
546/// emission), while the dynamic `register_node` path keeps pool-backed cells at
547/// the knob caps. Two cell layouts, ONE dispatch implementation — this trait is
548/// the seam that makes the split affordable. `Ref<'_, [T]>` erases the
549/// `heapless::Vec` capacity; `try_with_slot_mut` erases the slot's storage
550/// shape (borrowed `&'static mut dyn` today, fused inline in the per-class
551/// static tomorrow).
552///
553/// Object-safe on purpose: the FFI trampolines carry a THIN `*mut c_void`, so
554/// each concrete cell type casts back to itself and only then widens to
555/// `&dyn CellView`.

560    /// one. Holds the registry's shared borrow for the returned `Ref`'s life.

562    fn service_clients(&self) -> Ref<'_, [(IdStr, crate::HandleId)]>;
563    fn action_clients(&self) -> Ref<'_, [(IdStr, usize)]>;
564    fn action_servers(&self) -> Ref<'_, [(IdStr, crate::internals::ActionServerRawHandle)]>;
565    /// Run `f` over the component slot unless it is already mutably borrowed
566    /// (a re-entrant dispatch on the same cell) — in which case the dispatch
567    /// is dropped, exactly as the pre-trait `try_borrow_mut` spelled it.
568    fn try_with_slot_mut(&self, f: &mut dyn FnMut(&mut dyn ComponentSlot));
569    /// The executor's parameter store, or `None` until `apply_param_services`
570    /// threads it in.

573    /// phase-426 W3 — the node key this component's parameter reads are about.
574    /// Travels with the store, because a store keyed by node and a reader that
575    /// cannot name one is how every component read the primary's values.

577    fn view_param_node(&self) -> nros_params::NodeKey;
578    /// The `(callbacks, messages)` dispatch counters. Only the dynamic
579    /// runtime's stats fold reads them through the trait; the macro path's
580    /// fold reads the `CellHeader` directly.
581    #[cfg(feature = "alloc")]

583    /// Registration-time registry appends. `Err(())` = registry full — the
584    /// component declared more entities of that kind than its cell's bound,
585    /// which the caller reports loudly (never a silent drop).
586    fn push_publisher(&self, id: IdStr, handle: EmbeddedRawPublisher) -> Result<(), ()>;
587    fn push_service_client(&self, id: IdStr, handle: crate::HandleId) -> Result<(), ()>;
588    fn push_action_client(&self, id: IdStr, entry_index: usize) -> Result<(), ()>;

597    /// `register()` creates). The entry lives — and is dropped — with the cell.
598    fn place_service_ctx(&self, ctx: ServiceServerCtx) -> Result<*mut core::ffi::c_void, ()>;
599    fn place_action_server_ctx(&self, ctx: ActionServerCtx) -> Result<*mut core::ffi::c_void, ()>;
600    fn place_action_client_ctx(&self, ctx: ActionClientCtx) -> Result<*mut core::ffi::c_void, ()>;
601    /// Issue 2075 — file a declared timer's handle under its id's key.
602    /// `Ok(true)` = filed; `Ok(false)` = the table is full, so the timer runs
603    /// but cannot be controlled by id; `Err(())` = another timer of this

752        }

847/// Issue 2075 — apply the timer ops this cell's contexts requested, against
848/// the executor. Runs on both sides of the component's tick: before it for
849/// the ops callbacks requested during the pass's dispatch, after it for those
850/// the tick itself requested. The executor's own `reset_timer` /
851/// `cancel_timer` are the ops (a reset re-seeds a clock timer's reading, so
852/// a timer declared stopped starts a FRESH period). A timer already released
853/// answers an error, which is dropped: there is nothing left to control.
854fn apply_timer_ops(cell: &dyn CellView, exec_ptr: *mut Executor<'static>) {
855    cell.drain_timer_ops(&mut |handle, op| {
856        // SAFETY: as `tick_one_cell`'s other executor uses: no callback is
857        // running and nothing else holds the executor between dispatches.
858        let executor = unsafe { &mut *exec_ptr };
859        let _ = match op {
860            TimerOp::Reset => executor.reset_timer(handle),
861            TimerOp::Cancel => executor.cancel_timer(handle),
862        };
863    });

877/// Owns the [`Executor`] and one slot per registered component. The
878/// register / spin lifecycle:
879///
880/// 1. [`from_executor`](Self::from_executor) wraps an open
881///    [`Executor`].

888///    [`CallbackId`].
889/// 3. [`spin`](Self::spin) / [`spin_once`](Self::spin_once) drive the
890///    executor; between iterations every registered component's
891///    [`ExecutableNode::tick`] runs.

897    /// index (`Executor::registering_level` at registration; `None` before a
898    /// level runner set one). Only that level's pass ticks the component.
899    component_levels: Vec<Option<u8>>,

906/// Bump view over the runtime's slot backing. Raw pointer, not a slice, so the
907/// runtime stays movable — the SLOTS never move; they live in the backing.
908#[cfg(feature = "alloc")]
909struct ComponentPool {
910    base: *mut MaybeUninit<u8>,
911    len_bytes: usize,
912    per_slot: usize,
913    capacity: usize,
914    next: usize,
915}
916
917// SAFETY: the pool is confined to the runtime (`&mut self` on every use); the
918// raw pointer targets `'static` backing handed over at construction.
919#[cfg(feature = "alloc")]
920unsafe impl Send for ComponentPool {}
921
922#[cfg(feature = "alloc")]
923impl ComponentPool {
924    /// Next disjoint slot window, or `None` when the pool is full.
925    fn next_slot(&mut self) -> Option<&'static mut [MaybeUninit<u8>]> {
926        if self.next >= self.capacity {
927            return None;
928        }
929        let off = self.next * self.per_slot;
930        debug_assert!(off + self.per_slot <= self.len_bytes);
931        self.next += 1;
932        // SAFETY: `base` points at `'static` backing of `len_bytes`; the
933        // window is in-bounds by the checks above and DISJOINT from every
934        // earlier one because `next` only increases.
935        Some(unsafe { core::slice::from_raw_parts_mut(self.base.add(off), self.per_slot) })
936    }
937}

942    ///
943    /// The convenience constructor — allocates `RuntimeSizing::DEFAULT`'s
944    /// backing once and never frees it, exactly the relationship
945    /// `Executor::from_session` has to `Executor::open_in`. Use
946    /// [`new_in`](Self::new_in) on an image that must not allocate.
947    pub fn from_executor(executor: Executor<'static>) -> Self {
948        let sizing = crate::runtime_storage::RuntimeSizing::DEFAULT;
949        let backing: &'static mut [MaybeUninit<u64>] =
950            Vec::leak(alloc::vec![MaybeUninit::uninit(); sizing.u64_len()]);
951        // SAFETY: freshly leaked — `'static`, uniquely owned, exactly
952        // `u64_len()` words.
953        unsafe { Self::new_in(executor, backing, sizing) }

956    /// Wrap an already-built [`Executor`] over CALLER-SUPPLIED slot storage.
957    ///
958    /// Size `backing` with [`crate::runtime_storage::RuntimeSizing::u64_len`]; a short one panics,
959    /// naming both sizes (fail-loud on every profile — a short backing is
960    /// silent corruption, the `executor::storage::carve` / issue #131 lesson).
961    ///
962    /// # Safety
963    /// `backing` must be uniquely owned by this runtime for its whole life:
964    /// slots carved from it are handed out as `&'static mut`, so aliasing it
965    /// anywhere else is undefined behaviour.
966    pub unsafe fn new_in(
967        executor: Executor<'static>,
968        backing: &'static mut [MaybeUninit<u64>],
969        sizing: crate::runtime_storage::RuntimeSizing,
970    ) -> Self {
971        let need = sizing.u64_len();
972        assert!(
973            backing.len() >= need,
974            "component pool backing too small: {} u64 words < {} required for {} slot(s) \
975             of {} bytes — size it with RuntimeSizing::u64_len()",
976            backing.len(),
977            need,
978            sizing.components,
979            sizing.slot_bytes,
980        );
981        Self {
982            executor,
983            components: Vec::new(),
984            component_levels: Vec::new(),
985            pool: ComponentPool {
986                base: backing.as_mut_ptr() as *mut MaybeUninit<u8>,
987                len_bytes: backing.len() * 8,
988                per_slot: (sizing.slot_bytes.div_ceil(8) * 8).max(1),
989                capacity: sizing.components,
990                next: 0,
991            },
992        }
993    }

1282    /// Phase 216.B.3 / C.3 follow-up — route a signaled callback to
1283    /// every registered component slot.
1284    ///
1285    /// The RTIC (`nros-board-rtic-stm32f4`) and Embassy
1286    /// (`nros-board-embassy-stm32f4`) dispatch tasks dequeue a
1287    /// [`nros_platform::SignaledCallback`] envelope from their SPSC
1288    /// queue / Embassy channel and need a routing entry point that
1289    /// hands the callback off to the right Node's `on_callback`
1290    /// trampoline. This method is that entry point.
1291    ///
1292    /// # Strategy — linear scan
1293    ///
1294    /// Each registered slot dispatches through the codegen-emitted
1295    /// `d()` trampoline from `nros::node!()` (a ~~`dispatch_fn`~~ pointer once;
1296    /// see
1297    /// `packages/core/nros-macros/src/lib.rs`). That trampoline calls
1298    /// `<NodeTy as ExecutableNode>::on_callback`, whose body
1299    /// `match`es on the callback's own tag set
1300    /// (`Subscription` / `Timer` / `Service` / `Action` ids) and is a
1301    /// no-op for non-matching `cb_id`s. So a linear scan across every
1302    /// slot is correct — each slot self-filters and at most one
1303    /// component actually acts on a given `cb_id`. A focused
1304    /// `cb_id → slot` index is a separate follow-up; the trampoline's
1305    /// tag dispatch already gates the real work cheaply (string
1306    /// compare on statically known literals), so the linear scan is
1307    /// the minimum-viable wiring that closes the conceptual gap left
1308    /// by the B.3 / C.3 skeleton emits.
1309    ///
1310    /// # Borrow semantics
1311    ///
1312    /// Each `ComponentCell`'s slot lives behind a [`RefCell`]; the
1313    /// per-slot dispatch takes `try_borrow_mut` and is a no-op on
1314    /// re-entrancy. The runtime is single-threaded by construction
1315    /// (the dispatch task owns it via `&mut self`), so the borrow
1316    /// always succeeds in normal flow.
1317    ///
1318    /// # Both populations
1319    ///
1320    /// Issue 1865 — a component reaches this runtime one of two ways, and the
1321    /// two are disjoint (see the `observed_callback_counts` fold, issue #140):
1322    /// [`register_node`](Self::register_node) pushes it to `components`, while
1323    /// the install seam every `nros::node!` `register(runtime)` goes through
1324    /// (`install_node_typed*` → `register_node_borrowed`) ENROLLS it in the
1325    /// executor and never pushes. This scanned `components` alone, so since
1326    /// the RTIC entry moved onto `register(runtime)` (phase-289) every signaled
1327    /// callback the RTIC/Embassy dispatch task forwarded reached no component.
1328    pub fn dispatch_callback(&mut self, cb_id: &str, ctx: &mut CallbackCtx<'_>) {
1329        for cell in &self.components {
1330            if let Ok(mut slot) = cell.slot.try_borrow_mut() {
1331                slot.dispatch(cb_id, ctx);
1332            }
1333        }
1334        for state in self.executor.enrolled_component_states() {
1335            // SAFETY: every enrolled state is a placed `ComponentCell` from
1336            // `register_node_borrowed` (the only enroll site), kept alive until
1337            // `Executor::drop`; `CellPrefix` is its layout-checked prefix.
1338            let cell = unsafe { &*(state as *const CellPrefix) };
1339            if let Ok(mut slot) = cell.slot.try_borrow_mut() {
1340                slot.dispatch(cb_id, ctx);
1341            }
1342        }
1343    }

1602    fn observed_callback_counts(&self) -> (usize, usize) {
1603        let direct = self
1604            .components
1605            .iter()
1606            .fold((0, 0), |(callbacks, messages), cell| {
1607                let (c, m) = cell.dispatch_counts();
1608                (callbacks + c, messages + m)
1609            });
1610        // issue #140 — install-seam components (`nros::node!` →
1611        // `install_node_typed*` → `register_node_borrowed`) never enter
1612        // `self.components`; their cells live only as the executor's enrolled
1613        // component slots (leaked `Arc<ComponentCell>`s). Without folding them
1614        // the hosted spin reported callbacks=0 for every macro-baked entry
1615        // (multihost robot2 et al.) while dispatch demonstrably ran. The two
1616        // populations are disjoint by construction: `register_node` pushes to
1617        // `components` and does not enroll; `register_node_borrowed` enrolls
1618        // and does not push.
1619        self.executor
1620            .enrolled_component_states()
1621            .fold(direct, |(callbacks, messages), state| {
1622                // SAFETY: every enrolled state is a leaked `Arc<ComponentCell>`
1623                // from `register_node_borrowed` (the only enroll site); the
1624                // executor keeps it alive until `Executor::drop`, and we only
1625                // read its atomic counters here.
1626                let header = unsafe { &*(state as *const CellHeader) };
1627                (
1628                    callbacks + header.callback_dispatches.load(Ordering::Relaxed),
1629                    messages + header.message_dispatches.load(Ordering::Relaxed),
1630                )
1631            })
1632    }

1699    /// `register` call. That is the condition the flat seed had.
1700    ///
1701    /// A `false` from the store is a refusal only if the name is still absent
1702    /// afterwards: the store also answers `false` for "already declared".
1703    #[cfg(feature = "param-store")]
1704    fn seed_launch_params(
1705        &mut self,
1706        executor: &mut Executor<'static>,
1707        node: nros_node::executor::NodeId,
1708    ) -> NodeResult<()> {
1709        if executor.params().is_none() {
1710            return Ok(());
1711        }
1712        let params = self.launch_params;
1713        for (name, raw) in params {
1714            // phase-446 seam: the value's type is inferred from the launch
1715            // string; a type declared in the contract replaces this call.
1716            let value = infer_param_value(raw);
1717            if !executor.declare_parameter_on(node, name, value)
1718                && executor.get_parameter_on(node, name).is_none()
1719            {
1720                nros_log::log_error!(
1721                    nros_log::get_logger("nros"),
1722                    "launch parameter '{name}' for node index {} was refused by the \
1723                     parameter store",
1724                    node.raw()
1725                );
1726                return Err(NodeDeclError::Runtime);
1727            }
1728        }
1729        Ok(())
1730    }

2723    // `cfg(feature = "std")` arm never delivered.
2724    if !matches!(e, nros_node::NodeError::ExecutorFull) {
2725        nros_log::log_error!(
2726            nros_log::get_logger("nros"),
2727            "node declaration failed — NodeError::{e:?}"

2729    }

2731    // it); without this the collapse left only an opaque `NodeRegister("<pkg>")`

2737    if !matches!(e, nros_node::NodeError::ExecutorFull) {
2738        log::error!("nros: node declaration failed — NodeError::{e:?}");
2739    }
