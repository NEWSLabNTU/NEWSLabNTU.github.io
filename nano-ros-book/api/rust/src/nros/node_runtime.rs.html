1//! Phase 212.M.5.a.2 — Executor-backed `NodeRuntime` /
2//! `DeclaredNodeRuntime` for nano-ros.
3//!
4//! [`MetadataRecorder`](crate::node_metadata::MetadataRecorder)
5//! (the planner sink) binds the
6//! [`Node`](crate::node::Node) /
7//! [`ExecutableNode`](crate::node::ExecutableNode)
8//! traits to a pure metadata target. This module is the missing twin:
9//! it binds the same traits to a live [`Executor`](crate::Executor) so
10//! a Node pkg can actually run — nodes, publishers,
11//! subscriptions, timers materialise as real executor handles, and
12//! every fired callback dispatches into
13//! [`ExecutableNode::on_callback`] with the right
14//! [`CallbackId`].
15//!
16//! Shape:
17//!
18//! ```ignore
19//! use nros::{Executor, ExecutorConfig};
20//! use nros::node_runtime::ExecutorNodeRuntime;
21//!
22//! let cfg = ExecutorConfig::from_env().node_name("talker_main");
23//! let executor = Executor::open(&cfg).unwrap();
24//! let mut runtime = ExecutorNodeRuntime::from_executor(executor);
25//! let _handle = runtime.register_node::<Talker>().unwrap();
26//! runtime.spin().unwrap();
27//! ```
28//!
29//! Owned-spin / board consumer: the macro-emitted `<pkg>::register(runtime)`
30//! wrapper (Phase 258, Track 2) installs each Node onto the runtime's executor
31//! through the uniform `install_node_typed` seam — the same seam the C/C++
32//! typed entries use. The codegen `run_plan(runtime)` body + `nros::main!`
33//! owned-spin loop drive one `register(runtime)?` per launch-XML `<node>`,
34//! then `runtime.spin()`. (The retired Phase 212.M.5.a four-fn-ptr BSP-baker
35//! ABI — `register_dispatch_slot` / `nros_run_components` — is gone.)
36//!
37//! ## Coverage today (Phase 212.M.5.a.2)
38//!
39//! Publishers, subscriptions, and repeating timers wire end-to-end:
40//! the live executor delivers callbacks; the bound
41//! [`ExecutableNode::on_callback`] body runs with a
42//! [`CallbackCtx`] backed by the per-component publisher resolver.
43//! Service servers / clients and action servers / clients wire
44//! end-to-end too (Phase 212.M-F.23): `create_entity` registers them on
45//! the executor with C-ABI trampolines that route inbound requests /
46//! goals into the component's `on_callback`, and the tick-time client /
47//! action surface ([`TickCtx`]) is backed by `RuntimeClientDispatch` /
48//! `RuntimeActions` over the live executor. Parameters are still a
49//! follow-up (registration succeeds; param callbacks don't fire yet).

148/// Shared per-component cell. Subscription / timer closures registered
149/// against the executor hold an `Arc` clone so they can dispatch +
150/// publish back through the resolver.
151struct ComponentCell {
152    slot: RefCell<Box<dyn ComponentSlot>>,
153    publishers: RefCell<Vec<(String, EmbeddedRawPublisher)>>,
154    // Phase 212.M-F.23 — declarative service/action CLIENT + action-SERVER
155    // handles, keyed by stable entity id, resolved during tick dispatch.
156    // Mirror of the orchestration `GenClientDispatch`/`GenActionExec` arrays,
157    // but built at registration time on the single-node runtime. Service- and
158    // action-SERVER request/goal dispatch is owned by the executor (the
159    // trampolines registered in `create_entity`); only the action-server
160    // handle is kept here so the tick can complete goals / publish feedback.
161    service_clients: RefCell<Vec<(String, crate::HandleId)>>,
162    action_clients: RefCell<Vec<(String, usize)>>,
163    action_servers: RefCell<Vec<(String, crate::ActionServerRawHandle)>>,
164    callback_dispatches: AtomicUsize,
165    message_dispatches: AtomicUsize,
166    // Phase 264 W4c — raw pointer to the executor's volatile parameter store, so a
167    // subscription/timer/service/action callback can read `ctx.parameter::<T>(name)`.
168    // The callback closures + leaked trampolines hold only an `Arc<ComponentCell>` (the
169    // executor is unreachable when they fire), so the store address is threaded HERE by
170    // `apply_param_services`' post-pass (mirrors the `run_ticks` disjoint borrow). Null
171    // until param services are registered. Stable for the executor's life: the server
172    // lives in a `Box<ParamState>`, and since phase-382 W2' its SLOTS live in
173    // caller-owned storage it merely borrows — a fixed-length table either way, so a
174    // declare never moves anything this pointer or a stored `&ParameterValue` names.
175    #[cfg(feature = "param-services")]
176    param_server: core::cell::Cell<*const nros_params::ParameterServer<'static>>,
177}
178
179impl ComponentCell {
180    /// Phase 264 W4c — the executor's parameter store, or `None` until
181    /// `apply_param_services` threads it in. The deref is sound: single-threaded
182    /// executor, param services mutate the server only outside callback dispatch.
183    #[cfg(feature = "param-services")]
184    fn param_server(&self) -> Option<&nros_params::ParameterServer<'static>> {
185        let ptr = self.param_server.get();
186        if ptr.is_null() {
187            None
188        } else {
189            // SAFETY: `ptr` is the address of the executor's boxed `ParameterServer`
190            // (stable for the executor's life); param services mutate it before/after
191            // dispatch (`spin.rs` pre/post), never during, so no aliasing `&mut` is live.
192            Some(unsafe { &*ptr })
193        }
194    }
195
196    fn lookup_publisher<R>(
197        &self,
198        entity_id: &str,
199        f: impl FnOnce(&EmbeddedRawPublisher) -> R,
200    ) -> Option<R> {
201        let pubs = self.publishers.borrow();
202        pubs.iter()
203            .find(|(id, _)| id == entity_id)
204            .map(|(_, p)| f(p))
205    }
206}

213impl PublisherResolver for CellResolver<'_> {
214    fn publish_raw(&self, entity_id: &str, data: &[u8]) -> NodeResult<()> {
215        // issue 0736 — the two arms report DIFFERENT failures. A miss on
216        // `lookup_publisher` means this component declared no publisher for the
217        // entity, so nothing ever reached the transport; a `publish_raw` error
218        // means the transport had the sample and refused it. Both used to
219        // return `Runtime`, which is why a console full of "publish FAILED"
220        // could not distinguish a wiring bug from a congested link.
221        self.cell
222            .lookup_publisher(entity_id, |p| {
223                p.publish_raw(data).map_err(|_| NodeDeclError::Runtime)
224            })
225            .unwrap_or(Err(NodeDeclError::UnknownPublisher))
226    }

238/// Executor-backed component runtime.
239///
240/// Owns the [`Executor`] and one slot per registered component. The
241/// register / spin lifecycle:
242///
243/// 1. [`from_executor`](Self::from_executor) wraps an open
244///    [`Executor`].
245/// 2. [`register_node`](Self::register_node) builds the
246///    component's `State`, runs [`Node::register`](crate::node::Node::register) over an
247///    internal [`NodeRuntime`] adapter that materialises nodes /
248///    pubs / subs / timers on the real executor, and wires each
249///    subscription + timer callback to dispatch into
250///    [`ExecutableNode::on_callback`] with the right
251///    [`CallbackId`].
252/// 3. [`spin`](Self::spin) / [`spin_once`](Self::spin_once) drive the
253///    executor; between iterations every registered component's
254///    [`ExecutableNode::tick`] runs.
255pub struct ExecutorNodeRuntime {
256    executor: Executor<'static>,
257    components: Vec<Arc<ComponentCell>>,
258}

282    /// RFC-0052 / phase-296 W5.4 — lower a tier's RTOS-agnostic scheduling
283    /// policy onto this executor's DEFAULT scheduling context. One `Executor`
284    /// per tier means "the tier's policy" == "this executor's default SC";
285    /// per-group/per-handle bindings still take precedence.
286    ///
287    /// **Portable** across every board — call from each board's `run_tiers`
288    /// after building the runtime. Takes the tier fields as primitives (not a
289    /// `TierSpec`) so `nros` needs no board/platform dependency; a board passes
290    /// `tier.class`, `tier.period_us`, … straight through.
291    ///
292    /// `real_time` + `budget_us` + `period_us` → [`SchedClass::Sporadic`](crate::SchedClass::Sporadic);
293    /// `best_effort` → [`SchedClass::BestEffort`](crate::SchedClass::BestEffort); `time_triggered` +
294    /// `period_us` → the cyclic dispatcher (major frame = period, window =
295    /// `budget_us` or the whole frame); `deadline_us` sets the SC deadline and
296    /// `deadline_policy` its action. A tier with no class/budget/deadline
297    /// leaves the default `Fifo` SC untouched (byte-identical pre-W3 behavior).
298    pub fn apply_tier_sched_policy(
299        &mut self,
300        class: Option<&str>,
301        period_us: Option<u64>,
302        budget_us: Option<u64>,
303        deadline_us: Option<u64>,
304        deadline_policy: Option<&str>,
305    ) {
306        use crate::SchedContext;
307        // Common backend (RFC-0052): the tier→SchedContext lowering lives ONCE
308        // in `SchedContext::from_tier_policy`, shared with the C / C++ entries
309        // (`nros_{c,cpp}_create_sched_context_from_policy`) so the mapping never
310        // drifts between languages. `None` → keep the default `Fifo` SC.
311        let Some((sc, tt_frame)) = SchedContext::from_tier_policy(
312            class,
313            period_us,
314            budget_us,
315            deadline_us,

318            return;
319        };
320        if let Some(frame_us) = tt_frame {
321            self.executor_mut()
322                .register_time_triggered_dispatcher(frame_us);
323        }
324        self.executor_mut().set_default_sched_context(sc);
325    }

332    /// Register a [`Node`](crate::node::Node) (which must also be
333    /// [`ExecutableNode`]) into this runtime. Builds the
334    /// component's `State` (via [`ExecutableNode::init`]) and
335    /// walks [`Node::register`](crate::node::Node::register) over the live executor — every
336    /// declared node / pub / sub / timer materialises as a real
337    /// executor handle, and subscription + timer callbacks are wired
338    /// to dispatch into [`ExecutableNode::on_callback`].
339    pub fn register_node<C: ExecutableNode + 'static>(&mut self) -> NodeResult<RegisteredNode<C>>
340    where
341        C::State: 'static,
342    {
343        let cell = Arc::new(ComponentCell {
344            slot: RefCell::new(Box::new(TypedSlot::<C> {
345                state: C::init(),
346                _phantom: PhantomData,
347            })),
348            publishers: RefCell::new(Vec::new()),
349            service_clients: RefCell::new(Vec::new()),
350            action_clients: RefCell::new(Vec::new()),
351            action_servers: RefCell::new(Vec::new()),
352            callback_dispatches: AtomicUsize::new(0),
353            message_dispatches: AtomicUsize::new(0),
354            // W4c — set by `apply_param_services` once the store exists.
355            #[cfg(feature = "param-services")]
356            param_server: core::cell::Cell::new(core::ptr::null()),
357        });
358        let component_idx = self.components.len();
359        self.components.push(cell.clone());
360
361        let mut sink = ExecutorSink {
362            executor: &mut self.executor,
363            cell: cell.clone(),
364            nodes: Vec::new(),
365            node_identity: None, // direct API — no launch injection
366            remaps: &[],         // direct API — no launch remaps
367            qos_overrides: &[],  // direct API — no plan overrides
368        };
369        let sink_dyn: &mut dyn NodeRuntime = &mut sink;
370        let mut context = NodeContext::new(C::NAME, sink_dyn);
371        let result = C::register(&mut context);
372        if result.is_err() {
373            // Roll back the slot push so `component_count` stays
374            // consistent with what users observe.
375            self.components.pop();
376        }
377        result?;
378
379        // W4c — capture the executor's volatile param store on the cell (if param
380        // services were registered before this call) so the node's callbacks can read
381        // `ctx.parameter::<T>(name)`. Mirrors the install-seam path.
382        #[cfg(feature = "param-services")]
383        if let Some(server) = self.executor.params() {
384            cell.param_server.set(server as *const _);
385        }
386
387        Ok(RegisteredNode {
388            component_idx,
389            _phantom: PhantomData,
390        })
391    }

423    /// Phase 216.B.3 / C.3 follow-up — route a signaled callback to
424    /// every registered component slot.
425    ///
426    /// The RTIC (`nros-board-rtic-stm32f4`) and Embassy
427    /// (`nros-board-embassy-stm32f4`) dispatch tasks dequeue a
428    /// [`nros_platform::SignaledCallback`] envelope from their SPSC
429    /// queue / Embassy channel and need a routing entry point that
430    /// hands the callback off to the right Node's `on_callback`
431    /// trampoline. This method is that entry point.
432    ///
433    /// # Strategy — linear scan
434    ///
435    /// Each registered slot's `dispatch_fn` is the codegen-emitted
436    /// `d()` trampoline from `nros::node!()` (see
437    /// `packages/core/nros-macros/src/lib.rs`). That trampoline calls
438    /// `<NodeTy as ExecutableNode>::on_callback`, whose body
439    /// `match`es on the callback's own tag set
440    /// (`Subscription` / `Timer` / `Service` / `Action` ids) and is a
441    /// no-op for non-matching `cb_id`s. So a linear scan across every
442    /// slot is correct — each slot self-filters and at most one
443    /// component actually acts on a given `cb_id`. A focused
444    /// `cb_id → slot` index is a separate follow-up; the trampoline's
445    /// tag dispatch already gates the real work cheaply (string
446    /// compare on statically known literals), so the linear scan is
447    /// the minimum-viable wiring that closes the conceptual gap left
448    /// by the B.3 / C.3 skeleton emits.
449    ///
450    /// # Borrow semantics
451    ///
452    /// Each `ComponentCell`'s slot lives behind a [`RefCell`]; the
453    /// per-slot dispatch takes `try_borrow_mut` and is a no-op on
454    /// re-entrancy. The runtime is single-threaded by construction
455    /// (the dispatch task owns it via `&mut self`), so the borrow
456    /// always succeeds in normal flow.
457    pub fn dispatch_callback(&mut self, cb_id: &str, ctx: &mut CallbackCtx<'_>) {
458        for cell in &self.components {
459            if let Ok(mut slot) = cell.slot.try_borrow_mut() {
460                slot.dispatch(cb_id, ctx);
461            }
462        }
463    }

465    /// Spin until the executor's halt flag is raised.
466    ///
467    /// phase-359 W10 — this used to be `#[cfg(feature = "std")]` and described
468    /// itself as hosted-only, but the body is `Duration` + `spin_once` +
469    /// `is_halted`, none of which need an OS. The gate was describing a
470    /// CONVENTION (a BSP usually wants its own loop so it can interleave
471    /// board work) as if it were a requirement, and a bare-metal image that
472    /// wants exactly this loop had to hand-roll it to get it.
473    ///
474    /// It does need `alloc`: the halt flag is an `Arc`, so a core-only image
475    /// has no flag to poll.
476    #[cfg(feature = "alloc")]
477    pub fn spin(&mut self) -> Result<(), ExecutorError> {
478        // 10 ms tick cadence — matches the existing executor spin
479        // budgeting (see `Executor::spin_default`); short enough that
480        // component `tick` hooks observe latency under one cycle.
481        let tick = Duration::from_millis(10);
482        while !self.executor.is_halted() {
483            let _ = self.executor.spin_once(tick);
484            self.run_ticks();
485        }
486        Ok(())
487    }

495    fn run_ticks(&mut self) {
496        // Per-component tick — each component's resolver is its own cell.
497        // Phase 212.M-F.23: the tick reaches the executor (service-client
498        // call_raw poll, action-server complete/feedback) through a raw
499        // pointer so `&self.components` and `&mut self.executor` (disjoint
500        // fields) can be live at once.
501        let exec_ptr: *mut Executor<'static> = &mut self.executor;
502        for cell in &self.components {
503            tick_one_cell(cell.as_ref(), exec_ptr);
504        }
505    }

508/// Phase 258 (Track 2, 2a) — drive one component cell's `tick` against the
509/// executor. The single source of truth for the per-component tick body,
510/// shared by [`ExecutorNodeRuntime::run_ticks`] (the owned-runtime path) and
511/// the executor-enrolled [`component_tick_trampoline`] (the `install` path).
512///
513/// `exec_ptr` is reached through a raw `*mut Executor<'static>` so the caller can hold
514/// the component (`&ComponentCell`) and the executor live at once — they are
515/// disjoint, and `RuntimeActions` / `RuntimeClientDispatch` reborrow `&mut`
516/// per call (see their docs).
517fn tick_one_cell(cell: &ComponentCell, exec_ptr: *mut Executor<'static>) {
518    let resolver = CellResolver { cell };
519    let service_clients = cell.service_clients.borrow();
520    let action_clients = cell.action_clients.borrow();
521    let action_servers = cell.action_servers.borrow();
522    let mut actions = RuntimeActions {
523        executor: exec_ptr,
524        handles: &action_servers,
525    };
526    let mut clients = RuntimeClientDispatch {
527        executor: exec_ptr,
528        services: &service_clients,
529        actions: &action_clients,
530    };
531    let mut ctx = TickCtx::new(&resolver, &mut actions, &mut clients);
532    // W4c — `tick` reads `ctx.parameter::<T>(name)` from the store via the cell pointer
533    // (the store is a separate `Box<ParamState>` allocation, so this does NOT alias the
534    // `&mut Executor<'static>` the action/client tick calls reborrow through `exec_ptr`).
535    #[cfg(feature = "param-services")]
536    ctx.set_param_server(cell.param_server());
537    if let Ok(mut slot) = cell.slot.try_borrow_mut() {
538        slot.tick(&mut ctx);
539    }
540}

542/// Phase 258 (Track 2, 2a) — executor `ComponentSlot.tick` trampoline. Casts
543/// the enrolled state back to the component cell + `exec_ctx` back to the
544/// executor and drives one tick. The layering-clean `extern "C"` shim the
545/// `nros-node` [`Executor`] calls each `spin_once` (it can't name `nros`'s
546/// [`ComponentCell`] — see [`register_node_borrowed`]'s enroll).
547///
548/// # Safety
549/// `state` must be the leaked `Arc<ComponentCell>` enrolled via
550/// [`Executor::enroll_component`] from [`register_node_borrowed`] (borrowed
551/// here, **not** consumed); `exec_ctx` must be the live `*mut Executor<'static>` the
552/// executor passes itself.
553unsafe extern "C" fn component_tick_trampoline(
554    state: *mut core::ffi::c_void,
555    exec_ctx: *mut core::ffi::c_void,
556) {
557    // SAFETY: `state` is a live, leaked `Arc<ComponentCell>` ptr (kept alive
558    // until `component_drop_trampoline`); borrow it without taking ownership.
559    let cell = unsafe { &*(state as *const ComponentCell) };
560    tick_one_cell(cell, exec_ctx as *mut Executor<'static>);
561}

564/// Reconstitutes + drops the leaked `Arc<ComponentCell>` enrolled by
565/// [`register_node_borrowed`]. Run exactly once on `Executor::drop`.
566///

591    fn executor_handle(&mut self) -> *mut core::ffi::c_void {
592        // Phase 258 (Track 2, 2a) — hand the owned-spin entry a raw pointer to
593        // the executor this runtime owns, so a Node pkg's `register(runtime)`
594        // can install through `nros::install_node_typed` (same seam as the
595        // C/C++ typed entries). The pointer is valid for the runtime's life
596        // (the executor is an inline field); the install call uses it only
597        // during registration, before any concurrent spin.
598        &mut self.executor as *mut Executor<'static> as *mut core::ffi::c_void
599    }
600
601    // Phase 264 W2 — register the REP-2002 lifecycle services + drive boot
602    // autostart on the owned executor (mirrors `generate.rs::render_lifecycle_fn`).
603    // Only compiled with `lifecycle-services`; without it the trait default no-op
604    // applies, so a `[lifecycle]` block is silently inert (the Entry opts in by
605    // enabling `nros/lifecycle-services`). `nros::main!` calls this when
606    // `system.toml` declares `[lifecycle]`.
607    #[cfg(feature = "lifecycle-services")]
608    fn apply_lifecycle(&mut self, autostart: u8) -> Result<(), &'static str> {
609        // issue 0460 — carry WHY. The caller can only say which capability
610        // failed; without this the whole diagnostic was
611        // `NodeRegister("lifecycle")`.
612        self.executor
613            .register_lifecycle_services()
614            .map_err(|e| capability_reason(&e))?;
615        if autostart >= 1
616            && let Some(sm) = self.executor.lifecycle_state_machine_mut()
617        {
618            // No transition callbacks registered → each transition takes the
619            // default-success path (REP-2002 skeleton), as the bake does.
620            unsafe {
621                let _ = sm.trigger_transition(crate::LifecycleTransition::Configure);
622                if autostart >= 2 {
623                    let _ = sm.trigger_transition(crate::LifecycleTransition::Activate);
624                }
625            }
626        }
627        Ok(())
628    }
629
630    // Phase 264 W4b — register the 6 ROS 2 parameter services on the owned executor
631    // + seed the volatile param store with the launch-baked `<param>` initials
632    // (mirrors `generate.rs::render_param_persistence_fn`, minus persistence). Only
633    // compiled with `param-services`; without it the trait default no-op applies, so a
634    // `[param_services]` block is silently inert (the Entry opts in by enabling
635    // `nros/param-services`). `nros::main!` calls this when `system.toml` declares
636    // `[param_services]`. Reconfigured values (via `ros2 param set`) live in RAM until
637    // the next boot — persistence is out of scope (issue 0080).
638    // W4c note: `nros::main!` emits this BEFORE the per-node `register` calls, so the
639    // store exists when each cell is created — `register_node_borrowed` / `register_node`
640    // then capture the (stable, boxed) `ParameterServer` address on the cell, letting a
641    // callback read `ctx.parameter::<T>(name)`. (The macro-path cells live in the
642    // executor's tick registry, not `self.components`, so a post-pass here wouldn't reach
643    // them — capture-at-registration does.)
644    #[cfg(feature = "param-services")]
645    fn apply_param_services(&mut self, params: &[(&str, &str)]) -> Result<(), &'static str> {
646        self.executor
647            .register_parameter_services()
648            .map_err(|e| capability_reason(&e))?;
649        for (name, raw) in params {
650            self.executor
651                .declare_parameter(name, infer_param_value(raw));
652        }
653        Ok(())
654    }
655
656    fn observed_callback_counts(&self) -> (usize, usize) {
657        let direct = self
658            .components
659            .iter()
660            .fold((0, 0), |(callbacks, messages), cell| {
661                (
662                    callbacks + cell.callback_dispatches.load(Ordering::Relaxed),
663                    messages + cell.message_dispatches.load(Ordering::Relaxed),
664                )
665            });
666        // issue #140 — install-seam components (`nros::node!` →
667        // `install_node_typed*` → `register_node_borrowed`) never enter
668        // `self.components`; their cells live only as the executor's enrolled
669        // component slots (leaked `Arc<ComponentCell>`s). Without folding them
670        // the hosted spin reported callbacks=0 for every macro-baked entry
671        // (multihost robot2 et al.) while dispatch demonstrably ran. The two
672        // populations are disjoint by construction: `register_node` pushes to
673        // `components` and does not enroll; `register_node_borrowed` enrolls
674        // and does not push.
675        self.executor
676            .enrolled_component_states()
677            .fold(direct, |(callbacks, messages), state| {
678                // SAFETY: every enrolled state is a leaked `Arc<ComponentCell>`
679                // from `register_node_borrowed` (the only enroll site); the
680                // executor keeps it alive until `Executor::drop`, and we only
681                // read its atomic counters here.
682                let cell = unsafe { &*(state as *const ComponentCell) };
683                (
684                    callbacks + cell.callback_dispatches.load(Ordering::Relaxed),
685                    messages + cell.message_dispatches.load(Ordering::Relaxed),
686                )
687            })
688    }

1065/// Lower a source-recorded [`ParameterDefault`] into the executor-facing
1066/// [`nros_params::ParameterValue`] used to seed a declared parameter. Scalar
1067/// defaults carry their value directly; the array variants record only the
1068/// declared type (no element data at the source layer) so they seed as
1069/// `NotSet` — the parameter is still declared, just without a concrete array
1070/// default. A `Double` default is stored as a string at the metadata layer and
1071/// parsed here (unparseable → `0.0`).

1074    default: Option<&crate::node_metadata::ParameterDefault>,

1077    use nros_params::ParameterValue;
1078    match default {
1079        None => ParameterValue::NotSet,
1080        Some(ParameterDefault::Bool(b)) => ParameterValue::Bool(*b),
1081        Some(ParameterDefault::Integer(i)) => ParameterValue::Integer(*i),
1082        Some(ParameterDefault::Double(s)) => {
1083            ParameterValue::Double(s.as_str().parse::<f64>().unwrap_or(0.0))
1084        }
1085        Some(ParameterDefault::String(s)) => {
1086            ParameterValue::from_string(s.as_str()).unwrap_or(ParameterValue::NotSet)
1087        }
1088        Some(
1089            ParameterDefault::BoolArray
1090            | ParameterDefault::IntegerArray
1091            | ParameterDefault::DoubleArray
1092            | ParameterDefault::StringArray,
1093        ) => ParameterValue::NotSet,
1094    }

1097/// Phase 264 W4b — infer a [`nros_params::ParameterValue`] from a raw launch
1098/// `<param value=…/>` string. ROS 2 launch `<param>` values are untyped strings; the
1099/// macro path has no per-param type attribute, so infer: `true`/`false` → `Bool`, an
1100/// `i64` literal → `Integer`, an `f64` literal → `Double`, otherwise `String`. This

1120fn dispatch_into_cell(cell: &Arc<ComponentCell>, cb_id: &str, payload: &[u8]) {
1121    cell.callback_dispatches.fetch_add(1, Ordering::Relaxed);
1122    if !payload.is_empty() {
1123        cell.message_dispatches.fetch_add(1, Ordering::Relaxed);
1124    }
1125    let resolver = CellResolver {
1126        cell: cell.as_ref(),
1127    };
1128    let mut ctx = CallbackCtx::new(payload, &resolver);
1129    // W4c — let the callback read `ctx.parameter::<T>(name)` from the executor's store
1130    // (threaded onto the cell by `apply_param_services`; `None` until then).
1131    #[cfg(feature = "param-services")]
1132    ctx.set_param_server(cell.param_server());
1133    // If the slot is already borrowed (a re-entrant publish from a
1134    // tick hook on the same cell, etc.) we drop this dispatch. In
1135    // practice `try_borrow_mut` succeeds because subscription / timer
1136    // callbacks run sequentially under the single-threaded executor.
1137    if let Ok(mut slot) = cell.slot.try_borrow_mut() {
1138        slot.dispatch(cb_id, &mut ctx);
1139    }
1140}

1233/// Service-server request callback: deserialize-side runs in the component's
1234/// `on_callback` via `CallbackCtx::with_reply`; the executor sends the reply

1236unsafe extern "C" fn service_server_trampoline(
1237    req: *const u8,
1238    req_len: usize,
1239    resp: *mut u8,
1240    resp_cap: usize,
1241    resp_len: *mut usize,
1242    ctx: *mut core::ffi::c_void,

1247    let mut written = 0usize;
1248    let resolver = CellResolver {

1256        slot.dispatch(&sctx.callback_id, &mut cb);
1257    }

1350impl ClientDispatch for RuntimeClientDispatch<'_> {
1351    fn call_raw(
1352        &mut self,
1353        service_entity: &str,
1354        request_cdr: &[u8],
1355        response_buf: &mut [u8],
1356    ) -> NodeResult<usize> {
1357        use crate::ClientTrait;
1358        let hid = self.service(service_entity)?;
1359        {
1360            let executor = unsafe { &mut *self.executor };
1361            let entry = unsafe { executor.service_client_entry_mut(hid.0) }
1362                .ok_or(NodeDeclError::Runtime)?;
1363            entry
1364                .handle
1365                .send_request_raw(request_cdr)
1366                .map_err(|_| NodeDeclError::Runtime)?;
1367        }
1368        // Bounded wait — caps total time so the tick loop stays responsive.
1369        for _ in 0..200 {
1370            let executor = unsafe { &mut *self.executor };
1371            executor.spin_once(core::time::Duration::from_millis(10));
1372            let entry = unsafe { executor.service_client_entry_mut(hid.0) }
1373                .ok_or(NodeDeclError::Runtime)?;
1374            match entry.handle.try_recv_reply_raw(response_buf) {
1375                // Issue 0778 — one call in flight on this blocking path, so
1376                // the sequence id is dropped deliberately.
1377                Ok(Some((len, _seq))) => return Ok(len),
1378                Ok(None) => continue,
1379                Err(_) => return Err(NodeDeclError::Runtime),
1380            }
1381        }
1382        Err(NodeDeclError::Runtime)
1383    }
1384
1385    fn send_goal_raw(&mut self, action_entity: &str, goal_cdr: &[u8]) -> NodeResult<GoalId> {
1386        let entry_index = self.action_entry(action_entity)?;
1387        let executor = unsafe { &mut *self.executor };
1388        let core = unsafe { executor.action_client_core_mut(entry_index) }
1389            .ok_or(NodeDeclError::Runtime)?;
1390        let goal_id = core.send_goal_raw(goal_cdr).map_err(decl_err_from_node)?;
1391        // rclcpp-style: request the result immediately. The server queues the
1392        // get_result request until the goal terminates, then replies — the
1393        // executor's spin auto-delivers it to the bound result callback (the
1394        // executor never auto-sends this request, so the declarative client
1395        // must). Best-effort: a transport hiccup just means no result callback.
1396        let _ = core.send_get_result_request(&goal_id);
1397        Ok(goal_id)
1398    }
1399}

1467/// Phase 257 (W0-B) — register an [`ExecutableNode`] `C` against a **borrowed**
1468/// executor (the shared cffi `Executor` a foreign-language typed entry hands in via
1469/// its `nros::global_handle()` / `Node::executor_handle()`), returning the live
1470/// [`ComponentCell`].
1471///
1472/// Unlike [`ExecutorNodeRuntime::register_node`] this owns neither the executor nor a
1473/// components list: the executor's per-entity callbacks hold `Arc<ComponentCell>`
1474/// clones (see [`ExecutorSink`]), so the cell stays alive for the executor's lifetime
1475/// via dispatch alone — the caller may drop the returned cell (pub/sub/timer nodes, the
1476/// W0-B target) or stash it to drive `tick` (service-client/action nodes; phase-257 D2).
1477/// The node self-creates its node (its `Node::NAME`) on the shared executor (phase-257
1478/// D7 Option C — Rust nodes in a foreign entry self-name, no entry-side qos-override).
1479/// Issue 0095 — preserve executor callback-table exhaustion through the
1480/// `NodeError → NodeDeclError` collapse so the register seam (and ultimately
1481/// the `nros::main!` entry) can name `NROS_EXECUTOR_MAX_CBS` instead of an
1482/// opaque `NodeRegister`. Every other `NodeError` stays `Runtime`.
1483fn decl_err_from_node(e: nros_node::NodeError) -> NodeDeclError {
1484    // Issue 0428 — `NodeDeclError::Runtime` collapses every non-`ExecutorFull`
1485    // cause into one opaque `NodeRegister("<pkg>")`, "four collapses away from the
1486    // cause" (node.rs). That is how a Cyclone descriptor/transport failure was
1487    // mis-diagnosed twice. Surface the real variant on the error path (rare — this
1488    // only runs when a declaration is already failing), so the register seam names
1489    // WHAT failed, not just WHERE.
1490    // issue 0589 — `nros_log`, not `std::eprintln!`: std stdio is fatal on Zephyr
1491    // native_sim, and this crate cannot gate on the platform. Routing it here
1492    // also means a `no_std` image gets the diagnostic, which the old
1493    // `cfg(feature = "std")` arm never delivered.
1494    if !matches!(e, nros_node::NodeError::ExecutorFull) {
1495        nros_log::nros_error!(
1496            nros_log::get_logger("nros"),
1497            "node declaration failed — NodeError::{e:?}"
1498        );
1499    }
1500    // no_std targets get the same diagnostic through the `log` facade (every
1501    // RTOS board bridges it); without this the collapse left only an opaque
1502    // `NodeRegister("<pkg>")` on embedded — e.g. a static subscriber-pool
1503    // exhaustion (NROS_RMW_SUBSCRIBER_SLOTS) surfaced with no cause at all.
1504    #[cfg(not(feature = "std"))]
1505    if !matches!(e, nros_node::NodeError::ExecutorFull) {
1506        log::error!("nros: node declaration failed — NodeError::{e:?}");
1507    }
1508    match e {
1509        nros_node::NodeError::ExecutorFull => NodeDeclError::ExecutorFull,
1510        _ => NodeDeclError::Runtime,
1511    }
1512}

1514fn register_node_borrowed<'p, C: ExecutableNode + 'static>(
1515    executor: &mut Executor<'static>,
1516    params: &'p [(&'p str, &'p str)],
1517    node_identity: Option<(&'static str, &'static str)>,
1518    remaps: &'p [(&'p str, &'p str)],

1520) -> NodeResult<Arc<ComponentCell>>
1521where

1523{

1525        slot: RefCell::new(Box::new(TypedSlot::<C> {
1526            state: C::init(),
1527            _phantom: PhantomData,
1528        })),
1529        publishers: RefCell::new(Vec::new()),
1530        service_clients: RefCell::new(Vec::new()),
1531        action_clients: RefCell::new(Vec::new()),
1532        action_servers: RefCell::new(Vec::new()),

1534        message_dispatches: AtomicUsize::new(0),

1537        param_server: core::cell::Cell::new(core::ptr::null()),
1538    });

1540        // Reborrow so `executor` stays usable for `enroll_component` after the
1541        // sink (which holds `&mut Executor<'static>`) is dropped below.

1543        cell: cell.clone(),

1545        // Phase 268 W1 — thread the per-component identity bake (RFC-0046).
1546        node_identity,
1547        // Phase 305 W3 (issue 0255) — thread the per-component remap bake.
1548        remaps,
1549        // Issue #52 — thread the per-component QoS-override bake.
1550        qos_overrides,
1551    };

1553    let mut context = NodeContext::new(C::NAME, sink_dyn);

1555    // them via `NodeContext::param`.
1556    context.set_params(params);
1557    C::register(&mut context)?;
1558
1559    // Phase 258 (Track 2, 2a) — enroll the cell in the executor's component
1560    // tick registry so `install`'d nodes tick (closes phase-257 D2: poll-only
1561    // service-client/action nodes have no callbacks keeping the cell alive AND
1562    // never ran `tick`). Leak one `Arc` clone for the executor to own; it runs
1563    // `tick` each `spin_once` and drops the clone on `Executor::drop`. Harmless
1564    // for pub/sub/timer-only nodes — their tick body is a no-op. Registry-full
1565    // (`MAX_NODES`) reclaims the clone and proceeds without a tick slot.

1567    // SAFETY: `raw` is a freshly-leaked `Arc<ComponentCell>`; the trampolines
1568    // match its provenance (borrow on tick, reclaim on drop).

1570        executor.enroll_component(raw, component_tick_trampoline, component_drop_trampoline)
1571    }
1572    .is_err()
1573    {
1574        // SAFETY: enroll rejected the slot, so the executor never took `raw`;
1575        // reclaim the leaked clone here to avoid a permanent leak.
1576        drop(unsafe { Arc::from_raw(raw as *const ComponentCell) });
1577    }
1578
1579    // W4c — capture the executor's volatile param store on the cell so this node's
1580    // callbacks can read `ctx.parameter::<T>(name)`. Non-null only when `nros::main!`
1581    // emitted `apply_param_services` BEFORE this register call (`[param_services]`
1582    // declared); otherwise the store is absent and the field stays null.
1583    #[cfg(feature = "param-services")]
1584    if let Some(server) = executor.params() {
1585        cell.param_server.set(server as *const _);
1586    }
1587
1588    Ok(cell)
1589}

1656/// All slices must outlive the call (`'static` promoted in the macro emit).
1657///
1658/// # Safety
1659/// `executor` must be the live `*mut Executor<'static>` handle a typed entry passes, valid for the call.
1660pub unsafe fn install_node_typed_with_launch<C: ExecutableNode + 'static>(
1661    executor: *mut core::ffi::c_void,
1662    params: &[(&str, &str)],
1663    node_identity: Option<(&'static str, &'static str)>,
1664    remaps: &[(&str, &str)],
1665    qos_overrides: &'static [nros_node::executor::node_record::QosOverrideCode],
1666) -> i32
1667where
1668    C::State: 'static,
1669{
1670    if executor.is_null() {
1671        return -1;
1672    }
1673    // SAFETY: per the fn contract, `executor` is the live `*mut Executor<'static>` handle.
1674    let exec: &mut Executor<'static> = unsafe { &mut *(executor as *mut Executor<'static>) };
1675    match register_node_borrowed::<C>(exec, params, node_identity, remaps, qos_overrides) {

1677        // Issue 0095 — distinct code for executor-table exhaustion so the macro
1678        // register seam can name `NROS_EXECUTOR_MAX_CBS` instead of an opaque
1679        // `NodeRegister`. Every other failure stays the generic `-1`.
1680        Err(NodeDeclError::ExecutorFull) => -2,
1681        Err(_) => -1,
1682    }
1683}

1732/// issue 0460 — map a `NodeError` to a static reason a capability failure can
