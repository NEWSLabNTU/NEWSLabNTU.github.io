1//! The hosted edge of configuration: the process environment.
2//!
3//! issue 0687 / phase-359 W10. Every env var nano-ros honours is read HERE and
4//! nowhere else. The core (`nros-node`) takes values — `ExecutorConfig::
5//! resolve_with` accepts an [`EnvRung`](nros_node::EnvRung) of already-resolved fields — so it
6//! needs no `std`, no `env` capability, and no stub on the five ports that have
7//! no environment to read.
8//!
9//! That is the difference between this and the ABI ports W10 made for the
10//! clock, sleep, tasks and the log sink: those name capabilities every RTOS
11//! HAS. A process environment is not one. Modelling it as an ABI entry would
12//! have put `return 0` in five ports and permanent surface in the header, for a
13//! facility that exists on one platform family.
14//!
15//! **One reader, through the C library** (phase-359 W10 stage B). The
16//! environment is read with `getenv`, not `std::env`. A process environment is
17//! a property of the PROCESS, not of the Rust standard library: threadx-linux
18//! and freertos-posix are `#![no_std]` images that are still Linux processes
19//! with a C library, and issue 1763 measured what happens when they stop
20//! honouring `$ROS_DOMAIN_ID` (parallel cells shared a domain and heard each
21//! other). There used to be two readers — this one on `std::env` and
22//! `host_env` on `getenv` — with the same variables and rules; they are one
23//! now, and `nros` needs no `std` to be a hosted edge. The edge is "a C library
24//! is linked", which on this tree is `target_os = "linux"`; a cross target has
25//! no `getenv` and this module does not exist there.
26//!
27//! Strings are borrowed from the process environment, which outlives the boot,
28//! so the resolved config's `&'static str` fields point straight into it. No
29//! cache is needed (the old `std::env` reader kept a `OnceLock` only to turn an
30//! owned `String` into a `&'static str`, and its staleness was issue 0607).
31//! Nothing in nano-ros calls `setenv` after boot.
32//!
33//! What lives here:
34//!
35//! * [`ExecutorConfigEnvExt::from_env`] — the constructor that used to be an
36//!   inherent method on `ExecutorConfig`. Bring the trait into scope (the
37//!   prelude has it) and the call site spelling is unchanged. `env` feature.
38//! * [`resolve_hosted`](crate::env::resolve_hosted) / [`try_resolve_hosted`](crate::env::try_resolve_hosted) — RFC-0045 precedence model A
39//!   with the environment rung on top. The C / C++ entries call these on every
40//!   Linux build, `env` or not.
41//! * [`rmw_selector`] — the one `$NROS_RMW` reader (issue 0687's first half).
42//! * `var` / `read_file` (crate-private) — the two C-library reads the hosted `init*`
43//!   entries need (`$RMW_IMPLEMENTATION`, `--params-file`).
44

55/// `$NAME` as a C string, or `None` when unset — the one `getenv` call.
56fn raw(name: &CStr) -> Option<&'static CStr> {
57    // SAFETY: `name` is NUL-terminated; `getenv` returns NULL or a pointer to
58    // a NUL-terminated string in the process environment, which lives for the
59    // process, so a non-NULL result is a valid `'static` C string.
60    unsafe {
61        let p = getenv(name.as_ptr());
62        (!p.is_null()).then(|| CStr::from_ptr(p))
63    }
64}

66/// `$NAME` as UTF-8, or `None` when unset or not UTF-8 (`std::env::var`'s two
67/// `Err` cases, which the old std reader also treated as unset).
68///
69/// `name` is a Rust string; one longer than 63 bytes, or one holding a NUL,
70/// is no variable this tree reads and answers `None`.
71pub(crate) fn var(name: &str) -> Option<&'static str> {
72    let mut buf = [0u8; 64];
73    if name.len() >= buf.len() || name.as_bytes().contains(&0) {
74        return None;
75    }
76    buf[..name.len()].copy_from_slice(name.as_bytes());
77    let cname = CStr::from_bytes_until_nul(&buf).ok()?;
78    raw(cname)?.to_str().ok()
79}

81fn non_empty(v: Option<&'static str>) -> Option<&'static str> {
82    v.filter(|s| !s.is_empty())
83}

85/// A whole file, through the C library — `std::fs::read_to_string` for a
86/// crate with no `std`. `--params-file` is its one caller.
87///
88/// The error is a short reason (`"cannot be opened"`, `"is not UTF-8"`, ...),
89/// for the caller to put beside the path it already holds.
90#[cfg(all(feature = "env", feature = "param-store"))]
91pub(crate) fn read_file(path: &str) -> Result<alloc::string::String, &'static str> {
92    use core::ffi::c_void;
93    unsafe extern "C" {
94        fn fopen(path: *const c_char, mode: *const c_char) -> *mut c_void;
95        fn fread(ptr: *mut c_void, size: usize, n: usize, f: *mut c_void) -> usize;
96        fn ferror(f: *mut c_void) -> i32;
97        fn fclose(f: *mut c_void) -> i32;
98    }
99    let mut cpath = alloc::vec::Vec::with_capacity(path.len() + 1);
100    cpath.extend_from_slice(path.as_bytes());
101    if cpath.contains(&0) {
102        return Err("names a path holding a NUL byte");
103    }
104    cpath.push(0);
105    // SAFETY: both strings are NUL-terminated; `f` is checked before use and
106    // closed exactly once; `fread` writes at most `chunk.len()` bytes.
107    let bytes = unsafe {
108        let f = fopen(cpath.as_ptr().cast(), c"rb".as_ptr());
109        if f.is_null() {
110            return Err("cannot be opened");
111        }
112        let mut out = alloc::vec::Vec::new();
113        let mut chunk = [0u8; 4096];
114        loop {
115            let n = fread(chunk.as_mut_ptr().cast(), 1, chunk.len(), f);
116            out.extend_from_slice(&chunk[..n]);
117            if n < chunk.len() {
118                break;
119            }
120        }
121        let failed = ferror(f) != 0;
122        fclose(f);
123        if failed {
124            return Err("cannot be read");
125        }
126        out
127    };
128    alloc::string::String::from_utf8(bytes).map_err(|_| "is not UTF-8")
129}

131/// **The** answer to "which RMW backend did the user select" — one reader, one
132/// semantic, for every consumer.
133///
134/// phase-359 W10 / issue 0687. This variable had FOUR readers with THREE
135/// semantics: `Executor::open` read it as raw OS bytes, `nros`'s
136/// `open_session` as a UTF-8 string filtered for empty, `nros-c`'s entry as a
137/// string passed through EVEN WHEN EMPTY, and `nros::init` as a string with an
138/// `RMW_IMPLEMENTATION` fallback the other three did not have. "Which backend
139/// did the user ask for" had four answers in one process.
140///
141/// Two decisions are baked in, and both are deliberate:
142///
143/// * **`$NROS_RMW` only.** `$RMW_IMPLEMENTATION` is NOT folded in, though it
144///   was tempting and one reader did it. The two carry different vocabularies:
145///   this selector is matched against the cffi registry's canonical names
146///   (`zenoh`, `dds`, `cyclonedds`), while `RMW_IMPLEMENTATION` holds ROS names
147///   (`rmw_cyclonedds_cpp`). Feeding a ROS name to `resolve_backend` yields
148///   `Unknown` — an ERROR — where today it is ignored and the unique-backend
149///   path runs. Unifying them without a mapping would convert "ignored" into
150///   "fails to start". The hosted `init` keeps its fallback for the
151///   `Context.rmw` HINT, which is a different quantity.
152/// * **Empty or non-UTF-8 means unset.** A name that is not UTF-8 cannot match
153///   a registry entry, so treating it as absent is what the caller wants.
154///
155/// The return is `heapless::String` rather than `&str` because
156/// `RMW_SELECTOR_CAP` is a real bound, not a guess: it is the capacity of
157/// `Executor::primary_rmw_name`, so a longer value cannot name a registry slot
158/// and is reported as unset rather than truncated into a different backend's
159/// name.
160pub fn rmw_selector() -> Option<nros_core::heapless::String<RMW_SELECTOR_CAP>> {
161    let s = non_empty(var("NROS_RMW"))?;
162    nros_core::heapless::String::try_from(s).ok()
163}

165/// The environment as an [`EnvRung`], for [`ExecutorConfig::resolve_with`].
166fn env_rung() -> Result<EnvRung<'static>, BootConfigError> {
167    // phase-484 W8c — a retired alias is REFUSED, never ignored: an ignored
168    // `ZENOH_LOCATOR` dials the default router and reads as "no delivery".
169    // Presence alone refuses (an empty value is still somebody's intent), and
170    // the refusal is logged here so the C / C++ shims, which only get a
171    // return code, still name the replacement. Only presence is tested, so a
172    // non-UTF-8 value still refuses.
173    for &(name, replacement) in RETIRED_ENV_VARS {
174        let mut cname = [0u8; 32];
175        cname[..name.len()].copy_from_slice(name.as_bytes());
176        let cname = CStr::from_bytes_until_nul(&cname).expect("retired env name fits");
177        if raw(cname).is_some() {
178            let err = BootConfigError::RetiredEnvVar { name, replacement };
179            nros_log::log_error!(nros_log::get_logger("nros"), "{}", err);
180            return Err(err);
181        }
182    }
183    // Issue 0330 — a SET locator is the env's answer even when empty; the
184    // active backend substitutes its own default for an empty one.
185    let locator = var("NROS_LOCATOR");
186    let mode = match var("NROS_SESSION_MODE") {
187        Some("peer") => SessionMode::Peer,
188        _ => SessionMode::Client,
189    };
190    let domain_id = match non_empty(var("ROS_DOMAIN_ID")) {
191        // #206 — malformed is an ERROR, not a silent skip. The parse lives
192        // here rather than in the core because only this side has the text.
193        Some(s) => Some(
194            s.trim()
195                .parse::<u32>()
196                .map_err(|_| BootConfigError::DomainIdParse)?,
197        ),
198        None => None,
199    };
200    let rmw = non_empty(var("NROS_RMW")).filter(|s| s.len() <= RMW_SELECTOR_CAP);
201    Ok(EnvRung {
202        locator,
203        domain_id,
204        // Session mode has no baked rung to fall through to, so `Client` is
205        // always the answer when the variable says nothing.
206        mode: Some(mode),
207        // Issue 0794 — an EMPTY name or namespace is not a declaration: `""`
208        // is what an unset one already resolves to.
209        node_name: non_empty(var("NROS_NODE_NAME")),
210        namespace: non_empty(var("NROS_NODE_NAMESPACE")),
211        rmw,
212    })
213}

215/// Which rung answered one field, for the boot line.
216fn source(env: bool, baked: bool) -> &'static str {
217    if env {
218        "env"
219    } else if baked {
220        "baked"
221    } else {
222        "default"
223    }
224}

226/// The boot line: what each identity field resolved to, and FROM WHERE.
227///
228/// A misspelled variable does nothing, and without this line nothing says so:
229/// the image runs on its baked domain and reads as "no delivery". One line,
230/// short enough for `nros_log`'s 256-byte buffer (a body that does not fit is
231/// dropped, not truncated), so the locator is the field that can be elided.
232fn report(env: &EnvRung<'_>, baked: &BootConfig<'_>, cfg: &ExecutorConfig<'_>) {
233    let locator = if cfg.locator.is_empty() {
234        "(backend default)"
235    } else if cfg.locator.len() > 96 {
236        "(long)"
237    } else {
238        cfg.locator
239    };
240    nros_log::log_info!(
241        nros_log::get_logger("nros"),
242        "boot config: domain {} ({}), locator {} ({}), rmw {} ({})",
243        cfg.domain_id,
244        source(env.domain_id.is_some(), baked.domain_id.is_some()),
245        locator,
246        source(env.locator.is_some(), baked.locator.is_some()),
247        cfg.rmw.unwrap_or("(unique)"),
248        source(env.rmw.is_some(), baked.rmw.is_some()),
249    );
250}

252/// RFC-0045 precedence model A with the environment on top:
253/// `env (var set) > baked > compiled default`.
254///
255/// Fails loud on invalid identity input (repo rule: a bad domain id at boot is
256/// a configuration error, never a silent domain-0 node). FFI shims that need a
257/// return code call [`try_resolve_hosted`].
258pub fn resolve_hosted<'a>(baked: BootConfig<'a>) -> ExecutorConfig<'a> {
259    match try_resolve_hosted(baked) {
260        Ok(cfg) => cfg,
261        Err(e) => panic!("nros boot-config resolution failed: {e}"),
262    }
263}
264
265/// Fallible [`resolve_hosted`] — returns [`BootConfigError`] instead of
266/// panicking, so the C / C++ FFI shims can surface a return code.
267///
268/// - `ROS_DOMAIN_ID` set but non-numeric → `DomainIdParse` (the pre-#206 C++
269///   header silently collapsed this to domain 0; the resolver silently ignored
270///   it — both were wrong).
271/// - any resolved domain id > `DOMAIN_ID_MAX` → `DomainIdRange`, INCLUDING a
272///   baked one (the DDS backend would only fail later).
273///
274/// Logs one line naming each field's source (`report`).
275pub fn try_resolve_hosted<'a>(
276    baked: BootConfig<'a>,
277) -> Result<ExecutorConfig<'a>, BootConfigError> {
278    let env = env_rung()?;
279    let cfg = ExecutorConfig::try_resolve_with(baked, Some(env))?;
280    report(&env, &baked, &cfg);
281    Ok(cfg)
282}

284/// `ExecutorConfig::from_env()`, as an extension trait.
285///
286/// issue 0687 — this was an inherent method on `ExecutorConfig`, which is
287/// defined in `nros-node`; an inherent method cannot be moved to another crate
288/// and keep its spelling, and the spelling is what ~26 call sites (most of them
289/// user-facing native examples) are written against. A trait keeps
290/// `ExecutorConfig::from_env()` working wherever it is in scope — the
291/// [`prelude`](crate::prelude) carries it, so a consumer that already writes
292/// `use nros::prelude::*` changes nothing at all.
293#[cfg(feature = "env")]
294pub trait ExecutorConfigEnvExt {
295    /// Create a configuration from environment variables.
296    ///
297    /// Reads:
298    /// - `NROS_LOCATOR` — Middleware locator. Unset ⇒ empty (issue 0330: the
299    ///   active RMW backend applies its own default; e.g. zenoh dials
300    ///   `nros_rmw_zenoh::DEFAULT_LOCATOR`). The retired name `ZENOH_LOCATOR`
301    ///   is refused (phase-484 W8c) — see [`BootConfigError::RetiredEnvVar`].
302    /// - `ROS_DOMAIN_ID` — ROS 2 domain ID (default: `0`).
303    /// - `NROS_SESSION_MODE` — `"client"` or `"peer"` (default: `"client"`).
304    ///   The retired name `ZENOH_MODE` is refused (phase-484 W8c).
305    /// - `NROS_RMW` — the backend selector, through [`rmw_selector`]. issue
306    ///   0687: `Executor::open` used to read this itself; it now takes
307    ///   `ExecutorConfig::rmw`, so a config built here still selects the
308    ///   backend the user named.
309    ///
310    /// The returned `&'static str` fields point into the process environment,
311    /// read live on every call.
312    ///
313    /// **Panics** on a malformed or out-of-range `$ROS_DOMAIN_ID` or a set
314    /// retired alias (`ZENOH_LOCATOR` / `ZENOH_MODE`), like
315    /// [`resolve_hosted`] and the C / C++ entries. A boot identity that cannot
316    /// be resolved is a configuration error; the alternative is a node silently
317    /// running on domain 0, which is what issue #206 removed everywhere else.
318    fn from_env() -> ExecutorConfig<'static>;
319}
320
321#[cfg(feature = "env")]
322impl ExecutorConfigEnvExt for ExecutorConfig<'static> {
323    fn from_env() -> ExecutorConfig<'static> {
324        // issue 0687 — `from_env` IS `resolve_hosted` with nothing baked. The
325        // one field that does NOT come from the rung is `node_name`: `from_env`
326        // has never honoured `$NROS_NODE_NAME` (its doc lists what it reads),
327        // and callers chain `.node_name(..)` immediately.
328        ExecutorConfig {
329            node_name: "node",
330            ..resolve_hosted(BootConfig::default())
331        }
332    }
333}

335/// Process-wide mutex that serialises every test that touches the process
336/// environment — the ones that MUTATE it (`EnvGuard` below) and the ones that
337/// only READ it through `init()` / `Context::from_env()` (`init.rs`).
338///
339/// `cargo test` runs `#[test]`s in parallel within a single binary by default,
340/// and the reader reads the live environment on every call. So a reader that
341/// does not hold this lock can
342/// observe a sibling's `ROS_DOMAIN_ID=abc` halfway through: `init.rs`'s
343/// `plain_args_pass_through_to_init` compared two reads and got `Ok(..)` vs
344/// `Err(EnvParseFailed)` under `just ci gate`. It lived in `env.rs`'s private
345/// test module, so the readers in `init.rs` could not take it. (`cargo
346/// nextest` runs each test in its own process, so the lock is uncontended
347/// there, but taking it is still correct.)
348///
349/// Poison-TOLERANT, deliberately: three tests assert that a bad
350/// `$ROS_DOMAIN_ID` panics, and a panic while this guard is alive poisons the
351/// mutex — which would then fail every LATER test for a reason that has
352/// nothing to do with what it asserts (measured: 3 intended failures became
353/// 8). The data behind the lock is `()`; there is no invariant a panic could
354/// have broken.
355#[cfg(all(test, feature = "env"))]
356pub(crate) fn test_env_lock() -> std::sync::MutexGuard<'static, ()> {
357    static LOCK: std::sync::OnceLock<std::sync::Mutex<()>> = std::sync::OnceLock::new();
358    LOCK.get_or_init(|| std::sync::Mutex::new(()))
359        .lock()
360        .unwrap_or_else(|e| e.into_inner())
361}

371    /// RAII guard that saves and restores a single env var.
372    struct EnvGuard {
373        key: &'static str,
374        prev: Option<std::ffi::OsString>,
375    }
376
377    impl EnvGuard {
378        fn set(key: &'static str, value: &str) -> Self {
379            let prev = std::env::var_os(key);
380            // SAFETY: serialised via env_lock().
381            unsafe { std::env::set_var(key, value) };
382            Self { key, prev }
383        }
384
385        fn unset(key: &'static str) -> Self {
386            let prev = std::env::var_os(key);
387            // SAFETY: serialised via env_lock().
388            unsafe { std::env::remove_var(key) };
389            Self { key, prev }
390        }
391    }
392
393    impl Drop for EnvGuard {
394        fn drop(&mut self) {
395            // SAFETY: serialised via env_lock().
396            unsafe {
397                match &self.prev {
398                    Some(v) => std::env::set_var(self.key, v),
399                    None => std::env::remove_var(self.key),
400                }
401            }
402        }
403    }

578    /// issue 0794 — the hosted edge fills the namespace rung.
579    ///
580    /// Before, `EnvRung` had no namespace field, so this variable could not
581    /// exist: a hosted node's namespace came from the baked rung or from a
582    /// hand-written `ExecutorConfig::namespace(..)` — the per-board lore
583    /// RFC-0045 exists to abolish.
584    #[test]
585    fn try_resolve_namespace_env_rung() {
586        let _l = env_lock();
587        let baked = BootConfig {
588            namespace: Some("/baked"),
589            ..BootConfig::default()
590        };
591
592        let g = EnvGuard::set("NROS_NODE_NAMESPACE", "/robot1");
593        let cfg = try_resolve_hosted(baked).expect("resolve ok");
594        assert_eq!(
595            cfg.namespace, "/robot1",
596            "env rung must override the baked namespace with its own value"
597        );
598        drop(g);
599
600        // Unset falls through to baked, which is the rung below it.
601        let _g = EnvGuard::unset("NROS_NODE_NAMESPACE");
602        let cfg = try_resolve_hosted(baked).expect("resolve ok");
603        assert_eq!(cfg.namespace, "/baked");
604
605        // EMPTY is unset, not "override back to root": `""` is already what an
606        // unset namespace resolves to, so honouring it would give `NROS_NODE_
607        // NAMESPACE=` a meaning no caller can distinguish from absence.
608        let _g = EnvGuard::set("NROS_NODE_NAMESPACE", "");
609        let cfg = try_resolve_hosted(baked).expect("resolve ok");
610        assert_eq!(cfg.namespace, "/baked");
611    }
