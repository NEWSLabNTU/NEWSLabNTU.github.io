1//! Phase 212.L.5 — top-level init API; phase-427 W9 — on every target.
2//!
3//! [`Context`] is WHERE this image is connected — locator, domain, RMW,
4//! session mode, and the source those came from (RFC-0089 "Context and
5//! `init`, settled"). One per process or image; a resolved value, not an
6//! entity on the graph. A `Node` is a named participant with its own
7//! entities, and one image holds several of those on one `Context`
8//! (RFC-0047).
9//!
10//! ## Where the values come from
11//!
12//! * **Hosted** (feature `env`): the process environment (`ROS_DOMAIN_ID`,
13//!   `NROS_LOCATOR`, `NROS_SESSION_MODE`, `NROS_RMW` / `RMW_IMPLEMENTATION`)
14//!   through the one hosted resolver, `crate::env::try_resolve_hosted`. A
15//!   launcher projects per-node values into that environment before exec.
16//! * **Freestanding** (no `env`): there is no process and no environment; the
17//!   same values are BAKED at compile time — `option_env!("NROS_LOCATOR")` and
18//!   `option_env!("NROS_DOMAIN_ID")`, exported into this crate's build by
19//!   `nros_zephyr_build::bake_nros_config()` from Kconfig (`build.rs`), or
20//!   present in the process environment of the build for the other RTOS lanes
21//!   (the board crates read the same two names the same way). The semantics
22//!   survive with the source moved from run time to build time: "the
23//!   environment this image was built for or launched in".
24//!
25//! The `env` feature gates ONLY the process-environment reader and the
26//! argv/launch entry points; `Context`, [`ContextSource`], [`InitError`],
27//! [`InitOptions`] and [`Context::config`] compile on every target. The floor
28//! for `Context` itself is `alloc`: `locator` and `rmw` are owned `String`s
29//! so the value outlives whatever transient produced it (an env cache, a
30//! parsed launch file, a harness-supplied override).
31//!
32//! ## The rclrs family
33//!
34//! [`Context::default_from_env`], [`Context::from_env`] and (hosted only)
35//! [`Context::new`] are the `rclrs::Context` constructors under the same
36//! names; [`InitOptions`] is rclrs's, with the one option it has. [`init()`] is
37//! the C++-symmetric anchor (`rclcpp::init`) and equals `default_from_env()`.
38//!
39//! Three patterns are supported (per the Phase 212.L canonical pkg shape):
40//!
41//! 1. **Node pkg** — register via the [`nros::node!`](crate::node!)
42//!    macro (Phase 172 W.3); the generated runtime owns the spin loop and
43//!    builds its `Context` through [`Context::default_from_env`].
44//! 2. **Application pkg + launch-aware** — call [`init_with_launch_auto`] (or
45//!    [`init_with_launch`] for an explicit path). The returned [`Context`]
46//!    carries launch-resolved fields (domain id, locator, RMW choice). User
47//!    code drives its own spin via `Executor::open` +
48//!    `Executor::spin`.
49//! 3. **Application pkg + custom spin** — call [`init()`] /
50//!    [`Context::default_from_env`] (or [`init_with_args`] / [`Context::new`]
51//!    for argv-style entry). Launch file is ignored; env vars +
52//!    `ExecutorConfig::from_env()` semantics still apply.
53//!
54//! To actually open a session, materialise an [`crate::ExecutorConfig`] via
55//! [`Context::config`] and pass it to `Executor::open`.
56//!
57//! ## Launch overlay (current limitation)
58//!
59//! `init_with_launch_auto` / `init_with_launch` currently consume the
60//! launch-resolved knobs the parent `nros launch` process exports via env
61//! vars (`ROS_DOMAIN_ID`, `NROS_LOCATOR`, `NROS_SESSION_MODE`,
62//! `RMW_IMPLEMENTATION`, plus the placeholder `NROS_RUNTIME_OVERLAY` for
63//! the future structured overlay path). The launch XML is NOT parsed
64//! in-process; the runtime trusts the launcher to project the relevant
65//! params / remaps / env into the child environment. A follow-up wave wires
66//! the structured overlay (Option A — `nros launch --emit-runtime-overlay`
67//! → JSON sidecar consumed here). See Phase 212.L.5 notes.

86/// Errors returned by the init API.
87#[derive(Debug, Clone, PartialEq, Eq)]
88pub enum InitError {
89    /// `init_with_launch(path)` was passed a path that does not exist or
90    /// could not be read.
91    LaunchFileNotFound,
92    /// The launch file existed but could not be parsed.
93    ///
94    /// Phase 212.L.5 ships a stub — actual XML parsing arrives with the
95    /// runtime-overlay wave. Until then this variant is unused.
96    LaunchParseFailed,
97    /// A launch-derived env var (`ROS_DOMAIN_ID`, etc.) failed to parse —
98    /// or, freestanding, the baked `NROS_DOMAIN_ID` is not a decimal integer.
99    EnvParseFailed,
100    /// A domain id — from the environment, the bake, or an
101    /// [`InitOptions::with_domain_id`] override — exceeds
102    /// [`crate::DOMAIN_ID_MAX`]. Rejected here rather than
103    /// handed to a backend that can only fail later (the #206 rule).
104    DomainIdOutOfRange,
105}
106
107impl core::fmt::Display for InitError {
108    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
109        match self {
110            InitError::LaunchFileNotFound => f.write_str("launch file not found"),
111            InitError::LaunchParseFailed => f.write_str("launch file parse failed"),
112            InitError::EnvParseFailed => f.write_str("env var parse failed"),
113            InitError::DomainIdOutOfRange => f.write_str("domain id out of range"),
114        }
115    }
116}

123/// Phase 212.L.5 — resolved init context.
124///
125/// Returned by every `init*` entry point and every `Context::*` constructor.
126/// Carries the fields the user needs to construct an [`ExecutorConfig`] and
127/// open a session.
128///
129/// Fields are owned (`String`) so the `Context` can outlive transient parents
130/// (env caches, parsed launch files, a harness-supplied locator). That is why
131/// the type's feature floor is `alloc`.
132#[cfg(feature = "alloc")]
133#[derive(Debug, Clone)]
134pub struct Context {
135    /// ROS 2 domain ID (`ROS_DOMAIN_ID`, default 0; baked `NROS_DOMAIN_ID`
136    /// freestanding).
137    pub domain_id: u32,
138    /// Middleware locator (`NROS_LOCATOR` / legacy `ZENOH_LOCATOR`; baked
139    /// `NROS_LOCATOR` freestanding). Empty means ABSENT: `nros` is
140    /// RMW-agnostic, so the linked backend substitutes its own default
141    /// (issue 0330).
142    pub locator: alloc::string::String,
143    /// Session mode (`NROS_SESSION_MODE` / legacy `ZENOH_MODE`, default `Client`).
144    pub mode: SessionMode,
145    /// RMW implementation hint (`RMW_IMPLEMENTATION` /  `NROS_RMW`).
146    ///
147    /// Empty when neither var is set, and always empty freestanding (an image
148    /// links exactly one backend). The runtime uses this to pick a primary
149    /// backend when multiple are linked; see `crate::internals::open_session`.
150    pub rmw: alloc::string::String,
151    /// Source of this context — useful for diagnostics + tests.
152    pub source: ContextSource,
153}

155/// Where the [`Context`] came from. Diagnostics only.
156#[derive(Debug, Clone, Copy, PartialEq, Eq)]
157pub enum ContextSource {
158    /// Built from env vars by [`init()`] / [`init_with_args`] /
159    /// [`Context::new`], or by [`Context::default_from_env`] /
160    /// [`Context::from_env`] on a hosted build.
161    Env,
162    /// Built from a launch file (path supplied to [`init_with_launch`]) or
163    /// auto-discovered via [`init_with_launch_auto`]. The launch XML itself
164    /// is NOT yet parsed (see module docs); the launcher's projected env
165    /// is the source of truth for now.
166    Launch,
167    /// phase-427 W9 — built from the constants baked into this crate at
168    /// compile time (`NROS_LOCATOR`, `NROS_DOMAIN_ID`) by [`Context::baked`],
169    /// which is what [`Context::default_from_env`] / [`Context::from_env`]
170    /// resolve to on a freestanding build. The entry macros
171    /// (`nros::main!`'s Zephyr arm, `nros::zephyr_component_main!`) produce
172    /// this one.
173    Baked,
174}

176/// phase-427 W9 — `rclrs::InitOptions`, with the one option it has.
177///
178/// Consumed by [`Context::from_env`] and [`Context::new`]; `None` for the
179/// domain keeps whatever the environment (hosted) or the bake (freestanding)
180/// resolved.
181#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
182pub struct InitOptions {
183    domain_id: Option<u32>,
184}
185
186impl InitOptions {
187    /// Default options: nothing overridden.
188    pub const fn new() -> Self {
189        Self { domain_id: None }
190    }
191
192    /// Override the domain id (`Some`) or keep the resolved one (`None`).
193    ///
194    /// `usize` because that is rclrs's signature. A value above `u32::MAX`
195    /// cannot be a domain id at all; it saturates here and the constructor
196    /// that consumes the options then rejects it as
197    /// [`InitError::DomainIdOutOfRange`], which is the loud place — a builder
198    /// method has no `Result` to carry the refusal in.
199    pub const fn with_domain_id(mut self, domain_id: Option<usize>) -> Self {
200        self.domain_id = match domain_id {
201            Some(d) if d > u32::MAX as usize => Some(u32::MAX),
202            Some(d) => Some(d as u32),
203            None => None,
204        };
205        self
206    }
207
208    /// The in-place twin of [`with_domain_id`](Self::with_domain_id) — rclrs
209    /// has both, and a ported `opts.set_domain_id(Some(7))` compiles as
210    /// written. Same saturation.
211    pub fn set_domain_id(&mut self, domain_id: Option<usize>) {
212        *self = self.with_domain_id(domain_id);
213    }
214
215    /// The domain override, if any.
216    pub const fn domain_id(&self) -> Option<u32> {
217        self.domain_id
218    }
219}

221#[cfg(feature = "alloc")]
222/// The range check every constructor applies to a domain id, whatever its
223/// source (env, bake, override): the same bound `try_resolve_hosted` enforces
224/// ("INCLUDING a baked one"), so a freestanding image cannot boot on a domain
225/// a hosted process would have refused.
226fn check_domain_id(domain_id: u32) -> Result<u32, InitError> {
227    if domain_id > DOMAIN_ID_MAX {
228        Err(InitError::DomainIdOutOfRange)
229    } else {
230        Ok(domain_id)
231    }
232}

234#[cfg(feature = "alloc")]
235impl Context {
236    /// Materialise an [`ExecutorConfig`] for a node with the given name.
237    ///
238    /// The returned config borrows from `self`, so callers usually do:
239    ///
240    /// ```ignore
241    /// let ctx = nros::Context::default_from_env()?;
242    /// let cfg = ctx.config("talker");
243    /// let mut executor = nros::Executor::open(&cfg)?;
244    /// ```
245    pub fn config<'a>(&'a self, node_name: &'a str) -> ExecutorConfig<'a> {
246        ExecutorConfig::new(self.locator.as_str())
247            .node_name(node_name)
248            .domain_id(self.domain_id)
249            .mode(self.mode)
250    }
251
252    /// `rclrs::Context::default_from_env()` — the environment, default
253    /// options.
254    ///
255    /// Hosted (feature `env`) this reads the process environment and is
256    /// exactly [`init()`]; freestanding it is [`Context::baked`], the build
257    /// environment this image was compiled for. RFC-0089 "Context and `init`,
258    /// settled": the semantics survive with the source moved from run time to
259    /// build time, which is the bounded half of this adoption.
260    pub fn default_from_env() -> Result<Context, InitError> {
261        #[cfg(feature = "env")]
262        {
263            init()
264        }
265        #[cfg(not(feature = "env"))]
266        {
267            Self::baked()
268        }
269    }
270
271    /// `rclrs::Context::from_env(options)` — [`default_from_env`](Self::default_from_env)
272    /// with the [`InitOptions`] applied on top: a
273    /// `Some` domain replaces the resolved one, `None` keeps it.
274    pub fn from_env(options: InitOptions) -> Result<Context, InitError> {
275        let mut ctx = Self::default_from_env()?;
276        if let Some(domain_id) = options.domain_id {
277            ctx.domain_id = check_domain_id(domain_id)?;
278        }
279        Ok(ctx)
280    }

282    /// `rclrs::Context::new(args, options)` — the argv constructor. HOSTED
283    /// ONLY: a freestanding image has no argv, so the name is absent there and
284    /// the compiler says so.
285    ///
286    /// **What it does with the arguments:** the same as [`init_with_args`] —
287    /// it does NOT parse them, and it does not silently ignore them either. A
288    /// vector carrying `--ros-args` is REFUSED LOUDLY at the call (logged
289    /// through `nros_log`, then a panic naming the flag —
290    /// [`REFUSE_INIT_ARGS`]); any other vector passes through to
291    /// [`Context::from_env`] untouched, because nothing was dropped. Parsing
292    /// `--ros-args` is remap resolution (RFC-0020 class 4) and belongs beside
293    /// `nros::resolve_name`, not in a constructor. RFC-0089: only the VALUE
294    /// carries the defect, so the call is the earliest point it is knowable.
295    ///
296    /// `S: AsRef<str>` is a superset of rclrs's `Item = String`, so
297    /// `Context::new(std::env::args(), InitOptions::new())` compiles as
298    /// written.
299    #[cfg(feature = "env")]
300    pub fn new<I, S>(args: I, options: InitOptions) -> Result<Context, InitError>
301    where
302        I: IntoIterator<Item = S>,
303        S: AsRef<str>,
304    {
305        refuse_ros_args(args);
306        Self::from_env(options)
307    }

309    /// phase-427 W9 — the freestanding constructor: the constants baked into
310    /// this crate at compile time.
311    ///
312    /// * `NROS_LOCATOR` — the middleware locator. Unset or empty ⇒ empty ⇒
313    ///   the linked backend's own default (issue 0330; for zenoh-pico that is
314    ///   multicast scouting, which native_sim NSOS cannot satisfy — hence the
315    ///   bake).
316    /// * `NROS_DOMAIN_ID` — the domain. Unset ⇒ 0; non-numeric ⇒
317    ///   [`InitError::EnvParseFailed`]; above `DOMAIN_ID_MAX` ⇒
318    ///   [`InitError::DomainIdOutOfRange`].
319    ///
320    /// Both names reach this crate's build the same way they reach the board
321    /// crates: on Zephyr, `build.rs` re-exports `CONFIG_NROS_ZENOH_LOCATOR` /
322    /// `CONFIG_NROS_DOMAIN_ID` from `$DOTCONFIG` (`nros_zephyr_build::
323    /// bake_nros_config`, the issue-0460 channel); on the other RTOS lanes the
324    /// fixture build exports them into the process environment. This is the
325    /// ONE `option_env!` site for the pair — the entry macros used to carry
326    /// their own copies (and `nros::main!`'s Zephyr arm had lost the domain
327    /// half, issue 0161's class), which is what "one source of the baked
328    /// shape" replaces.
329    ///
330    /// Mode is `Client` and `rmw` is empty: an image links exactly one
331    /// backend and nothing bakes a session mode today.
332    ///
333    /// Hosted builds have this too — it answers "what was baked", which a
334    /// hosted process rarely wants; [`Context::default_from_env`] is the
335    /// constructor that picks the right source for the build.
336    pub fn baked() -> Result<Context, InitError> {
337        Self::from_baked(option_env!("NROS_LOCATOR"), option_env!("NROS_DOMAIN_ID"))
338    }
339
340    /// The parse behind [`Context::baked`], with the two constants injected
341    /// so a unit test can exercise every arm without a build environment.
342    /// Public because the macros in `nros-macros` expand in another crate;
343    /// not API.
344    #[doc(hidden)]
345    pub fn from_baked(
346        baked_locator: Option<&str>,
347        baked_domain_id: Option<&str>,
348    ) -> Result<Context, InitError> {
349        let domain_id = match baked_domain_id {
350            Some(d) => check_domain_id(d.trim().parse().map_err(|_| InitError::EnvParseFailed)?)?,
351            None => 0,
352        };
353        Ok(Context {
354            domain_id,
355            locator: alloc::string::String::from(baked_locator.unwrap_or("")),
356            mode: SessionMode::Client,
357            rmw: alloc::string::String::new(),
358            source: ContextSource::Baked,
359        })
360    }
361
362    /// Replace the locator with one handed in at run time, when there is one.
363    ///
364    /// #166 / phase-286 W1 — native_sim test parallelism: the harness launches
365    /// an image with `-testargs --nros-locator=<loc>` and a per-test router on
366    /// that ephemeral port, and `nros_runtime_locator_override()`
367    /// (`nros-platform-zephyr`, argv-backed) hands it back here. `None` or an
368    /// empty string keeps the baked value, so on real hardware — where the hook
369    /// returns NULL — the bake stands. The entry macros call this after
370    /// [`Context::default_from_env`]; a hand-written entry may too.
371    pub fn with_locator_override(mut self, locator: Option<&str>) -> Self {
372        if let Some(loc) = locator {
373            if !loc.is_empty() {
374                self.locator = alloc::string::String::from(loc);
375            }
376        }
377        self
378    }
379}

381#[cfg(feature = "env")]
382fn read_env_context(source: ContextSource) -> Result<Context, InitError> {
383    // issue 0687 — through the ONE resolver, not a third parse of the same four
384    // variables. This function had its own copy, and the copies had drifted:
385    // it did not warn on the legacy `$ZENOH_LOCATOR`/`$ZENOH_MODE` spellings
386    // the other reader deprecates, and it range-checked nothing, so
387    // `ROS_DOMAIN_ID=300` reached a backend that could only fail later.
388    //
389    // Issue 0330 — no backend default for the locator: `nros` is RMW-agnostic,
390    // so unset env leaves it EMPTY (= absent) and the linked backend
391    // substitutes its own (zenoh: `nros_rmw_zenoh::DEFAULT_LOCATOR`; xrce: its
392    // agent default; cyclonedds ignores the locator entirely). That is what
393    // `resolve_hosted` does with an empty `BootConfig` too.
394    let resolved =
395        crate::env::try_resolve_hosted(nros_node::BootConfig::default()).map_err(|e| match e {
396            nros_node::BootConfigError::DomainIdRange => InitError::DomainIdOutOfRange,
397            nros_node::BootConfigError::DomainIdParse => InitError::EnvParseFailed,
398        })?;
399    let locator = alloc::string::String::from(resolved.locator);
400    let domain_id = resolved.domain_id;
401    let mode = resolved.mode;
402    // issue 0687 — the `$NROS_RMW` half comes from the shared selector; the
403    // `RMW_IMPLEMENTATION` fallback stays HERE and only here. `Context.rmw` is
404    // a ROS-vocabulary HINT (`rmw_cyclonedds_cpp`), not the cffi registry
405    // selector (`cyclonedds`) — folding the two together would hand a ROS name
406    // to `resolve_backend`, which answers `Unknown` and fails the open.
407    let rmw = crate::rmw_selector()
408        .map(|s| alloc::string::String::from(s.as_str()))
409        .or_else(|| std::env::var("RMW_IMPLEMENTATION").ok())
410        .unwrap_or_default();
411    Ok(Context {
412        domain_id,
413        locator,
414        mode,
415        rmw,
416        source,
417    })
418}

420/// Pattern 3 — raw init, launch file ignored. The C++-symmetric anchor:
421/// `rclcpp::init()` in the Rust spelling, and EQUAL to
422/// [`Context::default_from_env`] on a hosted build (RFC-0089 "Context and
423/// `init`, settled" — each language follows its own upstream; rclrs spells
424/// this one on the type, rclcpp as a free function, and both are here).
425///

427/// `NROS_RMW` / `RMW_IMPLEMENTATION`) and returns a [`Context`]. The

435/// The refusal [`init_with_args`] and [`Context::new`] emit when handed
436/// `--ros-args`. The same text `NROS_RCLCPP_REFUSE_INIT_ARGV` carries for the
437/// C++ twin (`packages/api/nros-cpp/include/nros/log.hpp`), with the Rust
438/// spellings.
439#[cfg(feature = "env")]
440pub const REFUSE_INIT_ARGS: &str = "nros::init_with_args / nros::Context::new was given --ros-args, which nano-ros cannot \
441honour (RFC-0089, phase-417 W3.b). Proceeding would DISCARD it, so `-r chatter:=/other` would \
442silently become a wrong-topic bug at runtime -- the 'compiles and differs' the rule \
443forbids. Nothing in this process parses --ros-args yet, and honouring them is remap \
444resolution -- RFC-0020 violation class 4 -- so the parser belongs beside nros::resolve_name, \
445not in this wrapper. Today remaps and parameter overrides come from the LAUNCHER, which \
446projects them into the environment before exec. Call nros::init() / \
447nros::Context::default_from_env(), or nros::init_with_launch_auto() for the launch-aware \
448entry point.";
449
450/// The refusal's predicate, separately checkable (RFC-0089 §"where the
451/// refusal fires": an abort inlined into `init` can only be observed by a
452/// process that then dies, which is the shape of a check nothing runs).
453///
454/// Exact match only. `--ros-args-extra` is not the flag, and a prefix match
455/// would refuse an argument nano-ros never drops — the mutation the C++
456/// predicate's `static_assert`s pin, pinned here by a unit test.
457#[cfg(feature = "env")]
458pub fn args_have_ros_args<I, S>(args: I) -> bool
459where
460    I: IntoIterator<Item = S>,
461    S: AsRef<str>,
462{
463    args.into_iter().any(|a| a.as_ref() == "--ros-args")
464}
465
466/// The one refusal site, shared by [`init_with_args`] and [`Context::new`]:
467/// logged through `nros_log` (never `std::println!`, issue 0589), then a
468/// panic naming the flag — the same shape `rclcpp::init(argc, argv)` has in
469/// `nros.hpp`.
470#[cfg(feature = "env")]
471fn refuse_ros_args<I, S>(args: I)
472where
473    I: IntoIterator<Item = S>,
474    S: AsRef<str>,
475{
476    if args_have_ros_args(args) {
477        nros_log::log_error!(nros_log::get_logger("nros"), "{}", REFUSE_INIT_ARGS);
478        panic!("{}", REFUSE_INIT_ARGS);
479    }
480}

482/// Pattern 3 — like [`init()`] but accepts a `[--arg=value, ...]`-style argv
483/// iterator. The free-function twin of [`Context::new`] with default options.
484///
485/// **What it does with the arguments:** it does NOT parse them, and it does
486/// not silently ignore them either. The structured argv parse (`--ros-args -p
487/// foo:=42`, remaps) is remap resolution, which belongs beside
488/// `nros::resolve_name` and lands with the runtime-overlay wave. Until it
489/// does, an argument vector that carries `--ros-args` is REFUSED LOUDLY at the
490/// call: the refusal is logged through `nros_log` (never `std::println!`,
491/// issue 0589) and the process panics naming the flag — the same shape
492/// `rclcpp::init(argc, argv)` has in `nros.hpp`. A vector with no ROS
493/// arguments is unaffected, because nothing was dropped.
494///
495/// RFC-0089: only the VALUE carries the defect, so the call is the earliest
496/// point it is knowable; silence there is the "compiles and differs" the
497/// compile-or-conform rule forbids.
498#[cfg(feature = "env")]
499pub fn init_with_args<I, S>(args: I) -> Result<Context, InitError>
500where
501    I: IntoIterator<Item = S>,
502    S: AsRef<str>,
503{
504    refuse_ros_args(args);
505    init()
506}

803/// Pattern 2 — launch-aware init.
804///
805/// Resolves the launch file via:
806///
807/// 1. `$NROS_RUNTIME_OVERLAY` — when set, the path points at a JSON sidecar
808///    written by `nros launch --emit-runtime-overlay`. (NOT yet consumed;
809///    placeholder for the follow-up wave.)
810/// 2. `<CARGO_MANIFEST_DIR>/launch/<pkg>.launch.xml` or
811///    `<CARGO_MANIFEST_DIR>/launch/system.launch.xml`. (NOT yet parsed;
812///    placeholder.)
813/// 3. The env vars described in [`init()`] — the launcher projects launch
814///    params into the child env before `exec()`, so the env path is the
815///    de-facto launch overlay today.
816///
817/// Returns a [`Context`] whose `source = ContextSource::Launch` so callers
818/// can introspect whether the run is launch-driven.
819#[cfg(feature = "env")]
820pub fn init_with_launch_auto() -> Result<Context, InitError> {
821    // TODO (Phase 212.L.5 follow-up):
822    //   1. If $NROS_RUNTIME_OVERLAY is set, read the JSON sidecar and fold
823    //      its params/remaps/env into the Context.
824    //   2. Else walk <CARGO_MANIFEST_DIR>/launch/* and parse the XML
825    //      in-process (Option B — only if Option A overhead is rejected).
826    // For now the env path is the only overlay channel.
827    read_env_context(ContextSource::Launch)
828}

830/// Pattern 2 — explicit-path variant of [`init_with_launch_auto`].
831///
832/// Verifies the file exists (so misspelled paths fail fast at init time)
833/// but does NOT yet parse the XML; the launcher's projected env is the
834/// active overlay. See the module-level notes for the follow-up plan.
835#[cfg(feature = "env")]
836pub fn init_with_launch(path: impl AsRef<Path>) -> Result<Context, InitError> {
837    let p = path.as_ref();
838    if !p.exists() {
839        return Err(InitError::LaunchFileNotFound);
840    }
841    // TODO (Phase 212.L.5 follow-up): parse the launch XML and fold params
842    // / remaps / env into the returned Context. Today we only verify the
843    // file exists and fall through to the env overlay path.
844    read_env_context(ContextSource::Launch)
