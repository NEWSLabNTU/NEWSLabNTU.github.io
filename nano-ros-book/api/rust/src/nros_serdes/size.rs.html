1//! Phase 380 W1 — a message's serialized size, computed instead of guessed.
2//!
3//! `NROS_SUBSCRIPTION_BUFFER_SIZE` defaults to 1024 bytes and is a GUESS.
4//! Nothing checks it against the messages an image actually subscribes to, so a
5//! sample that does not fit is dropped AFTER the transport ACKed it, and
6//! `report_dropped_take` can only say "raise the knob" because the runtime does
7//! not know what value would have worked. On a target that knob is static RAM
8//! nobody can spare. Issue 0776 is the gap; this module is the calculator.
9//!
10//! # Why this is not a vtable slot
11//!
12//! Settled in phase-376 W4: nothing about a size bound varies by backend, and
13//! upstream proves it by not varying at all — both `librmw_cyclonedds_cpp.so`
14//! and `librmw_fastrtps_cpp.so` answer `rmw_get_serialized_message_size` with
15//! `RMW_RET_UNSUPPORTED`, and nothing in a Humble install calls it. Upstream can
16//! afford that because its serialized buffer RESIZES; the bound is a hint that
17//! saves a realloc. Ours cannot resize, so the same number is load-bearing here.
18//!
19//! # Thread the offset — do not sum the maxima
20//!
21//! Padding is a function of WHERE a field starts, so a calculation that sums
22//! per-field maxima is wrong the moment a variable-length field shifts what
23//! follows. [`size_bound`] therefore takes `current_alignment` in and returns
24//! the size from there, which is also what makes nested structs compose with no
25//! special case — the same signature rosidl's generated Fast-RTPS support uses
26//! (`max_serialized_size_T(full_bounded, is_plain, current_alignment)`).
27//!
28//! # Agreement with the writer is the only thing that matters
29//!
30//! Every rule below was read out of `CdrWriter`, not out of the CDR spec:
31//! `align()` caps alignment at 4 under XCDR2 and honours 8 under XCDR1;
32//! `begin_dheader()` aligns to 4 and reserves 4 bytes for EVERY struct under
33//! XCDR2 (a generated `serialize` opens with it, so nested structs get one too);
34//! `write_string` writes `len + 1` and then the NUL. A bound that merely looks
35//! right is worthless — see `size_tests.rs`, which checks it against the bytes
36//! the writer actually produced.
37

43/// What one walk of a schema can say about size.
44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub struct SizeBound {
46    /// Bytes, given the starting offset the walk was handed.
47    pub bytes: usize,
48    /// False once an unbounded `String` / `WString` / `Sequence` is reached —
49    /// `bytes` is then a FLOOR, not a bound, and callers must not size a buffer
50    /// from it.
51    pub bounded: bool,
52    /// No variable-length member anywhere, so the layout is fixed: `bytes` is
53    /// EXACT rather than an upper bound, and the type is loan-eligible
54    /// (phase-380 W5 wires this to `borrow_loaned_message` /
55    /// `subscription_supports_in_place` rather than letting a second notion of
56    /// "fixed layout" grow).
57    pub plain: bool,
58}

60/// The encapsulation header the payload crossing our vtable carries: 2 bytes of
61/// representation id + 2 of options.
62///
63/// TOP LEVEL ONLY, not per nested struct — which is why the Cyclone backend
64/// computes `total = paylen + 4`. [`size_bound`] deliberately does not include
65/// it; [`max_serialized_size`] does.
66pub const ENCAPSULATION_HEADER_BYTES: usize = 4;
67
68/// Padding needed to reach `alignment` from `offset`, under `version`.
69///
70/// Mirrors `CdrWriter::align` exactly, including the XCDR2 cap at 4 — a message
71/// containing an `int64` therefore has TWO different bounds, and a single
72/// constant would be silently wrong for one encoding.
73const fn pad_to(offset: usize, alignment: usize, version: EncodingVersion) -> usize {
74    let alignment = match version {
75        EncodingVersion::Xcdr2 => {
76            if alignment > 4 {
77                4
78            } else {
79                alignment
80            }
81        }
82        EncodingVersion::Xcdr1 => alignment,
83    };
84    if alignment == 0 {
85        return 0;
86    }
87    (alignment - (offset % alignment)) % alignment
88}

90/// Size of one field, starting at `offset`. Returns the new offset plus the two
91/// flags, folded by the caller.
92const fn field_bound(
93    ty: &FieldType,
94    version: EncodingVersion,
95    offset: usize,
96) -> (usize, bool, bool) {
97    match ty {
98        FieldType::Bool | FieldType::Uint8 | FieldType::Int8 => (offset + 1, true, true),
99        FieldType::Uint16 | FieldType::Int16 => {
100            let o = offset + pad_to(offset, 2, version);
101            (o + 2, true, true)
102        }
103        FieldType::Uint32 | FieldType::Int32 | FieldType::Float32 => {
104            let o = offset + pad_to(offset, 4, version);
105            (o + 4, true, true)
106        }
107        FieldType::Uint64 | FieldType::Int64 | FieldType::Float64 => {
108            let o = offset + pad_to(offset, 8, version);
109            (o + 8, true, true)
110        }
111        // `write_string` writes `len + 1` as the u32 prefix and then the NUL,
112        // so a bound of `n` payload bytes costs `4 + n + 1`.
113        FieldType::BoundedString(n) => {
114            let o = offset + pad_to(offset, 4, version);
115            (o + 4 + *n + 1, true, false)
116        }
117        FieldType::BoundedWString(n) => {
118            let o = offset + pad_to(offset, 4, version);
119            (o + 4 + 2 * *n, true, false)
120        }
121        // No bound exists. Account the length prefix so `bytes` is an honest
122        // floor, and clear `bounded` so nobody sizes a buffer from it.
123        FieldType::String | FieldType::WString | FieldType::Sequence(_) => {
124            let o = offset + pad_to(offset, 4, version);
125            (o + 4, false, false)
126        }
127        FieldType::Array(n, inner) => {
128            let mut o = offset;
129            let mut bounded = true;
130            let mut plain = true;
131            let mut i = 0;
132            while i < *n {
133                let (next, b, p) = field_bound(inner, version, o);
134                o = next;
135                bounded &= b;
136                plain &= p;
137                i += 1;
138            }
139            // A fixed array of plain elements is itself plain; of anything else,
140            // not — the element count is fixed but each element's size is not.
141            (o, bounded, plain)
142        }
143        FieldType::BoundedSequence(n, inner) => {
144            let mut o = offset + pad_to(offset, 4, version);
145            o += 4;
146            let mut bounded = true;
147            let mut i = 0;
148            while i < *n {
149                let (next, b, _) = field_bound(inner, version, o);
150                o = next;
151                bounded &= b;
152                i += 1;
153            }
154            // Never plain: the wire length varies with the actual element count
155            // even though its maximum is known.
156            (o, bounded, false)
157        }
158        FieldType::Nested(nested) => {
159            let inner = size_bound(nested.fields, version, offset);
160            (inner.bytes, inner.bounded, inner.plain)
161        }
162    }
163}

165/// Walk `fields` from `current_alignment`, returning the size bound of the
166/// struct they describe.
167///
168/// The returned `bytes` is an ABSOLUTE offset — the position after the last
169/// field, measured from the same origin `current_alignment` was measured from —
170/// so a nested struct composes by being handed the parent's current offset.
171/// Subtract the starting offset if a length is what you want.
172///
173/// Excludes the encapsulation header; see [`max_serialized_size`].
174pub const fn size_bound(
175    fields: &'static [Field],
176    version: EncodingVersion,
177    current_alignment: usize,
178) -> SizeBound {
179    let mut offset = current_alignment;
180    let mut bounded = true;
181    let mut plain = true;
182
183    // XCDR2 delimits EVERY appendable struct, nested ones included: a generated
184    // `serialize` opens with `begin_dheader()`, which aligns to 4 and reserves
185    // 4. Under XCDR1 the call is a no-op. Missing this UNDER-reports, which is
186    // the dangerous direction — an under-reported bound sizes a buffer too
187    // small and reintroduces the very drop this exists to stop.
188    if matches!(version, EncodingVersion::Xcdr2) {
189        offset += pad_to(offset, 4, version);
190        offset += 4;
191        // A DHEADER does not make a struct non-plain: its width is fixed.
192    }
193
194    let mut i = 0;
195    while i < fields.len() {
196        let (next, b, p) = field_bound(&fields[i].ty, version, offset);
197        offset = next;
198        bounded &= b;
199        plain &= p;
200        i += 1;
201    }
202
203    SizeBound {
204        bytes: offset,
205        bounded,
206        plain,
207    }
208}

210/// The whole payload a publisher hands the transport: encapsulation header plus
211/// the struct's body, starting from offset 0.
212///
213/// `None` when the type is unbounded — the honest answer, and the one that keeps
214/// a caller from sizing a buffer off a floor. Reach for `serialized_size(&self)`
215/// (phase-380 W3) when an unbounded type still needs a number for THIS message.
216pub const fn max_serialized_size(
217    fields: &'static [Field],
218    version: EncodingVersion,
219) -> Option<usize> {
220    // The body's offsets are measured from after the encapsulation header —
221    // that is where `CdrWriter`'s `origin` sits — so the walk starts at 0 and
222    // the header is added once, at the top.
223    let bound = size_bound(fields, version, 0);
224    if bound.bounded {
225        Some(ENCAPSULATION_HEADER_BYTES + bound.bytes)
226    } else {
227        None
228    }
229}

231/// Phase 380 W3 — the EXACT serialized size of THIS message.
232///
233/// Two questions get asked about size and they are not the same one:
234///
235/// | question | asked by | this module |
236/// | --- | --- | --- |
237/// | how large can this TYPE ever be? | build-time buffer sizing | [`max_serialized_size`] |
238/// | how large is THIS message? | a publisher before publishing; a drop report | `serialized_size` |
239///
240/// The second is the only honest answer for an unbounded type, where the first
241/// is `None` — a `String` field has no maximum, but the string in hand has a
242/// length. It is what lets a drop report name the number that would have worked
243/// instead of saying "raise the knob".
244///
245/// Exact by construction: it runs the REAL writer with its stores disabled
246/// (`CdrWriter::measuring`), so the count comes from the same code that emits
247/// the bytes. A second walk of the schema could not see the actual string
248/// lengths and sequence counts, and would be a second implementation to keep in
249/// step besides.
250///
251/// Includes the encapsulation header, matching [`max_serialized_size`], so the
252/// two are directly comparable — which is the whole point at a call site
253/// deciding whether a message fits.
254pub fn serialized_size<T: crate::traits::Serialize>(
255    value: &T,
256    version: EncodingVersion,
257) -> Result<usize, crate::error::SerError> {
258    let mut w = crate::cdr::CdrWriter::measuring(&mut [], version);
259    value.serialize(&mut w)?;
260    Ok(ENCAPSULATION_HEADER_BYTES + w.position())
261}

263/// Phase 380 W4 — does a receive buffer of `rx_buf` bytes fit every message of
264/// this type?
265///
266/// `true` when the type is bounded and the bound fits. **`false` when the type
267/// is UNBOUNDED**, deliberately: no finite buffer fits a `String`, so the
268/// honest answer to "is this guaranteed to fit" is no. A caller that wants
269/// "fits unless proven otherwise" is asking a different question and should
270/// look at [`serialized_size`] per message.
271///
272/// Const, so a call site can put it in a `const { assert!(...) }` and turn a
273/// runtime drop into a build error:
274///
275/// ```ignore
276/// const { assert!(buffer_fits::<Odometry>(RX_BUF, EncodingVersion::Xcdr1)) };
277/// ```
278///
279/// # Why this is not simply a bound on `Subscription`
280///
281/// `Subscription<M, RX_BUF>` bounds `M: RosMessage`, which is a DIFFERENT trait
282/// from [`crate::schema::Message`] — and measured 2026-08-26, hand-written
283/// `RosMessage` types with no schema DO exist (`nros-core/src/service.rs`, the
284/// component-runtime tests). Tightening that bound would break them, so the
285/// assertion is opt-in at sites where the schema is known rather than universal
286/// at a site where it is not. See the phase doc for what closing that gap needs.
287pub const fn buffer_fits<M: crate::schema::Message>(
288    rx_buf: usize,
289    version: EncodingVersion,
290) -> bool {
291    let bound = match version {
292        EncodingVersion::Xcdr1 => M::MAX_SERIALIZED_SIZE_XCDR1,
293        EncodingVersion::Xcdr2 => M::MAX_SERIALIZED_SIZE_XCDR2,
294    };
295    match bound {
296        Some(n) => rx_buf >= n,
297        None => false,
298    }
299}
300
301/// Phase 380 W4 — the BUILD-ASSERTION predicate: `false` only when the type is
302/// provably too large for `rx_buf`.
303///
304/// Distinct from [`buffer_fits`], and the difference is the whole reason both
305/// exist:
306///
307/// * `buffer_fits` answers "is this GUARANTEED to fit", so an unbounded type is
308///   `false` — no finite buffer fits a `String`.
309/// * this answers "can we PROVE it will not fit", so an unbounded type is
310///   `true` — there is nothing to check, and failing the build for every
311///   `std_msgs/String` subscription would be absurd.
312///
313/// Asserting `buffer_fits` at a subscription site was my first attempt and
314/// would have refused the most common message in ROS. Kept as two named
315/// functions rather than one with a flag, because the wrong one is silently
316/// plausible at either call site.
317///
318/// Checks BOTH encodings and takes the larger: the encoding is a runtime
319/// property of the peer, so a buffer sized from XCDR1 alone is a trap — XCDR2
320/// adds a DHEADER per struct and is the bigger of the two for nested types.
321pub const fn bound_fits<M: crate::schema::Message>(rx_buf: usize) -> bool {
322    let x1 = match M::MAX_SERIALIZED_SIZE_XCDR1 {
323        Some(n) => n,
324        None => return true, // unbounded: nothing to prove
325    };
326    let x2 = match M::MAX_SERIALIZED_SIZE_XCDR2 {
327        Some(n) => n,
328        None => return true,
329    };
330    let largest = if x1 > x2 { x1 } else { x2 };
331    rx_buf >= largest
332}

334/// Phase 380 W5 — is this type loan-eligible?
335///
336/// A loan hands out a pointer into the transport's own memory, which is sound
337/// only when the layout is fixed: no length prefix to chase, no variable
338/// member. That is exactly [`SizeBound::plain`], so the answer falls out of W1
339/// rather than becoming a second notion of "fixed layout" maintained by hand —
340/// which is the drift `borrow_loaned_message` and
341/// `subscription_supports_in_place` would otherwise each grow their own version
342/// of.
343pub const fn is_loan_eligible<M: crate::schema::Message>() -> bool {
344    M::IS_PLAIN
345}

363    /// Serialize a MAXIMAL instance of a schema with the real `CdrWriter`, and
364    /// return the byte length the writer produced (header included).
365    ///
366    /// This is the whole point of the test module: a bound that is merely
367    /// self-consistent proves nothing. Every assertion below compares
368    /// [`max_serialized_size`] against bytes this function actually wrote.
369    fn write_maximal(fields: &[Field], version: EncodingVersion, buf: &mut [u8]) -> usize {
370        let mut w = match version {
371            EncodingVersion::Xcdr1 => CdrWriter::new_with_header(buf).unwrap(),
372            EncodingVersion::Xcdr2 => CdrWriter::new_with_header_xcdr2(buf).unwrap(),
373        };
374        let dh = w.begin_dheader().unwrap();
375        write_fields(&mut w, fields);
376        w.end_dheader(dh).unwrap();
377        w.position()
378    }
379
380    fn write_fields(w: &mut CdrWriter<'_>, fields: &[Field]) {
381        for field in fields {
382            write_one(w, &field.ty);
383        }
384    }

386    fn write_one(w: &mut CdrWriter<'_>, ty: &FieldType) {
387        match ty {
388            FieldType::Bool => w.write_bool(true).unwrap(),
389            FieldType::Uint8 => w.write_u8(0xAB).unwrap(),
390            FieldType::Int8 => w.write_i8(-1).unwrap(),
391            FieldType::Uint16 => w.write_u16(0xBEEF).unwrap(),
392            FieldType::Int16 => w.write_i16(-2).unwrap(),
393            FieldType::Uint32 => w.write_u32(0xDEADBEEF).unwrap(),
394            FieldType::Int32 => w.write_i32(-3).unwrap(),
395            FieldType::Float32 => w.write_f32(1.5).unwrap(),
396            FieldType::Uint64 => w.write_u64(u64::MAX).unwrap(),
397            FieldType::Int64 => w.write_i64(-4).unwrap(),
398            FieldType::Float64 => w.write_f64(2.5).unwrap(),
399            // Maximal = exactly `n` payload bytes, which is what the bound
400            // claims room for.
401            FieldType::BoundedString(n) => {
402                let s = "x".repeat(*n);
403                w.write_string(&s).unwrap()
404            }
405            FieldType::String => w.write_string("").unwrap(),
406            FieldType::Array(n, inner) => {
407                for _ in 0..*n {
408                    write_one(w, inner);
409                }
410            }
411            FieldType::BoundedSequence(n, inner) => {
412                w.write_u32(*n as u32).unwrap();
413                for _ in 0..*n {
414                    write_one(w, inner);
415                }
416            }
417            FieldType::Nested(nested) => {
418                let dh = w.begin_dheader().unwrap();
419                write_fields(w, nested.fields);
420                w.end_dheader(dh).unwrap();
421            }
422            other => panic!("test writer has no maximal value for {other:?}"),
423        }
424    }

462    /// The defect issue 0776 calls out first: a message containing an `int64`

580        static WRAPPED: &[Field] = &[f("inner", FieldType::Nested(&INNER))];

582        let flat = max_serialized_size(FLAT, EncodingVersion::Xcdr2).unwrap();
583        let wrapped = max_serialized_size(WRAPPED, EncodingVersion::Xcdr2).unwrap();
584        assert_eq!(
585            wrapped,
586            flat + 4,
587            "the nested struct's own DHEADER must be counted under XCDR2"
588        );
589        // ...and must NOT be under XCDR1, where begin_dheader is a no-op.
590        assert_eq!(
591            max_serialized_size(WRAPPED, EncodingVersion::Xcdr1).unwrap(),
592            max_serialized_size(FLAT, EncodingVersion::Xcdr1).unwrap(),
593            "XCDR1 has no DHEADER; wrapping must cost nothing"
594        );
595    }
