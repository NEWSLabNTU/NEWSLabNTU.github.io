1//! Phase 380 W1 — a message's serialized size, computed instead of guessed.
2//!
3//! `NROS_SUBSCRIPTION_BUFFER_SIZE` defaults to 1024 bytes and is a GUESS.
4//! Nothing checks it against the messages an image actually subscribes to, so a
5//! sample that does not fit is dropped AFTER the transport ACKed it, and
6//! `report_dropped_take` can only say "raise the knob" because the runtime does
7//! not know what value would have worked. On a target that knob is static RAM
8//! nobody can spare. Issue 0776 is the gap; this module is the calculator.
9//!
10//! # Why this is not a vtable slot
11//!
12//! Settled in phase-376 W4: nothing about a size bound varies by backend, and
13//! upstream proves it by not varying at all — both `librmw_cyclonedds_cpp.so`
14//! and `librmw_fastrtps_cpp.so` answer `rmw_get_serialized_message_size` with
15//! `RMW_RET_UNSUPPORTED`, and nothing in a Humble install calls it. Upstream can
16//! afford that because its serialized buffer RESIZES; the bound is a hint that
17//! saves a realloc. Ours cannot resize, so the same number is load-bearing here.
18//!
19//! # Thread the offset — do not sum the maxima
20//!
21//! Padding is a function of WHERE a field starts, so a calculation that sums
22//! per-field maxima is wrong the moment a variable-length field shifts what
23//! follows. [`size_bound`] therefore takes `current_alignment` in and returns
24//! the size from there, which is also what makes nested structs compose with no
25//! special case — the same signature rosidl's generated Fast-RTPS support uses
26//! (`max_serialized_size_T(full_bounded, is_plain, current_alignment)`).
27//!
28//! # Agreement with the writer is the only thing that matters
29//!
30//! Every rule below was read out of `CdrWriter`, not out of the CDR spec:
31//! `align()` caps alignment at 4 under XCDR2 and honours 8 under XCDR1;
32//! `begin_dheader()` aligns to 4 and reserves 4 bytes for EVERY struct under
33//! XCDR2 (a generated `serialize` opens with it, so nested structs get one too);
34//! `write_string` writes `len + 1` and then the NUL. A bound that merely looks
35//! right is worthless — see `size_tests.rs`, which checks it against the bytes
36//! the writer actually produced.
37

45/// What one walk of a schema can say about size.
46#[derive(Debug, Clone, Copy, PartialEq, Eq)]
47pub struct SizeBound {
48    /// Bytes, given the starting offset the walk was handed.
49    pub bytes: usize,
50    /// False once an unbounded `String` / `WString` / `Sequence` is reached —
51    /// `bytes` is then a FLOOR, not a bound, and callers must not size a buffer
52    /// from it.
53    pub bounded: bool,
54    /// No variable-length member anywhere, so the WIRE length is fixed: `bytes`
55    /// is EXACT rather than an upper bound.
56    ///
57    /// **Nothing dispatches on this, by decision** (issues 1369 and 1422). It is
58    /// the walk's exactness property, asserted as such by
59    /// `plain_struct_bound_is_exact` here and by
60    /// `bound_holds_against_the_writer_for_every_generated_type` over the whole
61    /// generated corpus, and it is surfaced as
62    /// [`crate::schema::Message::IS_PLAIN`] / [`is_loan_eligible`] so there is
63    /// ONE definition of "fixed layout" for whoever finally needs one.
64    ///
65    /// The earlier text here said phase-380 W5 "wires this to
66    /// `borrow_loaned_message` / `subscription_supports_in_place`". It does not:
67    /// W5 shipped the definition and left the wiring to whoever owns those slots
68    /// (`docs/roadmap/archived/phase-380-serialized-size-bound.md:228-232`), and
69    /// **issue 0814 step 5 is where the wiring now waits** — deliberately, since
70    /// 0814 measures the loan surface as a cost on three of four backends and
71    /// the natural consumer of this predicate is Cyclone-with-iceoryx, which no
72    /// target we ship can reach. Do not gate the in-place path on it: that path
73    /// hands the callback raw CDR to deserialize, which needs no fixed layout, so
74    /// the flag would only take it away from every type with a `String`.
75    ///
76    /// It is NOT "POD-blit eligible", and a second flag under that name in the
77    /// codegen IR was deleted by issue 1422 for disagreeing with this one on
78    /// `bool`, on nested all-`float64` structs and on mixed alignment. Nor does
79    /// it make the bound version-INDEPENDENT: XCDR2's DHEADER is fixed-width, so
80    /// a plain type stays plain while its bound still differs per encoding
81    /// (`Time` measures 12 under XCDR1 and 16 under XCDR2).
82    pub plain: bool,
83}

85/// The encapsulation header the payload crossing our vtable carries: 2 bytes of
86/// representation id + 2 of options.
87///
88/// TOP LEVEL ONLY, not per nested struct — which is why the Cyclone backend
89/// computes `total = paylen + 4`. [`size_bound`] deliberately does not include
90/// it; [`max_serialized_size`] does.
91pub const ENCAPSULATION_HEADER_BYTES: usize = 4;

93/// Padding needed to reach `alignment` from `offset`, under `version`.
94///
95/// Mirrors `CdrWriter::align` exactly, including the XCDR2 cap at 4 — a message
96/// containing an `int64` therefore has TWO different bounds, and a single
97/// constant would be silently wrong for one encoding.
98const fn pad_to(offset: usize, alignment: usize, version: EncodingVersion) -> usize {
99    let alignment = match version {
100        EncodingVersion::Xcdr2 => {
101            if alignment > 4 {
102                4
103            } else {
104                alignment
105            }
106        }
107        EncodingVersion::Xcdr1 => alignment,
108    };
109    if alignment == 0 {
110        return 0;
111    }
112    (alignment - (offset % alignment)) % alignment
113}

115/// Size of one field, starting at `offset`. Returns the new offset plus the two
116/// flags, folded by the caller.
117const fn field_bound(
118    ty: &FieldType,
119    version: EncodingVersion,
120    offset: usize,
121) -> (usize, bool, bool) {
122    match ty {
123        FieldType::Bool | FieldType::Uint8 | FieldType::Int8 => (offset + 1, true, true),
124        FieldType::Uint16 | FieldType::Int16 => {
125            let o = offset + pad_to(offset, 2, version);
126            (o + 2, true, true)
127        }
128        FieldType::Uint32 | FieldType::Int32 | FieldType::Float32 => {
129            let o = offset + pad_to(offset, 4, version);
130            (o + 4, true, true)
131        }
132        FieldType::Uint64 | FieldType::Int64 | FieldType::Float64 => {
133            let o = offset + pad_to(offset, 8, version);
134            (o + 8, true, true)
135        }
136        // `write_string` writes `len + 1` as the u32 prefix and then the NUL,
137        // so a bound of `n` payload bytes costs `4 + n + 1`.
138        FieldType::BoundedString(n) => {
139            let o = offset + pad_to(offset, 4, version);
140            (o + 4 + *n + 1, true, false)
141        }
142        FieldType::BoundedWString(n) => {
143            let o = offset + pad_to(offset, 4, version);
144            (o + 4 + 2 * *n, true, false)
145        }
146        // No bound exists. Account the length prefix so `bytes` is an honest
147        // floor, and clear `bounded` so nobody sizes a buffer from it.
148        FieldType::String | FieldType::WString | FieldType::Sequence(_) => {
149            let o = offset + pad_to(offset, 4, version);
150            (o + 4, false, false)
151        }
152        FieldType::Array(n, inner) => {
153            let mut o = offset;
154            let mut bounded = true;
155            let mut plain = true;
156            let mut i = 0;
157            while i < *n {
158                let (next, b, p) = field_bound(inner, version, o);
159                o = next;
160                bounded &= b;
161                plain &= p;
162                i += 1;
163            }
164            // A fixed array of plain elements is itself plain; of anything else,
165            // not — the element count is fixed but each element's size is not.
166            (o, bounded, plain)
167        }
168        FieldType::BoundedSequence(n, inner) => {
169            let mut o = offset + pad_to(offset, 4, version);
170            o += 4;
171            let mut bounded = true;
172            let mut i = 0;
173            while i < *n {
174                let (next, b, _) = field_bound(inner, version, o);
175                o = next;
176                bounded &= b;
177                i += 1;
178            }
179            // Never plain: the wire length varies with the actual element count
180            // even though its maximum is known.
181            (o, bounded, false)
182        }
183        FieldType::Nested(nested) => {
184            let inner = size_bound(nested.fields, version, offset);
185            (inner.bytes, inner.bounded, inner.plain)
186        }
187    }
188}

190/// Walk `fields` from `current_alignment`, returning the size bound of the
191/// struct they describe.
192///
193/// The returned `bytes` is an ABSOLUTE offset — the position after the last
194/// field, measured from the same origin `current_alignment` was measured from —
195/// so a nested struct composes by being handed the parent's current offset.
196/// Subtract the starting offset if a length is what you want.
197///
198/// Excludes the encapsulation header; see [`max_serialized_size`].
199pub const fn size_bound(
200    fields: &'static [Field],
201    version: EncodingVersion,
202    current_alignment: usize,
203) -> SizeBound {
204    let mut offset = current_alignment;
205    let mut bounded = true;
206    let mut plain = true;
207
208    // XCDR2 delimits EVERY appendable struct, nested ones included: a generated
209    // `serialize` opens with `begin_dheader()`, which aligns to 4 and reserves
210    // 4. Under XCDR1 the call is a no-op. Missing this UNDER-reports, which is
211    // the dangerous direction — an under-reported bound sizes a buffer too
212    // small and reintroduces the very drop this exists to stop.
213    if matches!(version, EncodingVersion::Xcdr2) {
214        offset += pad_to(offset, 4, version);
215        offset += 4;
216        // A DHEADER does not make a struct non-plain: its width is fixed.
217    }
218
219    let mut i = 0;
220    while i < fields.len() {
221        let (next, b, p) = field_bound(&fields[i].ty, version, offset);
222        offset = next;
223        bounded &= b;
224        plain &= p;
225        i += 1;
226    }
227
228    SizeBound {
229        bytes: offset,
230        bounded,
231        plain,
232    }
233}

235/// How deep a field path [`first_unbounded`] will name before it stops.
236///
237/// ROS message nesting is shallow in practice (`PoseStamped.header.stamp.sec`
238/// is three), and this walk runs on `no_std` with no allocator, so the path is
239/// a fixed array rather than a `Vec`. A deeper type still reports — the path is
240/// simply truncated, and [`UnboundedField::truncated`] says so, because a path
241/// that silently stops short would point at the wrong member.
242pub const MAX_FIELD_PATH_DEPTH: usize = 8;
243
244/// Which unbounded member kind was reached.
245#[derive(Debug, Clone, Copy, PartialEq, Eq)]
246pub enum UnboundedKind {
247    /// `string` with no IDL bound.
248    String,
249    /// `wstring` with no IDL bound.
250    WString,
251    /// `sequence<T>` with no IDL bound.
252    Sequence,
253}
254
255impl UnboundedKind {
256    /// The IDL spelling, for a diagnostic.
257    pub const fn as_str(self) -> &'static str {
258        match self {
259            UnboundedKind::String => "string",
260            UnboundedKind::WString => "wstring",
261            UnboundedKind::Sequence => "sequence<T>",
262        }
263    }
264}

266/// The first member that makes a type unbounded, named.
267///
268/// [`max_serialized_size`] answers `None`, which is honest and useless on its
269/// own: a user told their type has no bound cannot act without knowing WHICH
270/// member costs it, and the offender is routinely three structs down in a type
271/// they did not write. This is that answer.
272#[derive(Debug, Clone, Copy, PartialEq, Eq)]
273pub struct UnboundedField {
274    /// Field names from the root outward; only `depth` entries are meaningful.
275    pub path: [&'static str; MAX_FIELD_PATH_DEPTH],
276    /// How many entries of `path` are set.
277    pub depth: usize,
278    /// True when nesting exceeded [`MAX_FIELD_PATH_DEPTH`] and `path` names a
279    /// prefix rather than the whole route.
280    pub truncated: bool,
281    /// What kind of member it is.
282    pub kind: UnboundedKind,
283}
284
285impl core::fmt::Display for UnboundedField {
286    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
287        for (i, seg) in self.path.iter().take(self.depth).enumerate() {
288            if i > 0 {
289                f.write_str(".")?;
290            }
291            f.write_str(seg)?;
292        }
293        if self.truncated {
294            f.write_str(".…")?;
295        }
296        write!(f, " ({})", self.kind.as_str())
297    }
298}

300/// Report EVERY member that makes `fields` unbounded, in declaration order.
301///
302/// [`first_unbounded`] answers "which member costs the bound" one member at a
303/// time, and that is the wrong shape for the caller that actually asks.
304/// phase-403 W0 made an unbounded type a BUILD ERROR, and a stock ROS type is
305/// routinely unbounded in several places at once — `nav_msgs/Odometry` has
306/// `header.frame_id` and `child_frame_id`, and `sensor_msgs/PointCloud2` has
307/// four. Naming only the first turns "bound your types" into cap, rebuild,
308/// discover the next one, repeat, once per member, with a whole codegen run
309/// between each step. One build should name everything that needs a bound.
310///
311/// THE walk. [`first_unbounded`] is expressed on top of this rather than beside
312/// it: two walks of one schema is the shape the sizes-header mirror defect keeps
313/// taking (issues 0088 -> 0268), and "the first thing this reports" is exactly
314/// what "the first unbounded member" means, so there is nothing left for a
315/// second implementation to say.
316///
317/// `visit` returns [`ControlFlow`] so a caller that wants only the first can
318/// stop the walk rather than let it enumerate a whole type and discard all but
319/// one answer. The return value is `Break` iff the visitor broke.
320///
321/// `&mut dyn FnMut` rather than `impl FnMut`: the walk recurses, and a recursive
322/// generic function cannot name the closure type it would instantiate itself
323/// with. `no_std` and allocation-free either way — nothing is collected here,
324/// which is what lets an all-members form exist at all on a target with no
325/// allocator. The COLLECTING is the caller's, and only codegen (`std`) does it.
326pub fn visit_unbounded(
327    fields: &'static [Field],
328    visit: &mut dyn FnMut(UnboundedField) -> ControlFlow<()>,
329) -> ControlFlow<()> {
330    fn walk(
331        fields: &'static [Field],
332        prefix: &mut [&'static str; MAX_FIELD_PATH_DEPTH],
333        depth: usize,
334        visit: &mut dyn FnMut(UnboundedField) -> ControlFlow<()>,
335    ) -> ControlFlow<()> {
336        for field in fields {
337            let truncated = depth >= MAX_FIELD_PATH_DEPTH;
338            if !truncated {
339                prefix[depth] = field.name;
340            }
341            let here = |kind: UnboundedKind| UnboundedField {
342                path: *prefix,
343                depth: if truncated {
344                    MAX_FIELD_PATH_DEPTH
345                } else {
346                    depth + 1
347                },
348                truncated,
349                kind,
350            };
351            match &field.ty {
352                FieldType::String => visit(here(UnboundedKind::String))?,
353                FieldType::WString => visit(here(UnboundedKind::WString))?,
354                FieldType::Sequence(_) => visit(here(UnboundedKind::Sequence))?,
355                // A fixed array or a bounded sequence is bounded only if its
356                // ELEMENT is, and the element can itself be an unbounded
357                // string — `string[4]` has no bound. The element carries no
358                // name of its own, so it reports at the field's own path, and
359                // ONCE: several offenders inside one unnamed element would all
360                // print the same path, which reads as a repeated line rather
361                // than as more information.
362                FieldType::Array(_, inner) | FieldType::BoundedSequence(_, inner) => {
363                    if let Some(kind) = element_unbounded(inner) {
364                        visit(here(kind))?;
365                    }
366                }
367                FieldType::Nested(nested) => {
368                    if truncated {
369                        // Cannot record another segment; report the deepest
370                        // path we can name rather than descending silently.
371                        walk(nested.fields, prefix, depth, &mut |mut u| {
372                            u.truncated = true;
373                            visit(u)
374                        })?;
375                    } else {
376                        walk(nested.fields, prefix, depth + 1, visit)?;
377                    }
378                }
379                _ => {}
380            }
381        }
382        ControlFlow::Continue(())
383    }
384
385    /// An element type is not a field, so it has no name — report only its kind.
386    fn element_unbounded(ty: &'static FieldType) -> Option<UnboundedKind> {
387        match ty {
388            FieldType::String => Some(UnboundedKind::String),
389            FieldType::WString => Some(UnboundedKind::WString),
390            FieldType::Sequence(_) => Some(UnboundedKind::Sequence),
391            FieldType::Array(_, inner) | FieldType::BoundedSequence(_, inner) => {
392                element_unbounded(inner)
393            }
394            FieldType::Nested(nested) => {
395                // A nested struct inside an array: any unbounded member counts.
396                first_unbounded(nested.fields).map(|u| u.kind)
397            }
398            _ => None,
399        }
400    }
401
402    let mut prefix = [""; MAX_FIELD_PATH_DEPTH];
403    walk(fields, &mut prefix, 0, visit)
404}

406/// Find the first member that makes `fields` unbounded, if any.
407///
408/// Returns `None` exactly when [`size_bound`] reports `bounded` — the two walk
409/// the same schema by the same rules, so a type that has a bound has no
410/// offender to name and vice versa. That agreement is asserted by test, not
411/// assumed: two walks of one schema is the shape the sizes-header mirror defect
412/// keeps taking (issues 0088 -> 0268), so the second one exists only because it
413/// answers a question the first cannot (WHICH member) and must be checked
414/// against it.
415///
416/// The walk is [`visit_unbounded`]; this is its "stop at the first one" caller.
417/// Reach for [`visit_unbounded`] when the diagnostic should name every member a
418/// user has to fix, which is what codegen wants now that an unbounded type is a
419/// build error (phase-403 W0).
420///
421/// Not `const`: it recurses through `&'static NestedType`, and the array
422/// bookkeeping is not worth expressing in a const walk when every caller is a
423/// diagnostic path.
424pub fn first_unbounded(fields: &'static [Field]) -> Option<UnboundedField> {
425    let mut found = None;
426    let _ = visit_unbounded(fields, &mut |u| {
427        found = Some(u);
428        ControlFlow::Break(())
429    });
430    found
431}

433/// The whole payload a publisher hands the transport: encapsulation header plus
434/// the struct's body, starting from offset 0.
435///
436/// `None` when the type is unbounded — the honest answer, and the one that keeps
437/// a caller from sizing a buffer off a floor. Reach for `serialized_size(&self)`
438/// (phase-380 W3) when an unbounded type still needs a number for THIS message.
439pub const fn max_serialized_size(
440    fields: &'static [Field],
441    version: EncodingVersion,
442) -> Option<usize> {
443    // The body's offsets are measured from after the encapsulation header —
444    // that is where `CdrWriter`'s `origin` sits — so the walk starts at 0 and
445    // the header is added once, at the top.
446    let bound = size_bound(fields, version, 0);
447    if bound.bounded {
448        Some(ENCAPSULATION_HEADER_BYTES + bound.bytes)
449    } else {
450        None
451    }
452}

454/// Phase 380 W3 — the EXACT serialized size of THIS message.
455///
456/// Two questions get asked about size and they are not the same one:
457///
458/// | question | asked by | this module |
459/// | --- | --- | --- |
460/// | how large can this TYPE ever be? | build-time buffer sizing | [`max_serialized_size`] |
461/// | how large is THIS message? | a publisher before publishing; a drop report | `serialized_size` |
462///
463/// The second is the only honest answer for an unbounded type, where the first
464/// is `None` — a `String` field has no maximum, but the string in hand has a
465/// length. It is what lets a drop report name the number that would have worked
466/// instead of saying "raise the knob".
467///
468/// Exact by construction: it runs the REAL writer with its stores disabled
469/// (`CdrWriter::measuring`), so the count comes from the same code that emits
470/// the bytes. A second walk of the schema could not see the actual string
471/// lengths and sequence counts, and would be a second implementation to keep in
472/// step besides.
473///
474/// Includes the encapsulation header, matching [`max_serialized_size`], so the
475/// two are directly comparable — which is the whole point at a call site
476/// deciding whether a message fits.
477pub fn serialized_size<T: crate::traits::Serialize>(
478    value: &T,
479    version: EncodingVersion,
480) -> Result<usize, crate::error::SerError> {
481    let mut w = crate::cdr::CdrWriter::measuring(&mut [], version);
482    value.serialize(&mut w)?;
483    Ok(ENCAPSULATION_HEADER_BYTES + w.position())
484}

486/// Phase 380 W4 — does a receive buffer of `rx_buf` bytes fit every message of
487/// this type?
488///
489/// `true` when the type is bounded and the bound fits. **`false` when the type
490/// is UNBOUNDED**, deliberately: no finite buffer fits a `String`, so the
491/// honest answer to "is this guaranteed to fit" is no. A caller that wants
492/// "fits unless proven otherwise" is asking a different question and should
493/// look at [`serialized_size`] per message.
494///
495/// Const, so a call site can put it in a `const { assert!(...) }` and turn a
496/// runtime drop into a build error:
497///
498/// ```ignore
499/// const { assert!(buffer_fits::<Odometry>(RX_BUF, EncodingVersion::Xcdr1)) };
500/// ```
501///
502/// # Why this is not simply a bound on `Subscription`
503///
504/// `Subscription<M, RX_BUF>` bounds `M: RosMessage`, which is a DIFFERENT trait
505/// from [`crate::schema::Message`] — and measured 2026-08-26, hand-written
506/// `RosMessage` types with no schema DO exist (`nros-core/src/service.rs`, the
507/// component-runtime tests). Tightening that bound would break them, so the
508/// assertion is opt-in at sites where the schema is known rather than universal
509/// at a site where it is not. See the phase doc for what closing that gap needs.
510pub const fn buffer_fits<M: crate::schema::Message>(
511    rx_buf: usize,
512    version: EncodingVersion,
513) -> bool {
514    let bound = match version {
515        EncodingVersion::Xcdr1 => M::MAX_SERIALIZED_SIZE_XCDR1,
516        EncodingVersion::Xcdr2 => M::MAX_SERIALIZED_SIZE_XCDR2,
517    };
518    match bound {
519        Some(n) => rx_buf >= n,
520        None => false,
521    }
522}

524/// Phase 392 W3a — the type's own receive-buffer bound, or `None` when it has
525/// none.
526///
527/// The VALUE behind [`bound_fits`]'s predicate. A subscription that knows its
528/// type's bound can be routed to a size class instead of forcing the GLOBAL
529/// buffer knob up: `NROS_SUBSCRIBER_BUFFER_SIZE` multiplies across
530/// `MAX_SUBSCRIBERS x RING_DEPTH`, so raising it from 1024 to 4096 for one
531/// 4 KiB topic costs 98,304 bytes, while the large class that topic belongs in
532/// is already reserved and empty.
533///
534/// Takes the larger of the two encodings, for the same reason `bound_fits`
535/// does: the peer picks the encoding at runtime, so sizing from XCDR1 alone is
536/// a trap.
537///
538/// `None` means "no bound EXISTS", never "unknown" — phase 380 is explicit that
539/// a buffer must not be sized from a fallback. A caller routing by size class
540/// must treat `None` as "keep the default", not as "assume small".
541pub const fn max_serialized_bound<M: crate::schema::Message>() -> Option<usize> {
542    match (M::MAX_SERIALIZED_SIZE_XCDR1, M::MAX_SERIALIZED_SIZE_XCDR2) {
543        (Some(x1), Some(x2)) => Some(if x1 > x2 { x1 } else { x2 }),
544        // One encoding unbounded makes the type unbounded on the wire, because
545        // the peer chooses. Not `Some(the_other)`.
546        _ => None,
547    }
548}

550/// Phase 380 W4 — the BUILD-ASSERTION predicate: `false` only when the type is
551/// provably too large for `rx_buf`.
552///
553/// Distinct from [`buffer_fits`], and the difference is the whole reason both
554/// exist:
555///
556/// * `buffer_fits` answers "is this GUARANTEED to fit", so an unbounded type is
557///   `false` — no finite buffer fits a `String`.
558/// * this answers "can we PROVE it will not fit", so an unbounded type is
559///   `true` — there is nothing to check, and failing the build for every
560///   `std_msgs/String` subscription would be absurd.
561///
562/// Asserting `buffer_fits` at a subscription site was my first attempt and
563/// would have refused the most common message in ROS. Kept as two named
564/// functions rather than one with a flag, because the wrong one is silently
565/// plausible at either call site.
566///
567/// Checks BOTH encodings and takes the larger: the encoding is a runtime
568/// property of the peer, so a buffer sized from XCDR1 alone is a trap — XCDR2
569/// adds a DHEADER per struct and is the bigger of the two for nested types.
570pub const fn bound_fits<M: crate::schema::Message>(rx_buf: usize) -> bool {
571    let x1 = match M::MAX_SERIALIZED_SIZE_XCDR1 {
572        Some(n) => n,
573        None => return true, // unbounded: nothing to prove
574    };

576        Some(n) => n,
577        None => return true,

948    fn write_one(w: &mut CdrWriter<'_>, ty: &FieldType) {
949        match ty {
950            FieldType::Bool => w.write_bool(true).unwrap(),
951            FieldType::Uint8 => w.write_u8(0xAB).unwrap(),
952            FieldType::Int8 => w.write_i8(-1).unwrap(),
953            FieldType::Uint16 => w.write_u16(0xBEEF).unwrap(),
954            FieldType::Int16 => w.write_i16(-2).unwrap(),
955            FieldType::Uint32 => w.write_u32(0xDEADBEEF).unwrap(),
956            FieldType::Int32 => w.write_i32(-3).unwrap(),
957            FieldType::Float32 => w.write_f32(1.5).unwrap(),
958            FieldType::Uint64 => w.write_u64(u64::MAX).unwrap(),
959            FieldType::Int64 => w.write_i64(-4).unwrap(),
960            FieldType::Float64 => w.write_f64(2.5).unwrap(),
961            // Maximal = exactly `n` payload bytes, which is what the bound
962            // claims room for.
963            FieldType::BoundedString(n) => {
964                let s = "x".repeat(*n);
965                w.write_string(&s).unwrap()
966            }
967            FieldType::String => w.write_string("").unwrap(),
968            FieldType::Array(n, inner) => {
969                for _ in 0..*n {
970                    write_one(w, inner);
971                }
972            }
973            FieldType::BoundedSequence(n, inner) => {
974                w.write_u32(*n as u32).unwrap();
975                for _ in 0..*n {
976                    write_one(w, inner);
977                }
978            }
979            FieldType::Nested(nested) => {
980                let dh = w.begin_dheader().unwrap();
981                write_fields(w, nested.fields);
982                w.end_dheader(dh).unwrap();
983            }
984            other => panic!("test writer has no maximal value for {other:?}"),
985        }
986    }

1163    /// XCDR2 delimits every struct, nested ones included. Missing the nested
1164    /// DHEADER under-reports — the dangerous direction.
1165    #[test]
1166    fn xcdr2_counts_a_dheader_per_struct() {
1167        static INNER: NestedType = NestedType {
1168            type_name: "t/Inner",
1169            fields: TIME,
1170        };
1171        static FLAT: &[Field] = &[f("sec", FieldType::Int32), f("nanosec", FieldType::Uint32)];
1172        static WRAPPED: &[Field] = &[f("inner", FieldType::Nested(&INNER))];
1173        both(WRAPPED);
1174        let flat = max_serialized_size(FLAT, EncodingVersion::Xcdr2).unwrap();
1175        let wrapped = max_serialized_size(WRAPPED, EncodingVersion::Xcdr2).unwrap();
1176        assert_eq!(
1177            wrapped,
1178            flat + 4,
1179            "the nested struct's own DHEADER must be counted under XCDR2"
1180        );

1182        assert_eq!(
1183            max_serialized_size(WRAPPED, EncodingVersion::Xcdr1).unwrap(),
1184            max_serialized_size(FLAT, EncodingVersion::Xcdr1).unwrap(),
1185            "XCDR1 has no DHEADER; wrapping must cost nothing"
1186        );
