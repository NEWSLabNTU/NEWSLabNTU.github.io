1// Issue 2638.
2//! The ONE reading of a launch parameter's baked text.
3//!
4//! An embedded image has no launch runtime to hand `--params-file` to, so the
5//! build bakes every launch parameter as TEXT and the image types it when it
6//! seeds the store. That text is written by
7//! `nros_orchestration_ir::launch_param_text` and read here; the two are one
8//! convention and are tested as a round trip.
9//!
10//! The reading follows what ROS gives the same value:
11//!
12//! - a scalar is `true` / `false` (any of the three cases rcl accepts), then
13//!   an `i64`, then an `f64`, then a string -- the order rcl's YAML reading
14//!   gives an unquoted scalar;
15//! - a whole text in single quotes is a STRING, with `''` read as `'`. The
16//!   writer quotes a string that would otherwise read as something else;
17//! - a YAML flow sequence `[a, b, c]` is an ARRAY. The first element decides
18//!   the element type and every other element must have it, which is rcl's
19//!   rule (`parse.c`: "Sequence should be of same type"): `[1, 2.5]` is
20//!   refused, not widened. A quoted element is a string.
21//!
22//! The list used to be baked as `a,b,c` and read as one string, so a
23//! `[40.0, 50.0, 60.0]` from a `config/*.param.yaml` reached the node as the
24//! string `"40.0,50.0,60.0"` -- and a one-element list as a scalar.
25//!
26//! No allocation: the element type is decided in one pass over the text and
27//! the value is filled in place in a second, so no temporary array exists.
28

32/// Why a baked text has no value to seed.
33#[derive(Debug, Clone, Copy, PartialEq, Eq)]
34pub enum LaunchTextError {
35    /// `[]`. ROS gives an empty sequence no value: rcl's file parser
36    /// allocates a sequence on its first element and never for `[]`, and
37    /// launch_ros refuses an inline `value="[]"` ("Expected a non-empty
38    /// sequence"). There is no element type to give it, so a seeder leaves
39    /// the declaration's default standing.
40    EmptyList,
41    /// A list whose elements are not all one type (`[1, 2.5]`, `[1, a]`).
42    /// rcl refuses the file and launch_ros refuses the inline value.
43    MixedList,
44    /// A string past `MAX_STRING_VALUE_LEN`, or a list past `MAX_ARRAY_LEN`
45    /// (`MAX_STRING_ARRAY_LEN` for a list of strings).
46    DoesNotFit,
47    /// An unterminated quote or a stray character after a quoted element.
48    Malformed,
49}

51#[derive(Clone, Copy, PartialEq, Eq)]
52enum Kind {

54    Integer,
55    Double,
56    String,

59fn boolean(t: &str) -> Option<bool> {
60    match t {
61        "true" | "True" | "TRUE" => Some(true),
62        "false" | "False" | "FALSE" => Some(false),
63        _ => None,
64    }
65}
66
67fn kind_of_unquoted(t: &str) -> Kind {
68    if boolean(t).is_some() {
69        Kind::Bool
70    } else if t.parse::<i64>().is_ok() {
71        Kind::Integer
72    } else if t.parse::<f64>().is_ok() {
73        Kind::Double
74    } else {
75        Kind::String
76    }
77}

79/// One element of a flow sequence, as written.
80#[derive(Clone, Copy)]
81enum Element<'a> {
82    /// Unquoted, trimmed.
83    Plain(&'a str),
84    /// Single-quoted; the slice is BETWEEN the quotes, `''` still doubled.
85    Single(&'a str),
86    /// Double-quoted; the slice is between the quotes, escapes not read.
87    Double(&'a str),
88}
89
90impl Element<'_> {
91    fn kind(self) -> Kind {
92        match self {
93            Element::Plain(t) => kind_of_unquoted(t),
94            Element::Single(_) | Element::Double(_) => Kind::String,
95        }
96    }
97}

99/// Walk the elements of the flow sequence body `inner` (the text between
100/// `[` and `]`), calling `f` on each. `Err(Malformed)` on a broken quote.
101fn for_each_element<'a>(
102    inner: &'a str,
103    mut f: impl FnMut(Element<'a>) -> Result<(), LaunchTextError>,
104) -> Result<usize, LaunchTextError> {
105    let bytes = inner.as_bytes();
106    let mut i = 0;
107    let mut n = 0;
108    let skip_ws = |i: &mut usize| {
109        while *i < bytes.len() && bytes[*i].is_ascii_whitespace() {
110            *i += 1;
111        }
112    };
113    skip_ws(&mut i);
114    if i == bytes.len() {
115        return Ok(0);
116    }
117    loop {
118        skip_ws(&mut i);
119        let element = match bytes.get(i) {
120            Some(b'\'') => {
121                let start = i + 1;
122                let mut j = start;
123                loop {
124                    match bytes.get(j) {
125                        None => return Err(LaunchTextError::Malformed),
126                        Some(b'\'') if bytes.get(j + 1) == Some(&b'\'') => j += 2,
127                        Some(b'\'') => break,
128                        Some(_) => j += 1,
129                    }
130                }
131                i = j + 1;
132                Element::Single(&inner[start..j])
133            }
134            Some(b'"') => {
135                let start = i + 1;
136                let mut j = start;
137                loop {
138                    match bytes.get(j) {
139                        None => return Err(LaunchTextError::Malformed),
140                        Some(b'\\') => j += 2,
141                        Some(b'"') => break,
142                        Some(_) => j += 1,
143                    }
144                }
145                i = j + 1;
146                Element::Double(&inner[start..j])
147            }
148            _ => {
149                let start = i;
150                while i < bytes.len() && bytes[i] != b',' {
151                    i += 1;
152                }
153                Element::Plain(inner[start..i].trim())
154            }
155        };
156        f(element)?;
157        n += 1;
158        skip_ws(&mut i);
159        match bytes.get(i) {
160            None => return Ok(n),
161            Some(b',') => {
162                i += 1;
163                // YAML allows one trailing comma: `[a, b,]` is two elements.
164                skip_ws(&mut i);
165                if i == bytes.len() {
166                    return Ok(n);
167                }
168            }
169            Some(_) => return Err(LaunchTextError::Malformed),
170        }
171    }
172}

174/// Push `'...'`'s content into `out`, reading `''` as `'`.
175fn push_single_quoted<const N: usize>(out: &mut String<N>, body: &str) -> bool {
176    let mut rest = body;
177    while let Some(at) = rest.find("''") {
178        if out.push_str(&rest[..=at]).is_err() {
179            return false;
180        }
181        rest = &rest[at + 2..];
182    }
183    out.push_str(rest).is_ok()
184}
185
186fn list_body(text: &str) -> Option<&str> {
187    text.strip_prefix('[')?.strip_suffix(']')
188}
189
190fn single_quoted_body(text: &str) -> Option<&str> {
191    let body = text.strip_prefix('\'')?.strip_suffix('\'')?;
192    // `'a' 'b'` is not one quoted scalar: every inner quote must be doubled.
193    let mut rest = body;
194    while let Some(at) = rest.find('\'') {
195        if rest.as_bytes().get(at + 1) != Some(&b'\'') {
196            return None;
197        }
198        rest = &rest[at + 2..];
199    }
200    Some(body)
201}

203/// A flow sequence body's element type and element count: the first element
204/// decides the type and every other must have it (rcl's rule). The ONE pass both
205/// readers below take.
206fn list_kind(inner: &str) -> Result<(Kind, usize), LaunchTextError> {
207    let mut kind: Option<Kind> = None;
208    let n = for_each_element(inner, |e| {
209        let k = e.kind();
210        match kind {
211            None => kind = Some(k),
212            Some(seen) if seen != k => return Err(LaunchTextError::MixedList),
213            Some(_) => {}
214        }
215        Ok(())
216    })?;
217    match kind {
218        Some(kind) => Ok((kind, n)),
219        None => Err(LaunchTextError::EmptyList),
220    }
221}

223/// The TYPE a launch parameter's baked text seeds the store with.
224#[derive(Debug, Clone, Copy, PartialEq, Eq)]
225pub enum LaunchTextType {
226    Bool,
227    Integer,
228    Double,
229    String,
230    BoolArray,
231    IntegerArray,
232    DoubleArray,
233    StringArray,
234}

236// Issue 2768 (a build sizing the store counts launch seeds).
237/// The type [`ParameterValue::assign_launch_text`] would give
238/// `text`, without filling a value and without a capacity check.
239///
240/// For a BUILD that sizes the store before any value exists: a launch seed
241/// takes a slot of this type, so the host counts it by the device's own
242/// reading rather than a second one that could disagree. No capacity, because
243/// the build is what decides the capacities: a seed too long for today's store
244/// is exactly what the count must grow it for, not a reason to skip it.
245/// `Err` only where the device refuses the text whatever the capacity: an
246/// empty list, a mixed list, a broken quote.
247pub fn launch_text_type(text: &str) -> Result<LaunchTextType, LaunchTextError> {
248    if let Some(inner) = list_body(text.trim()) {
249        let (kind, _) = list_kind(inner)?;
250        return Ok(match kind {
251            Kind::Bool => LaunchTextType::BoolArray,
252            Kind::Integer => LaunchTextType::IntegerArray,
253            Kind::Double => LaunchTextType::DoubleArray,
254            Kind::String => LaunchTextType::StringArray,
255        });
256    }
257    if single_quoted_body(text).is_some() {
258        return Ok(LaunchTextType::String);
259    }
260    Ok(match kind_of_unquoted(text) {
261        Kind::Bool => LaunchTextType::Bool,
262        Kind::Integer => LaunchTextType::Integer,
263        Kind::Double => LaunchTextType::Double,
264        Kind::String => LaunchTextType::String,
265    })
266}

268impl ParameterValue {
269    /// Type a launch parameter's baked text and write it into `self`, IN
270    /// PLACE (a value is ~8 KiB; see `assign_string`). The reading is the
271    /// module's: scalar inference, a single-quoted string, or a flow
272    /// sequence typed by its elements.
273    ///
274    /// On `Err`, `self` is unspecified and must not be stored.
275    pub fn assign_launch_text(&mut self, text: &str) -> Result<(), LaunchTextError> {
276        if let Some(inner) = list_body(text.trim()) {
277            return self.assign_flow_list(inner);
278        }
279        if let Some(body) = single_quoted_body(text) {
280            if !matches!(self, Self::String(_)) {
281                *self = Self::String(String::new());
282            }
283            return match self {
284                Self::String(s) => {
285                    s.clear();
286                    if push_single_quoted(s, body) {
287                        Ok(())
288                    } else {
289                        Err(LaunchTextError::DoesNotFit)
290                    }
291                }
292                _ => Err(LaunchTextError::DoesNotFit),
293            };
294        }
295        if let Some(b) = boolean(text) {
296            *self = Self::Bool(b);
297        } else if let Ok(i) = text.parse::<i64>() {
298            *self = Self::Integer(i);
299        } else if let Ok(d) = text.parse::<f64>() {
300            *self = Self::Double(d);
301        } else if !self.assign_string(text) {
302            return Err(LaunchTextError::DoesNotFit);
303        }
304        Ok(())
305    }
306
307    fn assign_flow_list(&mut self, inner: &str) -> Result<(), LaunchTextError> {
308        // Pass 1: the element type, which the first element decides.
309        let (kind, n) = list_kind(inner)?;
310        // A string list has its own element bound (issue 2647).
311        let cap = if kind == Kind::String {
312            crate::types::MAX_STRING_ARRAY_LEN
313        } else {
314            crate::types::MAX_ARRAY_LEN
315        };
316        if n > cap {
317            return Err(LaunchTextError::DoesNotFit);
318        }
319        // Pass 2: fill the value of that type in place.
320        *self = match kind {
321            Kind::Bool => Self::BoolArray(Vec::new()),
322            Kind::Integer => Self::IntegerArray(Vec::new()),
323            Kind::Double => Self::DoubleArray(Vec::new()),
324            Kind::String => Self::StringArray(Vec::new()),
325        };
326        let full = LaunchTextError::DoesNotFit;
327        for_each_element(inner, |e| {
328            let pushed = match (&mut *self, e) {
329                (Self::BoolArray(v), Element::Plain(t)) => {
330                    v.push(boolean(t).unwrap_or(false)).is_ok()
331                }
332                (Self::IntegerArray(v), Element::Plain(t)) => {
333                    v.push(t.parse::<i64>().unwrap_or(0)).is_ok()
334                }
335                (Self::DoubleArray(v), Element::Plain(t)) => {
336                    v.push(t.parse::<f64>().unwrap_or(0.0)).is_ok()
337                }
338                (Self::StringArray(v), e) => {
339                    let mut s = String::new();
340                    let ok = match e {
341                        Element::Plain(t) | Element::Double(t) => s.push_str(t).is_ok(),
342                        Element::Single(t) => push_single_quoted(&mut s, t),
343                    };
344                    ok && v.push(s).is_ok()
345                }
346                _ => false,
347            };
348            if pushed { Ok(()) } else { Err(full) }
349        })?;
350        Ok(())
351    }
352}

363    /// Issue 2768 -- the build's type reading and the device's value reading
364    /// agree on every text the device can store, and refuse the same texts.
365    #[test]
366    fn the_type_reader_agrees_with_the_value_reader() {
367        let corpus = [
368            "true",
369            "FALSE",
370            "42",
371            "-7",
372            "1.0",
373            "1e3",
374            "hello",
375            "",
376            "40.0,50.0",
377            "'1.0'",
378            "'[a]'",
379            "'it''s'",
380            "''",
381            "'a' 'b'",
382            "[true, false]",
383            "[1, 2, 3]",
384            "[1.5, 2.5]",
385            "[a, b]",
386            "['x', y]",
387            "[1, 2.5]",
388            "[]",
389            "[1, a]",
390            "['broken]",
391            " [1, 2] ",
392        ];
393        for text in corpus {
394            let ty = launch_text_type(text);
395            match read(text) {
396                Ok(v) => {
397                    let want = match v {
398                        ParameterValue::Bool(_) => LaunchTextType::Bool,
399                        ParameterValue::Integer(_) => LaunchTextType::Integer,
400                        ParameterValue::Double(_) => LaunchTextType::Double,
401                        ParameterValue::String(_) => LaunchTextType::String,
402                        ParameterValue::BoolArray(_) => LaunchTextType::BoolArray,
403                        ParameterValue::IntegerArray(_) => LaunchTextType::IntegerArray,
404                        ParameterValue::DoubleArray(_) => LaunchTextType::DoubleArray,
405                        ParameterValue::StringArray(_) => LaunchTextType::StringArray,
406                        other => panic!("{text:?}: unexpected {other:?}"),
407                    };
408                    assert_eq!(ty, Ok(want), "{text:?}");
409                }
410                Err(LaunchTextError::DoesNotFit) => {
411                    assert!(ty.is_ok(), "{text:?}: a capacity is not a type error")
412                }
413                Err(e) => assert_eq!(ty, Err(e), "{text:?}"),
414            }
415        }
416        // A list longer than any store is still typed: the build grows for it.
417        extern crate std;
418        let long = std::format!("[{}]", std::vec!["1"; 4096].join(", "));
419        assert_eq!(launch_text_type(&long), Ok(LaunchTextType::IntegerArray));
420    }
