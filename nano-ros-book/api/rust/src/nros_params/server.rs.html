15/// Entry in the parameter storage.
16///
17/// phase-382 W2' — crate-private ON PURPOSE. The storage is caller-owned now,
18/// but the SLOT layout is not part of the deal: consumers name
19/// [`ParameterTable`] / [`ParameterStorage`], never this.
20#[derive(Debug, Clone)]
21pub(crate) struct ParameterEntry {
22    /// phase-426 W1 — which node declared it. The slot arena is shared (one
23    /// fixed table per executor, no per-node allocation); this is what keeps
24    /// two nodes' identically-named parameters apart inside it.
25    node: NodeKey,
26    /// The parameter (name + value)
27    param: Parameter,
28    /// Optional descriptor with constraints
29    descriptor: Option<ParameterDescriptor>,
30}

32/// The slot table a [`ParameterServer`] borrows.
33///
34/// phase-382 W2' — the server's storage is CALLER-OWNED, and this newtype is
35/// what makes that expressible without publishing the slot type. A slot is a
36/// `ParameterEntry`, which stays PRIVATE (it is the store's internal slot
37/// layout, not an API); a consumer that only needs to *name* the storage —
38/// `nros-node`, which carves it out of the executor backing in W3' — names a
39/// `ParameterTable` instead.
40///
41/// **The table's LENGTH is the server's capacity.** [`MAX_PARAMETERS`] is the
42/// DEFAULT capacity (see [`ParameterStorage`]), not a bound the borrow
43/// re-imposes: a caller who hands over a longer or shorter table gets exactly
44/// that many slots, and every capacity question the server answers
45/// ([`is_full`](ParameterServer::is_full), the declare path) reads it from
46/// here.
47pub struct ParameterTable<'s> {
48    entries: &'s mut [Option<ParameterEntry>],
49}
50
51impl<'s> ParameterTable<'s> {
52    /// Wrap a caller-owned slice of slots.
53    pub(crate) fn new(entries: &'s mut [Option<ParameterEntry>]) -> Self {
54        Self { entries }
55    }
56
57    /// Number of slots — the capacity of a server built on this table.
58    pub fn capacity(&self) -> usize {
59        self.entries.len()
60    }
61
62    /// Number of slots currently holding a parameter.
63    pub fn occupied(&self) -> usize {
64        self.entries.iter().filter(|slot| slot.is_some()).count()
65    }
66}

69    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
70        f.debug_struct("ParameterTable")
71            .field("capacity", &self.capacity())
72            .finish_non_exhaustive()
73    }

76/// Backing storage a caller places, then lends to a [`ParameterServer`].
77///
78/// `N` defaults to [`MAX_PARAMETERS`], the build-time knob
79/// (`NROS_MAX_PARAMETERS` / `CONFIG_NROS_MAX_PARAMETERS`). It is a DEFAULT,
80/// not a cap — `ParameterStorage::<8>::new()` and `ParameterStorage::<256>`
81/// are equally valid, and the server takes its capacity from whatever table it
82/// is handed.
83///
84/// The intended shapes are a `static`, a caller-owned struct field, or (with
85/// `alloc`) one allocation made once at start-up:
86///
87/// ```
88/// use nros_params::{NodeKey, ParameterServer, ParameterStorage, ParameterValue};
89///
90/// let mut storage: ParameterStorage<8> = ParameterStorage::new();
91/// let mut server = ParameterServer::new_in(storage.as_table());
92/// assert!(server.declare(NodeKey::PRIMARY, "max_speed", ParameterValue::Double(1.0)));
93/// ```
94///
95/// # Why this type exists at all — issue 0756
96///
97/// This is where the size lives, and it is enormous: `ParameterValue` is sized
98/// by its largest variant
99/// (`StringArray(Vec<String<MAX_STRING_VALUE_LEN>, MAX_ARRAY_LEN>)`), so every
100/// slot costs ~8.5 KiB whatever it actually holds. Measured: **285,184 bytes**
101/// at the default 32 slots, **2,281,472** at 256. (The eight extra bytes in
102/// issue 0756's 285,192 were the `count` that stayed behind on
103/// [`ParameterServer`], which is 24 bytes total now.)
104///
105/// Before phase-382 W2' that bulk was INSIDE `ParameterServer`, so
106/// `Box::new(ParameterServer::new())` built a multi-hundred-KiB value on the
107/// STACK before copying it into the allocation — Rust has no placement-new —
108/// which silently overran a thread stack sized for anything smaller. On the
109/// Zephyr lane an image built with 256 slots booted to
110/// `dds_create_participant` and hung with no fault and no output, while 32 —
111/// which fits the 512 KiB main stack the cyclonedds snippet asks for — ran
112/// clean. The mitigation then was `ParameterServer::init_in_place`, a
113/// placement initialiser that bounded the largest temporary at one
114/// `Option<ParameterEntry>`.
115///
116/// W2' dissolves the *server* half of that: `ParameterServer` is now a table
117/// borrow plus a count (tens of bytes), so nothing about constructing a server
118/// depends on the knob. The bulk did not vanish, it MOVED here — and so did
119/// the hazard, verbatim: `Box::new(ParameterStorage::new())` is the same
120/// stack-first copy the issue was about. [`init_in_place`] is therefore kept,
121/// on the type that is actually big, and any heap-placed storage must go
122/// through it. A caller who places the storage as a `static` or lets W3' carve
123/// it out of the executor backing never materialises it by value and does not
124/// need it.
125///
126/// [`init_in_place`]: ParameterStorage::init_in_place
127pub struct ParameterStorage<const N: usize = MAX_PARAMETERS> {
128    entries: [Option<ParameterEntry>; N],
129}
130
131impl<const N: usize> ParameterStorage<N> {
132    /// Create empty storage.
133    ///
134    /// `const`, so a `static NROS_PARAMS: ParameterStorage = ParameterStorage::new();`
135    /// lands in `.bss` with no initialiser and no stack temporary.
136    // No `Default`: it would spell this same construction as a by-value
137    // expression, which is exactly the `Box::new(Default::default())` shape
138    // issue 0756 is about (see the type docs). Callers name `new()` — or
139    // `init_in_place` when the home is a heap allocation.
140    #[allow(clippy::new_without_default)]
141    #[allow(clippy::large_stack_arrays)] // Intentional: this IS the storage.
142    pub const fn new() -> Self {
143        Self {
144            entries: [const { None }; N],
145        }
146    }
147
148    /// Slots in this storage.
149    pub const fn capacity(&self) -> usize {
150        N
151    }
152
153    /// Lend the storage to a [`ParameterServer`].
154    pub fn as_table(&mut self) -> ParameterTable<'_> {
155        ParameterTable::new(&mut self.entries)
156    }
157
158    /// Initialise storage directly into `dst`, without ever materialising one
159    /// by value.
160    ///
161    /// Issue 0756 — see this type's documentation for why that matters. The
162    /// short version: this type is 285,184 bytes at the default 32 slots and
163    /// 2,281,472 at 256, and Rust has no placement-new, so
164    /// `Box::new(ParameterStorage::new())` builds all of it on the caller's
165    /// stack first. Initialising through the allocation bounds the largest
166    /// stack temporary at one `Option<ParameterEntry>`, so the
167    /// `NROS_MAX_PARAMETERS` knob no longer decides whether boot survives.
168    ///
169    /// # Safety
170    ///
171    /// `dst` must be non-null, correctly aligned, and valid for writes of
172    /// `size_of::<Self>()` bytes. The pointee may be uninitialised on entry;
173    /// it is fully initialised on return.
174    pub unsafe fn init_in_place(dst: *mut Self) {
175        // addr_of_mut! throughout: `(*dst).entries` as a place expression
176        // would require the pointee to already be initialised.
177        // Safety: delegated to this function's own contract on `dst`.
178        unsafe {
179            let entries = core::ptr::addr_of_mut!((*dst).entries).cast::<Option<ParameterEntry>>();
180            for i in 0..N {
181                entries.add(i).write(None);
182            }
183        }
184    }
185}

187/// Parameter server over caller-owned storage
188///
189/// Stores parameters in a [`ParameterTable`] the caller lends it; the table's
190/// length is the capacity. [`ParameterStorage`] is the usual way to produce
191/// one, defaulting to the build-time `MAX_PARAMETERS` slots.
192///
193/// # Keyed by node — phase-426 W1
194///
195/// Every per-parameter method takes a [`NodeKey`] first. The slot arena is
196/// shared (one table per executor), but a name is only ever looked up within
197/// one node, so two nodes composed onto one executor (RFC-0047) hold
198/// independent values for the same name — which is what upstream does, where
199/// each node owns a table and a set of six services.
200///
201/// # Example
202///
203/// ```
204/// use nros_params::{NodeKey, ParameterServer, ParameterStorage, ParameterValue};
205///
206/// let mut storage = ParameterStorage::<8>::new();
207/// let mut server = ParameterServer::new_in(storage.as_table());
208/// let talker = NodeKey::new(0);
209/// let listener = NodeKey::new(1);
210///
211/// // Declare a parameter with initial value
212/// server.declare(talker, "rate", ParameterValue::Double(1.0));
213/// server.declare(listener, "rate", ParameterValue::Double(2.0));
214///
215/// // Get parameter value
216/// assert_eq!(
217///     server.get(talker, "rate").and_then(|v| v.as_double()),
218///     Some(1.0)
219/// );
220/// assert_eq!(
221///     server.get(listener, "rate").and_then(|v| v.as_double()),
222///     Some(2.0)
223/// );
224///
225/// // Set parameter value
226/// server.set(talker, "rate", ParameterValue::Double(3.0));
227/// ```
228pub struct ParameterServer<'s> {
229    /// Borrowed parameter storage
230    table: ParameterTable<'s>,
231    /// Number of parameters currently stored, across every node.
232    count: usize,
233    /// May a wire-facing set ([`apply`](Self::apply)) DECLARE a name the node
234    /// does not hold? Off by default, as upstream
235    /// (`allow_undeclared_parameters` in rclcpp's `NodeOptions`,
236    /// `allow_undeclared` in rclrs) — issue 1151.
237    ///
238    /// phase-426 W1 — PER NODE, because upstream's is a node option. One node
239    /// opting in must not decide what a remote set may do to its sibling.
240    allow_undeclared: NodeFlags,
241    /// phase-446 F2 -- a declared descriptor holds a truncated description
242    /// nobody has reported yet. Lets [`Self::take_truncated_descriptions`]
243    /// answer "nothing" without walking the table, which is what lets the
244    /// executor ask on every spin.
245    truncation_pending: bool,
246}

249    /// phase-446 F2 -- report each declared parameter whose description was
250    /// truncated to `NROS_MAX_PARAM_DESCRIPTION_LEN`, ONCE, and forget it.
251    ///
252    /// This crate has no logger, so it records the truncation on the
253    /// descriptor and the executor, which has one, drains it here and logs.
254    /// The truncation itself happens where the text is given
255    /// ([`ParameterDescriptor::set_description`]), and a declaration can come
256    /// through several builders; the store is where every one of them lands.
257    /// Returns how many were reported.
258    pub fn take_truncated_descriptions(&mut self, mut report: impl FnMut(NodeKey, &str)) -> usize {
259        if !self.truncation_pending {
260            return 0;
261        }
262        self.truncation_pending = false;
263        let mut n = 0;
264        for entry in self.table.entries.iter_mut().flatten() {
265            if let Some(d) = entry.descriptor.as_mut()
266                && d.description_truncated()
267            {
268                d.clear_description_truncated();
269                report(entry.node, entry.param.name.as_str());
270                n += 1;
271            }
272        }
273        n
274    }

279        if descriptor.is_some_and(ParameterDescriptor::description_truncated) {

284    /// Create a parameter server over `table`.
285    ///
286    /// The server ADOPTS whatever the table already holds: `count` is
287    /// recovered from the occupied slots rather than assumed zero. That is
288    /// what makes a table safe to hand to a second server after the first is
289    /// dropped (and what lets storage arrive pre-seeded), where assuming empty
290    /// would desynchronise the count from the slots — `is_full` wrong in one
291    /// direction, `remove` underflowing in the other. Fresh
292    /// [`ParameterStorage`] is all-`None`, so the usual path costs one scan
293    /// and starts at zero.
294    pub fn new_in(table: ParameterTable<'s>) -> Self {
295        let count = table.occupied();
296        Self {
297            table,
298            count,
299            allow_undeclared: NodeFlags::new(),
300            // An adopted table may already hold a truncated description, so
301            // the first drain looks rather than assuming none.
302            truncation_pending: true,
303        }
304    }

396    /// Declare a new parameter with value and descriptor on `node`
397    ///
398    /// The descriptor provides metadata like description, constraints, and read-only flag.
399    ///
400    /// phase-428 W6 remainder: see [`Self::declare`]. Every `false` arm below
401    /// is a REFUSAL (name taken, table full, value rejected), never a question.
402    #[must_use]
403    pub fn declare_with_descriptor(
404        &mut self,
405        node: NodeKey,
406        name: &str,
407        value: ParameterValue,
408        descriptor: Option<ParameterDescriptor>,
409    ) -> bool {
410        // Check if already exists
411        if self.find_index(node, name).is_some() {
412            return false;
413        }
414
415        // Issue 1150 — an ill-formed range, or a default the range rejects,
416        // is refused at declaration, as upstream. Same predicate as every set.
417        if !Self::declaration_is_consistent(descriptor.as_ref(), &value) {
418            return false;
419        }
420
421        // Find an empty slot
422        let slot = match self.find_empty_slot() {
423            Some(idx) => idx,
424            None => return false,
425        };
426
427        // Create the parameter
428        let param = match Parameter::new(name, value) {
429            Some(p) => p,
430            None => return false,
431        };
432
433        self.note_descriptor(descriptor.as_ref());
434        self.table.entries[slot] = Some(ParameterEntry {
435            node,
436            param,
437            descriptor,
438        });
439        self.count += 1;
440        true
441    }

443    /// The declare-time half of issue 1150: the range must be well-formed
444    /// and the initial value must satisfy it (rclrs `InvalidRange` /

446    fn declaration_is_consistent(
447        descriptor: Option<&ParameterDescriptor>,
448        value: &ParameterValue,
449    ) -> bool {
450        match descriptor {
451            Some(desc) => desc.range.is_valid() && desc.validate_range(value),
452            None => true,

456    /// Would `set(name, value)` succeed on an EXISTING entry? Read-only,
457    /// then type, then range — the one ordering every set path uses.
458    fn check_set_existing(entry: &ParameterEntry, value: &ParameterValue) -> SetParameterResult {
459        if let Some(ref desc) = entry.descriptor {
460            if desc.read_only {
461                return SetParameterResult::ReadOnly;
462            }
463            if !desc.dynamic_typing && desc.param_type != value.param_type() {
464                return SetParameterResult::TypeMismatch;
465            }
466            if !desc.validate_range(value) {
467                return SetParameterResult::OutOfRange;
468            }
469        }
470        SetParameterResult::Success
471    }

473    /// Dry-run of [`apply`](Self::apply): the verdict a wire-facing set would
474    /// get, without changing anything. `SetParametersAtomically` runs this
475    /// over the whole batch before applying any of it.
476    pub fn check_apply(
477        &self,
478        node: NodeKey,
479        name: &str,
480        value: &ParameterValue,
481    ) -> SetParameterResult {
482        match self
483            .find_index(node, name)
484            .and_then(|idx| self.table.entries[idx].as_ref())
485        {
486            Some(entry) => Self::check_set_existing(entry, value),
487            None if !self.allows_undeclared(node) => SetParameterResult::Undeclared,
488            None if self.is_full() => SetParameterResult::StorageFull,
489            None if name.len() > MAX_PARAM_NAME_LEN => SetParameterResult::StorageFull,
490            None => SetParameterResult::Success,
491        }
492    }

494    /// THE wire-facing set — what one element of `SetParameters` /
495    /// `SetParametersAtomically` does (issue 1151).
496    ///
497    /// An existing parameter is set through [`set`](Self::set). An unknown
498    /// name is [`Undeclared`](SetParameterResult::Undeclared) unless
499    /// [`set_allow_undeclared`](Self::set_allow_undeclared) was switched on
500    /// FOR THAT NODE, in which case it is declared (or `StorageFull`). Every
501    /// service handler and every facade writer goes through here so the
502    /// by-value oracle, the streaming path, the atomic pre-check and the C
503    /// setters cannot disagree about what "undeclared" means.
504    pub fn apply(
505        &mut self,
506        node: NodeKey,
507        name: &str,
508        value: ParameterValue,
509    ) -> SetParameterResult {
510        let verdict = self.check_apply(node, name, &value);
511        if !verdict.is_success() {
512            return verdict;
513        }
514        if self.find_index(node, name).is_some() {
515            self.set(node, name, value)
516        } else if self.declare(node, name, value) {
517            SetParameterResult::Success
518        } else {
519            SetParameterResult::StorageFull
520        }
521    }

544    /// Set a parameter value
545    ///
546    /// Returns the result of the set operation.
547    pub fn set(&mut self, node: NodeKey, name: &str, value: ParameterValue) -> SetParameterResult {
548        let idx = match self.find_index(node, name) {
549            Some(idx) => idx,
550            None => return SetParameterResult::NotFound,
551        };
552
553        let entry = match self.table.entries[idx].as_mut() {
554            Some(e) => e,
555            None => return SetParameterResult::NotFound,
556        };
557
558        // Read-only, type, range (min/max/step) — one predicate, shared with
559        // the atomic pre-check.
560        let verdict = Self::check_set_existing(entry, &value);
561        if !verdict.is_success() {
562            return verdict;
563        }
564
565        entry.param.value = value;
566        SetParameterResult::Success
567    }

704    /// Declare a parameter with a descriptor and optional initial value.
705    ///
706    /// This is a more comprehensive declaration method used by the typed parameter API.
707    ///
708    /// # Arguments
709    /// - `node`: The node the parameter belongs to.
710    /// - `descriptor`: The metadata for the parameter.
711    /// - `initial_value`: An optional initial value for the parameter. If `None`,
712    ///   the parameter will be initialized to `ParameterValue::NotSet`.
713    pub fn declare_parameter(
714        &mut self,
715        node: NodeKey,
716        descriptor: ParameterDescriptor,
717        initial_value: Option<&ParameterValue>,
718    ) -> Result<(), SetParameterResult> {
719        let name = descriptor.name.as_str();
720
721        // Check if already exists
722        if self.find_index(node, name).is_some() {
723            return Err(SetParameterResult::TypeMismatch); // Indicates already declared
724        }
725
726        let param_value = initial_value.cloned().unwrap_or_default();
727
728        // Issue 1150 — same declare-time check as `declare_with_descriptor`.
729        if !Self::declaration_is_consistent(Some(&descriptor), &param_value) {
730            return Err(SetParameterResult::InvalidRange);
731        }
732
733        // Find an empty slot
734        let slot = match self.find_empty_slot() {
735            Some(idx) => idx,
736            None => return Err(SetParameterResult::StorageFull),
737        };
738
739        let param = Parameter::new(name, param_value).ok_or(SetParameterResult::StorageFull)?; // name too long
740
741        self.note_descriptor(Some(&descriptor));
742        self.table.entries[slot] = Some(ParameterEntry {
743            node,
744            param,
745            descriptor: Some(descriptor),
746        });
747        self.count += 1;
748        Ok(())
749    }

773/// Builder for declaring parameters with a fluent API
774pub struct LegacyParameterBuilder<'a, 's> {
775    server: &'a mut ParameterServer<'s>,
776    /// phase-426 W1 — the node this declaration belongs to, fixed when the
777    /// builder is made so no chained call can lose it.
778    node: NodeKey,
779    name: String<MAX_PARAM_NAME_LEN>,
780    value: ParameterValue,
781    descriptor: Option<ParameterDescriptor>,
782}
783
784impl<'a, 's> LegacyParameterBuilder<'a, 's> {
785    /// Create a new parameter builder
786    pub fn new(
787        server: &'a mut ParameterServer<'s>,
788        node: NodeKey,
789        name: &str,
790        value: ParameterValue,
791    ) -> Option<Self> {
792        let mut n = String::new();
793        n.push_str(name).ok()?;
794        Some(Self {
795            server,
796            node,
797            name: n,
798            value,
799            descriptor: None,
800        })
801    }
802
803    /// Set the parameter description
804    pub fn description(mut self, desc: &str) -> Self {
805        let param_type = self.value.param_type();
806        if self.descriptor.is_none() {
807            self.descriptor = ParameterDescriptor::new(self.name.as_str(), param_type);
808        }
809        // phase-446 F2 -- truncated at a character boundary and recorded,
810        // where `let _ = push_str` stored nothing for a too-long description.
811        if let Some(ref mut d) = self.descriptor {
812            d.set_description(desc);
813        }
814        self
815    }
816
817    /// Set the parameter as read-only
818    pub fn read_only(mut self) -> Self {
819        let param_type = self.value.param_type();
820        if self.descriptor.is_none() {
821            self.descriptor = ParameterDescriptor::new(self.name.as_str(), param_type);
822        }
823        if let Some(ref mut d) = self.descriptor {
824            d.read_only = true;
825        }
826        self
827    }
828
829    /// Set integer range constraints
830    pub fn integer_range(mut self, min: i64, max: i64, step: i64) -> Self {
831        let param_type = self.value.param_type();
832        if self.descriptor.is_none() {
833            self.descriptor = ParameterDescriptor::new(self.name.as_str(), param_type);
834        }
835        if let Some(ref mut d) = self.descriptor {
836            d.range = crate::types::ParameterRange::Integer(crate::types::IntegerRange::new(
837                min, max, step,
838            ));
839        }
840        self
841    }
842
843    /// Set float range constraints
844    pub fn float_range(mut self, min: f64, max: f64, step: f64) -> Self {
845        let param_type = self.value.param_type();
846        if self.descriptor.is_none() {
847            self.descriptor = ParameterDescriptor::new(self.name.as_str(), param_type);
848        }
849        if let Some(ref mut d) = self.descriptor {
850            d.range = crate::types::ParameterRange::FloatingPoint(
851                crate::types::FloatingPointRange::new(min, max, step),
852            );
853        }
854        self
855    }
856
857    /// Declare the parameter
858    ///
859    /// phase-428 W6 remainder: the terminal of a builder chain, so a dropped
860    /// answer is a parameter that was configured in full and then not declared.
861    #[must_use]
862    pub fn declare(self) -> bool {
863        self.server.declare_with_descriptor(
864            self.node,
865            self.name.as_str(),
866            self.value,
867            self.descriptor,
868        )
869    }
