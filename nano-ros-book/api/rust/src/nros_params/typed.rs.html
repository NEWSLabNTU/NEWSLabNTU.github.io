16/// Error type for typed parameter operations
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub enum ParameterError {
19    /// Parameter already declared with a different type
20    TypeMismatch,
21    /// Value is outside allowed range
22    OutOfRange,
23    /// Parameter is read-only
24    ReadOnly,
25    /// Parameter not found
26    NotFound,
27    /// Internal storage is full
28    StorageFull,
29    /// String conversion failed
30    StringConversion,
31    /// Invalid range for type
32    InvalidRange,
33    /// An on-set-parameters callback refused the write (phase-417 W4.a).
34    Rejected,
35    /// A set on a name that was never declared, and the server does not
36    /// allow undeclared parameters (issue 1151; rclrs `parameter.rs`).
37    ///
38    /// Distinct from [`NotFound`](Self::NotFound): that is what a DIRECT
39    /// `ParameterServer::set` says about a missing name; this is what the
40    /// wire-facing [`ParameterServer::apply`](crate::ParameterServer::apply)
41    /// says when the only remaining option — declaring it — is switched off.
42    /// Moved here from the retired `SetParameterResult` (phase-515).
43    Undeclared,
44}
45
46#[allow(deprecated)]
47impl From<crate::SetParameterResult> for ParameterError {
48    /// The retired verdict enum's error half, kept for one release beside the
49    /// deprecated type (phase-515).
50    fn from(result: crate::SetParameterResult) -> Self {
51        use crate::SetParameterResult as R;
52        match result {
53            R::TypeMismatch => ParameterError::TypeMismatch,
54            R::OutOfRange => ParameterError::OutOfRange,
55            R::ReadOnly => ParameterError::ReadOnly,
56            R::NotFound => ParameterError::NotFound,
57            R::StorageFull => ParameterError::StorageFull,
58            R::InvalidRange => ParameterError::InvalidRange,
59            R::Rejected => ParameterError::Rejected,
60            R::Undeclared => ParameterError::Undeclared,
61            R::Success => panic!("SetParameterResult::Success is not an error"),
62        }
63    }
64}

66/// Trait for types that can be converted to a parameter range from `RangeInclusive`
67pub trait RangeConvertible: ParameterVariant + Clone {
68    /// Convert a `RangeInclusive<Self>` to a `ParameterRange`
69    fn to_parameter_range(
70        range: core::ops::RangeInclusive<Self>,
71    ) -> Result<ParameterRange, ParameterError>;
72}
73
74impl RangeConvertible for i64 {
75    fn to_parameter_range(
76        range: core::ops::RangeInclusive<Self>,
77    ) -> Result<ParameterRange, ParameterError> {
78        Ok(ParameterRange::Integer(crate::IntegerRange::new(
79            *range.start(),
80            *range.end(),
81            1, // Default step of 1
82        )))
83    }
84}
85
86impl RangeConvertible for f64 {
87    fn to_parameter_range(
88        range: core::ops::RangeInclusive<Self>,
89    ) -> Result<ParameterRange, ParameterError> {
90        Ok(ParameterRange::FloatingPoint(
91            crate::FloatingPointRange::new(
92                *range.start(),
93                *range.end(),
94                0.0, // No step constraint (any value in range is valid)
95            ),
96        ))
97    }
98}

100/// Builder for declaring a typed parameter
101pub struct ParameterBuilder<'a, 's, T: ParameterVariant> {
102    /// Borrow of the parameter server. phase-382 W2' — the server itself
103    /// borrows its table, so every handle onto it carries BOTH lifetimes:
104    /// `'a` is this borrow, `'s` is the caller-owned storage underneath.
105    server: &'a mut ParameterServer<'s>,
106    /// phase-426 W1 — the node this parameter belongs to. Fixed when the
107    /// builder is made, so a chained call cannot lose it and land the
108    /// declaration on a sibling.
109    node: NodeKey,
110    /// Parameter name
111    name: &'a str,
112    /// Default value
113    default: Option<T>,
114    /// Human-readable description
115    description: Option<&'a str>,
116    /// phase-417 W4.a — the descriptor's other free text,
117    /// `rcl_interfaces/msg/ParameterDescriptor::additional_constraints`, which
118    /// `ros2 param describe` prints.
119    constraints: Option<&'a str>,
120    /// Range constraints
121    range: Option<ParameterRange>,
122    /// Whether the parameter is read-only
123    read_only: bool,
124    /// Phantom data to hold the type parameter
125    _phantom: core::marker::PhantomData<T>,
126}
127
128impl<'a, 's, T: ParameterVariant> ParameterBuilder<'a, 's, T> {
129    /// Create a new parameter builder
130    pub fn new(server: &'a mut ParameterServer<'s>, node: NodeKey, name: &'a str) -> Self {
131        Self {
132            server,
133            node,
134            name,
135            default: None,
136            description: None,
137            constraints: None,
138            range: None,
139            read_only: false,
140            _phantom: core::marker::PhantomData,
141        }
142    }
143
144    /// Set a default value for the parameter
145    ///
146    /// The ONE default setter. rclrs also has `default_from_iter(impl IntoIterator)`
147    /// and `default_string_array(&[&str])`; neither exists here, so pass the array
148    /// value itself. An array is a `heapless::Vec` bounded by `MAX_ARRAY_LEN` (a
149    /// string array is `Vec<String<MAX_STRING_VALUE_LEN>, MAX_ARRAY_LEN>`, two
150    /// compile-time bounds), and an over-capacity value is refused by
151    /// [`ParameterVariant::try_to_parameter_value`], never truncated. No allocator.
152    pub fn default(mut self, value: T) -> Self {
153        self.default = Some(value);
154        self
155    }
156
157    /// Set a human-readable description for the parameter
158    pub fn description(mut self, desc: &'a str) -> Self {
159        self.description = Some(desc);
160        self
161    }
162
163    /// Set the free-text `additional_constraints` — rclrs's
164    /// `ParameterBuilder::constraints`, and rclc's fourth argument to
165    /// `rclc_add_parameter_description` (phase-417 W4.a).
166    ///
167    /// Text longer than `NROS_MAX_PARAM_CONSTRAINTS_LEN` — whose default is
168    /// **0**, so an image pays for this field only when it asks for it — is
169    /// truncated at a character boundary and REPORTED once through
170    /// [`ParameterServer::take_truncated_descriptions`], never dropped in
171    /// silence.
172    ///
173    /// [`ParameterServer::take_truncated_descriptions`]: crate::ParameterServer::take_truncated_descriptions
174    pub fn constraints(mut self, text: &'a str) -> Self {
175        self.constraints = Some(text);
176        self
177    }
178
179    /// Set integer range constraints for the parameter
180    pub fn integer_range(mut self, min: i64, max: i64, step: i64) -> Result<Self, ParameterError> {
181        if T::parameter_type() != ParameterType::Integer {
182            return Err(ParameterError::InvalidRange);
183        }
184        let range = crate::IntegerRange::new(min, max, step);
185        // Issue 1150 — `min > max` or a negative step is refused here, where
186        // the caller can see it, not silently stored and published.
187        if !range.is_valid() {
188            return Err(ParameterError::InvalidRange);
189        }
190        self.range = Some(ParameterRange::Integer(range));
191        Ok(self)
192    }
193
194    /// Set floating point range constraints for the parameter
195    pub fn float_range(mut self, min: f64, max: f64, step: f64) -> Result<Self, ParameterError> {
196        if T::parameter_type() != ParameterType::Double {
197            return Err(ParameterError::InvalidRange);
198        }
199        let range = crate::FloatingPointRange::new(min, max, step);
200        if !range.is_valid() {
201            return Err(ParameterError::InvalidRange);
202        }
203        self.range = Some(ParameterRange::FloatingPoint(range));
204        Ok(self)
205    }
206
207    /// Set range constraints using an inclusive range
208    ///
209    /// This is a convenience method that works with `RangeInclusive`:
210    /// - For `i64` parameters: `range(0..=100)` sets an integer range with step 1
211    /// - For `f64` parameters: `range(0.0..=1.0)` sets a floating point range with step 0.0
212    ///
213    /// For more control (e.g., custom step), use `integer_range()` or `float_range()`.
214    pub fn range(mut self, range: core::ops::RangeInclusive<T>) -> Result<Self, ParameterError>
215    where
216        T: RangeConvertible,
217    {
218        self.range = Some(T::to_parameter_range(range)?);
219        Ok(self)
220    }
221
222    /// The descriptor the three terminal verbs declare with — ONE
223    /// construction (phase-417 W4.a).
224    ///
225    /// It was written out three times, identically but for the `read_only`
226    /// line, which is why `additional_constraints` would otherwise have needed
227    /// adding in three places and reached whichever two somebody remembered.
228    /// `read_only` stays at the call site because that is the one line the
229    /// three genuinely disagree about.
230    ///
231    /// phase-446 F2 -- a too-long description is truncated and reported, never
232    /// a reason to refuse the declaration: the contract derives the string
233    /// capacity to 0 for a scalar-only image, and this used to fail every
234    /// `.description(..)` there with `StringConversion`. The same holds for
235    /// the constraints text, whose capacity DEFAULTS to 0.
236    fn build_descriptor(&self) -> Result<ParameterDescriptor, DeclarationError> {
237        let mut descriptor = ParameterDescriptor::new(self.name, T::parameter_type())
238            .ok_or(DeclarationError::NameTooLong)?;
239        if let Some(desc) = self.description {
240            descriptor.set_description(desc);
241        }
242        if let Some(text) = self.constraints {
243            descriptor.set_additional_constraints(text);
244        }
245        descriptor.range = self.range.unwrap_or_default();
246        Ok(descriptor)
247    }
248
249    /// Declare a read-only parameter
250    ///
251    /// Read-only parameters cannot be changed after declaration.
252    /// A default value must be provided.
253    pub fn read_only(self) -> Result<ReadOnlyParameter<'a, 's, T>, DeclarationError> {
254        // Must have a default value for read-only parameters
255        let default_value = self
256            .default
257            .as_ref()
258            .ok_or(DeclarationError::NoValueAvailable)?
259            .clone();
260
261        let mut descriptor = self.build_descriptor()?;
262        descriptor.read_only = true;
263
264        let param_value = default_value.to_parameter_value();
265
266        declare(self.server, self.node, descriptor, Some(&param_value))?;
267
268        Ok(ReadOnlyParameter::new(self.server, self.node, self.name))
269    }
270
271    /// Declare a mandatory parameter
272    ///
273    /// If no default value is provided, it must be set externally before use.
274    pub fn mandatory(self) -> Result<MandatoryParameter<'a, 's, T>, DeclarationError> {
275        let mut descriptor = self.build_descriptor()?;
276        descriptor.read_only = self.read_only;
277
278        let default_value = self.default.map(|v| v.to_parameter_value());
279
280        declare(self.server, self.node, descriptor, default_value.as_ref())?;
281
282        Ok(MandatoryParameter::new(self.server, self.node, self.name))
283    }
284
285    /// Declare an optional parameter
286    pub fn optional(self) -> Result<OptionalParameter<'a, 's, T>, DeclarationError> {
287        let mut descriptor = self.build_descriptor()?;
288        descriptor.read_only = self.read_only;
289
290        let default_value = self.default.map(|v| v.to_parameter_value());
291
292        declare(self.server, self.node, descriptor, default_value.as_ref())?;
293
294        Ok(OptionalParameter::new(self.server, self.node, self.name))
295    }
296}

298/// The typed terminals' declaration, reporting rclrs's [`DeclarationError`].
299///
300/// `ParameterServer::declare_parameter` answers in `ParameterError`, where
301/// `TypeMismatch` means "already declared" and `InvalidRange` covers both an
302/// ill-formed range and a default outside it; this names each case. It keeps
303/// that call (rather than `try_declare_with_descriptor`) because the typed
304/// path emits no parameter event on declaration, and changing that is not
305/// part of a rename (phase-515 R8).
306fn declare(
307    server: &mut ParameterServer<'_>,
308    node: NodeKey,
309    descriptor: ParameterDescriptor,
310    value: Option<&crate::ParameterValue>,
311) -> Result<(), DeclarationError> {
312    let consistency = ParameterServer::declaration_consistency(
313        Some(&descriptor),
314        &value.cloned().unwrap_or_default(),
315    );
316    server
317        .declare_parameter(node, descriptor, value)
318        .map_err(|result| match result {
319            ParameterError::TypeMismatch => DeclarationError::AlreadyDeclared,
320            ParameterError::InvalidRange => {
321                consistency.err().unwrap_or(DeclarationError::InvalidRange)
322            }
323            _ => DeclarationError::StorageFull,
324        })
325}

327/// A parameter that must always have a value
328pub struct MandatoryParameter<'a, 's, T: ParameterVariant> {
329    server: &'a mut ParameterServer<'s>,
330    /// phase-426 W1 — the node whose table this handle reads and writes.
331    node: NodeKey,
332    name: String<{ crate::MAX_PARAM_NAME_LEN }>,
333    _phantom: core::marker::PhantomData<T>,
334}
335
336impl<'a, 's, T: ParameterVariant> MandatoryParameter<'a, 's, T> {
337    pub(crate) fn new(server: &'a mut ParameterServer<'s>, node: NodeKey, name: &'a str) -> Self {
338        let mut n = String::new();
339        n.push_str(name).unwrap();
340        Self {
341            server,
342            node,
343            name: n,
344            _phantom: core::marker::PhantomData,
345        }
346    }
347
348    /// Get the current value of the parameter
349    pub fn get(&self) -> T {
350        self.server
351            .get_parameter_value(self.node, self.name.as_str())
352            .and_then(|val| T::from_parameter_value(&val))
353            .expect("Mandatory parameter must have a value")
354    }
355
356    /// Set the value of the parameter
357    pub fn set(&mut self, value: T) -> Result<(), ParameterError> {
358        // issue 0323 — reject an over-capacity value here rather than letting
359        // `to_parameter_value`'s `unwrap_or_default()` turn it into `NotSet`
360        // (a TYPE change) or an EMPTY array and then reporting success.
361        let converted = value
362            .try_to_parameter_value()
363            .map_err(|_| ParameterError::StringConversion)?;
364        let result = self
365            .server
366            .set_parameter_value(self.node, self.name.as_str(), converted);
367        // The typed API never auto-declares, so an undeclared name is simply
368        // not there.
369        result.map_err(|e| match e {
370            ParameterError::Undeclared => ParameterError::NotFound,
371            e => e,
372        })
373    }
374}

451/// Provides access to undeclared parameters in a ParameterServer
452///
453/// This struct is returned by `Node::use_undeclared_parameters()` and allows
454/// for dynamic retrieval of parameter values by name without explicit declaration.
455///
456/// rclrs's `Parameters` (phase-515 R8): read AND write, by name. It was
457/// `UndeclaredParameters`, which survives one release as a deprecated alias.
458pub struct Parameters<'a, 's> {
459    server: &'a mut ParameterServer<'s>,
460    /// phase-426 W1 — whose undeclared parameters. Without this the accessor
461    /// reads whatever node happened to declare the name first.
462    node: NodeKey,
463}
464
465/// The pre-phase-515 name of [`Parameters`].
466#[deprecated(
467    since = "0.6.0",
468    note = "renamed to `Parameters`, rclrs's name (phase-515 R8)"
469)]
470pub type UndeclaredParameters<'a, 's> = Parameters<'a, 's>;
471
472impl<'a, 's> Parameters<'a, 's> {
473    pub fn new(server: &'a mut ParameterServer<'s>, node: NodeKey) -> Self {
474        Self { server, node }
475    }
476
477    /// rclrs's `Parameters::get::<T>(name)` (phase-515 R8): the value of
478    /// `name` as `T`, or `None` when it is unset or holds another type.
479    pub fn get<T: crate::ParameterVariant>(&self, name: &str) -> Option<T> {
480        self.server
481            .get(self.node, name)
482            .and_then(|value| T::from_parameter_value(value))
483    }
484
485    /// rclrs's `Parameters::set(name, value)` (phase-515 R8), declaring
486    /// `name` if absent. ADOPT-BOUNDED: it takes `&mut self` where rclrs's
487    /// takes `&self` (this store has no interior mutability; the handle
488    /// borrows the server mutably), and a value past this node's compile-time
489    /// capacity is `ParameterError::StringConversion` rather than rclrs's
490    /// `ParameterValueError`.
491    pub fn set<T: crate::ParameterVariant>(
492        &mut self,
493        name: &str,
494        value: T,
495    ) -> Result<(), ParameterError> {
496        let value = value
497            .try_to_parameter_value()
498            .map_err(|_| ParameterError::StringConversion)?;
499        self.set_value(name, value)
500    }
501
502    /// Try to get the value of an undeclared boolean parameter
503    pub fn get_bool(&self, name: &str) -> Option<bool> {
504        self.server.get_bool(self.node, name)
505    }
506
507    /// Try to get the value of an undeclared integer parameter
508    pub fn get_integer(&self, name: &str) -> Option<i64> {
509        self.server.get_integer(self.node, name)
510    }
511
512    /// Try to get the value of an undeclared double parameter
513    pub fn get_double(&self, name: &str) -> Option<f64> {
514        self.server.get_double(self.node, name)
515    }
516
517    /// Try to get the value of an undeclared string parameter
518    pub fn get_string(&self, name: &str) -> Option<&str> {
519        self.server.get_string(self.node, name)
520    }
521
522    // ---- writes (phase-417 W4.a) -------------------------------------------
523    //
524    // rclrs's `Parameters` handle reads AND writes; ours only read, which made
525    // `use_undeclared_parameters` half a handle. The writes go through
526    // `ParameterServer::set_or_declare` rather than `apply`, and that is the
527    // whole point of the type: `apply` refuses a name the node never declared
528    // unless the node opted in, and this handle IS the opt-in — a caller
529    // reaches it by naming it, exactly as the store's own doc comment for
530    // `set_or_declare` describes.
531
532    /// Set (declaring if absent) an undeclared boolean parameter.
533    pub fn set_bool(&mut self, name: &str, value: bool) -> Result<(), ParameterError> {
534        self.set_value(name, crate::ParameterValue::Bool(value))
535    }
536
537    /// Set (declaring if absent) an undeclared integer parameter.
538    pub fn set_integer(&mut self, name: &str, value: i64) -> Result<(), ParameterError> {
539        self.set_value(name, crate::ParameterValue::Integer(value))
540    }
541
542    /// Set (declaring if absent) an undeclared double parameter.
543    pub fn set_double(&mut self, name: &str, value: f64) -> Result<(), ParameterError> {
544        self.set_value(name, crate::ParameterValue::Double(value))
545    }
546
547    /// Set (declaring if absent) an undeclared string parameter.
548    pub fn set_string(&mut self, name: &str, value: &str) -> Result<(), ParameterError> {
549        let value =
550            crate::ParameterValue::from_string(value).ok_or(ParameterError::StringConversion)?;
551        self.set_value(name, value)
552    }
553
554    fn set_value(
555        &mut self,
556        name: &str,
557        value: crate::ParameterValue,
558    ) -> Result<(), ParameterError> {
559        self.server
560            .set_or_declare(self.node, name, value)
561            .map_err(|e| match e {
562                ParameterError::Undeclared => ParameterError::NotFound,
563                e => e,
564            })
565    }
566}

748    /// phase-417 W4.a -- the builder's `constraints` reaches the stored
749    /// descriptor through the ONE `build_descriptor`, from every terminal
750    /// verb, and a text that does not fit the (default 0) capacity is
751    /// REPORTED rather than dropped in silence.
752    // The comparison and the `min` below read as "always true" to clippy on a
753    // build that leaves `NROS_MAX_PARAM_CONSTRAINTS_LEN` at its default of 0 --
754    // and that is the point: the constant is a KNOB, the test is written
755    // against it so it holds at 0, at 64 and at 256, and a version specialised
756    // to whichever value this build resolved would read as coverage on the
757    // other two.
758    #[allow(clippy::absurd_extreme_comparisons, clippy::unnecessary_min_or_max)]
759    #[test]
760    fn builder_constraints_reach_the_descriptor_from_every_terminal() {
761        let text = "positive, hz";
762        let fits = text.len() <= crate::MAX_PARAM_CONSTRAINTS_LEN;
763        let stored = &text[..crate::MAX_PARAM_CONSTRAINTS_LEN.min(text.len())];
764
765        for which in 0..3u8 {
766            let mut storage: ParameterStorage = ParameterStorage::new();
767            let mut server = ParameterServer::new_in(storage.as_table());
768            {
769                let b = ParameterBuilder::<f64>::new(&mut server, NODE, "rate")
770                    .default(1.0)
771                    .description("publish rate")
772                    .constraints(text);
773                match which {
774                    0 => {
775                        b.read_only().expect("read_only");
776                    }
777                    1 => {
778                        b.mandatory().expect("mandatory");
779                    }
780                    _ => {
781                        b.optional().expect("optional");
782                    }
783                }
784            }
785            let desc = server.get_descriptor(NODE, "rate").expect("descriptor");
786            assert_eq!(desc.description.as_str(), "publish rate");
787            assert_eq!(
788                desc.additional_constraints.as_str(),
789                stored,
790                "terminal {which} lost the constraints text"
791            );
792            let mut reported = 0usize;
793            server.take_truncated_descriptions(|_, _| reported += 1);
794            assert_eq!(
795                reported,
796                usize::from(!fits),
797                "terminal {which}: a constraints string that did not fit went unreported"
798            );
799        }
800    }
