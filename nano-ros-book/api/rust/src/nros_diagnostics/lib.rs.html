1//! RFC-0052 / phase-296 W3b.1 — the on-target contract-violation reporter.
2//!
3//! Thin-wrapper discipline (RFC-0019): this crate BUILDS `DiagnosticArray`
4//! messages — it owns no publisher, no executor, no aggregation. The entry
5//! glue (fixture, codegen entry, `run_tiers` loop) creates one
6//! `Publisher<DiagnosticArray>` on `/diagnostics` and publishes whatever
7//! [`DiagnosticReporter::report`] returns. `no_std` + heapless throughout.
8//!
9//! Rule ids are the play_launch runtime-enforcement vocabulary — the SAME
10//! contract violated on either runtime reports in the SAME words
11//! (cross-runtime parity, RFC-0050/0052).
12
13#![forbid(unsafe_code)]
14#![no_std]

18/// Publisher rate below the declared `min_rate_hz` (pub-endpoint guarantee).
19pub const RULE_RATE_HIERARCHY: &str = "rate-hierarchy-runtime";
20/// Message age above the declared `max_age_ms` (sub-endpoint assumption).
21pub const RULE_MAX_AGE: &str = "max-age-runtime";
22/// Path latency above the declared `max_latency_ms` (path guarantee).
23pub const RULE_MAX_LATENCY: &str = "max-latency-runtime";
24/// phase-462 W2 -- a contracted subscription that took NOTHING for a whole
25/// `max_age_ms` window (sub-endpoint assumption, the on-target form of a
26/// liveliness lease).
27///
28/// Distinct from [`RULE_MAX_AGE`], and it has to be: that rule judges a
29/// message that ARRIVED and was too old, so an input that stops entirely
30/// leaves it silent forever -- the very failure a `max_age_ms` promise is
31/// made against. A declared bound on how old data may be is also a bound on
32/// how long there may be none.
33///
34/// phase-462 W5b -- also the on-target form of a contract `deadline` the
35/// subscription's backend cannot judge (it does not honour `DEADLINE`, or was
36/// never handed the deadline). The window is then the deadline, or the age
37/// bound when that is shorter; declared: the window, in ms.
38pub const RULE_SILENCE: &str = "silence-runtime";
39/// phase-462 W3b -- a contracted hazard's guard faulted (`contracts.hazards`):
40/// an input the safety mechanism depends on is gone. play_launch's own
41/// spelling, with no `-runtime` suffix, so one log reader serves the RTOS and
42/// the Linux side of a bridge. Measured: ms since the fault began; declared:
43/// the hazard's `ftti_ms` (0 when none).
44pub const RULE_HAZARD_DETECTED: &str = "hazard-detected";
45/// phase-462 W3b -- the hazard's `safe_state` publisher published after the
46/// fault. Measured: fault to reaction plus `settle_ms`; declared: `ftti_ms`.
47pub const RULE_HAZARD_REACTION: &str = "hazard-reaction";
48/// phase-462 W3b -- no guard of the hazard is faulted any more. Measured: how
49/// long the hazard lasted, ms.
50pub const RULE_HAZARD_RECOVERED: &str = "hazard-recovered";
51/// phase-462 W4c -- entering a contract mode took longer than its reaction
52/// path allows. An executor enters a requested mode at its next spin, so the
53/// measured value is the gap since its previous check (ms), an upper bound on
54/// how long the request waited; declared: the reaction path's `max_latency`.
55/// A GUARANTEE: how promptly this image switches is its own behaviour.
56pub const RULE_MODE_SWITCH: &str = "mode-switch-runtime";

58/// Which side of the contract the violated field belongs to — drives the
59/// 4-quadrant diagnosis (RFC-0050 §contracts): a violated GUARANTEE with
60/// met assumptions is a node bug; a violated ASSUMPTION is an upstream
61/// problem.
62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub enum ContractKind {
64    Assumption,
65    Guarantee,
66}
67
68impl ContractKind {
69    pub const fn as_str(self) -> &'static str {
70        match self {
71            ContractKind::Assumption => "assumption",
72            ContractKind::Guarantee => "guarantee",
73        }
74    }
75}

77/// Violation severity → `DiagnosticStatus` level.
78#[derive(Debug, Clone, Copy, PartialEq, Eq)]
79pub enum Severity {
80    Warn,
81    Error,
82}
83
84impl Severity {
85    /// `DiagnosticStatus` level bytes (generated consts are module-private;
86    /// values are the ROS-fixed 0=OK 1=WARN 2=ERROR 3=STALE).
87    pub const fn level(self) -> u8 {
88        match self {
89            Severity::Warn => 1,
90            Severity::Error => 2,
91        }
92    }
93}

95/// Builds rate-limited `DiagnosticArray` entries. One reporter per node
96/// (or per entry) is plenty — it carries only the rate-limit state.
97#[derive(Debug, Default)]
98pub struct DiagnosticReporter {
99    /// Minimum µs between emitted reports (monotonic clock supplied by the
100    /// caller). 0 = unlimited.
101    pub min_interval_us: u64,
102    last_report_us: u64,
103}
104
105impl DiagnosticReporter {
106    pub const fn new(min_interval_us: u64) -> Self {
107        Self {
108            min_interval_us,
109            last_report_us: 0,
110        }
111    }
112
113    /// Build one violation report, or `None` while rate-limited.
114    ///
115    /// `now_us` is the caller's monotonic clock (the executor's `clock_us`
116    /// on target). `fqn` names the violating entity (node FQN or
117    /// `node/endpoint` ref — the same key shape the SystemModel uses).
118    /// The stamp is left zero; the transport-side observer keys on
119    /// content, and stamping would drag the epoch clock into `no_std`
120    /// paths that don't have one.
121    pub fn report(
122        &mut self,
123        now_us: u64,
124        rule_id: &str,
125        severity: Severity,
126        kind: ContractKind,
127        fqn: &str,
128        message: &str,
129    ) -> Option<DiagnosticArray> {
130        if self.min_interval_us > 0
131            && self.last_report_us != 0
132            && now_us.saturating_sub(self.last_report_us) < self.min_interval_us
133        {
134            return None;
135        }
136        self.last_report_us = now_us;
137
138        let mut status = DiagnosticStatus {
139            level: severity.level(),
140            ..Default::default()
141        };
142        let _ = status.name.push_str(rule_id);
143        let _ = status.message.push_str(message);
144        let _ = status.hardware_id.push_str(fqn);
145        let mut kv = KeyValue::default();
146        let _ = kv.key.push_str("kind");
147        let _ = kv.value.push_str(kind.as_str());
148        let _ = status.values.push(kv);
149
150        let mut arr = DiagnosticArray::default();
151        let _ = arr.status.push(status);
152        Some(arr)
153    }
154}

156/// Issue 1635 — which side of the contract a runtime rule judges.
157///
158/// The rules about what ARRIVES (`max-age-runtime`, `silence-runtime`, and a
159/// hazard's guard faulting or recovering) are ASSUMPTIONS; every other rule --
160/// a hazard's reaction among them -- judges what this image itself does
161/// (its publish rate, its path latency, its deadlines, its own stack) and is a
162/// GUARANTEE. One table, so the fixture, the C/C++ entries and a Rust entry
163/// cannot classify the same rule two ways.
164pub fn kind_for_rule(rule: &str) -> ContractKind {
165    match rule {
166        RULE_MAX_AGE | RULE_SILENCE | RULE_HAZARD_DETECTED | RULE_HAZARD_RECOVERED => {
167            ContractKind::Assumption
168        }
169        _ => ContractKind::Guarantee,
170    }
171}

173impl DiagnosticReporter {
174    /// Issue 1635 — one drained executor violation, as a `DiagnosticArray`
175    /// (or `None` while rate-limited). The ONE mapping from the executor's
176    /// `Violation` fields to a report; `nros-node`'s type is not named here so
177    /// this crate stays below it.
178    ///
179    /// `rule` keeps the executor's own spelling — it is already the
180    /// play_launch vocabulary — and the message carries both numbers, whose
181    /// unit is the rule's (milli-Hz for rate, ms for age/latency, us for a
182    /// deadline miss).
183    pub fn report_violation(
184        &mut self,
185        now_us: u64,
186        rule: &str,
187        fqn: &str,
188        measured: u32,
189        declared: u32,
190    ) -> Option<DiagnosticArray> {
191        use core::fmt::Write as _;
192        let mut message = heapless::String::<64>::new();
193        let _ = write!(message, "measured {measured} vs declared {declared}");
194        self.report(
195            now_us,
196            rule,
197            Severity::Error,
198            kind_for_rule(rule),
199            fqn,
200            &message,
201        )
202    }
203}

205/// phase-474 I7 -- one violation report, streamed as CDR straight into `w`,
206/// byte-identical to serializing what [`DiagnosticReporter::report_violation`]
207/// returns (with no rate limit).
208///
209/// The reporter runs on the spin thread (at the end of a spin since phase-474
210/// I7, not at detection). The value form puts a
211/// whole `DiagnosticArray` on that stack -- 5,176 B on a 64-bit host: a
212/// four-slot vector of 1,224 B statuses, each with eight key/value slots,
213/// plus the status built before it -- to send one status with one key/value.
214/// On the safety island that frame ran a 16 KiB main stack into the idle
215/// thread's on the first stored violation. This writes the same bytes from the
216/// borrowed strings; the largest locals are the 64 B message and an empty
217/// `Header` (frame id capacity 256).
218///
219/// Field limits follow the value form exactly: a string longer than its
220/// field's capacity (name 64, message 128, hardware id 96) is sent EMPTY,
221/// because that is what `heapless::String::push_str` leaves.
222pub fn write_violation_report(
223    w: &mut nros_serdes::CdrWriter<'_>,
224    rule: &str,
225    fqn: &str,
226    measured: u32,
227    declared: u32,
228) -> Result<(), nros_serdes::SerError> {
229    write_violation_reports(w, &[(rule, fqn, measured, declared)])
230}
231
232/// phase-474 I7 -- several violation reports COALESCED into one
233/// `DiagnosticArray`: one status per `(rule, fqn, measured, declared)`, in
234/// order, each byte-identical to the status [`write_violation_report`] sends
235/// for it alone. The executor queues verdicts and hands them over off the
236/// judged dispatch; on a slow link one message for several verdicts is what
237/// keeps the report from costing a tick per verdict.
238///
239/// At most four statuses (the generated type's capacity) are meaningful to an
240/// nros subscriber; a caller passing more is refused with `BufferTooSmall`
241/// before anything is written.
242pub fn write_violation_reports(
243    w: &mut nros_serdes::CdrWriter<'_>,
244    reports: &[(&str, &str, u32, u32)],
245) -> Result<(), nros_serdes::SerError> {
246    use core::fmt::Write as _;
247    use nros_serdes::Serialize as _;
248    fn fit(s: &str, cap: usize) -> &str {
249        if s.len() <= cap {
250            s
251        } else {
252            ""
253        }
254    }
255    if reports.len() > 4 {
256        return Err(nros_serdes::SerError::BufferTooSmall);
257    }
258    // DiagnosticArray { header, status: [n] }
259    let array = w.begin_dheader()?;
260    nros_std_msgs::msg::Header::default().serialize(w)?;
261    w.write_u32(reports.len() as u32)?;
262    for &(rule, fqn, measured, declared) in reports {
263        let mut message = heapless::String::<64>::new();
264        let _ = write!(message, "measured {measured} vs declared {declared}");
265        // DiagnosticStatus { level, name, message, hardware_id, values: [one] }
266        let status = w.begin_dheader()?;
267        w.write_u8(Severity::Error.level())?;
268        w.write_string(fit(rule, 64))?;
269        w.write_string(fit(&message, 128))?;
270        w.write_string(fit(fqn, 96))?;
271        w.write_u32(1)?;
272        // KeyValue { "kind", assumption | guarantee }
273        let kv = w.begin_dheader()?;
274        w.write_string("kind")?;
275        w.write_string(kind_for_rule(rule).as_str())?;
276        w.end_dheader(kv)?;
277        w.end_dheader(status)?;
278    }
279    w.end_dheader(array)?;
280    Ok(())
281}

287    /// phase-474 I7 -- a coalesced report is the value form of one array
288    /// holding each verdict's status in order, under XCDR1 and XCDR2.
289    #[test]
290    fn a_coalesced_report_is_one_array_of_the_single_reports() {
291        use nros_serdes::{CdrWriter, Serialize as _};
292        let reports = [
293            (RULE_MAX_LATENCY, "/mrm_handler/mrm_state", 250u32, 206u32),
294            (
295                RULE_MAX_LATENCY,
296                "/mrm_handler/takeover_request_state",
297                250,
298                206,
299            ),
300            (RULE_SILENCE, "/n/in", 0, 500),
301        ];
302        let mut value = DiagnosticArray::default();
303        for (rule, fqn, m, d) in reports {
304            let one = DiagnosticReporter::new(0)
305                .report_violation(0, rule, fqn, m, d)
306                .unwrap();
307            for st in one.status.iter() {
308                value.status.push(st.clone()).unwrap();
309            }
310        }
311        for xcdr2 in [false, true] {
312            let mut a = [0u8; 1024];
313            let mut b = [0u8; 1024];
314            let mut wa = if xcdr2 {
315                CdrWriter::new_with_header_xcdr2(&mut a).unwrap()
316            } else {
317                CdrWriter::new_with_header(&mut a).unwrap()
318            };
319            value.serialize(&mut wa).unwrap();
320            let la = wa.position();
321            let mut wb = if xcdr2 {
322                CdrWriter::new_with_header_xcdr2(&mut b).unwrap()
323            } else {
324                CdrWriter::new_with_header(&mut b).unwrap()
325            };
326            write_violation_reports(&mut wb, &reports).unwrap();
327            let lb = wb.position();
328            assert_eq!(a[..la], b[..lb], "xcdr2={xcdr2}");
329        }
330        // More than the type's four statuses is refused, not truncated.
331        let mut c = [0u8; 2048];
332        let mut wc = CdrWriter::new_with_header(&mut c).unwrap();
333        let five = [reports[0]; 5];
334        assert!(write_violation_reports(&mut wc, &five).is_err());
335    }

340    fn the_streamed_report_is_the_value_reports_bytes() {
341        use nros_serdes::{CdrWriter, Serialize as _};
342        let long = "/a/very/long/node/name/that/does/not/fit/in/ninety/six/bytes/of/hardware/id/at/all/really/truly/not/even/close";

348        ];
349        for xcdr2 in [false, true] {
350            for (rule, fqn, m, d) in cases {

354                    CdrWriter::new_with_header_xcdr2(&mut a).unwrap()
355                } else {
356                    CdrWriter::new_with_header(&mut a).unwrap()
357                };
358                DiagnosticReporter::new(0)
359                    .report_violation(0, rule, fqn, m, d)
360                    .unwrap()
361                    .serialize(&mut wa)
362                    .unwrap();
363                let la = wa.position();

377    fn a_violation_report_keeps_the_rule_and_classifies_its_side() {
378        let mut r = DiagnosticReporter::new(0);
379        let arr = r
