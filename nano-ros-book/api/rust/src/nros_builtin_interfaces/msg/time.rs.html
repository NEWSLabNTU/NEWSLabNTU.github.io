8/// Time message type

15impl Serialize for Time {
16    fn serialize(&self, writer: &mut CdrWriter) -> Result<(), SerError> {
17        // phase-303 W4 (#0267) — DHEADER wrap for XCDR2 appendable structs.
18        // No-op under XCDR1 (byte-identical); under XCDR2 delimits this struct.
19        let __dh = writer.begin_dheader()?;
20        writer.write_i32(self.sec)?;
21        writer.write_u32(self.nanosec)?;
22        writer.end_dheader(__dh)?;
23        Ok(())
24    }
25}

27impl Deserialize for Time {
28    fn deserialize(reader: &mut CdrReader) -> Result<Self, DeserError> {
29        // phase-303 W4 (#0267) — read the XCDR2 DHEADER (no-op under XCDR1);
30        // end_dheader skips any unknown trailing members (forward compat).
31        let __dh = reader.begin_dheader()?;
32        let __value = Self {
33            sec: reader.read_i32()?,
34            nanosec: reader.read_u32()?,
35        };
36        reader.end_dheader(__dh)?;
37        Ok(__value)
38    }
39}

41// ── nros_core::SecNanosecMsg — what `Time::to_ros_msg()` returns ────────────
42// phase-467 Q3. This message's field set is `{ sec: i32, nanosec: u32 }`, so a
43// `nros_core::Time` reaches it by two field copies. `nros-core` cannot name a
44// message type — every generated crate depends on it, and message types are
45// generated per USER package — so the impl is emitted HERE and the trait is the
46// seam. It is the Rust stand-in for `nros::Time::to_msg`'s C++
47// `template <typename TimeMsgT>`, which binds structurally to the same shape.
48impl nros_core::SecNanosecMsg for Time {
49    fn from_sec_nanosec(sec: i32, nanosec: u32) -> Self {
50        Self { sec, nanosec }
51    }
52}
