1//! Unified platform abstraction traits for nros.
2//!
3//! This crate defines the backend-agnostic interface that platform
4//! implementations (POSIX, Zephyr, FreeRTOS, bare-metal, etc.) must satisfy.
5//! RMW backends consume these traits via thin shim crates that translate
6//! RMW-specific C symbols (e.g., `z_clock_now`, `uxr_millis`) into calls
7//! on the active platform implementation.
8//!
9//! # Trait hierarchy
10//!
11//! Capabilities are split into independent sub-traits so each RMW backend
12//! can declare exactly what it needs:
13//!
14//! - [`PlatformClock`] — monotonic clock (required by all backends)
15//! - [`PlatformAlloc`] — heap allocation (zenoh-pico only)
16//! - [`PlatformSleep`] — sleep / delay (zenoh-pico only)
17//! - [`PlatformYield`] — cooperative yield (zenoh-pico `socket_wait_event`)
18//! - [`PlatformRandom`] — pseudo-random number generation (zenoh-pico only)
19//! - [`PlatformTime`] — wall-clock time (zenoh-pico only)
20//! - [`PlatformThreading`] — tasks, mutexes, condvars (multi-threaded platforms)
21//!
22//! # Compile-time resolution
23//!
24//! Exactly one platform feature must be enabled. The `ConcretePlatform`
25//! type alias (gated on any `platform-*` feature) resolves to the active
26//! backend, eliminating generic parameters.
27

117/// Issue 1732 — a termination signal ENDS a hosted image's spin, so the boot
118/// funnel can close the RMW session before the process exits.
119///
120/// The one spelling for every hosted funnel: the Rust board
121/// (`nros-board-linux`, both its single-executor and its tiered path), the
122/// `nros::main!` hosted spin loops, and the native C/C++ runners in
123/// `nros-cpp`. The guard itself is C, in the POSIX port
124/// (`nros_posix_install_termination_guard` in `nros-platform-posix`), because
125/// that is the one translation unit every hosted image already links.
126///
127/// The contract, all of it in the port:
128///
129/// * installed only over the DEFAULT disposition of SIGTERM / SIGINT — an
130///   application's own handler and `SIG_IGN` are left alone;
131/// * the handler only sets a flag; nothing is torn down in a signal frame;
132/// * a SECOND signal ends the image by the default action, so a wedged
133///   teardown is still stoppable without SIGKILL.
134///
135/// On a build without the POSIX port nothing is installed and
136/// [`requested`](termination::requested) is always `false`, so a loop that
137/// polls it costs a constant.
138pub mod termination {
139    #[cfg(feature = "platform-posix")]
140    unsafe extern "C" {
141        fn nros_posix_install_termination_guard() -> i32;
142        fn nros_posix_termination_requested() -> i32;
143    }
144
145    /// Install the guard; returns how many of SIGTERM / SIGINT carry it (0 when
146    /// the application already handles both, or on a non-POSIX build).
147    /// Idempotent. Call it before the session opens, so a signal that arrives
148    /// during registration is not lost.
149    pub fn install_guard() -> usize {
150        #[cfg(feature = "platform-posix")]
151        {
152            // SAFETY: plain C call; `sigaction` on the process's own handlers.
153            let n = unsafe { nros_posix_install_termination_guard() };
154            if n > 0 { n as usize } else { 0 }
155        }
156        #[cfg(not(feature = "platform-posix"))]
157        {
158            0
159        }
160    }
161
162    /// `true` once a guarded termination signal has arrived. A spin loop that
163    /// sees it returns, and its funnel closes the session.
164    pub fn requested() -> bool {
165        #[cfg(feature = "platform-posix")]
166        {
167            // SAFETY: reads a `sig_atomic_t`.
168            unsafe { nros_posix_termination_requested() != 0 }
169        }
170        #[cfg(not(feature = "platform-posix"))]
171        {
172            false
173        }
174    }
175}

177/// The kernel's deadline reservation on the calling thread (issue 2018).
178///
179/// The reservation is C, in the POSIX port
180/// (`nros_posix_apply_current_deadline` in `nros-platform-posix`), for the
181/// reason [`termination`] gives. It is reached only on a build that selected
182/// that port. A `cfg(unix)` gate is not enough: Zephyr `native_sim`, and the
183/// host tiers of ThreadX and FreeRTOS, target a Linux triple without the POSIX
184/// port. Their port defines no `nros_posix_*` symbol, so a reference made the
185/// linker take the POSIX `platform.o` from whichever archive carried one, and
186/// every other symbol in it then collided with the image's own port
187/// (issue 2032; issue 1776 had the same shape).
188pub mod sched {
189    use core::ffi::c_char;
190
191    #[cfg(feature = "platform-posix")]
192    unsafe extern "C" {
193        fn nros_posix_apply_current_deadline(
194            name: *const c_char,
195            tier_class: *const c_char,
196            budget_us: u64,
197            period_us: u64,
198            deadline_us: u64,
199        ) -> i32;
200    }
201
202    /// Ask the kernel to reserve `budget_us` every `period_us` for the calling
203    /// thread. Returns `true` when the kernel took it, and `false` otherwise.
204    /// That covers a tier class that asks for no reservation, a refusal (no
205    /// `CAP_SYS_NICE`), and a build whose port has no such kernel facility.
206    /// On `false` the caller enforces the budget itself.
207    ///
208    /// # Safety
209    ///
210    /// `name` and `tier_class` are NUL-terminated strings or null.
211    pub unsafe fn apply_current_deadline(
212        name: *const c_char,
213        tier_class: *const c_char,
214        budget_us: u64,
215        period_us: u64,
216        deadline_us: u64,
217    ) -> bool {
218        #[cfg(feature = "platform-posix")]
219        {
220            // SAFETY: the caller's contract on the two strings.
221            unsafe {
222                nros_posix_apply_current_deadline(
223                    name,
224                    tier_class,
225                    budget_us,
226                    period_us,
227                    deadline_us,
228                ) == 1
229            }
230        }
231        #[cfg(not(feature = "platform-posix"))]
232        {
233            let _ = (name, tier_class, budget_us, period_us, deadline_us);
234            false
235        }
236    }
237}

243///
244/// On Zephyr's `native_sim`, the default network interface is assigned an IPv4

248/// before `Executor::open`. Mirrors the `nros_platform_zephyr_wait_network()` C
249/// helper the C/C++ examples use; the symbol is RMW-independent (defined in
250/// `nros-platform-zephyr`, compiled in every RMW build). Equivalent to
251/// `nros-board-zephyr`'s `ZephyrBoard::wait_link_up`.
252// phase-391 W3 — the rlsf arena behind Zephyr's C allocation funnel.
253#[cfg(feature = "platform-zephyr")]
254pub mod zephyr_heap;
255
256#[cfg(feature = "platform-zephyr")]
257pub mod zephyr {
258    unsafe extern "C" {
259        fn nros_platform_zephyr_wait_network(timeout_ms: i32) -> i32;
260    }
261
262    /// The default network interface did not come up within the timeout.
263    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
264    pub struct NetworkTimeout;
265
266    /// Block until the default Zephyr network interface is operational, or the
267    /// timeout expires. `Ok(())` on link-up, `Err(NetworkTimeout)` on timeout.
268    pub fn wait_network(timeout_ms: i32) -> Result<(), NetworkTimeout> {
269        // SAFETY: `nros_platform_zephyr_wait_network` has no preconditions beyond
270        // being called from a Zephyr thread context — always true in a Zephyr app.
271        let ret = unsafe { nros_platform_zephyr_wait_network(timeout_ms) };
272        if ret == 0 {
273            Ok(())
274        } else {
275            Err(NetworkTimeout)
276        }
277    }

280// ============================================================================
281// Phase 71.27 — opt-in `#[global_allocator]`
282// ============================================================================
283//
284// On bare-metal / RTOS targets DDS + heapless futures need a real
285// heap. Each `nros-platform-*` crate already implements `PlatformAlloc`
286// against its native heap (`pvPortMalloc` on FreeRTOS,
287// `tx_byte_allocate` on ThreadX, `kmm_malloc` on NuttX,
288// `k_malloc` on Zephyr, libc `malloc` on POSIX). This module promotes
289// that trait impl into a `#[global_allocator]` so application crates
290// don't have to write per-platform glue.
291//
292// Off by default — `platform-posix` users link against libstd's
293// allocator. Enable via `nros-platform/global-allocator` in the
294// example crate's `Cargo.toml` to wire it in.
295
296// phase-361 W8.c / issue 0594 — this is the ONE `#[global_allocator]` in the
297// tree. `nros-c` used to define a second one under an identical gate, reaching
298// the same heap by a different route (a direct `extern "C" nros_platform_alloc`
299// rather than the trait), and the two were kept apart only by a manifest
300// comment: `nros-c` deps `nros-platform` non-optionally, so any image that
301// enabled both features got a duplicate lang item. `nros-c/global-allocator`
302// now forwards here, which makes the duplication impossible rather than
303// merely discouraged — cargo unifies one crate's one feature into one unit.
304//
305// This adapter covers BOTH link shapes, which the `nros-c` copy did not:
306// every `platform-*` feature resolves `ConcretePlatform` to `CffiPlatform`
307// (see `resolve.rs`), whose `PlatformAlloc` impl IS `nros_platform_alloc`, and
308// the bare-metal Rust crates (mps2-an385, stm32f4, esp32-qemu) reach their own
309// arena through the same trait. One API, one arena, per RFC-0034 D6.
310#[cfg(all(feature = "global-allocator", not(feature = "std")))]
311mod global_allocator {
312    use core::{
313        alloc::{GlobalAlloc, Layout},
314        ffi::c_void,
315    };
316
317    use crate::ConcretePlatform;
318    use nros_platform_api::PlatformAlloc;
319
320    /// Alignment the platform ABI guarantees. `nros_platform_alloc` has no
321    /// alignment parameter, and every port behind it returns memory aligned
322    /// for the widest scalar — 8 bytes on every target nano-ros builds for.
323    const PLATFORM_ALIGN: usize = 8;
324
325    /// `GlobalAlloc` adapter over `<ConcretePlatform as PlatformAlloc>`.
326    pub struct PlatformGlobalAllocator;
327
328    unsafe impl GlobalAlloc for PlatformGlobalAllocator {
329        unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
330            // phase-361 W8.c — never return under-aligned memory: the previous
331            // `let _ = layout.align();` produced UB no build could see.
332            //
333            // An over-aligned request used to FAIL here (null, so
334            // `handle_alloc_error`). phase-397 W2 made the executor's
335            // `WakeCtx` `repr(align(64))` and put it in an `Arc`, so every
336            // embedded image with an executor panicked at boot with "memory
337            // allocation of 128 bytes failed" before registering a node
338            // (S32K344, Zephyr 4.4). The ABI still cannot express alignment,
339            // so an over-aligned block is carved out of a larger one instead
340            // (`over_align`).
341            let p = if layout.align() > PLATFORM_ALIGN {
342                match super::over_align::raw_size(layout) {
343                    // SAFETY: `raw_size` sized the raw block for `place`.
344                    Some(raw_size) => unsafe {
345                        super::over_align::place(
346                            <ConcretePlatform as PlatformAlloc>::alloc(raw_size) as *mut u8,
347                            layout,
348                        )
349                    },
350                    None => core::ptr::null_mut(),
351                }
352            } else {
353                <ConcretePlatform as PlatformAlloc>::alloc(layout.size()) as *mut u8
354            };
355            #[cfg(feature = "alloc-stats")]
356            if !p.is_null() {
357                super::heap_stats::STATS.on_alloc(layout.size());
358            }
359            p
360        }
361
362        unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
363            let raw = if layout.align() > PLATFORM_ALIGN {
364                // SAFETY: `ptr` came from the over-aligned branch of `alloc`
365                // (same layout), which stored the raw block below it.
366                unsafe { super::over_align::raw_of(ptr) }
367            } else {
368                ptr
369            };
370            <ConcretePlatform as PlatformAlloc>::dealloc(raw as *mut c_void);
371            #[cfg(feature = "alloc-stats")]
372            super::heap_stats::STATS.on_dealloc(layout.size());
373        }
374    }
375
376    #[global_allocator]
377    static ALLOCATOR: PlatformGlobalAllocator = PlatformGlobalAllocator;
378}

380/// An over-aligned block inside an 8-aligned one (the global allocator's
381/// `align > 8` path). The raw block is `size + align` bytes; the returned
382/// pointer is the first `align` boundary at least one `usize` past its start,
383/// and that `usize` just below it keeps the raw pointer for `dealloc`.
384///
385/// Room: the raw pointer is 8-aligned and `align` is a power of two above 8,
386/// so the gap to the boundary is a multiple of 8 in `8..=align`: the header
387/// fits in it and `gap + size <= align + size`.
388#[cfg_attr(
389    not(all(feature = "global-allocator", not(feature = "std"))),
390    allow(dead_code)
391)]
392mod over_align {
393    use core::alloc::Layout;
394
395    const HEADER: usize = core::mem::size_of::<usize>();
396
397    /// The raw block's size for `layout`, or `None` on overflow.
398    pub(crate) fn raw_size(layout: Layout) -> Option<usize> {
399        layout.size().checked_add(layout.align())
400    }
401
402    /// The aligned pointer inside `raw` (null stays null).
403    ///
404    /// # Safety
405    /// `raw` is null or an 8-aligned block of `raw_size(layout)` bytes, and
406    /// `layout.align() > 8`.
407    pub(crate) unsafe fn place(raw: *mut u8, layout: Layout) -> *mut u8 {
408        if raw.is_null() {
409            return raw;
410        }
411        let align = layout.align();
412        let start = raw as usize;
413        let aligned = (start + HEADER + align - 1) & !(align - 1);
414        let p = raw.wrapping_add(aligned - start);
415        // SAFETY: `aligned - HEADER >= start`, and it is 8-aligned (both
416        // `aligned` and `start` are), inside the raw block.
417        unsafe { (p.sub(HEADER) as *mut usize).write(start) };
418        p
419    }
420
421    /// The raw block `place` carved `p` out of.
422    ///
423    /// # Safety
424    /// `p` came from `place`.
425    pub(crate) unsafe fn raw_of(p: *mut u8) -> *mut u8 {
426        // SAFETY: `place` wrote the raw pointer there.
427        let start = unsafe { (p.sub(HEADER) as *const usize).read() };
428        p.wrapping_sub(p as usize - start)
429    }
430
431    #[cfg(test)]
432    mod tests {
433        extern crate std;
434        use super::*;
435        use std::{vec, vec::Vec};
436
437        #[test]
438        fn every_offset_fits_and_round_trips() {
439            for align in [16usize, 32, 64, 128] {
440                for size in [1usize, 8, 63, 64, 128, 200] {
441                    let layout = Layout::from_size_align(size, align).unwrap();
442                    let n = raw_size(layout).unwrap();
443                    // Every 8-aligned start an align-period can present.
444                    let mut buf: Vec<u64> = vec![0; (n + 2 * align) / 8 + 1];
445                    let base = buf.as_mut_ptr() as *mut u8;
446                    for off in (0..align).step_by(8) {
447                        let raw = base.wrapping_add(off);
448                        let p = unsafe { place(raw, layout) };
449                        assert_eq!(p as usize % align, 0);
450                        assert!(p as usize - raw as usize >= HEADER);
451                        assert!(p as usize + size <= raw as usize + n);
452                        assert_eq!(unsafe { raw_of(p) }, raw);
453                    }
454                }
455            }
456        }
457
458        #[test]
459        fn null_stays_null() {
460            let layout = Layout::from_size_align(128, 64).unwrap();
461            assert!(unsafe { place(core::ptr::null_mut(), layout) }.is_null());
462        }
463    }

466// phase-361 W8.c — the Rust-footprint heap counter, moved here with the
467// allocator it instruments. It counts only what passes through the
468// `#[global_allocator]`; the C side's direct `nros_platform_alloc` traffic
469// (zenoh-pico's `z_malloc` etc.) is not seen, so it under-reports true heap
470// pressure. The *unified* figure is the platform's own
471// `nros_platform_heap_used_bytes` (RFC-0034 D7).
472//
473// No `#[no_mangle]` here: the C names (`rclc_heap_used_bytes` …) belong to the
474// C/C++ API surface and stay exported by `nros-c` / `nros-cpp`, which read
475// these accessors. A pure-Rust image gets the counter without gaining C
476// symbols it never asked for.
477#[cfg(feature = "alloc-stats")]
478pub mod heap_stats {
479    use core::sync::atomic::{AtomicUsize, Ordering};
480
481    /// Bytes outstanding through the Rust global allocator, and the high-water
482    /// mark since boot. `Relaxed` throughout — this is instrumentation, and no
483    /// other state is ordered against it.
484    pub struct HeapStats {
485        used_bytes: AtomicUsize,
486        peak_bytes: AtomicUsize,
487    }
488
489    impl HeapStats {
490        /// Create a zeroed counter. `const` so it can back a `static`.
491        pub const fn new() -> Self {
492            Self {
493                used_bytes: AtomicUsize::new(0),
494                peak_bytes: AtomicUsize::new(0),
495            }
496        }
497
498        /// Record a successful allocation of `size` bytes and update the peak.
499        #[inline]
500        pub fn on_alloc(&self, size: usize) {
501            let used = self.used_bytes.fetch_add(size, Ordering::Relaxed) + size;
502            let _ = self.peak_bytes.fetch_max(used, Ordering::Relaxed);
503        }
504
505        /// Record a deallocation of `size` bytes.
506        #[inline]
507        pub fn on_dealloc(&self, size: usize) {
508            self.used_bytes.fetch_sub(size, Ordering::Relaxed);
509        }
510
511        /// Bytes currently outstanding.
512        #[inline]
513        pub fn used(&self) -> usize {
514            self.used_bytes.load(Ordering::Relaxed)
515        }
516
517        /// Peak outstanding bytes since boot.
518        #[inline]
519        pub fn peak(&self) -> usize {
520            self.peak_bytes.load(Ordering::Relaxed)
521        }
522    }
523
524    impl Default for HeapStats {
525        fn default() -> Self {
526            Self::new()
527        }
528    }
529
530    pub static STATS: HeapStats = HeapStats::new();
531
532    /// Bytes currently outstanding through the Rust global allocator.
533    #[inline]
534    pub fn used() -> usize {
535        STATS.used()
536    }
537
538    /// Peak outstanding bytes through the Rust global allocator since boot.
539    #[inline]
540    pub fn peak() -> usize {
541        STATS.peak()
542    }
