1//! [`RuntimeCtx`] — Phase 212.N.1.
2//!
3//! Per-invocation runtime context handed to `BoardEntry::run`'s
4//! `setup` callback. Carries the overlay knobs the codegen
5//! `run_plan(runtime)` body reads:
6//!
7//! - **params** — `(key, value)` pairs from launch XML
8//!   `<param name="…" value="…"/>`, projected here by `nros sync` at BUILD
9//!   time. NOT from `--ros-args -p k:=v`: a generated entry reads no argv,
10//!   and the hosted `nros::init_with_args` parse refuses `-p` rather than
11//!   pretending to honour it (runtime parameters are RFC-0015 §9's channel).
12//! - **remaps** — `(from, to)` topic/service renames, from launch
13//!   `<remap from= to=/>` by the same build-time projection.
14//! - **env** — environment-style key/value pairs (POSIX `getenv`
15//!   shape) accessible from no_std boards via this struct rather
16//!   than a `libc::getenv` call.
17//! - **runtime** — `&mut dyn NodeDispatchRuntime` sink the
18//!   codegen-emitted `run_plan(runtime)` body forwards each Node
19//!   pkg's `register(runtime)` call into (Phase 212.N.7 step-3.2).
20//!   Populated by each `BoardEntry::run` impl after opening its
21//!   executor; defaults to a no-op sink when constructed via
22//!   [`RuntimeCtx::with_runtime`].
23//!
24//! ## no_std-safe shape
25//!
26//! Slice-of-tuples kept on the boot stack. No allocation, no
27//! `core::collections`. Codegen owns the storage and passes a
28//! `&mut RuntimeCtx<'_>` whose backing slices live in `static`s.
29//!
30//! Hosted boards (POSIX) may instead build a longer-lived owned
31//! variant on the heap; the trait surface is slice-based so
32//! both shapes work.

36/// Layer-clean substitute for `nros::node_metadata::CallbackId` +
37/// `nros::node::CallbackCtx` at the [`NodeDispatchRuntime`] boundary
38/// (Phase 216.A.2). `nros-platform` sits below `nros` in the dep
39/// graph, so the trait surface cannot reference those types
40/// directly. The `nros`-side runtime impl wraps a real
41/// `(CallbackId, &mut CallbackCtx)` pair into this opaque shape
42/// before invoking [`NodeDispatchRuntime::signal_callback`]; the
43/// concrete dispatcher casts `ctx_ptr` back to
44/// `&mut nros::CallbackCtx<'_>` at the call site.
45///
46/// `#[repr(C)]` keeps the layout stable across the (same-language
47/// today, FFI-shaped tomorrow) `nros-platform` ↔ `nros` boundary.
48#[repr(C)]
49pub struct SignaledCallback<'a> {
50    /// Stable identifier string carried by `nros::CallbackId(&'a str)`.
51    pub cb_id: &'a str,
52
53    /// Erased pointer to the `nros::CallbackCtx<'_>` the dispatcher
54    /// will drive. The `nros`-side `NodeDispatchRuntime` impl casts
55    /// back to `&mut nros::CallbackCtx<'_>` before invoking the
56    /// component body.
57    pub ctx_ptr: *mut core::ffi::c_void,
58}

67/// Node runtime sink the codegen-emitted `run_plan(runtime)`
68/// body talks to (Phase 212.N.7 step-3.1).
69///
70/// Object-safe + `no_std`. The concrete impl
71/// (`ExecutorNodeRuntime` in `nros`) owns the live executor;
72/// `BoardEntry::run` installs it on the per-boot
73/// [`RuntimeCtx::runtime`] slot before invoking the user `setup`
74/// closure.
75///
76/// Phase 214.K.1 — renamed from `NodeRuntime` to disambiguate from
77/// the user-facing `nros::NodeRuntime` metadata-sink trait in
78/// `packages/api/nros/src/node.rs:112`. The two traits live at
79/// different layers (board-side dispatch sink vs user-side metadata
80/// declaration sink) and the previous shared name forced explicit
81/// `nros_platform::` / `nros::` qualification at every use site +
82/// produced confusing `impl NodeRuntime for X` ambiguity. The
83/// phase-214.K.1 `NodeRuntime` deprecation alias was removed in phase-313
84/// W1 (issue #0243); use `NodeDispatchRuntime` directly.
85pub trait NodeDispatchRuntime {
86    /// Drive the underlying executor for at most `timeout_ms`
87    /// milliseconds. `Ok(())` on a clean spin (including timeout);
88    /// `Err(())` if the executor surfaces a spin error.
89    ///
90    /// `Result<_, ()>` is deliberate: the board entry-point callers
91    /// (`nros-board-{freertos,nuttx,threadx}` spin loops) only
92    /// `{:?}`-print the error and `B::exit_failure()` — a typed enum
93    /// would carry no extra info across the trait boundary, since the
94    /// underlying `ExecutorError` from `nros::node_runtime` is mapped
95    /// to `()` at the impl site (`impl NodeDispatchRuntime for
96    /// ExecutorNodeRuntime`). `#[allow]` keeps the surface narrow.
97    #[allow(clippy::result_unit_err)]
98    fn spin_once(&mut self, timeout_ms: u32) -> Result<(), ()>;
99
100    /// Phase 264 W2 — register the REP-2002 lifecycle services on the underlying
101    /// executor + drive boot autostart. `autostart`: 0 = none, 1 = configure,
102    /// 2 = configure+activate. Default no-op (non-executor runtimes, or `nros`
103    /// built without `lifecycle-services`); the `ExecutorNodeRuntime` impl in
104    /// `nros` does the real work behind that feature. `nros::main!` calls this
105    /// (via [`RuntimeCtx::apply_lifecycle`]) when `system.toml` declares
106    /// `[lifecycle]`, mirroring the bake's `generate.rs::render_lifecycle_fn`.
107    /// issue 0460 — the `Err` carries a STATIC REASON, not `()`.
108    ///
109    /// A capability that fails to register is reported by the caller as
110    /// `RuntimeError::NodeRegister("lifecycle")`, which names WHICH capability
111    /// and nothing about WHY. On Zephyr that opaque string was the entire
112    /// diagnostic for three dead entries: the image printed nothing after
113    /// "Network ready" and the cause could only be guessed at. `&'static str`
114    /// rather than a typed error because this trait lives below `nros-node` and
115    /// cannot name `NodeError`.
116    fn apply_lifecycle(&mut self, autostart: u8) -> Result<(), &'static str> {
117        let _ = autostart;
118        Ok(())
119    }
120
121    /// Phase 264 W4b -- register the 6 ROS 2 parameter services on the underlying
122    /// executor and create its volatile RAM param store. Launch `<param>`
123    /// initials are NOT passed here (issue 1272): each node's are seeded on that
124    /// node as it is built, from the `params` slice `nros::main!` sets before
125    /// the node's `register` call. After this,
126    /// `ros2 param list/get/set` works against the running node; reconfigured values
127    /// live in RAM until the next boot (RFC-0004 §10; persistence is out of scope,
128    /// issue 0080). Default no-op (non-executor runtimes, or `nros` built without
129    /// `param-services`); the `ExecutorNodeRuntime` impl in `nros` does the real work
130    /// behind that feature. `nros::main!` calls this (via
131    /// [`RuntimeCtx::apply_param_services`]) when `system.toml` declares
132    /// `[param_services]`, mirroring the bake's `generate.rs::render_param_persistence_fn`.
133    /// issue 0460 — the `Err` carries a STATIC REASON, not `()`.
134    ///
135    /// A capability that fails to register is reported by the caller as
136    /// `RuntimeError::NodeRegister("lifecycle")`, which names WHICH capability
137    /// and nothing about WHY. On Zephyr that opaque string was the entire
138    /// diagnostic for three dead entries: the image printed nothing after
139    /// "Network ready" and the cause could only be guessed at. `&'static str`
140    /// rather than a typed error because this trait lives below `nros-node` and
141    /// cannot name `NodeError`.
142    fn apply_param_services(&mut self) -> Result<(), &'static str> {
143        Ok(())
144    }
145
146    /// Phase 258 (Track 2, 2a) — raw `*mut Executor` (as `void*`) for the
147    /// owned-spin entry, so a Node pkg's `register(runtime)` wrapper can call
148    /// the uniform `__nros_component_<pkg>_install(.., executor, ..)` seam
149    /// (`nros::install_node_typed`) instead of the retired opaque-fn-ptr
150    /// `register_dispatch_slot_dyn` bridge. A pointer crosses the
151    /// `nros-platform` → `nros` layering wall cleanly (the concrete
152    /// `ExecutorNodeRuntime` lives in `nros`; this trait can't name it).
153    ///
154    /// Default `null` — sinks without a live executor (e.g.
155    /// [`NullNodeRuntime`], framework-dispatch-only runtimes) report no
156    /// handle; the install path treats null as a registration error.
157    fn executor_handle(&mut self) -> *mut core::ffi::c_void {
158        core::ptr::null_mut()
159    }
160
161    /// phase-446 W6 -- hand the executor the parameters each node's contract
162    /// DECLARES, so a node whose code declares a parameter its contract does
163    /// not (or with another type) refuses the boot naming the node, the
164    /// parameter and the contract.
165    ///
166    /// `nodes` is `(node FQN, contract)`; `params` is `(node FQN, name,
167    /// rcl_interfaces type code)`. Plain tuples because this trait lives below
168    /// `nros-params` and cannot name its types. `nros::main!` calls this before
169    /// any node registers. Default no-op: a runtime with no parameter store
170    /// has nothing to check.
171    fn apply_declared_params(
172        &mut self,
173        nodes: &'static [(&'static str, &'static str)],
174        params: &'static [(&'static str, &'static str, u8)],
175    ) -> Result<(), &'static str> {
176        let _ = (nodes, params);
177        Ok(())
178    }
179
180    /// Observability counters from hosted/runtime tests.
181    ///
182    /// Returns `(all_callbacks, message_callbacks)`. Implementations
183    /// that cannot observe callback dispatch keep the default zeros.
184    fn observed_callback_counts(&self) -> (usize, usize) {
185        (0, 0)
186    }
187
188    /// Hand a signaled callback to the framework-side dispatcher
189    /// (Phase 216.A.2). Only meaningful for `DispatchStrategy::Deferred`
190    /// (RTIC / Embassy) runtimes — `Inline` runtimes drive callbacks
191    /// directly from `spin_once` and never call this. The default panic
192    /// surfaces the mis-wire loudly rather than silently dropping the
193    /// callback signal.
194    fn signal_callback(&mut self, _cb: SignaledCallback<'_>) {
195        panic!("signal_callback not implemented for Inline runtime");
196    }
197
198    /// Declare how this runtime delivers callbacks (Phase 216.A.2).
199    /// `nros check` (Phase 216.D.1) cross-validates each Node pkg's
200    /// `Node::DISPATCH` against this value. Defaults to `Inline` so
201    /// every existing impl reports the historical behavior unchanged.
202    fn dispatch_strategy(&self) -> DispatchStrategy {
203        DispatchStrategy::Inline
204    }
205}

223/// Runtime context handed to `BoardEntry::run(setup)`.
224///
225/// All three overlay slices may be empty. A board's launch overlay
226/// typically populates `params` + `remaps`; `env` is rarely set on
227/// embedded.
228pub struct RuntimeCtx<'a> {
229    /// `<param name=… value=…/>` from launch XML, baked here by
230    /// `nros::main!` from the resolved model at BUILD time.
231    ///
232    /// NOT `-p name:=value` CLI overrides: a generated entry reads no argv,
233    /// and the hosted `nros::init_with_args` parse refuses `-p` loudly, so
234    /// launch is the only producer of these.
235    pub params: &'a [(&'a str, &'a str)],
236
237    /// Topic / service / action remaps: `(from, to)`, set per component by
238    /// `nros::main!` from the model's launch `<remap>` rules (phase-306 W3,
239    /// issue 0255). `nros::node!`'s `register` forwards them into
240    /// `install_node_typed_with_launch`, where entity creation expands
241    /// `~`/relative names and substitutes matching rules (exact-FQN match,
242    /// first rule wins).
243    ///
244    /// These are AUTHORITATIVE rules: `ExecutorSink::create_entity` passes
245    /// them as `nros_node::names::resolve_name_layered`'s authoritative tier,
246    /// beneath which the executor's `--ros-args` rules (installed by a hosted
247    /// `nros::Context`) can only apply to a name the launch projected no rule
248    /// for.
249    pub remaps: &'a [(&'a str, &'a str)],
250
251    /// Environment-style key/value pairs (mostly POSIX). Empty on
252    /// embedded boards.
253    pub env: &'a [(&'a str, &'a str)],
254
255    /// Phase 268 W1 — launch-injected node identity `(name, namespace)` for
256    /// the NEXT `register()` call, set by `nros::main!` per component before
257    /// each `<pkg>::register(runtime)?` call. `None` → the node's own
258    /// `create_node(NodeOptions::new("…"))` default stands (backward-compatible).
259    /// Reset to `None` after every register in the self-bringup arm so a prior
260    /// component's identity never leaks into the next (RFC-0046).
261    pub node_identity: Option<(&'static str, &'static str)>,
262
263    /// Issue #52 — per-node QoS overrides for the NEXT `register()` call, baked
264    /// by `nros::main!` from the model's
265    /// `qos_overrides.<topic>.<role>.<policy>` params. Same reset discipline as
266    /// `params`/`remaps`: `&[]` when the node has none, so a prior component's
267    /// table never leaks into the next.
268    ///
269    /// Primitive `(topic, role, policy, value)` codes — the SAME wire form the
270    /// C and C++ ABIs use — because `nros-platform` sits below `nros-rmw` in the
271    /// layer graph and a typed `QoSOverride` field here would invert it. The
272    /// register seam installs them via `Executor::set_node_qos_overrides`, which
273    /// decodes at entity-create time.
274    pub qos_overrides: &'static [(&'static str, u8, u8, u32)],
275
276    /// Node runtime sink. `BoardEntry::run` populates this with
277    /// the live `ExecutorNodeRuntime`-backed impl before invoking
278    /// the user `setup` closure. The codegen-emitted
279    /// `run_plan(runtime)` body calls `<pkg>::register(runtime)` once per Node
280    /// pkg, which installs through `runtime.executor_handle()` +
281    /// `nros::install_node_typed` (Phase 258, Track 2).
282    ///
283    /// Defaults to a [`NullNodeRuntime`] when the context is
284    /// built via [`RuntimeCtx::with_runtime`]. That sink errors
285    /// every call so test fixtures that forget to wire a real runtime
286    /// fail loud.
287    pub runtime: &'a mut dyn NodeDispatchRuntime,
288}

303impl<'a> RuntimeCtx<'a> {
304    /// Build a [`RuntimeCtx`] with no params / remaps / env and the
305    /// given runtime sink. The common shape `BoardEntry::run`
306    /// constructs after opening its executor.
307    ///
308    /// For test fixtures that don't need a populated runtime, pass a
309    /// `&mut NullNodeRuntime` — every call against the sink
310    /// returns `Err(())`, surfacing the missing wiring.
311    pub fn with_runtime(runtime: &'a mut dyn NodeDispatchRuntime) -> Self {
312        Self {
313            params: &[],
314            remaps: &[],
315            env: &[],
316            node_identity: None,
317            qos_overrides: &[],
318            runtime,
319        }
320    }
321
322    /// Build a [`RuntimeCtx`] with explicit overlay slices + runtime
323    /// sink (Phase 212.N.7 step-3.2).
324    pub fn new(
325        runtime: &'a mut dyn NodeDispatchRuntime,
326        params: &'a [(&'a str, &'a str)],
327        remaps: &'a [(&'a str, &'a str)],
328        env: &'a [(&'a str, &'a str)],
329    ) -> Self {
330        Self {
331            params,
332            remaps,
333            env,
334            node_identity: None,
335            qos_overrides: &[],
336            runtime,
337        }
338    }
339
340    /// Phase 264 W2 — register lifecycle services + drive autostart on the runtime
341    /// sink (forwards to [`NodeDispatchRuntime::apply_lifecycle`]). `nros::main!`
342    /// emits this after the per-node `register` calls when `system.toml` declares
343    /// `[lifecycle]`. No-op unless `nros` is built with `lifecycle-services`.
344    pub fn apply_lifecycle(&mut self, autostart: u8) -> Result<(), &'static str> {
345        self.runtime.apply_lifecycle(autostart)
346    }
347
348    /// Phase 264 W4b -- register the ROS 2 parameter services and create the
349    /// volatile param store (forwards to
350    /// [`NodeDispatchRuntime::apply_param_services`]). `nros::main!` emits this
351    /// BEFORE the per-node `register` calls when `system.toml` declares
352    /// `[param_services]`, so the store exists when each node is built and that
353    /// node's launch `<param>` initials ([`RuntimeCtx::params`]) are seeded on it
354    /// (issue 1272). No-op unless `nros` is built with `param-services`.
355    pub fn apply_param_services(&mut self) -> Result<(), &'static str> {
356        self.runtime.apply_param_services()
357    }
358
359    /// Lookup a param by name; first match wins. Linear scan
360    /// because the slice is typically small (≤ a dozen entries).
361    pub fn param(&self, name: &str) -> Option<&'a str> {
362        self.params
363            .iter()
364            .find(|(k, _)| *k == name)
365            .map(|(_, v)| *v)
366    }
367
368    /// Lookup a remap by the original (`from`) name; returns the
369    /// rewritten name when remapped, else `None`.
370    pub fn remap(&self, from: &str) -> Option<&'a str> {
371        self.remaps
372            .iter()
373            .find(|(k, _)| *k == from)
374            .map(|(_, v)| *v)
375    }
376
377    /// Lookup an env entry by name.
378    pub fn env_var(&self, name: &str) -> Option<&'a str> {
379        self.env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v)
380    }
381
382    /// phase-446 W6 -- the contract's declared parameters (forwards to
383    /// [`NodeDispatchRuntime::apply_declared_params`]). `nros::main!` emits
384    /// this before the per-node `register` calls.
385    pub fn apply_declared_params(
386        &mut self,
387        nodes: &'static [(&'static str, &'static str)],
388        params: &'static [(&'static str, &'static str, u8)],
389    ) -> Result<(), &'static str> {
390        self.runtime.apply_declared_params(nodes, params)
391    }
392}

394/// Error returned by the codegen-emitted `run_plan(runtime)` body
395/// (Phase 212.N.4) and by Node pkg `register(runtime)` wrappers
396/// (Phase 212.N.7 step-2).
397///
398/// `no_std`-safe — variants are string-typed so embedded Entry pkgs
399/// don't need to pull `thiserror`/`anyhow` to print. The
400/// out-of-tree `nros-build` codegen library re-exports this type so
401/// emitted code references `::nros_platform::RuntimeError`, NOT
402/// `::nros_build::RuntimeError` — the embedded Entry pkg's runtime
403/// path then doesn't need `nros-build` as a runtime dep (build-dep
404/// only).
405#[derive(Debug)]
406#[non_exhaustive]
407pub enum RuntimeError {
408    /// A node's `register(runtime)` call failed. The string carries the
409    /// node pkg name.
410    ///
411    /// Phase 212.N.12 hard-renamed the legacy `ComponentRegister` variant
412    /// to `NodeRegister` to match the rclcpp_components / ROS 2 launch.xml
413    /// `<node pkg=…>` convention.
414    NodeRegister(&'static str),
415
416    /// issue 0460 — a CAPABILITY (`[lifecycle]`, `[param_services]`) failed to
417    /// register, with the reason.
418    ///
419    /// Distinct from [`NodeRegister`](Self::NodeRegister) because reusing it
420    /// forced the capability name to be the whole message: three Zephyr entries
421    /// died reporting `NodeRegister("lifecycle")`, which says WHICH capability
422    /// and nothing about WHY, and the image printed nothing else. Two static
423    /// strings rather than a log call, because not every entry crate depends on
424    /// `log` — the first attempt emitted `::log::error!` from the macro and
425    /// broke `native_rust_qos_entry` with `cannot find 'log' in the crate root`.
426    Capability {
427        /// `"lifecycle"` / `"param_services"`.
428        name: &'static str,
429        /// Why registration failed, from the runtime that attempted it.
430        reason: &'static str,
431    },
432
433    /// Issue 0095 — a node's `register` failed specifically because the
434    /// executor's fixed callback-entry table (`NROS_EXECUTOR_MAX_CBS`) is full.
435    /// Distinct from the opaque [`NodeRegister`](Self::NodeRegister) so the user
436    /// sees the actionable knob. The string carries the node pkg name.
437    ExecutorFull(&'static str),
438
439    /// The hosted Entry spin loop failed or did not observe the
440    /// requested runtime condition before its bounded test deadline.
441    Spin,
442}
443
444impl core::fmt::Display for RuntimeError {
445    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
446        match self {
447            Self::NodeRegister(msg) => write!(f, "node register failed: {msg}"),
448            Self::ExecutorFull(msg) => write!(
449                f,
450                "node '{msg}' register failed: executor callback table full — \
451                 raise NROS_EXECUTOR_MAX_CBS (build-time env, default 4)"
452            ),
453            Self::Capability { name, reason } => write!(
454                f,
455                "capability '[{name}]' failed to register: {reason} — the system \
456                 declares it, so the entry cannot run without it"
457            ),
458            Self::Spin => write!(f, "entry spin failed"),
459        }
460    }
