1//! The executor backing's placement arithmetic — issue 1197.
2//!
3//! # Why this is its own crate
4//!
5//! phase-392 W6 moved the executor's per-entry storage into a `.bss` static. On
6//! an RTOS the allocator reservation it used to come out of is itself fixed, so
7//! whatever sizes that reservation has to know how big the backing is.
8//!
9//! The size is `arena + tables`. The arena is plain arithmetic
10//! `nros-node/build.rs` already computes. The tables are `repr(Rust)` layout —
11//! field order, padding and niche packing are the compiler's to choose and to
12//! CHANGE between versions — so they can never be re-derived outside it. That is
13//! the sizes-header mirror class (0088 -> 0114 -> 0122 -> 0123 -> 0245 -> 0268)
14//! this tree has re-fixed six times.
15//!
16//! So the arithmetic takes the per-type facts as DATA. `nros-node` supplies what
17//! its compiler says while building for the target; a build script supplies
18//! units recovered for a target it is not itself compiling (the way `nros-c`
19//! recovers `EXECUTOR_OPAQUE_U64S` through `nros-sizes-build`). One
20//! implementation, two sources of the inputs.
21//!
22//! It is a separate crate rather than a `pub fn` in `nros-node` for one reason:
23//! a build script runs on the HOST, and a build-dependency on `nros-node` would
24//! give it a host-compiled `nros-node` — the wrong layout, silently. This crate
25//! is `no_std` and dependency-free so both sides can reach it.
26//!
27//! # What it does NOT decide
28//!
29//! The counts. `cbs`/`sc`/`nodes`/`arena` come from `nros-node/build.rs` and,
30//! on a synced leaf, from the derived env `nros sync` writes (issues 0827,
31//! 1061). This crate only places regions.
32

36/// Size and alignment of one region's element type.
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub struct RegionUnit {
39    /// `size_of::<T>()`.
40    pub size: usize,
41    /// `align_of::<T>()`. Must be a power of two; zero is rejected as 1.
42    pub align: usize,
43}
44
45/// Every element type the backing carves a table of, in declaration order.
46///
47/// The arena is absent on purpose: it is `[MaybeUninit<u8>]`, size 1 align 1 by
48/// definition, and its LENGTH is a count rather than a unit.
49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50pub struct RegionUnits {
51    pub callback_meta: RegionUnit,
52    pub sched_context: RegionUnit,
53    pub sched_context_id: RegionUnit,
54    pub sporadic_state: RegionUnit,
55    /// Placed only when [`Counts::alloc`] is set.
56    pub sporadic_atomic: RegionUnit,
57    pub remap_rule: RegionUnit,
58    pub node_record: RegionUnit,
59    pub concrete_session: RegionUnit,
60    pub extra_session_id: RegionUnit,
61    pub node_sched_entry: RegionUnit,
62    pub dispatch_slot: RegionUnit,
63    pub component_slot: RegionUnit,
64    pub group_name: RegionUnit,
65    pub group_sched_entry: RegionUnit,
66    pub violation: RegionUnit,
67    /// Alive supervision's per-SchedContext record, one per `sc`.
68    pub alive_slot: RegionUnit,
69    /// phase-476 W0 — the per-callback-slot lifetime tag (generation + owning
70    /// node), one per `cbs`. Placed after every older region so each keeps
71    /// its offset.
72    pub slot_tag: RegionUnit,
73    /// phase-382 W3' — one parameter-store slot (`nros_params::ParameterSlot`),
74    /// one per [`Counts::params`]. Placed LAST, for the reason `slot_tag` was:
75    /// an image that carves no store (`params == 0`) keeps every offset and
76    /// the total it had before the region existed.
77    pub parameter_slot: RegionUnit,
78}

80/// How many of each region the image needs.
81///
82/// `remaps` and `violations` are fixed counts in `nros-node` rather than knobs;
83/// they are parameters here so this crate states no policy.
84#[derive(Clone, Copy, Debug, PartialEq, Eq)]
85pub struct Counts {
86    pub cbs: usize,
87    pub sc: usize,
88    pub nodes: usize,
89    pub arena: usize,
90    pub remaps: usize,
91    pub violations: usize,
92    /// `nros-node`'s `alloc` feature — it adds the sporadic-atomic table, so it
93    /// changes the SIZE and cannot be inferred from the counts.
94    pub alloc: bool,
95    /// phase-382 W3' — parameter-store slots carved from the backing. ZERO for
96    /// an image the build does not know to build a store, which is what keeps
97    /// the region off every image that merely links the declare path (issue
98    /// 1702's rejected `.bss` static was unconditional). Policy is
99    /// `nros-params`'; this crate only places the region.
100    pub params: usize,
101}

129const fn align_up(off: usize, align: usize) -> usize {
130    off.div_ceil(align) * align
131}

133/// Place every region and report the offsets, total size and alignment.
134///
135/// Reproduces `#[repr(C)]` declaration-order layout: the arena first at offset
136/// 0, then each table aligned up to its own element alignment, and the total
137/// rounded up to the maximum alignment seen.
138pub const fn offsets(counts: Counts, units: RegionUnits) -> Offsets {
139    let Counts {
140        cbs,
141        sc,
142        nodes: node_slots,
143        arena,
144        remaps: remap_slots,
145        violations: violation_slots,
146        alloc,
147        params: param_slots,
148    } = counts;
149
150    let mut off = 0usize;
151    let mut max_align = 1usize;
152
153    // arena: [MaybeUninit<u8>; arena] — align 1, at offset 0.
154    let arena_off = 0usize;
155    off += arena;
156
157    macro_rules! place {
158        ($n:expr, $u:expr) => {{
159            let RegionUnit { size, align } = $u;
160            // A zero alignment is not a layout any compiler produces; treating
161            // it as 1 keeps this total rather than panicking in a `const fn`,
162            // and a caller that supplies one gets an answer it can compare.
163            let a = if align == 0 { 1 } else { align };
164            if a > max_align {
165                max_align = a;
166            }
167            off = align_up(off, a);
168            let at = off;
169            off += $n * size;
170            at
171        }};
172    }
173
174    let entries = place!(cbs, units.callback_meta);
175    let sched_contexts = place!(sc, units.sched_context);
176    let sched_context_bindings = place!(cbs, units.sched_context_id);
177    let sporadic_states = place!(sc, units.sporadic_state);
178    // Placed only under `alloc`, and the `else` must not move `off` — the
179    // no-alloc layout is the one every bare-metal image links.
180    let sporadic_atomic_states = if alloc {
181        place!(sc, units.sporadic_atomic)
182    } else {
183        off
184    };
185    let remaps = place!(remap_slots, units.remap_rule);
186    let nodes = place!(node_slots, units.node_record);
187    let extra_sessions = place!(node_slots, units.concrete_session);
188    let extra_session_ids = place!(node_slots, units.extra_session_id);
189    let node_sched_table = place!(node_slots, units.node_sched_entry);
190    let dispatch_slots = place!(node_slots, units.dispatch_slot);
191    let component_slots = place!(node_slots, units.component_slot);
192    let active_groups = place!(node_slots, units.group_name);
193    let group_sched_table = place!(cbs, units.group_sched_entry);
194    let monitor_violations = place!(violation_slots, units.violation);
195    let alive_slots = place!(sc, units.alive_slot);
196    let slot_tags = place!(cbs, units.slot_tag);
197    // Zero slots must not move `off` either: `place!` would still align it up
198    // to the slot's alignment, which is padding an image with no store would
199    // pay for nothing (and a total that moved for it).
200    let params = if param_slots > 0 {
201        place!(param_slots, units.parameter_slot)
202    } else {
203        off
204    };
205
206    Offsets {
207        arena: arena_off,
208        entries,
209        sched_contexts,
210        sched_context_bindings,
211        sporadic_states,
212        sporadic_atomic_states,
213        remaps,
214        nodes,
215        extra_sessions,
216        extra_session_ids,
217        node_sched_table,
218        dispatch_slots,
219        component_slots,
220        active_groups,
221        group_sched_table,
222        monitor_violations,
223        alive_slots,
224        slot_tags,
225        params,
226        size: align_up(off, max_align),
227        align: max_align,
228    }
229}

231/// The backing's total size in bytes — what a reservation has to hold.
232pub const fn size_of_backing(counts: Counts, units: RegionUnits) -> usize {
233    offsets(counts, units).size
