1//! phase-359 W10 — one place that spawns a platform task from Rust.
2//!
3//! Moved here from `nros-node::executor::platform_task`: three Rust callers now
4//! need it (`nros-node`'s worker pool and `open_threaded`, `nros-cpp`'s native
5//! tier runtime), and a helper for an ABI belongs beside the ABI rather than
6//! inside one of its consumers.
7//!
8//! Two executor-owned workers need a thread: the per-OS-priority pool
9//! (`os_priority` in `nros-node`'s executor) and the signalfd forwarder in `spin.rs`. Both were
10//! `std::thread`; both are platform tasks now, so the allocate-spawn-join
11//! sequence lives here rather than being written twice.
12//!
13//! The storage size is ASKED FOR (`nros_platform_task_storage_{size,align}`),
14//! never assumed. A hard-coded size is issue 0570 exactly: a Rust-side
15//! `pthread_attr_t` mirror was 36 bytes shorter than NuttX's, and
16//! `pthread_attr_init` wrote the difference into the caller's frame. The probes
17//! exist so no caller has to guess.
18
19// Both consumers are feature-selected AND one is `target_os = "linux"`, so
20// there are reachable combinations (e.g. `signal-fd-wake` off-Linux with no
21// `scheduler-os-priority`) where this module compiles with no caller. Enumerating
22// them in the `mod` gate would be a predicate nobody can keep correct; saying so
23// once here is the cheaper truth.
24#![allow(dead_code)]
25
26use core::ffi::c_void;
27
28/// The ABI's task attributes, mirrored for the `extern` declaration below.
29///
30/// phase-364 W3 — this is `nros_platform_task_attr_t` from `<nros/platform.h>`.
31/// It is declared here rather than imported because `nros-node` does not depend
32/// on `nros-platform-cffi` (where the generated bindings live) — the same reason
33/// the wake and task symbols are declared here by hand. The layout is checked
34/// against the header by `check-ffi-struct-mirrors`; see the note at the spawn
35/// site for what a drift would cost.
36#[repr(C)]
37struct TaskAttr {
38    name: *const core::ffi::c_char,
39    stack_bytes: usize,
40    stack_mem: *mut c_void,
41    priority: i32,
42    core: i8,
43    flags: u8,
44}
45
46/// `INT32_MIN` — inherit the creating task's priority.
47const PRIORITY_INHERIT: i32 = i32::MIN;
48
49unsafe extern "C" {
50    fn nros_platform_task_stack_unused_bytes() -> usize;
51    fn nros_platform_task_init(
52        task: *mut c_void,
53        attr: *mut c_void,
54        entry: unsafe extern "C" fn(*mut c_void) -> *mut c_void,
55        arg: *mut c_void,
56    ) -> i8;
57    fn nros_platform_task_join(task: *mut c_void) -> i8;
58    fn nros_platform_task_storage_size() -> usize;
59    fn nros_platform_task_storage_align() -> usize;
60}

62/// A spawned platform task plus the storage the platform tracks it in.
63///
64/// Joining is [`join`](Self::join) rather than `Drop`, because both callers
65/// have to signal their worker to stop BEFORE waiting for it — a `Drop` that
66/// joined implicitly would deadlock against a worker still blocked on its own
67/// wait.
68pub struct PlatformTask {
69    ptr: *mut u8,
70    layout: core::alloc::Layout,
71}
72
73impl PlatformTask {
74    /// Spawn `entry(arg)`, or `None` when this platform cannot host a task
75    /// (no storage sizing, allocation failure, or a refused spawn).
76    ///
77    /// # Safety
78    /// `arg` must remain valid and pointed-to until [`join`](Self::join)
79    /// returns — the task dereferences it.
80    pub unsafe fn spawn(
81        entry: unsafe extern "C" fn(*mut c_void) -> *mut c_void,
82        arg: *mut c_void,
83        stack_bytes: usize,
84        name: *const core::ffi::c_char,
85    ) -> Option<Self> {
86        // SAFETY: forwarded unchanged; the caller's contract is unchanged.
87        unsafe { Self::spawn_with(entry, arg, name, stack_bytes, PRIORITY_INHERIT as i64) }
88    }
89
90    /// Spawn `entry(arg)` stating a PRIORITY as well.
91    ///
92    /// phase-359 W10 — `spawn` inherits the creating task's priority, which is
93    /// right for the executor's own workers and wrong for a TIER: a tier's
94    /// priority is the thing its author declared. `priority <= 0` means
95    /// "unstated" and inherits, matching how the board descriptors spell an
96    /// absent priority; anything positive is the kernel's own number and is
97    /// passed through the band's RAW escape hatch, because a
98    /// `[tiers.<name>.<rtos>] priority` is already in the kernel's units.
99    ///
100    /// # Safety
101    /// Same as [`spawn`](Self::spawn).
102    pub unsafe fn spawn_with(
103        entry: unsafe extern "C" fn(*mut c_void) -> *mut c_void,
104        arg: *mut c_void,
105        name: *const core::ffi::c_char,
106        stack_bytes: usize,
107        priority: i64,
108    ) -> Option<Self> {
109        /// `NROS_PLATFORM_PRIORITY_RAW(n)` from `<nros/platform.h>`.
110        const fn raw(n: i32) -> i32 {
111            -0x4000_0000 - n
112        }
113        let priority = i32::try_from(priority)
114            .ok()
115            .filter(|p| *p > 0)
116            .map_or(PRIORITY_INHERIT, raw);
117        // SAFETY: both probes are documented pure functions, callable before
118        // any task exists.
119        let (size, align) = unsafe {
120            (
121                nros_platform_task_storage_size(),
122                nros_platform_task_storage_align(),
123            )
124        };
125        if size == 0 || align == 0 {
126            return None;
127        }
128        let layout = core::alloc::Layout::from_size_align(size, align).ok()?;
129        // SAFETY: `layout` has non-zero size.
130        let ptr = unsafe { alloc::alloc::alloc(layout) };
131        if ptr.is_null() {
132            return None;
133        }
134        // phase-364 W3 — a real attribute, not `NULL`.
135        //
136        // Passing `NULL` was correct on four ports and a guaranteed failure on
137        // ThreadX, whose `task_init` required an attr carrying the stack. W3
138        // made `NULL` mean "every default" everywhere, so `NULL` would work now
139        // — but the executor's workers do have a stack size to state, and
140        // stating it is what phase-359 W7 had to write a bespoke C shim to do.
141        let mut attr = TaskAttr {
142            name,
143            stack_bytes,
144            stack_mem: core::ptr::null_mut(),
145            priority,
146            core: -1,
147            flags: 0,
148        };
149        // SAFETY: `ptr` is storage of the size/alignment the platform asked
150        // for; `attr` outlives the call (the port copies what it needs);
151        // `entry`/`arg` are the caller's contract.
152        let rc = unsafe {
153            nros_platform_task_init(
154                ptr as *mut c_void,
155                (&raw mut attr) as *mut c_void,
156                entry,
157                arg,
158            )
159        };
160        if rc != 0 {
161            // SAFETY: same pair just returned by `alloc`; no task took it.
162            unsafe { alloc::alloc::dealloc(ptr, layout) };
163            return None;
164        }
165        Some(Self { ptr, layout })
166    }
167
168    /// Block until the task exits, then release its storage.
169    ///
170    /// The caller must already have told the task to stop; this only waits.
171    pub fn join(self) {
172        // SAFETY: `ptr` holds the handle `task_init` wrote.
173        unsafe { nros_platform_task_join(self.ptr as *mut c_void) };
174        // SAFETY: the task has exited, so nothing else references the storage;
175        // `ptr`/`layout` are the pair `spawn` allocated.
176        unsafe { alloc::alloc::dealloc(self.ptr, self.layout) };
177        core::mem::forget(self);
178    }
179}
180
181impl Drop for PlatformTask {
182    fn drop(&mut self) {
183        // Reached only if a caller dropped the handle without joining, which
184        // leaves a running task pointed at storage we are about to free. Leak
185        // the storage instead: a leak is recoverable, a use-after-free by a
186        // live task is not.
187        //
188        // `join` calls `mem::forget`, so the normal path never lands here.
189    }
190}

192/// Smallest number of bytes ever left unused on the CALLING task's stack, or
193/// `0` if this port does not instrument it.
194///
195/// The heap has had `heap_used_bytes` since RFC-0034 D7; the stack had
196/// nothing, and stack overflow is how one component corrupts another's state.
197/// A tier asks this about ITSELF -- both kernels answer for the calling task
198/// with no handle, and answering for an arbitrary one needs a native handle
199/// this ABI does not carry.
200///
201/// `0` means "not instrumented", not "no headroom".
202pub fn stack_unused_bytes() -> usize {
203    // SAFETY: a plain query with no arguments and no state; every port either
204    // answers or returns 0.
205    unsafe { nros_platform_task_stack_unused_bytes() }
206}
