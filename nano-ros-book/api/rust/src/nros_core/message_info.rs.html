1//! Message metadata for received messages
2//!
3//! This module provides the `MessageInfo` type which contains metadata
4//! about received messages, matching the rclrs pattern.
5//!
6//! # Example
7//!
8//! ```text
9//! node.create_subscription("/topic", |msg: &Int32, info: &MessageInfo| {
10//!     println!("Received at {:?} from {:?}", info.source_timestamp(), info.publisher_gid());
11//! });
12//! ```
13
14use crate::Time;
15
16/// Size of the publisher Global Identifier (GID), in bytes.
17///
18/// **24, which is upstream's `RMW_GID_STORAGE_SIZE`** — measured against Humble,
19/// `/opt/ros/humble/include/rmw/rmw/types.h:42` reads `24u` — and the width of
20/// the ABI's own `rmw_gid_t::data` (`RMW_GID_STORAGE_SIZE`,
21/// `packages/core/nros-rmw-abi/include/nros/rmw_entity.h`). So a gid read out of
22/// a [`MessageInfo`] and a gid written by the `get_gid_for_publisher` vtable
23/// slot are the same TYPE, comparable without a conversion.
24///
25/// It was 16 until the phase-467 RMW gap-closure design study's Q1(a). The 16
26/// is *zenoh's* wire width, not a property of the identifier — see
27/// [`MessageInfo::publisher_gid`] for what that leaves a reader owing.
28pub const PUBLISHER_GID_SIZE: usize = 24;
29
30/// Zero-extend a backend's narrower identity into a full-width publisher GID.
31///
32/// A backend whose identity is shorter than [`PUBLISHER_GID_SIZE`] must pad the
33/// TAIL with zeros rather than leave it undefined, or two gids naming the same
34/// publisher compare unequal on stack garbage. This is the one spelling of that
35/// padding on the Rust side; the Cyclone backend does the same thing in C++
36/// (`cyclone_get_gid_for_publisher` in
37/// `packages/rmw/cyclonedds/nros-rmw-cyclonedds/src/vtable.cpp` — a `memset`,
38/// then a `memcpy` of the 16-byte DDS writer GUID).
39///
40/// An `N` wider than [`PUBLISHER_GID_SIZE`] is a compile error, which is the
41/// Rust spelling of Cyclone's
42/// `static_assert(sizeof(guid.v) <= RMW_GID_STORAGE_SIZE)`.
43pub const fn pad_publisher_gid<const N: usize>(narrow: &[u8; N]) -> [u8; PUBLISHER_GID_SIZE] {
44    const {
45        assert!(
46            N <= PUBLISHER_GID_SIZE,
47            "a publisher identity wider than PUBLISHER_GID_SIZE cannot be zero-extended into it"
48        );
49    }
50    let mut wide = [0u8; PUBLISHER_GID_SIZE];
51    let mut i = 0;
52    while i < N {
53        wide[i] = narrow[i];
54        i += 1;
55    }
56    wide
57}

59/// Metadata about a received message
60///
61/// Contains information about the source and timing of a message.
62/// This matches the rclrs `MessageInfo` type.
63#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
64pub struct MessageInfo {
65    /// Timestamp when the message was published (from the publisher's clock)
66    source_timestamp: Time,
67    /// Timestamp when the message was received (from the subscriber's clock)
68    received_timestamp: Time,
69    /// Sequence number of the message from the publisher
70    publication_sequence_number: i64,
71    /// Sequence number of the message at the subscriber
72    reception_sequence_number: i64,
73    /// Global identifier of the publisher
74    publisher_gid: [u8; PUBLISHER_GID_SIZE],
75}
76
77impl MessageInfo {
78    /// Create a new MessageInfo with all fields set to defaults
79    pub const fn new() -> Self {
80        Self {
81            source_timestamp: Time::new(0, 0),
82            received_timestamp: Time::new(0, 0),
83            publication_sequence_number: 0,
84            reception_sequence_number: 0,
85            publisher_gid: [0u8; PUBLISHER_GID_SIZE],
86        }
87    }
88
89    /// Create a MessageInfo with the given timestamps
90    pub const fn with_timestamps(source: Time, received: Time) -> Self {
91        Self {
92            source_timestamp: source,
93            received_timestamp: received,
94            publication_sequence_number: 0,
95            reception_sequence_number: 0,
96            publisher_gid: [0u8; PUBLISHER_GID_SIZE],
97        }
98    }
99
100    /// Get the timestamp when the message was published
101    pub const fn source_timestamp(&self) -> Time {
102        self.source_timestamp
103    }
104
105    /// Get the timestamp when the message was received
106    pub const fn received_timestamp(&self) -> Time {
107        self.received_timestamp
108    }
109
110    /// Get the publication sequence number
111    pub const fn publication_sequence_number(&self) -> i64 {
112        self.publication_sequence_number
113    }
114
115    /// Get the reception sequence number
116    pub const fn reception_sequence_number(&self) -> i64 {
117        self.reception_sequence_number
118    }
119
120    /// Get the publisher's Global Identifier (GID).
121    ///
122    /// **The array is 24 bytes. How many of them MEAN anything is a backend
123    /// property, and today it is never all 24.**
124    ///
125    /// The width is upstream's ([`PUBLISHER_GID_SIZE`]), so this value and the
126    /// `rmw_gid_t` the `get_gid_for_publisher` vtable slot fills are one type.
127    /// What the width does NOT say:
128    ///
129    /// * **zenoh fills 16 and zero-pads 8.** `RMW_ATTACHMENT_SIZE` carries a
130    ///   16-byte gid because that is `rmw_zenoh_cpp`'s wire layout — its reader
131    ///   REJECTS any other length — so the trailing 8 bytes here are padding
132    ///   written by [`pad_publisher_gid`] and carry no information. Compare
133    ///   whole arrays anyway: the padding is deterministic, and truncating to
134    ///   16 by hand is how a future backend's extra bytes get silently dropped.
135    /// * **Cyclone never fills it.** A pure C/C++ backend writes no
136    ///   `MessageInfo` at all — the `message_info()` callback sees `None`
137    ///   there, not a gid — and nothing on the Cyclone receive path calls
138    ///   [`set_publisher_gid`](Self::set_publisher_gid). Wherever a
139    ///   `MessageInfo` does exist unpopulated it keeps its `Default`: all
140    ///   zeros, which is the ABI's spelling for "unknown", not an identity.
141    /// * **A gid from here is NOT today comparable with one from
142    ///   `get_gid_for_publisher`.** No backend produces both: on Cyclone the
143    ///   vtable slot returns a real DDS writer GUID and this field is never
144    ///   written; on zenoh this field is a per-publisher value and the slot is
145    ///   NULL. Making the two answers agree is issue 1495 — it changes a value
146    ///   a stock ROS 2 peer reads off our wire, so it is deliberately not part
147    ///   of the widening that made them the same type.
148    ///
149    /// An all-zero gid therefore means "this backend did not say", never "the
150    /// publisher's id is zero".
151    pub const fn publisher_gid(&self) -> &[u8; PUBLISHER_GID_SIZE] {
152        &self.publisher_gid
153    }
154
155    /// Set the source timestamp
156    pub fn set_source_timestamp(&mut self, ts: Time) {
157        self.source_timestamp = ts;
158    }
159
160    /// Set the received timestamp
161    pub fn set_received_timestamp(&mut self, ts: Time) {
162        self.received_timestamp = ts;
163    }
164
165    /// Set the publication sequence number
166    pub fn set_publication_sequence_number(&mut self, seq: i64) {
167        self.publication_sequence_number = seq;
168    }
169
170    /// Set the reception sequence number
171    pub fn set_reception_sequence_number(&mut self, seq: i64) {
172        self.reception_sequence_number = seq;
173    }
174
175    /// Set the publisher GID.
176    ///
177    /// Takes the full [`PUBLISHER_GID_SIZE`] width. A backend whose identity is
178    /// narrower zero-extends it through [`pad_publisher_gid`] rather than
179    /// building the array itself — see [`publisher_gid`](Self::publisher_gid)
180    /// for what a reader may and may not conclude from the result.
181    pub fn set_publisher_gid(&mut self, gid: [u8; PUBLISHER_GID_SIZE]) {
182        self.publisher_gid = gid;
183    }
184}

186/// Raw-subscription message info: [`MessageInfo`] metadata plus the
187/// sample's wire-level **attachment** bytes, borrowed for the callback
188/// scope.
189///
190/// Surfaced on the generic (type-erased) subscription path
191/// (`node.subscription(t).generic(..).message_info().build(cb)` — the
192/// `FnMut(&[u8], &RawMessageInfo)` callback). The attachment carries
193/// out-of-band tags such as the cross-RMW bridge's `bridge_origin`
194/// (read via [`attachment`](Self::attachment) for echo suppression).
195///
196/// The `'a` lifetime ties the borrowed attachment to the dispatch call;
197/// copy out what you need before the callback returns. Metadata
198/// accessors delegate to the inner [`MessageInfo`]; on backends without
199/// a combined raw+info+attachment take they read their defaults (the
200/// attachment is always populated).
201#[derive(Debug, Clone, Copy)]
202pub struct RawMessageInfo<'a> {
203    info: MessageInfo,
204    attachment: &'a [u8],
205}
206
207impl<'a> RawMessageInfo<'a> {
208    /// Build from an attachment slice (metadata defaulted).
209    pub const fn new(attachment: &'a [u8]) -> Self {
210        Self {
211            info: MessageInfo::new(),
212            attachment,
213        }
214    }
215
216    /// Build from explicit metadata + attachment.
217    pub const fn with_info(info: MessageInfo, attachment: &'a [u8]) -> Self {
218        Self { info, attachment }
219    }
220
221    /// The sample's wire-level attachment bytes (empty if none).
222    pub const fn attachment(&self) -> &'a [u8] {
223        self.attachment
224    }
225
226    /// The underlying metadata.
227    pub const fn info(&self) -> &MessageInfo {
228        &self.info
229    }
230
231    /// Timestamp when the message was published.
232    pub const fn source_timestamp(&self) -> Time {
233        self.info.source_timestamp()
234    }
235
236    /// Timestamp when the message was received.
237    pub const fn received_timestamp(&self) -> Time {
238        self.info.received_timestamp()
239    }
240
241    /// Publisher's Global Identifier (GID).
242    ///
243    /// Same bound as [`MessageInfo::publisher_gid`]: 24 bytes wide, with the
244    /// meaningful prefix decided by the backend and today never the whole
245    /// array.
246    pub const fn publisher_gid(&self) -> &[u8; PUBLISHER_GID_SIZE] {
247        self.info.publisher_gid()
248    }
249}

251#[cfg(test)]
252mod tests {
253    use super::*;
254
255    #[test]
256    fn test_message_info_default() {
257        let info = MessageInfo::new();
258        assert_eq!(info.source_timestamp(), Time::new(0, 0));
259        assert_eq!(info.publication_sequence_number(), 0);
260        assert_eq!(info.publisher_gid(), &[0u8; PUBLISHER_GID_SIZE]);
261    }
262
263    /// The width is upstream's, and that is the whole point of it: measured in
264    /// the `ros2` box, `/opt/ros/humble/include/rmw/rmw/types.h:42` reads
265    /// `#define RMW_GID_STORAGE_SIZE 24u`, and our own ABI header takes the
266    /// same number. A gid taken from a sample and a gid written into an
267    /// `rmw_gid_t` are one type only while this holds.
268    #[test]
269    fn publisher_gid_is_upstreams_storage_size() {
270        assert_eq!(PUBLISHER_GID_SIZE, 24);
271    }
272
273    /// zenoh's identity is 16 bytes and cannot move — that is
274    /// `rmw_zenoh_cpp`'s `RMW_ATTACHMENT_SIZE` wire layout, whose reader
275    /// rejects any other length. Padding is what lets it be stored at the
276    /// upstream width without either side moving.
277    #[test]
278    fn a_narrow_backend_identity_zero_extends_into_the_tail() {
279        let wire = [0xABu8; 16];
280        let wide = pad_publisher_gid(&wire);
281        assert_eq!(&wide[..16], &wire[..]);
282        assert_eq!(&wide[16..], &[0u8; PUBLISHER_GID_SIZE - 16]);
283
284        // Deterministic, so two samples from one publisher still compare equal
285        // over the WHOLE array — the property `test_gid_consistency` reads.
286        assert_eq!(pad_publisher_gid(&wire), wide);
287    }
288
289    #[test]
290    fn test_message_info_with_timestamps() {
291        let source = Time::new(1, 500_000_000);
292        let received = Time::new(1, 600_000_000);
293        let info = MessageInfo::with_timestamps(source, received);
294        assert_eq!(info.source_timestamp(), source);
295        assert_eq!(info.received_timestamp(), received);
296    }
