1//! Clock API for nros
2//!
3//! This module provides clock abstraction for different time sources:
4//! - **SystemTime**: Wall clock time (affected by system time changes)
5//! - **SteadyTime**: Monotonic time (not affected by system time changes)
6//! - **RosTime**: Simulation time (can be paused/scaled)
7//!
8//! # Example
9//!
10//! ```text
11//! use nros::clock::{Clock, ClockType};
12//!
13//! // Create a system clock
14//! let clock = Clock::system();
15//! let now = clock.now();
16//! println!("Current time: {} sec", now.sec);
17//!
18//! // Create a steady clock for measuring durations
19//! let clock = Clock::steady();
20//! let start = clock.now();
21//! // ... do work ...
22//! let elapsed = clock.now() - start;
23//! ```
24//!
25//! # Where the time comes from
26//!
27//! With a platform port linked (the `platform-clock` feature), every clock
28//! reads the PORT: `nros_platform_time_now_ns` for the wall, and
29//! `nros_platform_clock_ns` for the monotonic. Without one there is no clock
30//! to read, and all three types fall back to an internal counter the caller
31//! advances with `update_steady_time()` — suitable for a bare-metal RTIC or
32//! polling loop, which is the only build where that counter has an owner.
33

45/// Type of clock to use for time queries
46#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
47pub enum ClockType {
48    /// System time (wall clock)
49    ///
50    /// This clock reflects the system's real-time clock and may be
51    /// affected by NTP adjustments, user changes, or daylight saving time.
52    /// Use for timestamps that need to correlate with real-world time.
53    #[default]
54    SystemTime,
55
56    /// Steady/monotonic time
57    ///
58    /// This clock is guaranteed to be monotonically increasing and is
59    /// not affected by system time changes. Use for measuring durations
60    /// and timeouts. Reads the port's `nros_platform_clock_ns` where one is
61    /// linked, and the caller-advanced counter where none is.
62    SteadyTime,
63
64    /// ROS time (simulation time)
65    ///
66    /// This clock can be overridden for simulation purposes. When a
67    /// ROS time override is active — a simulator's or a bag player's
68    /// `/clock` — `now()` returns the overridden time. Otherwise it IS
69    /// [`SystemTime`](Self::SystemTime), which is what `rclcpp`'s
70    /// `ClockType::ROS_TIME` reads with `use_sim_time` false, so a node
71    /// written for simulation still runs standalone.
72    RosTime,
73}

110// On platforms without 64-bit atomics, use split 32-bit values
111// Note: This is not fully atomic but works for single-threaded embedded contexts
112#[cfg(not(target_has_atomic = "64"))]
113mod atomic_time {
114    use super::*;
115
116    // Split into high and low 32-bit parts
117    static ROS_TIME_OVERRIDE_LOW: AtomicI32 = AtomicI32::new(-1);
118    static ROS_TIME_OVERRIDE_HIGH: AtomicI32 = AtomicI32::new(-1);
119    static STEADY_TIME_LOW: AtomicI32 = AtomicI32::new(0);
120    static STEADY_TIME_HIGH: AtomicI32 = AtomicI32::new(0);
121
122    pub(super) fn get_ros_override() -> i64 {
123        let high = ROS_TIME_OVERRIDE_HIGH.load(Ordering::Relaxed);
124        let low = ROS_TIME_OVERRIDE_LOW.load(Ordering::Relaxed);
125        if high < 0 {
126            -1
127        } else {
128            ((high as i64) << 32) | (low as u32 as i64)
129        }
130    }
131
132    pub(super) fn set_ros_override(nanos: i64) {
133        if nanos < 0 {
134            ROS_TIME_OVERRIDE_HIGH.store(-1, Ordering::Relaxed);
135            ROS_TIME_OVERRIDE_LOW.store(-1, Ordering::Relaxed);
136        } else {
137            ROS_TIME_OVERRIDE_HIGH.store((nanos >> 32) as i32, Ordering::Relaxed);
138            ROS_TIME_OVERRIDE_LOW.store(nanos as i32, Ordering::Relaxed);
139        }
140    }
141
142    pub(super) fn get_steady() -> i64 {
143        let high = STEADY_TIME_HIGH.load(Ordering::Relaxed);
144        let low = STEADY_TIME_LOW.load(Ordering::Relaxed);
145        ((high as i64) << 32) | (low as u32 as i64)
146    }
147
148    pub(super) fn set_steady(nanos: i64) {
149        STEADY_TIME_HIGH.store((nanos >> 32) as i32, Ordering::Relaxed);
150        STEADY_TIME_LOW.store(nanos as i32, Ordering::Relaxed);
151    }
152
153    pub(super) fn add_steady(delta: i64) {
154        let current = get_steady();
155        set_steady(current.saturating_add(delta));
156    }
157}

174/// The platform's wall clock, or `None` when this build has no port to ask.
175///
176/// phase-359 W10 (backend tier). `nros-core` sits BELOW `nros-platform`, so it
177/// cannot depend on it — it declares the two ABI symbols directly, exactly as
178/// `nros-node` already does for `nros_platform_clock_ns` ("every platform port
179/// exports it through the same linkage contract"). The feature is what promises
180/// a port is linked; without it this is `None` and the caller keeps the counter
181/// it had.
182///
183/// ONE symbol since issue 0532 item 5 collapsed the wall clock; this function
184/// was named there as the one place that would change, and it was.
185// phase-359 W10 follow-up — NOT `not(std)`. This used to be gated away on a
186// `std` build, so an image with a port linked read `SystemTime` from
187// `Clock::system()` and `nros_platform_time_now_ns` from the executor's epoch
188// source: two wall clocks, one image. They agree on POSIX by coincidence (both
189// are CLOCK_REALTIME) and stop agreeing the moment a port has an opinion — an
190// RTC-backed or simulated one — because only the port is authoritative and only
191// one of the two readers asked it. The rule W10 set in `nros-node` applies
192// here: when a port is linked it IS the clock, and `std` is what a build
193// without one falls back to.
194#[cfg(feature = "platform-clock")]
195fn platform_wall_clock() -> Option<Time> {
196    unsafe extern "C" {
197        fn nros_platform_time_now_ns() -> u64;
198    }
199    // SAFETY: a bare wall-clock read, no pointer arguments, guaranteed by
200    // whichever port linked the binary — the same contract `nros-node` relies
201    // on for `nros_platform_clock_ns`.
202    //
203    // ONE symbol, so one sample: issue 0532 collapsed the former
204    // `time_since_epoch_{secs,nanos}` pair, which this was written against and
205    // which needed a bounded re-read to survive a second boundary landing
206    // between the two calls. That loop is what the collapse deletes.
207    let ns = unsafe { nros_platform_time_now_ns() };
208    // A port with no RTC returns 0. Reporting the Unix epoch as "now" would be
209    // a wrong answer stated confidently, so say nothing and let the caller's
210    // counter fallback stand.
211    if ns == 0 {
212        return None;
213    }
214    Some(Time::new(
215        (ns / 1_000_000_000) as i32,
216        (ns % 1_000_000_000) as u32,
217    ))
218}

220#[cfg(not(feature = "platform-clock"))]
221fn platform_wall_clock() -> Option<Time> {
222    None

225/// The platform's MONOTONIC clock, or `None` when this build has no port to ask.
226///
227/// issue 1334 — the sibling of [`platform_wall_clock`], declared the same way
228/// and under the same feature, because it is the same promise: `platform-clock`
229/// means a port is linked, and every port exports both halves
230/// (`nros/platform.h` "Wall clock" and RFC-0073's monotonic export are
231/// MANDATORY, not optional). `nros-node`, `nros-log`, `nros-c` and the `nros`
232/// umbrella already hand-declare this symbol under exactly this contract.
233///
234/// Unlike the wall clock there is no "port with no RTC" case to detect: a
235/// monotonic counter reading 0 is a legitimate answer at boot, so a zero is
236/// passed through rather than read as "no clock".
237#[cfg(feature = "platform-clock")]
238fn platform_steady_clock() -> Option<Time> {
239    unsafe extern "C" {
240        fn nros_platform_clock_ns() -> u64;
241    }
242    // SAFETY: a bare counter read, no pointer arguments, guaranteed by
243    // whichever port linked the binary — the same contract `platform_wall_clock`
244    // above relies on for `nros_platform_time_now_ns`.
245    let ns = unsafe { nros_platform_clock_ns() };
246    Some(Time::new(
247        (ns / 1_000_000_000) as i32,
248        (ns % 1_000_000_000) as u32,
249    ))
250}

252#[cfg(not(feature = "platform-clock"))]
253fn platform_steady_clock() -> Option<Time> {
254    None
255}

257impl Clock {
258    /// Create a new clock of the specified type
259    pub const fn new(clock_type: ClockType) -> Self {

261    }

263    /// Create a system time clock
264    ///
265    /// System time reflects the real-world wall clock time.
266    pub const fn system() -> Self {
267        Self {
268            clock_type: ClockType::SystemTime,
269        }
270    }
271
272    /// Create a steady (monotonic) time clock
273    ///
274    /// Steady time is guaranteed to only increase and is not affected
275    /// by system time changes.
276    pub const fn steady() -> Self {
277        Self {
278            clock_type: ClockType::SteadyTime,
279        }
280    }

282    /// Create a ROS time clock
283    ///
284    /// ROS time can be overridden for simulation. When no override is
285    /// active, it returns system time — the identical expression
286    /// [`Clock::system`] evaluates, so the two cannot drift (issue 1334).
287    pub const fn ros_time() -> Self {
288        Self {
289            clock_type: ClockType::RosTime,
290        }
291    }
292
293    /// Get the clock type
294    pub const fn clock_type(&self) -> ClockType {
295        self.clock_type
296    }
297
298    /// The wall clock, in ONE place — issue 1334.
299    ///
300    /// Two clock types answer "what is the wall time", and they must answer it
301    /// identically: `SystemTime` always, and `RosTime` whenever no `/clock`
302    /// override is installed. Before this they were two expressions, and only
303    /// the first was ever moved onto the port (phase-359 W10), so `RosTime`'s
304    /// fallback was left reading the counter — a value nothing in the tree
305    /// advances. One expression is what keeps the next move from splitting
306    /// them again.
307    ///
308    /// - **A platform port linked** (`platform-clock`): the port's wall clock.
309    ///   It is the authority, and an image must not hold two answers to "what
310    ///   time is it".
311    /// - **No port**: the internal counter, which the caller advances with
312    ///   `update_steady_time()`. A build with no port has no clock to read, so
313    ///   this is the honest answer rather than a fallback with an opinion.
314    fn wall_now() -> Time {
315        // phase-359 W10 (backend tier) — a wall clock that does not need the
316        // `std` FEATURE. Before this, `SystemTime` fell back to the STEADY
317        // counter unconditionally: the same value a monotonic clock returns,
318        // presented as time since the Unix epoch. That is not a degraded wall
319        // clock, it is a different quantity, and the only thing standing
320        // between a build and it was whether some crate in the graph happened
321        // to name `std`.
322        if let Some(t) = platform_wall_clock() {
323            return t;
324        }
325        Time::from_nanos(atomic_time::get_steady())
326    }
327
328    /// The monotonic clock, in ONE place — issue 1334's sibling site.
329    ///
330    /// Same rule as [`Self::wall_now`], one clock over: a linked port IS the
331    /// monotonic clock, and the counter is what a build with no port was
332    /// given.
333    ///
334    /// This overturns half of phase-359 W10, which moved `SystemTime` onto the
335    /// port and left this line reading "`SteadyTime` is the counter on every
336    /// flavour, deliberately: it is advanced by its owner … and the port's
337    /// monotonic export is the executor's business, not this type's". The
338    /// owner never arrived — nothing outside this file's own tests has ever
339    /// called `update_steady_time` — so on every shipped image
340    /// `Clock::steady().now()` was the constant 0, while the C surface
341    /// answered the SAME question (`NROS_CLOCK_STEADY_TIME`) with
342    /// `nros_platform_clock_ns`. That is issue 1334's complaint exactly, one
343    /// clock over: two surfaces, one image, different answers, and the Rust
344    /// one is not a clock. W10's own rule settles it — when a port is linked
345    /// it IS the clock — and the counter keeps its documented RTIC meaning in
346    /// the one build where it is the only source.
347    fn steady_now() -> Time {
348        if let Some(t) = platform_steady_clock() {
349            return t;
350        }
351        Time::from_nanos(atomic_time::get_steady())
352    }
353
354    /// Get the current time from this clock
355    ///
356    /// # Platform behavior
357    ///
358    /// - **A platform port linked** (`platform-clock`): the port's clocks —
359    ///   `nros_platform_time_now_ns` for the wall, `nros_platform_clock_ns`
360    ///   for the monotonic. The port is the authority, and an image must not
361    ///   hold two answers to "what time is it".
362    /// - **No port**: the internal counter, which the caller advances with
363    ///   `update_steady_time()`.
364    ///
365    /// phase-359 W10 — there is no `std::time` arm. The platform API IS the
366    /// clock: a build with a port reads it, a build without one has no clock to
367    /// read and says so with the counter it was given. `SystemTime` used to sit
368    /// here as a third answer for hosted builds, which made "what time is it"
369    /// depend on whether some crate in the graph happened to name `std`.
370    ///
371    /// `RosTime` with no override is `SystemTime`, in EVERY build shape and by
372    /// construction — both arms call [`Self::wall_now`]. That is rclcpp's
373    /// contract (`ClockType::ROS_TIME` with `use_sim_time` false reads the
374    /// system clock), what five doc sites across three languages already
375    /// promised, and what the C surface already did; issue 1334 is the four
376    /// phases in which the Rust arm said something else.
377    pub fn now(&self) -> Time {
378        match self.clock_type {
379            ClockType::SystemTime => Self::wall_now(),
380            ClockType::SteadyTime => Self::steady_now(),
381            ClockType::RosTime => {
382                let override_nanos = atomic_time::get_ros_override();
383                if override_nanos >= 0 {
384                    // A `/clock` source is driving simulated time.
385                    Time::from_nanos(override_nanos)
386                } else {
387                    // None is. rclcpp reads the system clock here and so do we
388                    // — the same expression `Clock::system()` evaluates, not a
389                    // copy of it (issue 1334).
390                    Self::wall_now()
391                }
392            }
393        }
394    }

396    /// Set a ROS time override
397    ///
398    /// When set, all `RosTime` clocks will return this time instead of
399    /// system time. This is useful for simulation.
400    ///
401    /// # Arguments
402    /// * `nanos` - Nanoseconds since epoch
403    pub fn set_ros_time_override(nanos: i64) {
404        atomic_time::set_ros_override(nanos);
405    }
406
407    /// Set a ROS time override from a Time value
408    pub fn set_ros_time_override_time(time: Time) {
409        Self::set_ros_time_override(time.to_nanos());
410    }
411
412    /// Clear the ROS time override
413    ///
414    /// After clearing, `RosTime` clocks will return system time again.
415    pub fn clear_ros_time_override() {
416        atomic_time::set_ros_override(-1);
417    }
418
419    /// Check if a ROS time override is active
420    pub fn is_ros_time_override_active() -> bool {
421        atomic_time::get_ros_override() >= 0
422    }
423
424    /// Get the current ROS time override value (if active)
425    pub fn get_ros_time_override() -> Option<Time> {
426        let nanos = atomic_time::get_ros_override();
427        if nanos >= 0 {
428            Some(Time::from_nanos(nanos))
429        } else {
430            None
431        }
432    }

434    /// Update the steady time counter
435    ///
436    /// For environments with NO platform port, call this periodically from
437    /// your main loop or RTIC task to advance the clocks. With a port linked
438    /// this counter is not what any clock reads — the port is (issue 1334) —
439    /// and advancing it has no observable effect.
440    ///
441    /// # Arguments
442    /// * `delta_nanos` - Nanoseconds elapsed since last call
443    pub fn update_steady_time(delta_nanos: i64) {
444        atomic_time::add_steady(delta_nanos);
445    }
446
447    /// Update the steady time counter (milliseconds version)
448    ///
449    /// Convenience method for RTIC tasks using millisecond intervals.
450    ///
451    /// # Arguments
452    /// * `delta_ms` - Milliseconds elapsed since last call
453    pub fn update_steady_time_ms(delta_ms: u64) {

456    }

458    /// Set the steady time counter to a specific value
459    ///
460    /// Use this to initialize the clock or synchronize with an external
461    /// time source.
462    pub fn set_steady_time(nanos: i64) {
463        atomic_time::set_steady(nanos);
464    }
465
466    /// Get the current steady time counter value
467    pub fn get_steady_time_nanos() -> i64 {
468        atomic_time::get_steady()
469    }

476    /// Serialises every test that touches the process-global clock state —
477    /// issue 1334.
478    ///
479    /// The override, the counter and (below) the fake port are all statics, and
480    /// `cargo test` runs these in parallel threads: a test that SETS one and
481    /// reads it back is racing every sibling that writes the same static. That
482    /// was already true before this guard existed (`test_ros_time_override` and
483    /// `test_steady_time_update` write two different globals that `Clock::now`
484    /// reads together) and it becomes load-bearing here, because the new tests
485    /// move the fake port's hands while another test asserts where they point.
486    ///
487    /// A spin lock rather than `std::sync::Mutex`: this crate is `#![no_std]`
488    /// and `std` is a FEATURE, so a `Mutex` here would confine the guard — and
489    /// therefore every test using it — to the `std` flavour, which is not where
490    /// the `platform-clock` tests need to live.
491    struct ClockGuard;
492
493    static CLOCK_LOCK: core::sync::atomic::AtomicBool = core::sync::atomic::AtomicBool::new(false);
494
495    impl ClockGuard {
496        fn acquire() -> Self {
497            while CLOCK_LOCK
498                .compare_exchange(false, true, Ordering::Acquire, Ordering::Relaxed)
499                .is_err()
500            {
501                core::hint::spin_loop();
502            }
503            Self
504        }
505    }
506
507    impl Drop for ClockGuard {
508        fn drop(&mut self) {
509            // Leave the globals as a fresh process has them, so a test that
510            // does NOT take the guard still sees the documented defaults.
511            Clock::clear_ros_time_override();
512            Clock::set_steady_time(0);
513            #[cfg(feature = "platform-clock")]
514            fake_port::reset();
515            CLOCK_LOCK.store(false, Ordering::Release);
516        }
517    }

519    /// The stand-in platform port for the `platform-clock` flavour.
520    ///
521    /// One definition of each symbol, at module level, because a linker sees
522    /// one test binary: the wall symbol used to be defined INSIDE the single
523    /// test that needed it, which works exactly until a second test needs it
524    /// too. Both hands MOVE (they are atomics, not constants), which is what
525    /// lets a test assert that a clock ADVANCES rather than merely that it
526    /// reads a magic number.
527    #[cfg(feature = "platform-clock")]
528    mod fake_port {
529        use core::sync::atomic::{AtomicU64, Ordering};
530
531        /// 2001-09-09T01:46:40Z — a fixed instant no real clock will return.
532        pub(super) const WALL_BASE_NS: u64 = 1_000_000_000 * 1_000_000_000;
533        /// 42 s of uptime — no relation to `WALL_BASE_NS`, deliberately: the
534        /// two clocks are different quantities and a test must not be able to
535        /// pass by reading the wrong one.
536        pub(super) const STEADY_BASE_NS: u64 = 42_000_000_000;
537
538        pub(super) static WALL_NS: AtomicU64 = AtomicU64::new(WALL_BASE_NS);
539        pub(super) static STEADY_NS: AtomicU64 = AtomicU64::new(STEADY_BASE_NS);
540
541        pub(super) fn reset() {
542            WALL_NS.store(WALL_BASE_NS, Ordering::Relaxed);
543            STEADY_NS.store(STEADY_BASE_NS, Ordering::Relaxed);
544        }
545
546        #[unsafe(no_mangle)]
547        extern "C" fn nros_platform_time_now_ns() -> u64 {
548            WALL_NS.load(Ordering::Relaxed)
549        }
550
551        #[unsafe(no_mangle)]
552        extern "C" fn nros_platform_clock_ns() -> u64 {
553            STEADY_NS.load(Ordering::Relaxed)
554        }
555    }

584    #[test]
585    fn test_ros_time_override() {
586        let _guard = ClockGuard::acquire();
587        // Clear any existing override
588        Clock::clear_ros_time_override();
589        assert!(!Clock::is_ros_time_override_active());
590        assert!(Clock::get_ros_time_override().is_none());

592        // Set override

594        Clock::set_ros_time_override_time(override_time);

596        assert_eq!(Clock::get_ros_time_override(), Some(override_time));
597
598        // ROS clock should return override time
599        let ros_clock = Clock::ros_time();
600        let now = ros_clock.now();
601        assert_eq!(now, override_time);
602

634    /// With NO port linked there is no clock, and all three types say so with
635    /// the one counter they were given — issue 1334.
636    ///
637    /// This replaces `test_system_clock_returns_nonzero`, which asserted
638    /// `now.sec > 0` for `Clock::system()` in exactly this shape. That claim
639    /// was false from the day phase-359 W10 removed the `std::time` arm — a
640    /// portless build reads the counter, which is 0 — and nobody saw it,
641    /// because its `all(std, not(platform-clock))` gate put it in no lane at
642    /// all: the one nros-core feature lane is `std,platform-clock`, and the
643    /// default lane has no `std`. A test in no lane is the shape CLAUDE.md's
644    /// `required-features` rule names, one cfg over.
645    ///
646    /// What it asserts instead is the property that IS true here, and it is the
647    /// portless half of this issue's fix: `RosTime` with no override reads
648    /// whatever `SystemTime` reads, whatever that turns out to be.
649    #[test]
650    #[cfg(not(feature = "platform-clock"))]
651    fn without_a_port_every_clock_is_the_counter() {

665    /// phase-359 W10 follow-up — a linked port OUTRANKS `std` for the wall
666    /// clock, on a `std` build too.
667    ///
668    /// This is the one configuration where the two disagree observably, and it
669    /// is why the test image defines the port symbol itself (`fake_port`): the
670    /// value returned is nothing like a real `SystemTime`, so a
671    /// `Clock::system()` that answered with `SystemTime::now()` — which is what
672    /// this file did before, because `platform_wall_clock` was gated
673    /// `not(std)` — fails loudly rather than coincidentally passing. On POSIX
674    /// both sources are CLOCK_REALTIME, so nothing short of a port with an
675    /// opinion can tell them apart.
676    #[test]
677    #[cfg(all(feature = "std", feature = "platform-clock"))]
678    fn platform_port_outranks_std_for_the_wall_clock() {
679        let _guard = ClockGuard::acquire();
680        let now = Clock::system().now();
681        assert_eq!(
682            now.sec, 1_000_000_000,
683            "Clock::system() must read the linked port, not SystemTime"
684        );
685        assert_eq!(now.nanosec, 0);
686    }

688    /// ROS time with no `/clock` source is the WALL clock, and it RUNS — issue
689    /// 1334, and the negative control for the whole fix.
690    ///
691    /// Before the fix this arm read `STEADY_TIME_NANOS`, a counter whose own
692    /// doc says it is "advanced by its owner" and which nothing outside this
693    /// file's tests has ever advanced. So the assertion that fails on the
694    /// pre-fix tree is the first one (the reading was 0, not the port's wall
695    /// clock) and the one after it (a clock that does not move is not a
696    /// clock).
697    ///
698    /// `rclcpp`'s `ClockType::ROS_TIME` with `use_sim_time` false reads the
699    /// system clock; five doc sites in this tree across Rust, C and C++ said we
700    /// did too; and the C surface (`nros_clock_get_now(NROS_CLOCK_ROS_TIME)`)
701    /// always did. This is the arm that disagreed with all of them.
702    #[test]
703    #[cfg(feature = "platform-clock")]
704    fn ros_time_with_no_override_reads_the_port_wall_clock() {
705        let _guard = ClockGuard::acquire();
706        Clock::clear_ros_time_override();
707        // A counter reading that is nothing like the wall clock, so reading the
708        // wrong source cannot pass by coincidence.
709        Clock::set_steady_time(1_234);
710
711        let ros = Clock::ros_time().now();
712        assert_eq!(
713            ros,
714            Clock::system().now(),
715            "with no override, ROS time IS system time"
716        );
717        assert_eq!(ros.sec, 1_000_000_000, "it must be the PORT's wall clock");
718
719        // ...and it advances with it. Two seconds on the port's hand.
720        fake_port::WALL_NS.fetch_add(2_000_000_000, core::sync::atomic::Ordering::Relaxed);
721        let later = Clock::ros_time().now();
722        assert_eq!(
723            later.to_nanos() - ros.to_nanos(),
724            2_000_000_000,
725            "ROS time with no source must ADVANCE with the wall clock"
726        );
727    }
