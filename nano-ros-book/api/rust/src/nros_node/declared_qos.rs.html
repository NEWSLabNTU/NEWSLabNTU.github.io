1//! phase-454 W10 — the DECLARED QoS history depth, and the check over it, for
2//! Rust.
3//!
4//! # What this is
5//!
6//! The QoS history DEPTH of a subscription multiplies the executor arena:
7//! `sizeof(entry) + (depth + 1) * bound + (depth + 1) * 8`. A build that sizes
8//! from a declared depth and an image that registers a different one are an
9//! image sized for one number and running another — so the two have to be held
10//! to one. C++ holds them with a `static_assert` inside `NROS_SUBSCRIBE`, and C
11//! with a `_Static_assert` inside `NROS_ASSERT_DECLARED_DEPTH`.
12//!
13//! # Why Rust's half is at REGISTRATION and not at compile time
14//!
15//! Both C surfaces get a compile-time answer because a preprocessor macro sees
16//! the call site's TOPIC as a token. Rust has neither of the two things that
17//! would make the same answer possible:
18//!
19//!   * **No macro seam at the subscribe site.** `nros::main!` emits the boot
20//!     scaffold and knows nothing about subscriptions; there is no
21//!     `nros::subscribe!`. Every call site is an ordinary method call.
22//!   * **The topic is a runtime `&str`**, not a const generic. A `const`
23//!     assertion would need it in the type system.
24//!
25//! And the descriptor road differs too: a RUST component is registered as an
26//! empty cmake INTERFACE library (`NanoRosNodeRegister.cmake`), compiled by the
27//! workspace runtime crate rather than by that target, so there is no C
28//! preprocessor in the lane and no include path to put a header on. That is why
29//! `_nros_declared_qos_arm()` returns early for `RUST` — the header mechanism
30//! is not available to it, not merely unimplemented.
31//!
32//! The facts arrive the way every other derived fact does on the cargo road:
33//! `NROS_ENTITY_DECLARED_DEPTHS` (`type|topic=depth`) reaches
34//! `nros-node/build.rs`, which writes [`crate::config::DECLARED_QOS_ROWS`].
35//! This module reads it, and a disagreement is
36//! [`NodeError::DeclaredDepthMismatch`] — which fails the registration, so the
37//! subscription never exists and the entry's `?` carries it out.
38//!
39//! # ABSENCE IS NOT ZERO
40//!
41//! A `(type, topic)` with no row is "nobody declared this endpoint". Nothing is
42//! checked for it and nothing is defaulted. An image with no contract sidecar
43//! has `DECLARED_QOS_ROWS == None` and behaves exactly as it did before.
44
45use crate::executor::NodeError;
46
47/// Every declared endpoint this image's contract states a depth for, or `None`
48/// when it declares none. `(type name, topic, depth)`; both the ROS and the
49/// DDS-mangled spelling of a type appear, because which one a generated message
50/// class carries is a property of the codegen that produced it.
51pub use crate::config::DECLARED_QOS_ROWS;
52
53/// Do two topic spellings name the same endpoint?
54///
55/// One leading `/` is ignored on each side: a contract states `/chatter` and a
56/// call site may write either `"/chatter"` or the relative `"chatter"` (
57/// `examples/workspaces/launch/src/listener_pkg` writes the second). Nothing
58/// deeper is normalised — a namespace is a different endpoint, and guessing at
59/// one here would make the check match rows nobody meant.
60fn topic_eq(a: &str, b: &str) -> bool {
61    let a = a.strip_prefix('/').unwrap_or(a);
62    let b = b.strip_prefix('/').unwrap_or(b);
63    a == b
64}
65
66/// The declared depth for `(type_name, topic)` in `rows`, or `None`.
67///
68/// Split out from [`declared_depth`] so the lookup is testable against a table
69/// this build did not generate: in-tree, `DECLARED_QOS_ROWS` is `None` (no
70/// contract reaches a unit-test build), so a test that could only call the
71/// public form would assert nothing while passing.
72fn depth_in(rows: &[(&str, &str, u32)], type_name: &str, topic: &str) -> Option<u32> {
73    rows.iter()
74        .find(|(t, tp, _)| *t == type_name && topic_eq(tp, topic))
75        .map(|(_, _, d)| *d)
76}
77
78/// The declared depth for `(type_name, topic)`, or `None` when nobody declared
79/// that endpoint.
80///
81/// Linear over a handful of rows, on a path that runs once per subscription at
82/// registration. `None` when this image declares nothing at all.
83pub fn declared_depth(type_name: &str, topic: &str) -> Option<u32> {
84    depth_in(DECLARED_QOS_ROWS?, type_name, topic)
85}
86
87/// Refuse a subscription whose QoS depth disagrees with what the contract
88/// declared for its topic.
89///
90/// `Ok(())` when they agree AND when nothing was declared — an image that has
91/// not opted in is not an image in error. On a disagreement the TOPIC and BOTH
92/// numbers go to the log first (a `NodeError` is a unit variant and can carry
93/// none of the three), then the registration is refused.
94///
95/// Called at every place a subscription is created from a topic and a QoS.
96/// `check-declared-qos-registration` holds that set to the whole set.
97pub fn check(type_name: &str, topic: &str, depth: u32) -> Result<(), NodeError> {
98    match declared_depth(type_name, topic) {
99        Some(declared) if declared != depth => {
100            nros_log::log_error!(
101                nros_log::get_logger("nros_node"),
102                "nros: subscription on topic `{}` (type `{}`) is being registered with QoS \
103                 depth {}, but this system's contract sidecar DECLARES depth {} for that \
104                 topic. Depth multiplies the executor arena, which was sized from the \
105                 declared number -- so the image is sized for {} and would run {}. Fix \
106                 whichever is wrong: the contract row \
107                 (<bringup>/launch/<stem>.contract.yaml, contracts.sub_endpoints.<ep>.qos) \
108                 or the QoS at the call site.",
109                topic,
110                type_name,
111                depth,
112                declared,
113                declared,
114                depth
115            );
116            Err(NodeError::DeclaredDepthMismatch)
117        }
118        _ => Ok(()),
119    }
120}

122#[cfg(test)]
123mod tests {
124    use super::*;
125
126    /// The shape `build.rs` writes: the ROS spelling and the DDS-mangled one,
127    /// for each declared endpoint.
128    const ROWS: &[(&str, &str, u32)] = &[
129        ("std_msgs/msg/Int32", "/chatter", 1),
130        ("std_msgs::msg::dds_::Int32_", "/chatter", 1),
131    ];
132
133    #[test]
134    fn either_type_spelling_finds_the_row() {
135        assert_eq!(depth_in(ROWS, "std_msgs/msg/Int32", "/chatter"), Some(1));
136        assert_eq!(
137            depth_in(ROWS, "std_msgs::msg::dds_::Int32_", "/chatter"),
138            Some(1),
139            "a generated Rust message class reports the DDS-mangled TYPE_NAME; a table that \
140             carried only the ROS spelling would match nothing and check nothing"
141        );
142    }
143
144    #[test]
145    fn a_relative_topic_matches_the_declared_absolute_one() {
146        assert_eq!(depth_in(ROWS, "std_msgs/msg/Int32", "chatter"), Some(1));
147    }
148
149    #[test]
150    fn absence_is_not_a_depth() {
151        assert_eq!(
152            depth_in(ROWS, "std_msgs/msg/Int32", "/undeclared"),
153            None,
154            "an endpoint nobody declared has no depth -- not 0, and not the ROS default of 10"
155        );
156        assert_eq!(
157            depth_in(ROWS, "std_msgs/msg/Bool", "/chatter"),
158            None,
159            "the table is keyed on the PAIR: a declared topic under another type is not a hit, \
160             or one declaration would size every type carried on that topic"
161        );
162        assert_eq!(depth_in(&[], "std_msgs/msg/Int32", "/chatter"), None);
163    }
164
165    #[test]
166    fn a_namespace_is_a_different_endpoint() {
167        assert_eq!(
168            depth_in(ROWS, "std_msgs/msg/Int32", "/ns/chatter"),
169            None,
170            "only ONE leading slash is ignored; matching a namespaced topic against a bare \
171             declaration would check a row nobody meant"
172        );
173    }
174
175    /// The public entry point over THIS build's table. In-tree that table is
176    /// `None` (no contract reaches a unit-test build), so this asserts the
177    /// no-declaration behaviour rather than a lookup -- and says so, because a
178    /// reader would otherwise take it for coverage of the lookup above.
179    #[cfg(not(nros_declared_qos_table))]
180    #[test]
181    fn an_image_that_declares_nothing_checks_nothing() {
182        assert!(
183            DECLARED_QOS_ROWS.is_none(),
184            "`nros_declared_qos_table` is unset, so the build wrote no table. If this fires, \
185             the cfg and the const have come apart and the declared-image test below is \
186             being skipped on a build that HAS a table."
187        );
188        assert_eq!(declared_depth("std_msgs/msg/Int32", "/chatter"), None);
189        assert_eq!(check("std_msgs/msg/Int32", "/chatter", 10), Ok(()));
190    }
191
192    /// THE NEGATIVE CONTROL, over the build's OWN table.
193    ///
194    /// Everything above tests `depth_in` against a table this module wrote,
195    /// which proves the search and nothing about DELIVERY: a carrier cmake
196    /// never forwards, a `build.rs` parse that drops every row, and a table
197    /// keyed on a spelling no call site writes all look exactly like "no
198    /// mismatch here". Six mechanisms in this campaign have been correct and
199    /// unreachable for precisely that reason.
200    ///
201    /// So this module is compiled only where `build.rs` wrote rows, and
202    /// `just check declared-qos-registration` builds this crate with
203    /// `NROS_ENTITY_DECLARED_DEPTHS` set to make that happen. It also asserts
204    /// that the test does NOT exist without the variable -- otherwise a cfg
205    /// that was always on would make the lane pass while proving nothing.
206    #[cfg(nros_declared_qos_table)]
207    mod declared_image {
208        use super::*;
209
210        #[test]
211        fn refuses_a_disagreeing_depth_and_accepts_an_agreeing_one() {
212            let rows = DECLARED_QOS_ROWS.expect("the cfg says this build wrote a table");
213            assert!(
214                !rows.is_empty(),
215                "an empty table with the cfg set is a parse that dropped every row -- which \
216                 would disable the check while reading as if it were on"
217            );
218            assert_eq!(
219                declared_depth("std_msgs/msg/Int32", "/chatter"),
220                Some(1),
221                "the lane forwards `std_msgs/msg/Int32|/chatter=1`; if this is None the \
222                 carrier, the parse or the key spelling is wrong, and every registration in \
223                 a real image would go unchecked"
224            );
225            assert_eq!(
226                declared_depth("std_msgs::msg::dds_::Int32_", "/chatter"),
227                Some(1),
228                "build.rs must emit the DDS-mangled spelling too -- it is what a generated \
229                 Rust message class reports as TYPE_NAME, and the only spelling a typed call \
230                 site ever hands `check`"
231            );
232            assert_eq!(check("std_msgs/msg/Int32", "/chatter", 1), Ok(()));
233            assert_eq!(
234                check("std_msgs::msg::dds_::Int32_", "/chatter", 10),
235                Err(NodeError::DeclaredDepthMismatch),
236                "THE control: declared 1, registered 10. If this passes, the registration \
237                 check does nothing on a real image and every lane above it stays green."
238            );
239            assert_eq!(
240                check("std_msgs/msg/Int32", "/undeclared", 10),
241                Ok(()),
242                "and an endpoint nobody declared is still not an error"
243            );
244        }
245    }
