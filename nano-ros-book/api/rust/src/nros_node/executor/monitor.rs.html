1//! RFC-0052 / phase-296 W3b.4/.5 — on-target contract monitors.
2//!
3//! The baked shape mirrors Phase 211.H's `qos_overrides`: codegen emits a
4//! `&'static [MonitorSpec]` table (plus one `static PubMonitorCell` per
5//! contracted publisher) from the SystemModel's contract layer; the entry
6//! installs it on the executor before entity creation. An uncontracted
7//! image bakes an empty table — every path below dead-code-eliminates.
8//!
9//! Publish counting is an atomic bump on the publisher handle (no clock,
10//! no lock on the hot path); the rate check runs on spin ticks over a
11//! ~[`RATE_CHECK_INTERVAL_US`] window and pushes violations into a small
12//! ring the entry glue drains into the `nros-diagnostics` reporter.
13//!
14//! W3b.5 adds three more rules on the same drain:
15//! - `max-age-runtime` — subscriber take-age (`epoch_now - header.stamp`
16//!   peeked from the raw CDR buffer at [`RosMessage::STAMP_OFFSET`],
17//!   recorded into a [`SubMonitorCell`] on the take path).
18//! - `max-latency-runtime` — node-path (take → publish) latency: the
19//!   dispatch elapsed time is attributed to every monitored publisher
20//!   whose counter advanced during that dispatch (an upper bound on
21//!   take → publish, measured on the executor's monotonic clock).
22//! - `deadline-miss-runtime` — a dispatched callback ran past its bound
23//!   SchedContext's `deadline_us`; what ELSE happens is the tier's
24//!   [`DeadlineAction`](super::sched_context::DeadlineAction).
25
26use core::sync::atomic::Ordering;
27// portable-atomic: RMW ops (fetch_add/fetch_max/swap) exist even on
28// riscv32imc / Cortex-M0+ that lack native CAS (same choice as
29// `SporadicState` / `AtomicSporadicState` in sched_context.rs).
30use portable_atomic::AtomicU32;
31
32/// One contracted publisher's counters. Baked as a `static` by codegen
33/// (or declared by the fixture); the publisher handle bumps `count` on
34/// every publish, the executor reads deltas on spin ticks.
35#[derive(Debug, Default)]
36pub struct PubMonitorCell {
37    pub count: AtomicU32,
38    /// W3b.5 — max observed take→publish latency (µs) in the current
39    /// check window. Written by the dispatch loop (fetch_max), drained
40    /// (swap 0) by the latency check.
41    pub max_latency_us: AtomicU32,
42}
43
44impl PubMonitorCell {
45    pub const fn new() -> Self {
46        Self {
47            count: AtomicU32::new(0),
48            max_latency_us: AtomicU32::new(0),
49        }
50    }
51}

53/// One contracted subscriber's take-age accumulator (W3b.5). The take
54/// path records `epoch_now - header.stamp` per message (fetch_max); the
55/// age check drains it (swap 0) per window.
56#[derive(Debug, Default)]
57pub struct SubMonitorCell {
58    /// Max observed take-age (ms) in the current check window.
59    pub max_age_ms: AtomicU32,
60}
61
62impl SubMonitorCell {
63    pub const fn new() -> Self {
64        Self {
65            max_age_ms: AtomicU32::new(0),
66        }
67    }
68
69    /// Take-path hook: record one message's age. `stamp_us` is the
70    /// peeked `header.stamp` as µs since the UNIX epoch, `epoch_now_us`
71    /// the receive-side wall clock. A stamp from the future clamps to 0.
72    pub fn observe(&self, stamp_us: u64, epoch_now_us: u64) {
73        let age_ms = (epoch_now_us.saturating_sub(stamp_us) / 1_000).min(u32::MAX as u64) as u32;
74        self.max_age_ms.fetch_max(age_ms, Ordering::Relaxed);
75    }
76}
77
78/// Peek `Time { i32 sec; u32 nanosec }` little-endian at `offset` in a
79/// raw CDR receive buffer (encapsulation header included) and return µs
80/// since the UNIX epoch. `None` when the buffer is too short or the
81/// stamp is pre-epoch/zero (unstamped messages never fire age monitors).
82pub fn peek_stamp_us(raw: &[u8], offset: usize) -> Option<u64> {
83    let sec_b = raw.get(offset..offset + 4)?;
84    let nsec_b = raw.get(offset + 4..offset + 8)?;
85    let sec = i32::from_le_bytes([sec_b[0], sec_b[1], sec_b[2], sec_b[3]]);
86    let nsec = u32::from_le_bytes([nsec_b[0], nsec_b[1], nsec_b[2], nsec_b[3]]);
87    if sec <= 0 {
88        return None;
89    }
90    Some(sec as u64 * 1_000_000 + nsec as u64 / 1_000)
91}

93/// One monitored publisher endpoint.
94#[derive(Debug, Clone, Copy)]
95pub struct MonitorSpec {
96    /// Topic name EXACTLY as the node passes it to `create_publisher`
97    /// (the SystemModel's wiring carries the same resolved name).
98    pub topic: &'static str,
99    /// Endpoint ref for violation reports (`<node FQN>/<endpoint>` — the
100    /// SystemModel contract key).
101    pub fqn: &'static str,
102    /// Declared publisher guarantee, in milli-Hz (fixed point: Hz × 1000).
103    /// 0 = no rate contract on this endpoint.
104    pub min_rate_hz_milli: u32,
105    /// W3b.5 — node-path budget (ms) for paths whose OUTPUT is this
106    /// endpoint (`contracts.node_paths[..].max_latency_ms`). 0 = no
107    /// latency contract.
108    pub max_latency_ms: u32,
109    /// The endpoint's counter cell.
110    pub cell: &'static PubMonitorCell,
111}

113/// One monitored subscriber endpoint (W3b.5 age contracts). Separate
114/// table from [`MonitorSpec`] — sub contracts key different endpoints

120    /// Endpoint ref for violation reports (the SystemModel contract key).
121    pub fqn: &'static str,
122    /// Declared max take-age (ms). 0 = no age contract.
123    pub max_age_ms: u32,
124    /// The endpoint's age accumulator.
125    pub cell: &'static SubMonitorCell,
126}

128/// Rate-check window (µs). Matches play_launch's ~5 s time-based trigger
129/// so both runtimes converge on comparable cadence.
130pub const RATE_CHECK_INTERVAL_US: u64 = 5_000_000;
131
132/// Max monitored endpoints per executor (const table, no_std).
133pub const MAX_MONITORS: usize = 8;
134/// Violation ring depth.
135pub const MAX_VIOLATIONS: usize = 8;

137/// A detected contract violation, in the play_launch rule-id vocabulary.
138#[derive(Debug, Clone)]
139pub struct Violation {
140    /// `"rate-hierarchy-runtime"` | `"max-age-runtime"` |
141    /// `"max-latency-runtime"` | `"deadline-miss-runtime"` |
142    /// `"timer-overrun-runtime"`.
143    pub rule: &'static str,
144    /// Violating endpoint ref (from the spec's `fqn`; the SC name for
145    /// deadline misses).
146    pub fqn: &'static str,
147    /// Measured value. Unit is per-rule: milli-Hz for the rate rule, ms
148    /// for age/latency, µs for deadline misses, dropped activations for
149    /// the timer-overrun rule.
150    pub measured: u32,
151    /// Declared bound, same unit as `measured`.
152    pub declared: u32,
153}

155/// Per-spec accounting state (parallel to the spec table).
156#[derive(Debug, Clone, Copy, Default)]
157pub(crate) struct MonitorState {
158    /// Window opened (a plain bool, not a 0-sentinel on the timestamp —
159    /// `now_us == 0` is a legitimate first sample on freshly-started
160    /// monotonic clocks).
161    pub(crate) opened: bool,
162    pub(crate) window_start_us: u64,
163    pub(crate) count_at_window_start: u32,
164    /// Suppress duplicate reports: only re-report after a clean window.
165    pub(crate) violated_last_window: bool,
166    /// W3b.5 — separate dedup for the latency rule on the same spec row.
167    pub(crate) latency_violated_last_window: bool,
168}
169
170/// Pure rate check over one window boundary. Returns `Some(violation)`
171/// when the window elapsed AND the measured rate is below the declared
172/// minimum (and we didn't already report last window).
173///
174/// Extracted from the executor so the math is unit-testable without a
175/// session: publish counting is injected via the cell, time via `now_us`.
176pub(crate) fn check_rate(
177    spec: &MonitorSpec,
178    state: &mut MonitorState,
179    now_us: u64,
180) -> Option<Violation> {
181    if spec.min_rate_hz_milli == 0 {
182        return None;
183    }
184    let count = spec.cell.count.load(Ordering::Relaxed);
185    if !state.opened {
186        // First observation: open the window, no verdict yet.
187        state.opened = true;
188        state.window_start_us = now_us;
189        state.count_at_window_start = count;
190        return None;
191    }
192    let window_us = now_us.saturating_sub(state.window_start_us);
193    if window_us < RATE_CHECK_INTERVAL_US {
194        return None;
195    }
196    let published = count.wrapping_sub(state.count_at_window_start) as u64;
197    // milli-Hz = published * 1e3 / window_s = published * 1e9 / window_us
198    let measured_milli_hz =
199        (published.saturating_mul(1_000_000_000) / window_us.max(1)).min(u32::MAX as u64) as u32;
200
201    // Roll the window.
202    state.window_start_us = now_us;
203    state.count_at_window_start = count;
204
205    if measured_milli_hz < spec.min_rate_hz_milli {
206        if state.violated_last_window {
207            return None; // still violated — already reported
208        }
209        state.violated_last_window = true;
210        Some(Violation {
211            rule: "rate-hierarchy-runtime",
212            fqn: spec.fqn,
213            measured: measured_milli_hz,
214            declared: spec.min_rate_hz_milli,
215        })
216    } else {
217        state.violated_last_window = false;
218        None
219    }
220}

222/// Pure latency check: drains the spec cell's window-max take→publish
223/// latency and fires when it exceeds the declared node-path budget.
224/// Same report-once-until-recovery semantics as the rate rule; runs on
225/// every monitor tick (the cell accumulates between ticks, so no window
226/// bookkeeping is needed — draining IS the window roll).
227pub(crate) fn check_latency(spec: &MonitorSpec, state: &mut MonitorState) -> Option<Violation> {
228    if spec.max_latency_ms == 0 {
229        return None;
230    }
231    let max_us = spec.cell.max_latency_us.swap(0, Ordering::Relaxed);
232    let max_ms = max_us / 1_000;
233    if max_ms > spec.max_latency_ms {
234        if state.latency_violated_last_window {
235            return None;
236        }
237        state.latency_violated_last_window = true;
238        Some(Violation {
239            rule: "max-latency-runtime",
240            fqn: spec.fqn,
241            measured: max_ms,
242            declared: spec.max_latency_ms,
243        })
244    } else {
245        // A quiet window (no dispatch attributed) also counts as clean —
246        // recovery resets the dedup like the rate rule's clean window.
247        state.latency_violated_last_window = false;
248        None
249    }
250}

258/// Pure age check: drains the sub cell's window-max take-age and fires
259/// when it exceeds the declared bound. Report-once-until-recovery.
260pub(crate) fn check_age(spec: &AgeMonitorSpec, state: &mut AgeState) -> Option<Violation> {
261    if spec.max_age_ms == 0 {
262        return None;
263    }

267            return None;

271            rule: "max-age-runtime",

273            measured: max_ms,
274            declared: spec.max_age_ms,
275        })
276    } else {
277        state.violated_last_window = false;
278        None
279    }

299    fn slow_publisher_fires_once_until_recovery() {
300        CELL.count.store(0, Ordering::Relaxed);
301        let s = spec(100_000); // 100 Hz declared
302        let mut st = MonitorState::default();
303
304        // t=0: opens window.
305        assert!(check_rate(&s, &mut st, 0).is_none());
306        // 5 publishes in 5 s = 1 Hz — violation.
307        CELL.count.store(5, Ordering::Relaxed);

309        assert_eq!(v.rule, "rate-hierarchy-runtime");
310        assert_eq!(v.fqn, "/demo/talker/chatter");
311        assert_eq!(v.measured, 1_000);
312        assert_eq!(v.declared, 100_000);
313        // Still slow next window — suppressed (no re-report spam).
314        CELL.count.store(10, Ordering::Relaxed);
315        assert!(check_rate(&s, &mut st, 2 * RATE_CHECK_INTERVAL_US).is_none());
316        // Recovers (500 publishes in 5 s = 100 Hz) — clean window resets.
317        CELL.count.store(510, Ordering::Relaxed);
318        assert!(check_rate(&s, &mut st, 3 * RATE_CHECK_INTERVAL_US).is_none());
319        // Degrades again — fires again.
320        CELL.count.store(511, Ordering::Relaxed);

426/// Issue #514 — emit one violation to the log.
427///
428/// A log line is deliberately the floor rather than a `/diagnostics`
429/// publication: it needs no publisher, no topic wiring, and no contract
430/// on the reporting path itself, so it works on a bare RTOS image and
431/// during boot. Publishing the same verdicts as `DiagnosticArray`
432/// belongs on top of this, not instead of it.
433///
434/// Free function rather than an `Executor` method because every call
435/// site sits inside a loop that already borrows the executor's spec
436/// tables.
437pub(crate) fn log_violation(v: &Violation) {
438    nros_log::nros_warn!(
439        nros_log::get_logger("nros"),
440        "contract violation: {} {} measured={} declared={}",
441        v.rule,
442        v.fqn,
443        v.measured,
444        v.declared
445    );
446}

448/// Issue #505 — periodic activations a timer dropped because its
449/// executor was blocked past the period boundary.
450///
451/// This rule exists because `check_rate` cannot see an isolated stall:
452/// it samples publish counts over a ~5 s window, so 20 missed
453/// activations of a 100 Hz loop are a 0.4% rate deficit — under any
454/// sane declared minimum, silence. (And under
455/// [`TimerOverrunPolicy::CatchUp`](super::arena::TimerOverrunPolicy)
456/// the replayed activations refill the window entirely, so the rate
457/// rule reports a HEALTHY loop while the tier is stalling.) The
458/// overrun counter is exact, needs no window, and does not depend on
459/// clock resolution.
460///
461/// `overruns` is the timer's monotonic saturating counter;
462/// `last_reported` is the value at the previous check, so the verdict
463/// is on the delta. Returns a violation when more than `tolerated`
464/// activations were dropped since the last check.
465pub(crate) fn check_timer_overrun(
466    overruns: u32,
467    last_reported: &mut u32,
468    tolerated: u32,
469) -> Option<Violation> {
470    let dropped = overruns.saturating_sub(*last_reported);
471    *last_reported = overruns;
472    if dropped <= tolerated {
473        return None;
474    }
475    Some(Violation {
476        rule: "timer-overrun-runtime",
477        // Timer entries carry no name at this altitude; same stand-in
478        // as `deadline-miss-runtime`.
479        fqn: "timer",
480        measured: dropped,
481        declared: tolerated,
482    })
483}
