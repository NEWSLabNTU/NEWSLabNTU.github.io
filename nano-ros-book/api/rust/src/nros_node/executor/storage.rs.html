1//! phase-271 / phase-409 — per-entry [`Executor`](super::spin::Executor)
2//! storage (issues 0110, 0563, 0936).
3//!
4//! The executor's sized tables (callback table + arena + scheduling-context
5//! tables + the node/session/dispatch tables) used to be inline fields sized by
6//! build-time consts baked into `nros-node` — one size for every entry sharing a
7//! compiled `nros-node`, and, worse, a size that every function MOVING an
8//! `Executor` had to make room for on its stack. Here the ENTRY supplies its own
9//! storage, sized to its topology, so a fat native entry and a lean embedded
10//! entry in one workspace each get the right size with no workspace-global env,
11//! and the `Executor` VALUE stays a small fixed header.
12//!
13//! Per the "C/C++ is a thin wrapper of Rust" principle the PUBLIC API stays
14//! generic-free: the entry hands a raw, 8-aligned `&mut [MaybeUninit<u64>]` backing
15//! (sized via [`executor_storage_u64_len`]); `nros-node` carves it privately into
16//! the typed sub-slices ([`carve`]). The only `unsafe` is that carve, validated
17//! against the `#[repr(C)]` reference [`ExecutorStorage`] layout by unit test.

55/// The node half of an [`ActiveGroup`] key: a hash of `(name, namespace)`.
56///
57/// FNV-1a, 64-bit, over `name` then `\0` then the normalised namespace. The
58/// separator matters: without it `("ab", "c")` and `("a", "bc")` hash alike,
59/// and a node name may legally contain any of the namespace's characters.
60///
61/// FNV because the tree already depends on it (heapless's `FnvIndexMap`), it
62/// is `const`-friendly and allocation-free, and this runs on the registration
63/// path of every entity on an MCU. A 64-bit collision across the handful of
64/// nodes in one image is not a risk worth a bigger key: at 32 entries the
65/// probability is around 3e-17.
66///
67/// ONE home, deliberately. Both ends of the comparison call this —
68/// `set_active_groups` hashing what the tier declared, and `create_entity`
69/// hashing the node an entity belongs to. Two hashers that agree by
70/// inspection is exactly the defect class issue 1172 is filed under.
71pub(crate) fn node_identity_hash(name: &str, namespace: &str) -> u64 {
72    const OFFSET: u64 = 0xcbf2_9ce4_8422_2325;
73    const PRIME: u64 = 0x0000_0100_0000_01b3;
74    let norm_ns = if namespace.is_empty() { "/" } else { namespace };
75    let mut h = OFFSET;
76    for b in name
77        .as_bytes()
78        .iter()
79        .chain(core::iter::once(&0u8))
80        .chain(norm_ns.as_bytes())
81    {
82        h ^= *b as u64;
83        h = h.wrapping_mul(PRIME);
84    }
85    h
86}

88/// One entry of a tier's active-group filter: `(node identity hash, group)`.
89///
90/// issue 1172 — the filter used to be the GROUP NAME ALONE, which cannot tell
91/// node A's `ctrl` from node B's `ctrl`. Two nodes may legitimately declare the
92/// same group id on different tiers and nothing rejects it, so a tier filter
93/// was admitting entities it did not own. The sibling table for the
94/// sched-context path (`group_sched_table`, `bind_group_sched`) has been
95/// node-qualified all along; only this one was not.
96///
97/// The node half is a HASH of `(name, namespace)` rather than the strings.
98/// Storing them would cost 184 B per entry against this table's 48 B, and the
99/// table is CARVED one entry per node slot — +144 B per slot, which is 4.6 KB
100/// of static executor arena at 32 slots. The group name stays readable because
101/// a refused filter should be able to say WHICH group it refused.
102///
103/// The hash has exactly one home, [`node_identity_hash`], and both ends of the
104/// comparison call it — codegen never hashes anything, it emits plain strings.
105/// A second hasher would be this phase's own defect in a new place.
106pub(crate) type ActiveGroup = (u64, GroupName);

109// CarvedVec — a `heapless::Vec` whose capacity lives in the caller's backing

112/// A push-only vector over CARVED storage: the elements live in the caller's
113/// backing, the `Executor` holds only a fat pointer and a fill cursor.
114///
115/// phase-409 (issue 0961) — this is the shape phase-271 gave the six sized
116/// tables, generalised so the other nine can use it too. It exists because
117/// `heapless::Vec<T, N>` puts `N * size_of::<T>()` bytes INSIDE the struct, so
118/// every knob that sets an `N` also sets the stack frame of every function that
119/// moves an `Executor`: `MAX_CBS` 14 -> 36 (a handle-count fix) added ~3.7 KiB
120/// to `Executor::open_in`'s prologue on a part whose main thread has 32 KiB in
121/// total.
122///
123/// Deliberately `MaybeUninit<T>` rather than `Option<T>`, for three reasons:
124/// [`as_slice`](Self::as_slice) can hand back the `&[T]` the public
125/// `Executor::nodes()` accessor already returns; `carve` writes NOTHING at open
126/// (the `memclr` in issue 0961's fault was the executor's tables being zeroed);
127/// and the layout matches `heapless::Vec`'s, so the storage cost is the same
128/// bytes in a different place.
129///
130/// Occupied slots are `[0, len)` and IN PUSH ORDER — several callers index by
131/// insertion position (`NodeId` IS an index into `nodes`; `session_idx` is an
132/// index into `extra_sessions`), so this must never gain a `swap_remove`.
133pub(crate) struct CarvedVec<'s, T> {
134    slots: &'s mut [MaybeUninit<T>],
135    len: usize,
136}
137
138impl<'s, T> CarvedVec<'s, T> {
139    /// Wrap carved, UNINITIALISED storage as an empty vector.
140    fn new(slots: &'s mut [MaybeUninit<T>]) -> Self {
141        Self { slots, len: 0 }
142    }
143
144    /// Slots this vector can hold — the entry's sizing, not a build-time const.
145    pub(crate) fn capacity(&self) -> usize {
146        self.slots.len()
147    }
148
149    /// Append `value`, returning it in `Err` when full (`heapless::Vec::push`).
150    pub(crate) fn push(&mut self, value: T) -> Result<(), T> {
151        if self.len == self.capacity() {
152            return Err(value);
153        }
154        self.slots[self.len].write(value);
155        self.len += 1;
156        Ok(())
157    }
158
159    /// Drop every element and reset the cursor.
160    pub(crate) fn clear(&mut self) {
161        let len = core::mem::replace(&mut self.len, 0);
162        if needs_drop::<T>() {
163            for slot in &mut self.slots[..len] {
164                // SAFETY: `[0, len)` was initialised by `push` and is dropped
165                // exactly once — `len` is reset above before this loop runs, so
166                // a panicking `T::drop` cannot make a second pass see it.
167                unsafe { slot.assume_init_drop() };
168            }
169        }
170    }
171
172    /// The initialised prefix.
173    pub(crate) fn as_slice(&self) -> &[T] {
174        // SAFETY: `push` initialises `[0, len)` and nothing shortens `len`
175        // without dropping (see `clear`).
176        unsafe { &*(&self.slots[..self.len] as *const [MaybeUninit<T>] as *const [T]) }
177    }
178
179    /// The initialised prefix, mutably.
180    pub(crate) fn as_mut_slice(&mut self) -> &mut [T] {
181        // SAFETY: as [`as_slice`](Self::as_slice).
182        unsafe { &mut *(&mut self.slots[..self.len] as *mut [MaybeUninit<T>] as *mut [T]) }
183    }
184}
185
186impl<T> Deref for CarvedVec<'_, T> {
187    type Target = [T];
188    fn deref(&self) -> &[T] {
189        self.as_slice()
190    }
191}
192
193impl<T> DerefMut for CarvedVec<'_, T> {
194    fn deref_mut(&mut self) -> &mut [T] {
195        self.as_mut_slice()
196    }
197}

199impl<T> Drop for CarvedVec<'_, T> {
200    fn drop(&mut self) {
201        // The backing is the CALLER's memory, so nothing drops these elements
202        // for us — and one of these tables is `extra_sessions`, whose elements
203        // close RMW sessions. A `Drop` impl on the vector (rather than a pass in
204        // `Executor::drop`) keeps the ORDER the inline `heapless::Vec`s had:
205        // `Executor`'s fields drop in declaration order, so the primary
206        // `session` still closes before the extras.
207        self.clear();
208    }
209}

347/// issue 1197 — publish the backing's TOTAL size as a symbol whose storage size
348/// IS the number, so a build script can recover it from the rlib without running
349/// anything. `nros-sizes-build::extract_sizes` reads exactly this shape, the way
350/// `nros-c` recovers `EXECUTOR_OPAQUE_U64S`.
351///
352/// **The total rather than the per-region units, and that is a decision.** The
353/// units alone would be more robust — they do not vary with the executor knobs,
354/// so a probe cache keyed by (target, features) is correct for them, and the
355/// caller supplies `alloc` and the counts explicitly. But the backing is
356/// `arena + tables`, and `ARENA_SIZE` is DERIVED in this crate's build script
357/// from `MAX_CBS`, `RX_BUF_SIZE` and `ACTION_CLIENTS`. A consumer computing from
358/// units would have to re-implement that derivation — a second implementation,
359/// which is the whole thing this design refuses. So the consumer probes the one
360/// number, and keys its probe directory by the executor env, because unlike the
361/// units this DOES vary with it.
362///
363/// Behind a feature and OFF by default: it is `#[used]`, and a shipped image
364/// should not carry a static that exists only to be measured. The probe enables
365/// it for its own nested build, which is a separate compilation from the linked
366/// one.
367#[cfg(feature = "layout-size-markers")]
368#[used]
369#[unsafe(no_mangle)]
370#[doc(hidden)]
371pub static __NROS_LU_SZ_executor_backing: [u8;
372    crate::executor::backing::EXECUTOR_BACKING_DEFAULT_U64S * 8] =
373    [0u8; crate::executor::backing::EXECUTOR_BACKING_DEFAULT_U64S * 8];
374
375const fn unit_of<T>() -> RegionUnit {
376    RegionUnit {
377        size: size_of::<T>(),
378        align: align_of::<T>(),
379    }
380}

428/// Byte [`Layout`] of the backing, evaluated against a SUPPLIED unit table.
429///
430/// issue 1197 — the public half of the split. `executor_storage_layout` is this
431/// with [`NATIVE_UNITS`]; an external consumer that has recovered the
432/// units for another target (a build script probing an rlib, the way `nros-c`
433/// recovers `EXECUTOR_OPAQUE_U64S`) gets the same arithmetic rather than a
434/// second implementation of it.
435///
436/// The OFFSETS stay private: only `carve` needs them, and it runs in the crate
437/// that owns the types. A consumer sizing a reservation needs the total.
438pub const fn executor_storage_layout_with(sizing: ExecutorSizing, units: RegionUnits) -> Layout {
439    let o = compute_offsets_with(sizing, units);
440    // SAFETY: as `executor_storage_layout` — `align` is a max of alignments and
441    // `size` is rounded up to it.
442    unsafe { Layout::from_size_align_unchecked(o.size, o.align) }
443}
444
445/// Byte [`Layout`] of the backing a `sizing`-sized executor needs.
446/// Public + non-generic so the macro / FFI can size a raw backing.
447pub const fn executor_storage_layout(sizing: ExecutorSizing) -> Layout {
448    let o = compute_offsets(sizing);
449    // SAFETY: `align` is a power of two (a `max` of `align_of` results) and `size`
450    // is rounded up to it; both are non-zero.
451    unsafe { Layout::from_size_align_unchecked(o.size, o.align) }
452}

454/// Number of `u64` words a backing must hold for a `sizing`-sized executor.
455/// `u64` backing is 8-aligned, which covers every field (all `align_of ≤ 8`;
456/// asserted in tests), so the entry never hand-aligns. The macro emits
457/// `[MaybeUninit<u64>; executor_storage_u64_len(sizing)]`.
458pub const fn executor_storage_u64_len(sizing: ExecutorSizing) -> usize {
459    executor_storage_layout(sizing).size().div_ceil(8)
460}

462/// Per-entry executor sizing — the entity counts an [`Executor`](super::spin::Executor)
463/// is built to hold. **Public + non-generic** (the "C/C++ is a thin wrapper"
464/// principle): the entry / macro / FFI supplies these as plain `usize`s rather
465/// than as type/const generics C can't name. Used to size + carve the backing.
466///
467/// `cbs` is capped at 64 by the executor's `u64` ready-set bitmask (asserted in
468/// [`carve`]-time / `open_in`).
469#[derive(Clone, Copy)]
470pub struct ExecutorSizing {
471    /// Callback-table slots (`entries`, the per-entry SC bindings, and the
472    /// per-callback-group sched table). ≤ 64.
473    pub cbs: usize,
474    /// Scheduling-context slots (`sched_contexts`, the sporadic state tables
475    /// and alive supervision's per-SC records).
476    pub sc: usize,
477    /// Bump-allocator arena size in bytes.
478    pub arena: usize,
479    /// phase-409 — Node slots. One worst-case extra session, extra-session id,
480    /// node-sched binding, dispatch slot, component slot and callback-group
481    /// filter entry per Node, so ONE count covers all seven tables (that is the
482    /// upper bound each of them already assumed under `MAX_NODES`).
483    pub nodes: usize,
484}

503    /// phase-412 #4 — refuse, at COMPILE time, a sizing whose arena is below
504    /// what this image's declared entities are modelled to need.
505    ///
506    /// For a caller that supplies its own `const` sizing for the image's ONE
507    /// executor (`open_in` over a `static` sized with [`u64_len`](Self::u64_len)),
508    /// so its arena is held to the same model the build-time default is:
509    ///

525/// The exact `#[repr(C)]` byte layout the C/C++ FFI's inline executor buffer must
526/// hold: an [`Executor`](super::spin::Executor)`<'static>` header immediately
527/// followed by a default-sized ([`ExecutorSizing::DEFAULT`]) storage backing.
528///
529/// The FFI keeps the executor inline (heap-free — matching the Rust no-alloc
530/// requirement) and carves its per-entry tables from the SAME buffer's
531/// [`backing`](Self::backing) tail. Because that buffer is **pinned** — the C
532/// caller allocates it, it is initialised in place, and it is only ever reached
533/// through a stable `nros_executor_t*` (never moved after init) — the resulting
534/// self-borrow (the header's slices pointing into the same struct's tail) is
535/// sound. The FFI probes `size_of` of THIS type (not bare `Executor`) to size
536/// its `_opaque` array, and reinterprets `_opaque` as `*mut ExecutorInlineStorage`
537/// (the executor stays at offset 0, so existing offset-0 accessors are unchanged).
538///
539/// phase-409 — moving the last nine tables into the backing moves bytes from
540/// `exec` to `backing` and leaves this total roughly where it was; what changes
541/// is the STACK, because the header is what `open_in` builds and returns by
542/// value.
543#[repr(C)]
544pub struct ExecutorInlineStorage {
545    /// The executor, written in place (offset 0) by `from_session_ptr_in`.
546    pub exec: MaybeUninit<super::spin::Executor<'static>>,
547    /// The carved backing the executor's slices borrow (the buffer's tail).
548    pub backing: [MaybeUninit<u64>; ExecutorSizing::DEFAULT.u64_len()],
549}
550
551/// Carve an 8-aligned `u64` backing into the typed executor slices.
552///
553/// # Safety
554/// - `backing.len() * 8` must be ≥ `executor_storage_layout(sizing).size()`.
555/// - The returned slices alias `backing`; it must outlive them (the `'s` bound)
556///   and not be otherwise accessed while they live.
557///
558/// The `Option`/binding tables are initialised here (`entries`/SC tables →
559/// `None`, bindings → `SchedContextId(0)`), so the returned `&mut [T]` reference
560/// validly-init memory. The [`CarvedVec`] tables are handed over empty and
561/// UNINITIALISED — nothing reads past their fill cursor.
562pub(crate) unsafe fn carve<'s>(
563    backing: &'s mut [MaybeUninit<u64>],
564    sizing: ExecutorSizing,
565) -> ExecutorSlices<'s> {
566    let ExecutorSizing {
567        cbs,
568        sc,
569        arena,
570        nodes: node_slots,
571    } = sizing;
572    let o = compute_offsets(sizing);
573    // Fail-loud on EVERY profile (not `debug_assert!`): embedded release builds
574    // strip debug-assertions, and a backing that is too small is silent memory
575    // corruption — the carved `entries`/`sched_contexts` tables run past the end
576    // of `backing` into whatever .bss follows (e.g. a C carrier's `__nros_c_inst`),
577    // leaving a NULL `drop_fn` that faults in `Executor::drop`. This is exactly
578    // how a STALE config-header mirror (C buffer sized from an out-of-date
579    // `NROS_*_STORAGE_SIZE`) manifested as a `jalr -> 0` on threadx-riscv64 (#131).
580    // Panic here instead, at open, with the two sizes named.
581    assert!(
582        backing.len() * 8 >= o.size,
583        "executor backing too small: {} bytes < {} required — the storage buffer \
584         (NROS_*_STORAGE_SIZE) disagrees with the executor layout; rebuild clean so \
585         the generated config header matches",
586        backing.len() * 8,
587        o.size
588    );
589    let base = backing.as_mut_ptr() as *mut u8;
590
591    // One `CarvedVec` over `$n` slots at byte offset `$at`. No element writes —
592    // the vector starts empty. Expanded inside the `unsafe` block below;
593    // SAFETY there: `compute_offsets` placed `$n` `$ty`s at `$at`, aligned and
594    // inside the asserted `o.size`, and each region is placed exactly once, so
595    // the slices do not alias.
596    macro_rules! carved {
597        ($at:expr, $n:expr, $ty:ty) => {{
598            let p = base.add($at) as *mut MaybeUninit<$ty>;
599            CarvedVec::new(core::slice::from_raw_parts_mut(p, $n))
600        }};
601    }
602
603    unsafe {
604        // arena — no init needed (MaybeUninit).
605        let arena_s =
606            core::slice::from_raw_parts_mut(base.add(o.arena) as *mut MaybeUninit<u8>, arena);
607
608        let entries_p = base.add(o.entries) as *mut Option<CallbackMeta>;
609        let mut i = 0;
610        while i < cbs {
611            entries_p.add(i).write(None);
612            i += 1;
613        }
614        let entries_s = core::slice::from_raw_parts_mut(entries_p, cbs);
615
616        let sc_p = base.add(o.sched_contexts) as *mut Option<SchedContext>;
617        let mut i = 0;
618        while i < sc {
619            sc_p.add(i).write(None);
620            i += 1;
621        }
622        let sched_contexts_s = core::slice::from_raw_parts_mut(sc_p, sc);
623
624        let bind_p = base.add(o.sched_context_bindings) as *mut SchedContextId;
625        let mut i = 0;
626        while i < cbs {
627            bind_p.add(i).write(SchedContextId(0));
628            i += 1;
629        }
630        let bindings_s = core::slice::from_raw_parts_mut(bind_p, cbs);
631
632        let sp_p = base.add(o.sporadic_states) as *mut Option<SporadicState>;
633        let mut i = 0;
634        while i < sc {
635            sp_p.add(i).write(None);
636            i += 1;
637        }
638        let sporadic_s = core::slice::from_raw_parts_mut(sp_p, sc);
639
640        #[cfg(feature = "alloc")]
641        let atomic_s = {
642            let ap = base.add(o.sporadic_atomic_states) as *mut Option<SporadicAtomic>;
643            let mut i = 0;
644            while i < sc {
645                ap.add(i).write(None);
646                i += 1;
647            }
648            core::slice::from_raw_parts_mut(ap, sc)
649        };
650
651        let remaps_p = base.add(o.remaps) as *mut Option<RemapRule>;
652        let mut i = 0;
653        while i < MAX_REMAPS {
654            remaps_p.add(i).write(None);
655            i += 1;
656        }
657        let remaps_s = core::slice::from_raw_parts_mut(remaps_p, MAX_REMAPS);
658
659        let alive_p = base.add(o.alive_slots) as *mut AliveSlot;
660        let mut i = 0;
661        while i < sc {
662            alive_p.add(i).write(AliveSlot::default());
663            i += 1;
664        }
665        let alive_s = core::slice::from_raw_parts_mut(alive_p, sc);
666
667        ExecutorSlices {
668            arena: arena_s,
669            entries: entries_s,
670            sched_contexts: sched_contexts_s,
671            sched_context_bindings: bindings_s,
672            sporadic_states: sporadic_s,
673            #[cfg(feature = "alloc")]
674            sporadic_atomic_states: atomic_s,
675            remaps: remaps_s,
676            nodes: carved!(o.nodes, node_slots, NodeRecord),
677            extra_sessions: carved!(o.extra_sessions, node_slots, ConcreteSession),
678            extra_session_ids: carved!(o.extra_session_ids, node_slots, ExtraSessionId),
679            node_sched_table: carved!(o.node_sched_table, node_slots, NodeSchedEntry),
680            dispatch_slots: carved!(o.dispatch_slots, node_slots, DispatchSlot),
681            component_slots: carved!(o.component_slots, node_slots, ComponentSlot),
682            active_groups: carved!(o.active_groups, node_slots, ActiveGroup),
683            group_sched_table: carved!(o.group_sched_table, cbs, GroupSchedEntry),
684            monitor_violations: carved!(o.monitor_violations, MAX_VIOLATIONS, Violation),
685            alive_slots: alive_s,
686        }
687    }
688}
