1//! phase-271 / phase-409 — per-entry [`Executor`](super::spin::Executor)
2//! storage (issues 0110, 0563, 0936).
3//!
4//! The executor's sized tables (callback table + arena + scheduling-context
5//! tables + the node/session/dispatch tables) used to be inline fields sized by
6//! build-time consts baked into `nros-node` — one size for every entry sharing a
7//! compiled `nros-node`, and, worse, a size that every function MOVING an
8//! `Executor` had to make room for on its stack. Here the ENTRY supplies its own
9//! storage, sized to its topology, so a fat native entry and a lean embedded
10//! entry in one workspace each get the right size with no workspace-global env,
11//! and the `Executor` VALUE stays a small fixed header.
12//!
13//! Per the "C/C++ is a thin wrapper of Rust" principle the PUBLIC API stays
14//! generic-free: the entry hands a raw, 8-aligned `&mut [MaybeUninit<u64>]` backing
15//! (sized via [`executor_storage_u64_len`]); `nros-node` carves it privately into
16//! the typed sub-slices ([`carve`]). The only `unsafe` is that carve, validated
17//! against the `#[repr(C)]` reference [`ExecutorStorage`] layout by unit test.

71/// The node half of an [`ActiveGroup`] key: a hash of `(name, namespace)`.
72///
73/// FNV-1a, 64-bit, over `name` then `\0` then the normalised namespace. The
74/// separator matters: without it `("ab", "c")` and `("a", "bc")` hash alike,
75/// and a node name may legally contain any of the namespace's characters.
76///
77/// FNV because the tree already depends on it (heapless's `FnvIndexMap`), it
78/// is `const`-friendly and allocation-free, and this runs on the registration
79/// path of every entity on an MCU. A 64-bit collision across the handful of
80/// nodes in one image is not a risk worth a bigger key: at 32 entries the
81/// probability is around 3e-17.
82///
83/// ONE home, deliberately. Both ends of the comparison call this —
84/// `set_active_groups` hashing what the tier declared, and `create_entity`
85/// hashing the node an entity belongs to. Two hashers that agree by
86/// inspection is exactly the defect class issue 1172 is filed under.
87pub(crate) fn node_identity_hash(name: &str, namespace: &str) -> u64 {
88    const OFFSET: u64 = 0xcbf2_9ce4_8422_2325;
89    const PRIME: u64 = 0x0000_0100_0000_01b3;
90    let norm_ns = if namespace.is_empty() { "/" } else { namespace };
91    let mut h = OFFSET;
92    for b in name
93        .as_bytes()
94        .iter()
95        .chain(core::iter::once(&0u8))
96        .chain(norm_ns.as_bytes())
97    {
98        h ^= *b as u64;
99        h = h.wrapping_mul(PRIME);
100    }
101    h
102}

104/// One entry of a tier's active-group filter: `(node identity hash, group)`.
105///
106/// issue 1172 — the filter used to be the GROUP NAME ALONE, which cannot tell
107/// node A's `ctrl` from node B's `ctrl`. Two nodes may legitimately declare the
108/// same group id on different tiers and nothing rejects it, so a tier filter
109/// was admitting entities it did not own. The sibling table for the
110/// sched-context path (`group_sched_table`, `bind_group_sched`) has been
111/// node-qualified all along; only this one was not.
112///
113/// The node half is a HASH of `(name, namespace)` rather than the strings.
114/// Storing them would cost 184 B per entry against this table's 48 B, and the
115/// table is CARVED one entry per node slot — +144 B per slot, which is 4.6 KB
116/// of static executor arena at 32 slots. The group name stays readable because
117/// a refused filter should be able to say WHICH group it refused.
118///
119/// The hash has exactly one home, [`node_identity_hash`], and both ends of the
120/// comparison call it — codegen never hashes anything, it emits plain strings.
121/// A second hasher would be this phase's own defect in a new place.
122pub(crate) type ActiveGroup = (u64, GroupName);

128/// A push-only vector over CARVED storage: the elements live in the caller's
129/// backing, the `Executor` holds only a fat pointer and a fill cursor.
130///
131/// phase-409 (issue 0961) — this is the shape phase-271 gave the six sized
132/// tables, generalised so the other nine can use it too. It exists because
133/// `heapless::Vec<T, N>` puts `N * size_of::<T>()` bytes INSIDE the struct, so
134/// every knob that sets an `N` also sets the stack frame of every function that
135/// moves an `Executor`: `MAX_CBS` 14 -> 36 (a handle-count fix) added ~3.7 KiB
136/// to `Executor::open_in`'s prologue on a part whose main thread has 32 KiB in
137/// total.
138///
139/// Deliberately `MaybeUninit<T>` rather than `Option<T>`, for three reasons:
140/// [`as_slice`](Self::as_slice) can hand back the `&[T]` the public
141/// `Executor::nodes()` accessor already returns; `carve` writes NOTHING at open
142/// (the `memclr` in issue 0961's fault was the executor's tables being zeroed);
143/// and the layout matches `heapless::Vec`'s, so the storage cost is the same
144/// bytes in a different place.
145///
146/// Occupied slots are `[0, len)` and IN PUSH ORDER — several callers index by
147/// insertion position (`NodeId` IS an index into `nodes`; `session_idx` is an
148/// index into `extra_sessions`), so this must never gain a `swap_remove`.
149pub(crate) struct CarvedVec<'s, T> {
150    slots: &'s mut [MaybeUninit<T>],
151    len: usize,
152}
153
154impl<'s, T> CarvedVec<'s, T> {
155    /// Wrap carved, UNINITIALISED storage as an empty vector.
156    fn new(slots: &'s mut [MaybeUninit<T>]) -> Self {
157        Self { slots, len: 0 }
158    }
159
160    /// Slots this vector can hold — the entry's sizing, not a build-time const.
161    pub(crate) fn capacity(&self) -> usize {
162        self.slots.len()
163    }
164
165    /// Append `value`, returning it in `Err` when full (`heapless::Vec::push`).
166    pub(crate) fn push(&mut self, value: T) -> Result<(), T> {
167        if self.len == self.capacity() {
168            return Err(value);
169        }
170        self.slots[self.len].write(value);
171        self.len += 1;
172        Ok(())
173    }
174
175    /// Remove and return the OLDEST element (index 0), shifting the rest down
176    /// one so push order is kept. O(len); phase-474 I1's violation ring is its
177    /// one caller, at `MAX_VIOLATIONS` (default 8) slots.
178    pub(crate) fn pop_front(&mut self) -> Option<T> {
179        if self.len == 0 {
180            return None;
181        }
182        // SAFETY: `[0, len)` is initialised; slot 0 is read out exactly once
183        // and then overwritten by the shift (or left outside `[0, len - 1)`).
184        let first = unsafe { self.slots[0].assume_init_read() };
185        let base = self.slots.as_mut_ptr();
186        // SAFETY: both ranges lie inside `slots`; `copy` handles the overlap.
187        unsafe { core::ptr::copy(base.add(1), base, self.len - 1) };
188        self.len -= 1;
189        Some(first)
190    }
191
192    /// Drop every element and reset the cursor.
193    pub(crate) fn clear(&mut self) {
194        let len = core::mem::replace(&mut self.len, 0);
195        if needs_drop::<T>() {
196            for slot in &mut self.slots[..len] {
197                // SAFETY: `[0, len)` was initialised by `push` and is dropped
198                // exactly once — `len` is reset above before this loop runs, so
199                // a panicking `T::drop` cannot make a second pass see it.
200                unsafe { slot.assume_init_drop() };
201            }
202        }
203    }
204
205    /// The initialised prefix.
206    pub(crate) fn as_slice(&self) -> &[T] {
207        // SAFETY: `push` initialises `[0, len)` and nothing shortens `len`
208        // without dropping (see `clear`).
209        unsafe { &*(&self.slots[..self.len] as *const [MaybeUninit<T>] as *const [T]) }
210    }
211
212    /// The initialised prefix, mutably.
213    pub(crate) fn as_mut_slice(&mut self) -> &mut [T] {
214        // SAFETY: as [`as_slice`](Self::as_slice).
215        unsafe { &mut *(&mut self.slots[..self.len] as *mut [MaybeUninit<T>] as *mut [T]) }
216    }
217}

220    type Target = [T];
221    fn deref(&self) -> &[T] {
222        self.as_slice()
223    }
224}

232impl<T> Drop for CarvedVec<'_, T> {
233    fn drop(&mut self) {
234        // The backing is the CALLER's memory, so nothing drops these elements
235        // for us — and one of these tables is `extra_sessions`, whose elements
236        // close RMW sessions. A `Drop` impl on the vector (rather than a pass in
237        // `Executor::drop`) keeps the ORDER the inline `heapless::Vec`s had:
238        // `Executor`'s fields drop in declaration order, so the primary
239        // `session` still closes before the extras.
240        self.clear();
241    }
242}

412/// issue 1197 — publish the backing's TOTAL size as a symbol whose storage size
413/// IS the number, so a build script can recover it from the rlib without running
414/// anything. `nros-sizes-build::extract_sizes` reads exactly this shape, the way
415/// `nros-c` recovers `EXECUTOR_OPAQUE_U64S`.
416///
417/// **The total rather than the per-region units, and that is a decision.** The
418/// units alone would be more robust — they do not vary with the executor knobs,
419/// so a probe cache keyed by (target, features) is correct for them, and the
420/// caller supplies `alloc` and the counts explicitly. But the backing is
421/// `arena + tables`, and `ARENA_SIZE` is DERIVED in this crate's build script
422/// from `MAX_CBS`, `RX_BUF_SIZE` and `ACTION_CLIENTS`. A consumer computing from
423/// units would have to re-implement that derivation — a second implementation,
424/// which is the whole thing this design refuses. So the consumer probes the one
425/// number, and keys its probe directory by the executor env, because unlike the
426/// units this DOES vary with it.
427///
428/// Behind a feature and OFF by default: it is `#[used]`, and a shipped image
429/// should not carry a static that exists only to be measured. The probe enables
430/// it for its own nested build, which is a separate compilation from the linked
431/// one.
432#[cfg(feature = "layout-size-markers")]
433#[used]
434#[unsafe(no_mangle)]
435#[doc(hidden)]
436pub static __NROS_LU_SZ_executor_backing: [u8;
437    crate::executor::backing::EXECUTOR_BACKING_DEFAULT_U64S * 8] =
438    [0u8; crate::executor::backing::EXECUTOR_BACKING_DEFAULT_U64S * 8];

524/// Byte [`Layout`] of the backing, evaluated against a SUPPLIED unit table.
525///
526/// issue 1197 — the public half of the split. `executor_storage_layout` is this
527/// with `NATIVE_UNITS`; an external consumer that has recovered the
528/// units for another target (a build script probing an rlib, the way `nros-c`
529/// recovers `EXECUTOR_OPAQUE_U64S`) gets the same arithmetic rather than a
530/// second implementation of it.
531///
532/// The OFFSETS stay private: only `carve` needs them, and it runs in the crate
533/// that owns the types. A consumer sizing a reservation needs the total.
534pub const fn executor_storage_layout_with(sizing: ExecutorSizing, units: RegionUnits) -> Layout {
535    let o = compute_offsets_with(sizing, &units);
536    // SAFETY: as `executor_storage_layout` — `align` is a max of alignments and
537    // `size` is rounded up to it.
538    unsafe { Layout::from_size_align_unchecked(o.size, o.align) }
539}

541/// Byte [`Layout`] of the backing a `sizing`-sized executor needs.
542/// Public + non-generic so the macro / FFI can size a raw backing.
543pub const fn executor_storage_layout(sizing: ExecutorSizing) -> Layout {
544    let o = compute_offsets(sizing);
545    // SAFETY: `align` is a power of two (a `max` of `align_of` results) and `size`
546    // is rounded up to it; both are non-zero.
547    unsafe { Layout::from_size_align_unchecked(o.size, o.align) }
548}

647/// The exact `#[repr(C)]` byte layout the C/C++ FFI's inline executor buffer must
648/// hold: an [`Executor`](super::spin::Executor)`<'static>` header immediately
649/// followed by a default-sized ([`ExecutorSizing::DEFAULT`]) storage backing.

651/// The FFI keeps the executor inline (heap-free — matching the Rust no-alloc

656/// self-borrow (the header's slices pointing into the same struct's tail) is

658/// its `_opaque` array, and reinterprets `_opaque` as `*mut ExecutorInlineStorage`
659/// (the executor stays at offset 0, so existing offset-0 accessors are unchanged).

661/// phase-409 — moving the last nine tables into the backing moves bytes from
662/// `exec` to `backing` and leaves this total roughly where it was; what changes
663/// is the STACK, because the header is what `open_in` builds and returns by
664/// value.

669    /// The carved backing the executor's slices borrow (the buffer's tail).
670    ///
671    /// Issue 1970 -- [`INLINE_BACKING_U64S`] words: the image's whole
672    /// executor, or none at all on a tiered image, whose tiers each carve a
673    /// backing of their own size ([`tier_sizing`]).
674    pub backing: [MaybeUninit<u64>; INLINE_BACKING_U64S],
675}

740/// phase-509 P9b item 4 (RFC-0108 "As built in P9b") -- the sizing of the
741/// ONE executor a level runner opens per core: every level's callbacks, nodes,
742/// arena and declared parameters, summed over the image's tiers (each row of
743/// the tier table is one level). `None` when the image sizes no tier.
744pub const fn levels_executor_sizing() -> Option<ExecutorSizing> {
745    let tiers = crate::config::TIER_SIZINGS;
746    if tiers.is_empty() {
747        return None;
748    }
749    let store = nros_params::IMPLIED_STORE_SLOTS;
750    let (mut cbs, mut arena, mut nodes, mut params) = (0usize, 0usize, 0usize, 0usize);
751    let mut i = 0;
752    while i < tiers.len() {
753        let t = &tiers[i];
754        cbs += t.cbs;
755        arena += t.arena;
756        nodes += t.nodes;
757        params += match t.params {
758            None => store,
759            Some(n) => n,
760        };
761        i += 1;
762    }

764    // once (`arena_model::BASE_OVERHEAD`), because issue 1970 sized a tier as
765    // its own executor. The levels share ONE executor, so the base is paid
766    // once: summing it per level handed the island's executor 29,820 B where
767    // the image's own model (`NROS_EXECUTOR_ARENA_SIZE`) is 25,724 -- exactly
768    // the two extra bases. See [`levels_arena_bytes`].
769    let arena = levels_arena_bytes(arena, tiers.len());
770    // One executor dispatches at most one ready word of slots; a level image
771    // past it is refused at registration, loudly, as any executor is.
772    let max_cbs = super::spin::MAX_CALLBACK_SLOTS;
773    Some(ExecutorSizing {
774        cbs: if cbs < max_cbs { cbs } else { max_cbs },
775        sc: crate::config::MAX_SC,
776        arena,
777        nodes,
778        params: if params < store { params } else { store },
779        remaps: MAX_REMAPS,
780    })

803/// Carve an 8-aligned `u64` backing into the typed executor slices.
804///
805/// # Safety
806/// - `backing.len() * 8` must be ≥ `executor_storage_layout(sizing).size()`.
807/// - The returned slices alias `backing`; it must outlive them (the `'s` bound)
808///   and not be otherwise accessed while they live.
809///
810/// The `Option`/binding tables are initialised here (`entries`/SC tables →
811/// `None`, bindings → `SchedContextId(0)`), so the returned `&mut [T]` reference
812/// validly-init memory. The [`CarvedVec`] tables are handed over empty and
813/// UNINITIALISED — nothing reads past their fill cursor.
814pub(crate) unsafe fn carve<'s>(
815    backing: &'s mut [MaybeUninit<u64>],
816    sizing: ExecutorSizing,
817) -> ExecutorSlices<'s> {
818    let ExecutorSizing {
819        cbs,
820        sc,
821        arena,
822        nodes: node_slots,
823        params: param_slots,
824        remaps: remap_slots,
825    } = sizing;
826    let extra_slots = extra_session_slots(sizing);
827    let o = compute_offsets(sizing);
828    // Fail-loud on EVERY profile (not `debug_assert!`): embedded release builds
829    // strip debug-assertions, and a backing that is too small is silent memory
830    // corruption — the carved `entries`/`sched_contexts` tables run past the end
831    // of `backing` into whatever .bss follows (e.g. a C carrier's `__nros_c_inst`),
832    // leaving a NULL `drop_fn` that faults in `Executor::drop`. This is exactly
833    // how a STALE config-header mirror (C buffer sized from an out-of-date
834    // `NROS_*_STORAGE_SIZE`) manifested as a `jalr -> 0` on threadx-riscv64 (#131).
835    // Panic here instead, at open, with the two sizes named.
836    assert!(
837        backing.len() * 8 >= o.size,
838        "executor backing too small: {} bytes < {} required — the storage buffer \
839         (NROS_*_STORAGE_SIZE) disagrees with the executor layout; rebuild clean so \
840         the generated config header matches",
841        backing.len() * 8,
842        o.size
843    );
844    let base = backing.as_mut_ptr() as *mut u8;
845
846    // One `CarvedVec` over `$n` slots at byte offset `$at`. No element writes —
847    // the vector starts empty. Expanded inside the `unsafe` block below;
848    // SAFETY there: `compute_offsets` placed `$n` `$ty`s at `$at`, aligned and
849    // inside the asserted `o.size`, and each region is placed exactly once, so
850    // the slices do not alias.
851    macro_rules! carved {
852        ($at:expr, $n:expr, $ty:ty) => {{
853            let p = base.add($at) as *mut MaybeUninit<$ty>;
854            CarvedVec::new(region_slice(p, $n))
855        }};
856    }
857
858    unsafe {
859        // arena — no init needed (MaybeUninit).
860        let arena_s = region_slice(base.add(o.arena) as *mut MaybeUninit<u8>, arena);
861
862        let entries_p = base.add(o.entries) as *mut Option<CallbackMeta>;
863        let mut i = 0;
864        while i < cbs {
865            entries_p.add(i).write(None);
866            i += 1;
867        }
868        let entries_s = region_slice(entries_p, cbs);
869
870        let sc_p = base.add(o.sched_contexts) as *mut Option<SchedContext>;
871        let mut i = 0;
872        while i < sc {
873            sc_p.add(i).write(None);
874            i += 1;
875        }
876        let sched_contexts_s = region_slice(sc_p, sc);
877
878        let bind_p = base.add(o.sched_context_bindings) as *mut SchedContextId;
879        let mut i = 0;
880        while i < cbs {
881            bind_p.add(i).write(SchedContextId(0));
882            i += 1;
883        }
884        let bindings_s = region_slice(bind_p, cbs);
885
886        let sp_p = base.add(o.sporadic_states) as *mut Option<SporadicState>;
887        let mut i = 0;
888        while i < sc {
889            sp_p.add(i).write(None);
890            i += 1;
891        }
892        let sporadic_s = region_slice(sp_p, sc);
893
894        #[cfg(feature = "alloc")]
895        let atomic_s = {
896            let ap = base.add(o.sporadic_atomic_states) as *mut Option<SporadicAtomic>;
897            let mut i = 0;
898            while i < sc {
899                ap.add(i).write(None);
900                i += 1;
901            }
902            region_slice(ap, sc)
903        };
904
905        let remaps_p = base.add(o.remaps) as *mut Option<RemapRule>;
906        let mut i = 0;
907        while i < remap_slots {
908            remaps_p.add(i).write(None);
909            i += 1;
910        }
911        let remaps_s = region_slice(remaps_p, remap_slots);
912
913        let alive_p = base.add(o.alive_slots) as *mut AliveSlot;
914        let mut i = 0;
915        while i < sc {
916            alive_p.add(i).write(AliveSlot::default());
917            i += 1;
918        }
919        let alive_s = region_slice(alive_p, sc);
920
921        let tags_p = base.add(o.slot_tags) as *mut SlotTag;
922        let mut i = 0;
923        while i < cbs {
924            tags_p.add(i).write(SlotTag::default());
925            i += 1;
926        }
927        let tags_s = region_slice(tags_p, cbs);
928
929        // phase-382 W3' — no init: the slots are `MaybeUninit` until the
930        // first declaration initialises them through `ParameterTable::init_in`.
931        let params_s = region_slice(
932            base.add(o.params) as *mut MaybeUninit<nros_params::ParameterSlot>,
933            param_slots,
934        );
935        // issue 1812 — no init either: `ParamHome::init_in` builds it in place.
936        let home_s = region_slice(
937            base.add(o.param_home) as *mut MaybeUninit<ParamHome>,
938            param_homes(sizing),
939        );
940
941        ExecutorSlices {
942            arena: arena_s,
943            entries: entries_s,
944            sched_contexts: sched_contexts_s,
945            sched_context_bindings: bindings_s,
946            sporadic_states: sporadic_s,
947            #[cfg(feature = "alloc")]
948            sporadic_atomic_states: atomic_s,
949            remaps: remaps_s,
950            nodes: carved!(o.nodes, node_slots, NodeRecord),
951            extra_sessions: carved!(o.extra_sessions, extra_slots, ConcreteSession),
952            extra_session_ids: carved!(o.extra_session_ids, extra_slots, ExtraSessionId),
953            node_sched_table: carved!(o.node_sched_table, node_slots, NodeSchedEntry),
954            dispatch_slots: carved!(o.dispatch_slots, node_slots, DispatchSlot),
955            component_slots: carved!(o.component_slots, node_slots, ComponentSlot),
956            active_groups: carved!(o.active_groups, node_slots, ActiveGroup),
957            group_sched_table: carved!(o.group_sched_table, cbs, GroupSchedEntry),
958            monitor_violations: carved!(o.monitor_violations, MAX_VIOLATIONS, Violation),
959            alive_slots: alive_s,
960            slot_tags: tags_s,
961            params: params_s,
962            param_home: home_s,
963        }
964    }
965}

967/// One carved region as a slice — the ONLY way `carve` builds one (issue 1771).
968///
969/// `slice::from_raw_parts_mut` requires an ALIGNED, non-null pointer even for
970/// a zero-length slice, and an empty region's offset is not always aligned:
971/// `nros_executor_layout::offsets` deliberately leaves `off` where it is for a
972/// region with no slots (the parameter store in an image without one, the
973/// atomic sporadic states without `alloc`), so the total does not grow by
974/// padding nothing uses. On thumbv7m that offset is misaligned for
975/// `ParameterSlot`, and the debug precondition check panicked at boot. An
976/// empty region needs no address at all, so it gets a dangling one.
977///
978/// # Safety
979/// For `n > 0`, `p` points at `n` properly aligned `T`s inside the backing
980/// that no other live slice covers.
981unsafe fn region_slice<'s, T>(p: *mut T, n: usize) -> &'s mut [T] {
982    if n == 0 {
983        return &mut [];
984    }
985    // SAFETY: the caller's contract, for a non-empty region.
986    unsafe { core::slice::from_raw_parts_mut(p, n) }
987}
