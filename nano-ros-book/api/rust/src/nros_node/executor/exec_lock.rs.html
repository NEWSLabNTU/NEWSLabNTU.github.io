1//! phase-509 P9b item 1 (RFC-0108 "As built in P9b") -- the executor's lock.
2//!
3//! RFC-0108 runs one executor per core with one worker thread per preemption
4//! level. Every path that touches the executor's state -- a worker's pass, and
5//! every C / C++ FFI entry a callback or another thread calls -- then has to be
6//! serialised, and that is this lock: a port-provided, RECURSIVE,
7//! PRIORITY-INHERITING mutex (`nros_platform_mutex_rec_*` on every port), which
8//! the runner installs on the executor before a second thread can reach it.
9//!
10//! An executor with no lock installed (every one-level image, every Rust
11//! `&mut` user) takes the `None` arm: a guard is one predictable branch and
12//! nothing else, so this seam changes no behaviour where it is not installed.
13//!
14//! Why recursive: until the pass releases the lock around each user callback
15//! (item 2), a callback that calls back into the executor (`timer_cancel`,
16//! `*_release`, a parameter call) re-enters the lock the spinning worker
17//! already holds, on the same thread.
18//!
19//! Why priority-inheriting: a less urgent worker holding the lock for a
20//! bookkeeping step must run at the waiter's priority, so a more urgent
21//! level's blocking is bounded by ONE critical section (RFC-0108 B_i).
22//!
23//! Interrupt handlers never take it: arrivals go through the scheduler's
24//! lock-free ready word and the wake object, exactly as before.
25
26use core::{ffi::c_void, ptr::NonNull};
27
28/// A port's lock or unlock entry, `nros_platform_mutex_rec_lock`'s shape:
29/// `0` on success, negative on failure.
30pub type ExecLockFn = unsafe extern "C" fn(*mut c_void) -> i8;
31
32/// The executor's lock: the port's recursive priority-inheriting mutex and
33/// the two calls that take and give it. Installed once, before any second
34/// thread can reach the executor ([`super::Executor::set_exec_lock`]), and
35/// never changed while one can.
36#[derive(Clone, Copy, Debug)]
37pub struct ExecLock {
38    ctx: NonNull<c_void>,
39    lock: ExecLockFn,
40    unlock: ExecLockFn,
41}
42
43// SAFETY: the context is the port's mutex, whose whole purpose is to be shared
44// between threads; the two entries are plain functions.
45unsafe impl Send for ExecLock {}
46// SAFETY: as above -- `lock` / `unlock` are the port's thread-safe calls.
47unsafe impl Sync for ExecLock {}
48
49impl ExecLock {
50    /// A lock over the port's mutex at `ctx`, already initialised by the
51    /// caller. `None` for a NULL `ctx`.
52    ///
53    /// # Safety
54    /// `ctx` is NULL or an initialised recursive mutex that `lock` / `unlock`
55    /// take and give, and it outlives every executor the lock is installed on.
56    pub unsafe fn new(ctx: *mut c_void, lock: ExecLockFn, unlock: ExecLockFn) -> Option<Self> {
57        Some(Self {
58            ctx: NonNull::new(ctx)?,
59            lock,
60            unlock,
61        })
62    }
63
64    fn take(&self) {
65        // SAFETY: `new`'s contract: the context is the initialised mutex these
66        // entries take. A failed take is not recoverable here (the port's
67        // mutex never fails on a valid handle); the guard proceeds, which is
68        // the behaviour the image had before the lock existed.
69        let _ = unsafe { (self.lock)(self.ctx.as_ptr()) };
70    }
71
72    fn give(&self) {
73        // SAFETY: as `take`; only ever called by the guard that took it.
74        let _ = unsafe { (self.unlock)(self.ctx.as_ptr()) };
75    }
76}
77
78/// Holds the executor's lock for its own lifetime: taken at
79/// [`ExecGuard::acquire`], given at drop. With no lock installed it holds
80/// nothing.
81#[must_use = "the lock is given back when the guard drops"]
82pub struct ExecGuard {
83    lock: Option<ExecLock>,
84}
85
86impl ExecGuard {
87    /// Take `lock`, if there is one.
88    pub fn take(lock: Option<ExecLock>) -> Self {
89        if let Some(l) = &lock {
90            l.take();
91        }
92        Self { lock }
93    }
94
95    /// Take the lock installed on the executor at `exec`, BEFORE any reference
96    /// to that executor exists -- the order that keeps a second thread's
97    /// reference from overlapping this one.
98    ///
99    /// # Safety
100    /// `exec` points at a live, initialised executor. Its lock field is read
101    /// without a reference: it is written only while no second thread can
102    /// reach the executor ([`super::Executor::set_exec_lock`]).
103    #[cfg(any(has_rmw, test))]
104    pub unsafe fn acquire(exec: *const super::Executor<'_>) -> Self {
105        // SAFETY: the caller's contract; a field read through the raw pointer
106        // creates no reference to the executor.
107        let lock = unsafe { core::ptr::read(core::ptr::addr_of!((*exec).exec_lock)) };
108        Self::take(lock)
109    }
110
111    /// Whether this guard holds a lock.
112    pub fn holds(&self) -> bool {
113        self.lock.is_some()
114    }
115}
116
117impl Drop for ExecGuard {
118    fn drop(&mut self) {
119        if let Some(l) = &self.lock {
120            l.give();
121        }
122    }
123}

125#[cfg(test)]
126mod tests {
127    use super::*;
128    use core::sync::atomic::{AtomicI32, Ordering};

130    /// A counting "mutex": the depth it is held at, and how often it was taken.
131    #[derive(Default)]
132    struct Counter {
133        depth: AtomicI32,
134        takes: AtomicI32,
135    }
136
137    unsafe extern "C" fn lock(c: *mut c_void) -> i8 {
138        // SAFETY: the test passes a `Counter` it owns as the context.
139        let c = unsafe { &*(c as *const Counter) };
140        c.depth.fetch_add(1, Ordering::SeqCst);
141        c.takes.fetch_add(1, Ordering::SeqCst);
142        0
143    }
144
145    unsafe extern "C" fn unlock(c: *mut c_void) -> i8 {
146        // SAFETY: as `lock`.
147        let c = unsafe { &*(c as *const Counter) };
148        c.depth.fetch_sub(1, Ordering::SeqCst);
149        0
150    }
151
152    /// A guard takes once and gives once, nests (the recursive case a
153    /// callback re-entering the executor is), and with no lock does nothing.
154    #[test]
155    fn a_guard_takes_once_and_gives_on_drop_and_nests() {
156        let c = Counter::default();
157        // SAFETY: `c` outlives every guard taken below.
158        let l = unsafe { ExecLock::new(&c as *const Counter as *mut c_void, lock, unlock) };
159        {
160            let g = ExecGuard::take(l);
161            assert!(g.holds());
162            assert_eq!(c.depth.load(Ordering::SeqCst), 1);
163            {
164                let _inner = ExecGuard::take(l);
165                assert_eq!(c.depth.load(Ordering::SeqCst), 2);
166            }
167            assert_eq!(c.depth.load(Ordering::SeqCst), 1);
168        }
169        assert_eq!(c.depth.load(Ordering::SeqCst), 0);
170        assert_eq!(c.takes.load(Ordering::SeqCst), 2);
171        let none = ExecGuard::take(None);
172        assert!(!none.holds());
173        // SAFETY: a NULL context is refused before anything is called.
174        assert!(unsafe { ExecLock::new(core::ptr::null_mut(), lock, unlock) }.is_none());
175    }

177    /// A spin lock standing in for the port mutex.
178    struct Spin(core::sync::atomic::AtomicBool);
179
180    unsafe extern "C" fn spin_lock(c: *mut c_void) -> i8 {
181        // SAFETY: the test passes the static `Spin` as the context.
182        let s = unsafe { &*(c as *const Spin) };
183        while s
184            .0
185            .compare_exchange_weak(false, true, Ordering::Acquire, Ordering::Relaxed)
186            .is_err()
187        {
188            core::hint::spin_loop();
189        }
190        0
191    }
192
193    unsafe extern "C" fn spin_unlock(c: *mut c_void) -> i8 {
194        // SAFETY: as `spin_lock`.
195        let s = unsafe { &*(c as *const Spin) };
196        s.0.store(false, Ordering::Release);
197        0
198    }

200    /// Two threads that touch shared state only under the guard never
201    /// interleave: every unsynchronised increment survives.
202    #[test]
203    fn guarded_sections_on_two_threads_do_not_interleave() {
204        struct Shared(core::cell::UnsafeCell<u64>);
205        // SAFETY: only touched under the guard below.
206        unsafe impl Sync for Shared {}
207        static LOCK: Spin = Spin(core::sync::atomic::AtomicBool::new(false));
208        static SHARED: Shared = Shared(core::cell::UnsafeCell::new(0));
209        let l =
210            // SAFETY: `LOCK` is a static, so it outlives both threads.
211            unsafe { ExecLock::new(&LOCK as *const Spin as *mut c_void, spin_lock, spin_unlock) };
212        const N: u64 = 100_000;
213        let worker = move || {
214            for _ in 0..N {
215                let _g = ExecGuard::take(l);
216                // SAFETY: the guard serialises every access to SHARED.
217                unsafe {
218                    let p = SHARED.0.get();
219                    let v = core::ptr::read_volatile(p);
220                    core::ptr::write_volatile(p, v + 1);
221                }
222            }
223        };
224        let a = std::thread::spawn(worker);
225        let b = std::thread::spawn(worker);
226        a.join().unwrap();
227        b.join().unwrap();
228        let _g = ExecGuard::take(l);
229        // SAFETY: both threads have joined and the guard is held.
230        assert_eq!(unsafe { *SHARED.0.get() }, 2 * N);
231    }
