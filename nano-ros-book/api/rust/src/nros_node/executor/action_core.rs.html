22/// DDS type name of an action's `send_goal` / `get_result` service.
23///
24/// ROS 2 names the action's two services by their per-channel service types —
25/// `<Action>_SendGoal` / `<Action>_GetResult` — **not** the bare action type
26/// (`<Action>`). A real `rcl_action` peer matches on those, so advertising the
27/// bare action type leaves our send_goal/get_result services undiscovered and
28/// every goal times out.
29///
30/// The codegen emits each channel's *request* envelope type name in DDS form,
31/// e.g. `example_interfaces::action::dds_::Fibonacci_SendGoal_Request_`. The
32/// service layer (`xrce_dds_request_type` / the Zenoh shim) extends a base type
33/// ending in `_` with `Request_` / `Response_`, so the base we must pass is the
34/// request type name with its trailing `Request_` stripped:
35/// `…Fibonacci_SendGoal_Request_` → `…Fibonacci_SendGoal_`. Falls back to the
36/// bare action type if the request name has an unexpected shape.
37pub(crate) fn action_service_base_type<'a>(
38    request_type_name: &'a str,
39    fallback_action_type: &'a str,
40) -> &'a str {
41    request_type_name
42        .strip_suffix("Request_")
43        .unwrap_or(fallback_action_type)
44}

46/// The per-channel DDS type name for a RAW-registered action server.
47///
48/// The typed path derives these from `A::SendGoalRequest::TYPE_NAME` via
49/// [`action_service_base_type`]. The raw path (`register_action_server_raw*`)
50/// has only the BARE action type, so it must construct them — and until
51/// phase-338 W3 it did not, advertising `…Fibonacci_` on send_goal / get_result
52/// / feedback where ROS 2 expects `…Fibonacci_SendGoal_`,
53/// `…Fibonacci_GetResult_` and `…Fibonacci_FeedbackMessage_`. The type name is
54/// baked into the keyexpr, so a client's query never matched the server's
55/// queryable and every goal timed out with `Transport(Timeout)` — exactly the
56/// failure [`action_service_base_type`]'s own doc warns about, on the other
57/// registration path.
58///
59/// `action_type` is DDS-form and ends in `_` (`…::dds_::Fibonacci_`); the
60/// result replaces that suffix with `_<Channel>_`. Returns the bare type
61/// unchanged if it has an unexpected shape, matching the typed path's
62/// fallback.
63pub fn action_channel_type<const N: usize>(
64    action_type: &str,
65    channel: &str,
66) -> heapless::String<N> {
67    let mut out: heapless::String<N> = heapless::String::new();
68    let base = action_type.strip_suffix('_').unwrap_or(action_type);
69    if out.push_str(base).is_err()
70        || out.push('_').is_err()
71        || out.push_str(channel).is_err()
72        || out.push('_').is_err()
73    {
74        let mut fallback: heapless::String<N> = heapless::String::new();
75        let _ = fallback.push_str(action_type);
76        return fallback;
77    }
78    out
79}

157/// Read a GoalId from a CDR reader as a fixed `uint8[16]` array.
158///
159/// ROS 2 actions carry the goal id as `unique_identifier_msgs/UUID`, whose
160/// single field is a **fixed-size** `uint8[16]` array — CDR fixed arrays have
161/// **no** length prefix. We must read exactly 16 bytes with no leading count,
162/// matching `unique_identifier_msgs::msg::UUID::deserialize`. (The pre-233.6
163/// framing wrote a `u32(16)` sequence prefix, which self-matched nano-ros peers
164/// but added 4 bytes a real `rcl_action` peer rejects.)
165fn read_goal_id(reader: &mut CdrReader<'_>) -> Result<GoalId, NodeError> {
166    let mut goal_id = GoalId::default();
167    for byte in &mut goal_id.uuid {
168        *byte = reader
169            .read_u8()
170            .map_err(|_| NodeError::Transport(TransportError::DeserializationError))?;
171    }
172    Ok(goal_id)
173}
174
175/// Write a GoalId into a CDR writer as a fixed `uint8[16]` array (no length
176/// prefix) — see [`read_goal_id`] for why the prefix must be absent.
177fn write_goal_id(writer: &mut CdrWriter<'_>, goal_id: &GoalId) -> Result<(), NodeError> {
178    for b in &goal_id.uuid {
179        writer.write_u8(*b).map_err(|_| NodeError::Serialization)?;
180    }
181    Ok(())
182}

188/// Type-agnostic action server core handling the raw-bytes protocol.
189///
190/// Manages active goal tracking (GoalId + status), completed result storage
191/// in a fixed-size slab, and all CDR framing for the action protocol.
192///
193/// The typed [`ActionServer`](super::handles::ActionServer) wraps this
194/// and adds `A::Goal` / `A::Feedback` / `A::Result` (de)serialization.
195pub struct ActionServerCore<
196    const GOAL_BUF: usize = { crate::config::DEFAULT_RX_BUF_SIZE },
197    const RESULT_BUF: usize = { crate::config::DEFAULT_RX_BUF_SIZE },
198    const FEEDBACK_BUF: usize = { crate::config::DEFAULT_RX_BUF_SIZE },
199    const MAX_GOALS: usize = 4,
200> {
201    pub(crate) send_goal_server: session::RmwServiceServer,
202    pub(crate) cancel_goal_server: session::RmwServiceServer,
203    pub(crate) get_result_server: session::RmwServiceServer,
204    pub(crate) feedback_publisher: session::RmwPublisher,
205    pub(crate) status_publisher: session::RmwPublisher,
206    pub(crate) active_goals: heapless::Vec<RawActiveGoal, MAX_GOALS>,
207    pub(crate) completed_results: heapless::Vec<CompletedResultEntry, MAX_GOALS>,
208    /// `get_result` requests that arrived while their goal was still active
209    /// (Phase 237). `rclcpp_action` sends `get_result` immediately after
210    /// acceptance and expects the reply only once the goal terminates, so we
211    /// hold the request's correlation token (`sequence_number`) here and flush
212    /// it in [`Self::complete_goal_raw`]. Deferral relies on the service
213    /// backend honoring `send_reply(seq)` after the handler returns — the
214    /// seq-keyed reply contract (Cyclone native; XRCE/Zenoh per Phase 237).
215    pub(crate) pending_get_results: heapless::Vec<PendingGetResult, MAX_GOALS>,
216    /// Slab storage for completed result CDR bytes.
217    pub(crate) result_slab: [u8; RESULT_BUF],
218    pub(crate) result_slab_used: usize,
219    pub(crate) goal_buffer: [u8; GOAL_BUF],
220    pub(crate) feedback_buffer: [u8; FEEDBACK_BUF],
221    pub(crate) cancel_buffer: [u8; CANCEL_BUF],
222}

259    /// Try to receive a goal request from the send_goal service.
260    ///
261    /// Returns the parsed GoalId, sequence number, and data length.
262    /// The full CDR data (including GoalId) remains in `goal_buffer`.
263    pub fn try_recv_goal_request(&mut self) -> Result<Option<RawGoalRequest>, NodeError> {
264        // Capture buf base ptr before borrowing through `try_recv_request`
265        // so we can recover the data offset after the borrow ends.
266        // DDS-style backends place a sequence-number prefix before the
267        // CDR payload; reading the buffer from offset 0 unconditionally
268        // would feed the prefix bytes to the deserializer.
269        let buf_start = self.goal_buffer.as_ptr() as usize;
270        // Phase 120: NoData (no pending request) is the steady-state
271        // expected condition — collapse it to `Ok(None)` instead of
272        // surfacing as ServiceRequestFailed. Any other transport
273        // error remains ServiceRequestFailed.
274        let request = match self
275            .send_goal_server
276            .try_recv_request(&mut self.goal_buffer)
277        {
278            Ok(opt) => opt,
279            Err(TransportError::NoData) => return Ok(None),
280            Err(_) => return Err(NodeError::Transport(TransportError::ServiceRequestFailed)),
281        };
282
283        let request = match request {
284            Some(r) => r,
285            None => return Ok(None),
286        };
287
288        let data_offset = (request.data.as_ptr() as usize).saturating_sub(buf_start);
289        let data_len = request.data.len();
290        let sequence_number = request.sequence_number;
291        #[allow(clippy::drop_non_drop)]
292        drop(request);
293
294        let mut reader =
295            CdrReader::new_with_header(&self.goal_buffer[data_offset..data_offset + data_len])
296                .map_err(|_| NodeError::Transport(TransportError::DeserializationError))?;
297
298        let goal_id = read_goal_id(&mut reader)?;
299
300        Ok(Some(RawGoalRequest {
301            goal_id,
302            sequence_number,
303            data_offset,
304            data_len,
305        }))
306    }

313    /// Accept a goal: records it, sends the acceptance reply, publishes status.
314    ///
315    /// # Ordering
316    ///
317    /// The goal is recorded in `active_goals` **before** the reply goes out,
318    /// and the recording is rolled back if the reply fails. Issue 0322: this
319    /// used to reply first and then `let _ = self.active_goals.push(...)`, so
320    /// once `MAX_GOALS` (default 4) were active, a 5th `send_goal` was
321    /// answered `accepted=true` and then dropped — no execution, no feedback,
322    /// no result, no terminal status. An rclcpp/rclpy client that saw
323    /// `accepted=true` waited on its result future forever.
324    ///
325    /// A full table is now answered truthfully with `accepted=false` via
326    /// [`Self::reject_goal`], which is a contract the client already handles.
327    pub fn accept_goal(&mut self, goal_id: GoalId, seq: i64) -> Result<(), NodeError> {
328        // Serialize first: a serialization failure here must leave no trace,
329        // and nothing below depends on the table.
330        let mut writer =
331            crate::tx_writer(&mut self.cancel_buffer).map_err(|_| NodeError::BufferTooSmall)?;
332        writer.write_u8(1).map_err(|_| NodeError::Serialization)?;
333        writer.write_i32(0).map_err(|_| NodeError::Serialization)?;
334        writer.write_u32(0).map_err(|_| NodeError::Serialization)?;
335        let reply_len = writer.position();
336
337        // Capacity is decided BEFORE anything reaches the wire, so a full
338        // table becomes an honest rejection rather than a lie.
339        if self
340            .active_goals
341            .push(RawActiveGoal {
342                goal_id,
343                status: GoalStatus::Accepted,
344            })
345            .is_err()
346        {
347            return self.reject_goal(seq);
348        }
349
350        if self
351            .send_goal_server
352            .send_reply(seq, &self.cancel_buffer[..reply_len])
353            .is_err()
354        {
355            // The client never learned it was accepted, so un-record it —
356            // otherwise the slot leaks and lowers the effective capacity for
357            // every later goal. `pop` removes the entry pushed just above:
358            // `&mut self` means nothing else can have touched the table.
359            self.active_goals.pop();
360            return Err(NodeError::ServiceReplyFailed);
361        }
362
363        // Past this point the acceptance is on the wire and irreversible.
364        //
365        // The status-array publish is therefore NOT propagated: both C and C++
366        // callers collapse `Err` to a generic error code
367        // (`nros-c/src/action/server.rs`, `nros-cpp/src/action.rs`), so
368        // returning one here would report "accept failed" for a goal that IS
369        // accepted and running — inviting the caller to reject or retry it.
370        // The client already holds `accepted=true` and will still receive the
371        // result; a missed status sample is degraded, not broken. Issue 0322
372        // proposed propagating this; see that issue for why it is deliberately
373        // not done.
374        let _status = self.publish_status_array();
375        Ok(())
376    }

378    /// Reject a goal: sends the rejection reply.
379    pub fn reject_goal(&mut self, seq: i64) -> Result<(), NodeError> {
380        // Serialize response: accepted=false + stamp
381        let mut writer =
382            crate::tx_writer(&mut self.cancel_buffer).map_err(|_| NodeError::BufferTooSmall)?;
383        writer.write_u8(0).map_err(|_| NodeError::Serialization)?;
384        writer.write_i32(0).map_err(|_| NodeError::Serialization)?;
385        writer.write_u32(0).map_err(|_| NodeError::Serialization)?;
386        let reply_len = writer.position();
387
388        self.send_goal_server
389            .send_reply(seq, &self.cancel_buffer[..reply_len])
390            .map_err(|_| NodeError::ServiceReplyFailed)
391    }

393    /// Publish feedback with raw CDR bytes.
394    ///
395    /// Writes GoalId framing + raw feedback bytes into the feedback buffer
396    /// and publishes.
397    pub fn publish_feedback_raw(
398        &mut self,
399        goal_id: &GoalId,
400        feedback_cdr: &[u8],
401    ) -> Result<(), NodeError> {
402        // GoalId framing (4 + 16 = 20 bytes) + feedback_cdr must fit in FEEDBACK_BUF
403        let needed = 4 + 20 + feedback_cdr.len(); // CDR header + GoalId + feedback
404        if needed > FEEDBACK_BUF {
405            return Err(NodeError::BufferTooSmall);
406        }
407
408        let mut writer =
409            crate::tx_writer(&mut self.feedback_buffer).map_err(|_| NodeError::BufferTooSmall)?;
410
411        write_goal_id(&mut writer, goal_id)?;
412
413        // Copy raw feedback bytes directly after GoalId
414        let pos = writer.position();
415        if pos + feedback_cdr.len() > FEEDBACK_BUF {
416            return Err(NodeError::BufferTooSmall);
417        }
418        self.feedback_buffer[pos..pos + feedback_cdr.len()].copy_from_slice(feedback_cdr);
419        let len = pos + feedback_cdr.len();
420
421        self.feedback_publisher
422            .publish_raw(&self.feedback_buffer[..len])
423            .map_err(|_| NodeError::Transport(TransportError::PublishFailed))
424    }

437    /// Complete a goal: remove from active, store raw result CDR in slab,
438    /// publish status.
439    pub fn complete_goal_raw(&mut self, goal_id: &GoalId, status: GoalStatus, result_cdr: &[u8]) {
440        // Remove from active goals
441        if let Some(pos) = self
442            .active_goals
443            .iter()
444            .position(|g| g.goal_id.uuid == goal_id.uuid)
445        {
446            self.active_goals.swap_remove(pos);
447        }
448
449        // Store result CDR in the slab
450        let offset = self.result_slab_used;
451        let end = offset + result_cdr.len();
452        let stored = if end <= RESULT_BUF {
453            self.result_slab[offset..end].copy_from_slice(result_cdr);
454            self.result_slab_used = end;
455            let _ = self.completed_results.push(CompletedResultEntry {
456                goal_id: *goal_id,
457                status,
458                offset,
459                len: result_cdr.len(),
460            });
461            true
462        } else {
463            false
464        };
465
466        // Phase 237 — flush any get_result requests that arrived while this goal
467        // was still active. The result is now in the slab; reply to each held
468        // requester via its retained `sequence_number`.
469        if stored {
470            let len = result_cdr.len();
471            let mut i = 0;
472            while i < self.pending_get_results.len() {
473                if self.pending_get_results[i].goal_id.uuid == goal_id.uuid {
474                    let seq = self.pending_get_results[i].sequence_number;
475                    // swap_remove moves the last entry into slot `i`; re-check `i`.
476                    let _ = self.pending_get_results.swap_remove(i);
477                    let _ = self.reply_get_result_from_slab(seq, status, offset, len);
478                } else {
479                    i += 1;
480                }
481            }
482        }
483
484        let _ = self.publish_status_array();
485    }

510    /// Phase 122.3.c.6.d — peek a pending cancel-goal request without
511    /// generating a reply. Returns the goal_id named in the request,
512    /// the matching service sequence number (use it with
513    /// [`send_cancel_reply`](Self::send_cancel_reply)), and the
514    /// goal's current status (`GoalStatus::Unknown` if no such
515    /// active goal). Returns `Ok(None)` when no cancel request is
516    /// pending.
517    ///
518    /// Used by L1 polling-mode action servers (nros-c / nros-cpp C
519    /// FFI) that want to drive cancel-decision policy without
520    /// passing a Rust closure across the C ABI. See the matching
521    /// [`send_cancel_reply`](Self::send_cancel_reply) for the reply
522    /// side. The high-level closure-based
523    /// [`try_handle_cancel`](Self::try_handle_cancel) keeps working
524    /// and now delegates to this pair.
525    pub fn try_recv_cancel_request(&mut self) -> Result<Option<PendingCancelRequest>, NodeError> {
526        let buf_start = self.cancel_buffer.as_ptr() as usize;
527        let request = match self
528            .cancel_goal_server
529            .try_recv_request(&mut self.cancel_buffer)
530        {
531            Ok(Some(r)) => r,
532            Ok(None) | Err(TransportError::NoData) => return Ok(None),
533            Err(_) => return Err(NodeError::Transport(TransportError::ServiceRequestFailed)),
534        };
535
536        let data_offset = (request.data.as_ptr() as usize).saturating_sub(buf_start);
537        let data_len = request.data.len();
538        let sequence_number = request.sequence_number;
539        #[allow(clippy::drop_non_drop)]
540        drop(request);
541
542        let mut reader =
543            CdrReader::new_with_header(&self.cancel_buffer[data_offset..data_offset + data_len])
544                .map_err(|_| NodeError::Transport(TransportError::DeserializationError))?;
545
546        let goal_id = read_goal_id(&mut reader)?;
547        let current_status = self.find_goal_status(&goal_id);
548
549        Ok(Some(PendingCancelRequest {
550            goal_id,
551            sequence_number,
552            current_status,
553        }))
554    }

556    /// Phase 122.3.c.6.d — send the reply to a previously-peeked
557    /// cancel-goal request. `sequence_number` must match the value
558    /// returned by [`try_recv_cancel_request`](Self::try_recv_cancel_request).
559    ///
560    /// `return_code` is the overall RPC status (`CancelResponse::Ok`
561    /// = at least one cancel honoured; other variants = whole-request
562    /// failure). `accepted` lists the goals that transition to
563    /// `Canceling`; this function flips their stored status before
564    /// publishing the status array.
565    pub fn send_cancel_reply(
566        &mut self,
567        sequence_number: i64,
568        return_code: nros_core::CancelResponse,
569        accepted: &[GoalId],
570    ) -> Result<(), NodeError> {
571        for id in accepted {
572            self.set_goal_status(id, GoalStatus::Canceling);
573        }
574
575        let mut writer =
576            crate::tx_writer(&mut self.goal_buffer).map_err(|_| NodeError::BufferTooSmall)?;
577        writer
578            .write_i8(return_code as i8)
579            .map_err(|_| NodeError::Serialization)?;
580        let count = u32::try_from(accepted.len()).unwrap_or(u32::MAX);
581        writer
582            .write_u32(count)
583            .map_err(|_| NodeError::Serialization)?;
584        for id in accepted {
585            write_goal_id(&mut writer, id)?;
586            // GoalInfo.stamp — zero timestamp.
587            writer.write_i32(0).map_err(|_| NodeError::Serialization)?;
588            writer.write_u32(0).map_err(|_| NodeError::Serialization)?;
589        }
590        let reply_len = writer.position();
591
592        self.cancel_goal_server
593            .send_reply(sequence_number, &self.goal_buffer[..reply_len])
594            .map_err(|_| NodeError::ServiceReplyFailed)?;
595
596        if !accepted.is_empty() {
597            let _ = self.publish_status_array();
598        }
599
600        Ok(())
601    }

603    /// Try to handle a cancel_goal request (type-agnostic).
604    pub fn try_handle_cancel(
605        &mut self,
606        cancel_handler: impl FnOnce(&GoalId, GoalStatus) -> nros_core::CancelResponse,
607    ) -> Result<Option<(GoalId, nros_core::CancelResponse)>, NodeError> {
608        let buf_start = self.cancel_buffer.as_ptr() as usize;
609        // Phase 120: NoData == steady-state idle; map to Ok(None).
610        let request = match self
611            .cancel_goal_server
612            .try_recv_request(&mut self.cancel_buffer)
613        {
614            Ok(Some(r)) => r,
615            Ok(None) | Err(TransportError::NoData) => return Ok(None),
616            Err(_) => return Err(NodeError::Transport(TransportError::ServiceRequestFailed)),
617        };
618
619        let data_offset = (request.data.as_ptr() as usize).saturating_sub(buf_start);
620        let data_len = request.data.len();
621        let sequence_number = request.sequence_number;
622        #[allow(clippy::drop_non_drop)]
623        drop(request);
624
625        let mut reader =
626            CdrReader::new_with_header(&self.cancel_buffer[data_offset..data_offset + data_len])
627                .map_err(|_| NodeError::Transport(TransportError::DeserializationError))?;
628
629        let goal_id = read_goal_id(&mut reader)?;
630
631        let current_status = self.find_goal_status(&goal_id);
632        let response = cancel_handler(&goal_id, current_status);
633
634        if response == nros_core::CancelResponse::Ok {
635            self.set_goal_status(&goal_id, GoalStatus::Canceling);
636        }
637
638        // Serialize response: return_code (i8) + goals_canceling (sequence of GoalInfo)
639        let mut writer =
640            crate::tx_writer(&mut self.goal_buffer).map_err(|_| NodeError::BufferTooSmall)?;
641        writer
642            .write_i8(response as i8)
643            .map_err(|_| NodeError::Serialization)?;
644
645        let num_canceling = if response == nros_core::CancelResponse::Ok {
646            1u32
647        } else {
648            0u32
649        };
650        writer
651            .write_u32(num_canceling)
652            .map_err(|_| NodeError::Serialization)?;
653        if response == nros_core::CancelResponse::Ok {
654            write_goal_id(&mut writer, &goal_id)?;
655            writer.write_i32(0).map_err(|_| NodeError::Serialization)?;
656            writer.write_u32(0).map_err(|_| NodeError::Serialization)?;
657        }
658        let reply_len = writer.position();
659
660        self.cancel_goal_server
661            .send_reply(sequence_number, &self.goal_buffer[..reply_len])
662            .map_err(|_| NodeError::ServiceReplyFailed)?;
663
664        Ok(Some((goal_id, response)))
665    }

667    /// Try to handle a get_result request using raw bytes.
668    ///
669    /// For completed goals, sends the stored raw result CDR from the slab.
670    /// For active/unknown goals, sends the provided `default_result_cdr` bytes.
671    ///
672    /// `default_result_cdr` should contain serialized result data (without CDR
673    /// header or status byte) — typically `A::Result::default()` serialized.
674    pub fn try_handle_get_result_raw(
675        &mut self,
676        default_result_cdr: &[u8],
677    ) -> Result<Option<GoalId>, NodeError> {
678        let buf_start = self.goal_buffer.as_ptr() as usize;
679        // Phase 120: NoData == steady-state idle; map to Ok(None).
680        let request = match self
681            .get_result_server
682            .try_recv_request(&mut self.goal_buffer)
683        {
684            Ok(Some(r)) => r,
685            Ok(None) | Err(TransportError::NoData) => return Ok(None),
686            Err(_) => return Err(NodeError::Transport(TransportError::ServiceRequestFailed)),
687        };
688
689        let data_offset = (request.data.as_ptr() as usize).saturating_sub(buf_start);
690        let data_len = request.data.len();
691        let sequence_number = request.sequence_number;
692        #[allow(clippy::drop_non_drop)]
693        drop(request);
694
695        let mut reader =
696            CdrReader::new_with_header(&self.goal_buffer[data_offset..data_offset + data_len])
697                .map_err(|_| NodeError::Transport(TransportError::DeserializationError))?;
698
699        let goal_id = read_goal_id(&mut reader)?;
700
701        // Look up in completed results
702        let completed = self
703            .completed_results
704            .iter()
705            .find(|c| c.goal_id.uuid == goal_id.uuid);
706
707        if let Some(entry) = completed {
708            // Completed: send status + stored result CDR from the slab.
709            let (off, len, status) = (entry.offset, entry.len, entry.status);
710            self.reply_get_result_from_slab(sequence_number, status, off, len)?;
711        } else if self
712            .active_goals
713            .iter()
714            .any(|g| g.goal_id.uuid == goal_id.uuid)
715        {
716            // Active goal → DEFER (Phase 237). `rclcpp_action` sends get_result
717            // right after acceptance and expects the reply only once the goal
718            // terminates; replying now with a non-terminal status makes the
719            // client treat an unfinished goal as done. Hold the request's
720            // correlation token; `complete_goal_raw` flushes it. The backend
721            // retains the reply token keyed by `sequence_number`.
722            if self
723                .pending_get_results
724                .push(PendingGetResult {
725                    goal_id,
726                    sequence_number,
727                })
728                .is_err()
729            {
730                // Table full — fail loud rather than silently strand the
731                // requester (caller surfaces it; the request is not re-queued).
732                return Err(NodeError::BufferTooSmall);
733            }
734        } else {
735            // Unknown goal → reply immediately with UNKNOWN + default result.
736            let mut writer =
737                crate::tx_writer(&mut self.goal_buffer).map_err(|_| NodeError::BufferTooSmall)?;
738            writer
739                .write_i8(GoalStatus::Unknown as i8)
740                .map_err(|_| NodeError::Serialization)?;
741            writer.align(4).map_err(|_| NodeError::Serialization)?;
742            let pos = writer.position();
743            if pos + default_result_cdr.len() > GOAL_BUF {
744                return Err(NodeError::BufferTooSmall);
745            }
746            self.goal_buffer[pos..pos + default_result_cdr.len()]
747                .copy_from_slice(default_result_cdr);
748            let reply_len = pos + default_result_cdr.len();
749
750            self.get_result_server
751                .send_reply(sequence_number, &self.goal_buffer[..reply_len])
752                .map_err(|_| NodeError::ServiceReplyFailed)?;
753        }
754
755        Ok(Some(goal_id))
756    }
757
758    /// Build + send a `get_result` reply — `[status i8][align(4)][result CDR]`
759    /// — copying the result bytes from the slab. Shared by the immediate
760    /// completed-goal path and the deferred flush in `complete_goal_raw`
761    /// (Phase 237). `goal_buffer` and `result_slab` are disjoint fields, so the
762    /// header build (into `goal_buffer`) and the slab copy don't alias.
763    fn reply_get_result_from_slab(
764        &mut self,
765        sequence_number: i64,
766        status: GoalStatus,
767        slab_offset: usize,
768        slab_len: usize,
769    ) -> Result<(), NodeError> {
770        let mut writer =
771            crate::tx_writer(&mut self.goal_buffer).map_err(|_| NodeError::BufferTooSmall)?;
772        writer
773            .write_i8(status as i8)
774            .map_err(|_| NodeError::Serialization)?;
775        // Align to 4 after the status byte: the result CDR starts with a u32
776        // sequence length the reader will `align(4)` to.
777        writer.align(4).map_err(|_| NodeError::Serialization)?;
778        let pos = writer.position();
779        if pos + slab_len > GOAL_BUF {
780            return Err(NodeError::BufferTooSmall);
781        }
782        self.goal_buffer[pos..pos + slab_len]
783            .copy_from_slice(&self.result_slab[slab_offset..slab_offset + slab_len]);
784        let reply_len = pos + slab_len;
785        self.get_result_server
786            .send_reply(sequence_number, &self.goal_buffer[..reply_len])
787            .map_err(|_| NodeError::ServiceReplyFailed)
788    }

809    /// Publish the current GoalStatusArray on the status topic.
810    pub fn publish_status_array(&self) -> Result<(), NodeError> {
811        let mut buf = [0u8; STATUS_ARRAY_BUF];
812        let mut writer = crate::tx_writer(&mut buf).map_err(|_| NodeError::BufferTooSmall)?;
813
814        writer
815            .write_u32(self.active_goals.len() as u32)
816            .map_err(|_| NodeError::Serialization)?;
817
818        for goal in &self.active_goals {
819            let stamped = GoalStatusStamped::new(GoalInfo::with_id(goal.goal_id), goal.status);
820            stamped
821                .serialize(&mut writer)
822                .map_err(|_| NodeError::Serialization)?;
823        }
824
825        let len = writer.position();
826        self.status_publisher
827            .publish_raw(&buf[..len])
828            .map_err(|_| NodeError::Transport(TransportError::PublishFailed))
829    }

914    /// Send a goal with raw CDR bytes. Returns the generated GoalId.
915    ///
916    /// The `goal_cdr` bytes are the serialized goal data (without GoalId framing).
917    /// This writes GoalId + goal_cdr into the goal buffer and sends the request.
918    ///
919    /// After calling, use `send_goal_client` and `result_buffer` to construct
920    /// a Promise for the acceptance reply.
921    pub fn send_goal_raw(&mut self, goal_cdr: &[u8]) -> Result<GoalId, NodeError> {
922        self.goal_counter += 1;
923        let mut goal_id = GoalId::default();
924        let counter_bytes = self.goal_counter.to_le_bytes();
925        goal_id.uuid[..8].copy_from_slice(&counter_bytes);
926
927        let mut writer =
928            crate::tx_writer(&mut self.goal_buffer).map_err(|_| NodeError::BufferTooSmall)?;
929
930        write_goal_id(&mut writer, &goal_id)?;
931
932        // Copy raw goal CDR bytes after GoalId
933        let pos = writer.position();
934        if pos + goal_cdr.len() > GOAL_BUF {
935            return Err(NodeError::BufferTooSmall);
936        }
937        self.goal_buffer[pos..pos + goal_cdr.len()].copy_from_slice(goal_cdr);
938        let req_len = pos + goal_cdr.len();
939
940        self.send_goal_client
941            .send_request_raw(&self.goal_buffer[..req_len])
942            .map_err(|_| NodeError::ServiceRequestFailed)?;
943
944        Ok(goal_id)
945    }
