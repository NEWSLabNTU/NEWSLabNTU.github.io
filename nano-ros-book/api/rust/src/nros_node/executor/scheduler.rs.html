4//! phase-509 P6 (RFC-0108 D3, D7; amended RFC-0107 step 4) -- the scheduler /
5//! worker split.
6//!
7//! RFC-0107's per-tier dispatcher was one value that both ORDERED ready work
8//! and was the unit of PREEMPTION. RFC-0108 splits it in two:
9//!
10//! - a per-core [`Scheduler`]: the ordered ready set over the core's callback
11//!   slots, each slot's dispatch level, and the band table that says which
12//!   worker owns which levels;
13//! - a per-level [`Worker`]: one contiguous [`Band`] of dispatch levels, its
14//!   park (RFC-0106: the deadline sources, the primitive that blocks, and
15//!   what the last park did) and its stack record (the headroom rule's
16//!   bound and state). A worker is the thread that runs its band.
17//!
18//! **One level is today's executor.** [`Executor`](super::Executor) stays the
19//! one-level facade: it holds one [`Worker`] whose band is every level
20//! ([`Band::ALL`]), so its dispatch order, its park and its stack rule are the
21//! ones it always had. The worker's fields are the park and stack fields the
22//! executor used to hold inline, moved and not changed.
23//!
24//! **Preemption between bands is the kernel's** (RFC-0108 D6). Each worker
25//! runs at its band's kernel priority and picks only its own band, most
26//! urgent level first, registration order within a level. On one core a less
27//! urgent worker runs only while every more urgent one is parked, so an
28//! arrival in a more urgent band preempts a running callback of a less urgent
29//! one, and the members of one band never preempt each other: an arrival in
30//! the band of the running worker waits for its current callback to return.
31//!
32//! **Waking (D7).** An arrival the transport attributes to its slot
33//! (RFC-0084) wakes only the worker of that slot's band
34//! ([`Scheduler::arrive`]). An unattributed arrival can only say "something
35//! on this session", so it wakes the most urgent parked worker, which
36//! re-picks ([`Scheduler::arrive_unattributed`]).
37//!
38//! The scheduler's run-time half takes `&self` and is lock-free (one atomic
39//! word of ready bits, one parked flag per worker), so the workers of one core
40//! and the transport's arrival path may all reach it at once. Its set-up half
41//! ([`Scheduler::add_worker`], [`Scheduler::bind_slot`]) takes `&mut self`
42//! and runs before any worker starts.
43//!
44//! Crate-private until the level runners (phase-509 P9) open a scheduler and
45//! its levels through the FFI; until then the executor uses only its one
46//! [`Worker`], and the [`Scheduler`] is driven by its own tests.
47
48// The multi-level half has no in-crate caller before P9.
49#![cfg_attr(not(test), allow(dead_code))]

59/// The least urgent dispatch level ([`Priority::dispatch_level`]: `Critical`
60/// 0, `Normal` 1 + rank, `BestEffort` 31).
61///
62/// [`Priority::dispatch_level`]: super::sched_context::Priority::dispatch_level
63pub const MAX_DISPATCH_LEVEL: u8 = 31;

65/// Callback slots one scheduler orders: one machine word of ready bits, as
66/// the entity wake's ready word (`entity_wake::SLOTS`).
67pub const SCHEDULER_SLOTS: usize = usize::BITS as usize;

69/// Workers (preemption levels) one scheduler can host: the
70/// `CONFIG_NROS_EXECUTOR_MAX_LEVELS` range's upper end.
71pub const MAX_WORKERS: usize = 32;

73/// A slot bound to no level yet.
74const UNBOUND: u8 = u8::MAX;

76/// A contiguous range of dispatch levels, inclusive at both ends; level 0 is
77/// the most urgent.
78#[derive(Clone, Copy, Debug, PartialEq, Eq)]
79pub struct Band {
80    /// The most urgent level in the band: its threshold (RFC-0108 D3).
81    pub most_urgent: u8,
82    /// The least urgent level in the band: its floor.
83    pub least_urgent: u8,
84}
85
86impl Band {
87    /// Every level: the one-level executor.
88    pub const ALL: Band = Band {
89        most_urgent: 0,
90        least_urgent: MAX_DISPATCH_LEVEL,
91    };
92
93    /// The band `[most_urgent, least_urgent]`, or `None` when it is empty or
94    /// reaches past the least urgent dispatch level (31).
95    pub const fn new(most_urgent: u8, least_urgent: u8) -> Option<Band> {
96        if most_urgent > least_urgent || least_urgent > MAX_DISPATCH_LEVEL {
97            None
98        } else {
99            Some(Band {
100                most_urgent,
101                least_urgent,
102            })
103        }
104    }
105
106    /// Whether `level` is in the band.
107    pub const fn contains(&self, level: u8) -> bool {
108        level >= self.most_urgent && level <= self.least_urgent
109    }
110}

112/// One preemption level: its band, its park and its stack record.
113///
114/// The park (RFC-0106: the platform deadline sources, the one primitive that
115/// blocks, and what the last park asked for and got) and the stack record
116/// (the headroom rule's declared bound and state, phase-436 E3/E4) moved here
117/// from [`Executor`](super::Executor) unchanged; the executor's park and
118/// stack methods read and write its worker's. The fields stay flat, not in a
119/// park and a stack sub-struct, so their small members share padding as they
120/// did inline in the executor.
121pub(crate) struct Worker {
122    /// The level's index in the image, 0 the most urgent (phase-509 P7):
123    /// what [`Executor::level`](super::Executor::level) answers and the
124    /// level its spins check in as.
125    pub(crate) level: u8,
126    /// The dispatch levels this worker runs.
127    pub(crate) band: Band,
128    // ---- the park (RFC-0106) ----
129    /// phase-436 -- the bound the last park was computed with, in us, and the
130    /// source that won it. See `WakeSourceId`.
131    pub(crate) last_park_bound_us: u64,
132    pub(crate) last_park_source: WakeSourceId,
133    /// phase-436 W2 -- the park bound after rounding UP to what the
134    /// platform's wake primitive can express. Kept beside the request rather
135    /// than replacing it: reporting both is what makes the rounding auditable
136    /// instead of silent (issue 1194).
137    pub(crate) last_park_achieved_us: u64,
138    /// phase-436 W6 -- registered platform deadline sources, and how many are
139    /// live. Inline table: no allocator on the targets this exists for.
140    pub(crate) wake_sources: [Option<WakeSourceSlot>; MAX_WAKE_SOURCES],
141    /// A `u8` (it was a `usize` inline in the executor) so it shares the
142    /// worker's small fields' padding: `MAX_WAKE_SOURCES` is far below 256.
143    pub(crate) wake_source_count: u8,
144    /// The single thing that actually blocks. Many sources say WHEN; one
145    /// primitive does the waiting.
146    pub(crate) park_primitive: Option<(ParkUntilFn, *mut c_void)>,
147    /// phase-436 W7.b (issue 1242) -- the finest park the INSTALLED primitive
148    /// can actually express, in microseconds. `0` = none installed, fall back
149    /// to the `wake_wait_ms` ABI floor.
150    pub(crate) park_granularity_declared_us: u64,
151    // ---- the stack record (phase-436 E3/E4) ----
152    /// Declared minimum stack headroom in bytes for the thread this worker
153    /// runs on; `0` (default) disables the rule. The entry that spawned the
154    /// thread is the one that knows what it gave it.
155    pub(crate) min_stack_headroom_bytes: usize,
156    /// Lowest headroom already reported, so the rule fires on new lows only.
157    /// `usize::MAX` = nothing reported yet.
158    pub(crate) stack_headroom_reported: usize,
159    /// phase-436 E3 -- a minimum is set but the port reports no stack data,
160    /// so the rule will skip every tick.
161    pub(crate) stack_headroom_blind: bool,
162    /// phase-436 E4 -- when the headroom query last ran (`None` = never; a
163    /// clockless build stores `Some(0)` to mean "has run"), spins since, and
164    /// how many queries actually reached the port.
165    pub(crate) stack_headroom_last_check_us: Option<u64>,
166    pub(crate) stack_headroom_spins_since_check: u32,
167    pub(crate) stack_headroom_checks: u32,
168}

194/// phase-509 P9b item 3 -- how a band worker waits on its own wake object:
195/// at most `us` microseconds, returning early when the object is signalled
196/// (`WorkerWakeFn`). The answer is ignored: either way the worker re-scans.
197pub type BandWaitFn = unsafe extern "C" fn(ctx: *mut c_void, us: u64) -> i32;
198
199/// phase-509 P9b item 3 (RFC-0108 D7, "As built in P9b") -- a band worker's
200/// wake object and the slots it answers for. The transport's arrival path
201/// reads it without the executor's lock, so it is `'static` and its fields
202/// are atomics or fixed at construction; it is linked into the executor's
203/// arrival routing by [`Executor::attach_band_worker`](super::Executor::attach_band_worker)
204/// and never unlinked.
205pub struct BandWaiter {
206    /// The band's slots, as the worker's last pass saw them. An arrival on
207    /// one of them wakes this worker only.
208    pub(crate) slots: AtomicUsize,
209    /// Signals the wake object (`k_sem_give`, a futex post).
210    #[cfg_attr(not(all(feature = "alloc", feature = "rmw-cffi")), allow(dead_code))]
211    // the arrival path
212    pub(crate) wake: WorkerWakeFn,
213    /// Waits on it, bounded.
214    pub(crate) wait: BandWaitFn,
215    pub(crate) ctx: *mut c_void,
216    /// The next waiter of the same executor.
217    #[cfg_attr(not(all(feature = "alloc", feature = "rmw-cffi")), allow(dead_code))]
218    // the arrival path
219    pub(crate) next: portable_atomic::AtomicPtr<BandWaiter>,
220}
221
222// SAFETY: `ctx` is the port's wake object, handed in with the two functions
223// that use it, callable from any thread (as `Scheduler::add_worker`'s wake);
224// everything else is atomic.
225unsafe impl Sync for BandWaiter {}
226// SAFETY: as above.
227unsafe impl Send for BandWaiter {}
228
229impl BandWaiter {
230    /// A waiter over the port's wake object at `ctx`.
231    ///
232    /// # Safety
233    /// `wake` and `wait` are callable from any thread with `ctx`, which
234    /// outlives every executor the waiter is attached to.
235    pub const unsafe fn new(wake: WorkerWakeFn, wait: BandWaitFn, ctx: *mut c_void) -> Self {
236        Self {
237            slots: AtomicUsize::new(0),
238            wake,
239            wait,
240            ctx,
241            next: portable_atomic::AtomicPtr::new(core::ptr::null_mut()),
242        }
243    }
244
245    /// Signal the worker.
246    #[cfg_attr(not(all(feature = "alloc", feature = "rmw-cffi")), allow(dead_code))] // the arrival path
247    pub(crate) fn signal(&self) {
248        // SAFETY: `new`'s contract.
249        unsafe { (self.wake)(self.ctx) };
250    }
251
252    /// Wait at most `us` microseconds, or until signalled.
253    pub(crate) fn wait_us(&self, us: u64) {
254        // SAFETY: `new`'s contract.
255        let _ = unsafe { (self.wait)(self.ctx, us) };
256    }
257}

259/// phase-509 P9b item 2 (RFC-0108 "As built in P9b") -- a worker of one band
260/// over an executor it shares with other workers: the record its own thread
261/// owns and hands to [`Executor::spin_band_once`](super::Executor::spin_band_once).
262///
263/// A band pass runs only the band's callbacks, and credits the band's timers
264/// with the time since this worker's own previous pass, so two workers never
265/// split one timer's elapsed time between them. With a [`BandWaiter`]
266/// (item 3) it parks on its own wake object until its band's next timer is
267/// due or an arrival in its band signals it.
268#[derive(Clone, Copy, Debug)]
269pub struct BandWorker {
270    /// The level's index in the image, 0 the most urgent: what its passes
271    /// check in as (`level_liveness`).
272    pub(crate) level: u8,
273    /// The dispatch levels this worker runs.
274    pub(crate) band: Band,
275    /// When this worker's previous pass read the clock; `None` before its
276    /// first.
277    pub(crate) last_spin_end_us: Option<u64>,
278    /// phase-509 P9b item 3 -- its wake object, if it parks.
279    pub(crate) waiter: Option<&'static BandWaiter>,
280}

321/// A worker's index in its scheduler, 0 the most urgent.
322#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]

325/// Why a scheduler refused a set-up step.
326#[derive(Clone, Copy, Debug, PartialEq, Eq)]
327pub enum LevelSetupError {
328    /// [`MAX_WORKERS`] workers are already added.
329    TooManyWorkers,
330    /// The band does not start right after the previous worker's (the first
331    /// at level 0): bands partition the levels, most urgent first.
332    NotContiguous { expected_most_urgent: u8, got: u8 },
333    /// The slot is past [`SCHEDULER_SLOTS`], or the level past
334    /// [`MAX_DISPATCH_LEVEL`].
335    OutOfRange,
336}
337
338struct Seat {
339    band: Band,
340    wake: Option<(WorkerWakeFn, *mut c_void)>,
341    parked: AtomicBool,
342}

344/// The per-core half: the ordered ready set and the band table.
345pub struct Scheduler {
346    /// Each slot's dispatch level; [`UNBOUND`] when no callback is bound.
347    slot_level: [AtomicU8; SCHEDULER_SLOTS],
348    /// One bit per slot with work queued.
349    ready: AtomicUsize,
350    seats: [Option<Seat>; MAX_WORKERS],
351    workers: u8,
352}
353
354// SAFETY: the raw wake contexts are handed in by the port together with the
355// function that reads them, and must be callable from any thread (the same
356// contract as a backend's wake callback). Everything else is atomic or set up
357// through `&mut self` before the workers start.
358unsafe impl Sync for Scheduler {}
359// SAFETY: as above; nothing in it is tied to the creating thread.
360unsafe impl Send for Scheduler {}
361
362impl Default for Scheduler {
363    fn default() -> Self {
364        Self::new()
365    }
366}
367
368impl Scheduler {
369    pub const fn new() -> Self {
370        Self {
371            slot_level: [const { AtomicU8::new(UNBOUND) }; SCHEDULER_SLOTS],
372            ready: AtomicUsize::new(0),
373            seats: [const { None }; MAX_WORKERS],
374            workers: 0,
375        }
376    }
377
378    /// Add the next worker, less urgent than every one added before it. Its
379    /// band must start right after the previous band's floor (the first at
380    /// level 0): the bands partition the levels and are contiguous (RFC-0108
381    /// D3). `wake` signals its wake object; `None` for a worker nothing has
382    /// to wake (a test, or a worker that polls).
383    pub fn add_worker(
384        &mut self,
385        band: Band,
386        wake: Option<(WorkerWakeFn, *mut c_void)>,
387    ) -> Result<WorkerId, LevelSetupError> {
388        let idx = self.workers as usize;
389        if idx >= MAX_WORKERS {
390            return Err(LevelSetupError::TooManyWorkers);
391        }
392        let expected = match idx {
393            0 => 0,
394            _ => match &self.seats[idx - 1] {
395                Some(prev) => prev.band.least_urgent + 1,
396                None => 0,
397            },
398        };
399        if band.most_urgent != expected {
400            return Err(LevelSetupError::NotContiguous {
401                expected_most_urgent: expected,
402                got: band.most_urgent,
403            });
404        }
405        self.seats[idx] = Some(Seat {
406            band,
407            wake,
408            parked: AtomicBool::new(false),
409        });
410        self.workers += 1;
411        Ok(WorkerId(idx as u8))
412    }
413
414    /// Workers added.
415    pub fn worker_count(&self) -> usize {
416        self.workers as usize
417    }
418
419    /// Bind callback `slot` to dispatch `level`.
420    pub fn bind_slot(&mut self, slot: usize, level: u8) -> Result<(), LevelSetupError> {
421        if slot >= SCHEDULER_SLOTS || level > MAX_DISPATCH_LEVEL {
422            return Err(LevelSetupError::OutOfRange);
423        }
424        self.slot_level[slot].store(level, SeqCst);
425        Ok(())
426    }
427
428    /// The worker whose band holds `level`; `None` when no band does.
429    pub fn worker_of(&self, level: u8) -> Option<WorkerId> {
430        self.seats[..self.workers as usize]
431            .iter()
432            .position(|s| s.as_ref().is_some_and(|s| s.band.contains(level)))
433            .map(|i| WorkerId(i as u8))
434    }
435
436    /// The band `worker` runs.
437    pub fn band(&self, worker: WorkerId) -> Option<Band> {
438        self.seat(worker).map(|s| s.band)
439    }
440
441    fn seat(&self, worker: WorkerId) -> Option<&Seat> {
442        self.seats
443            .get(worker.0 as usize)
444            .and_then(Option::as_ref)
445            .filter(|_| (worker.0) < self.workers)
446    }
447
448    fn wake(&self, worker: WorkerId) {
449        if let Some(seat) = self.seat(worker)
450            && let Some((wake, ctx)) = seat.wake
451        {
452            // SAFETY: the port handed `wake` and `ctx` in together, callable
453            // from any thread (see `add_worker`).
454            unsafe { wake(ctx) };
455        }
456    }
457
458    /// RFC-0108 D7: an arrival attributed to `slot` queues it and wakes only
459    /// the worker of its band. Returns that worker; `None` when the slot is
460    /// bound to no level a band holds, which is then woken as an
461    /// unattributed arrival would be.
462    pub fn arrive(&self, slot: usize) -> Option<WorkerId> {
463        let Some(level) = self.slot_level.get(slot).map(|l| l.load(SeqCst)) else {
464            return self.arrive_unattributed();
465        };
466        // The bit first, so a worker the wake reaches sees it.
467        self.ready.fetch_or(1usize << slot, SeqCst);
468        match self.worker_of(level) {
469            Some(w) => {
470                self.wake(w);
471                Some(w)
472            }
473            None => self.arrive_unattributed(),
474        }
475    }
476
477    /// RFC-0108 D7: an arrival nothing attributes wakes the most urgent
478    /// parked worker, which re-picks. `None` when no worker is parked: a
479    /// running worker re-picks after its current callback anyway.
480    pub fn arrive_unattributed(&self) -> Option<WorkerId> {
481        let w = self.seats[..self.workers as usize]
482            .iter()
483            .position(|s| s.as_ref().is_some_and(|s| s.parked.load(SeqCst)))
484            .map(|i| WorkerId(i as u8))?;
485        self.wake(w);
486        Some(w)
487    }
488
489    /// Record that `worker` is parked (`true`) or running (`false`).
490    pub fn set_parked(&self, worker: WorkerId, parked: bool) {
491        if let Some(seat) = self.seat(worker) {
492            seat.parked.store(parked, SeqCst);
493        }
494    }
495
496    /// Whether `worker` is parked.
497    pub fn is_parked(&self, worker: WorkerId) -> bool {
498        self.seat(worker).is_some_and(|s| s.parked.load(SeqCst))
499    }
500
501    /// Whether any slot in `worker`'s band has work queued.
502    pub fn has_work(&self, worker: WorkerId) -> bool {
503        let Some(band) = self.band(worker) else {
504            return false;
505        };
506        let ready = self.ready.load(SeqCst);
507        (0..SCHEDULER_SLOTS)
508            .any(|s| ready & (1usize << s) != 0 && band.contains(self.slot_level[s].load(SeqCst)))
509    }
510
511    /// Take the most urgent queued slot of `worker`'s band: the lowest
512    /// dispatch level, registration (slot) order within a level. `None`
513    /// when its band has nothing queued.
514    pub fn pick(&self, worker: WorkerId) -> Option<usize> {
515        let band = self.band(worker)?;
516        loop {
517            let ready = self.ready.load(SeqCst);
518            let mut best: Option<(u8, usize)> = None;
519            let mut bits = ready;
520            while bits != 0 {
521                let s = bits.trailing_zeros() as usize;
522                bits &= bits - 1;
523                let level = self.slot_level[s].load(SeqCst);
524                if band.contains(level) && best.is_none_or(|(l, _)| level < l) {
525                    best = Some((level, s));
526                }
527            }
528            let (_, slot) = best?;
529            let bit = 1usize << slot;
530            if self
531                .ready
532                .compare_exchange(ready, ready & !bit, SeqCst, SeqCst)
533                .is_ok()
534            {
535                return Some(slot);
536            }
537        }
538    }
539
540    /// Run `worker`'s band until it has nothing queued: pick, dispatch, pick
541    /// again from the ready set as it is now (RFC-0108 D2). Returns how many
542    /// callbacks ran.
543    pub fn run(&self, worker: WorkerId, mut dispatch: impl FnMut(usize)) -> usize {
544        let mut ran = 0;
545        while let Some(slot) = self.pick(worker) {
546            dispatch(slot);
547            ran += 1;
548        }
549        ran
550    }
551}

609    /// D7: an attributed arrival wakes only its band's worker; an
610    /// unattributed one the most urgent parked worker.
611    #[test]
612    fn an_arrival_wakes_only_its_bands_worker() {
613        static WOKEN: [AtomicU8; 2] = [AtomicU8::new(0), AtomicU8::new(0)];
614        unsafe extern "C" fn wake(ctx: *mut c_void) {
615            WOKEN[ctx as usize].fetch_add(1, SeqCst);
616        }
617        let mut s = Scheduler::new();
618        s.add_worker(Band::new(0, 1).unwrap(), Some((wake, ctx(0))))
619            .unwrap();
620        s.add_worker(Band::new(2, 31).unwrap(), Some((wake, ctx(1))))
621            .unwrap();
622        s.bind_slot(0, 1).unwrap();
623        s.bind_slot(1, 5).unwrap();
624        assert_eq!(s.arrive(1), Some(WorkerId(1)));
625        assert_eq!(
626            (WOKEN[0].load(SeqCst), WOKEN[1].load(SeqCst)),
627            (0, 1),
628            "a band-1 arrival does not wake band 0"
629        );
630        assert_eq!(s.arrive(0), Some(WorkerId(0)));
631        assert_eq!((WOKEN[0].load(SeqCst), WOKEN[1].load(SeqCst)), (1, 1));
632        // Nobody parked: an unattributed arrival wakes nobody (each running
633        // worker re-picks after its callback).
634        assert_eq!(s.arrive_unattributed(), None);
635        s.set_parked(WorkerId(1), true);
636        assert!(s.is_parked(WorkerId(1)) && !s.is_parked(WorkerId(0)));
637        assert_eq!(s.arrive_unattributed(), Some(WorkerId(1)));
638        s.set_parked(WorkerId(0), true);
639        assert_eq!(s.arrive_unattributed(), Some(WorkerId(0)));
640        // Each worker picks only its own band.
641        assert_eq!(s.pick(WorkerId(0)), Some(0));
642        assert_eq!(s.pick(WorkerId(0)), None);
643        assert!(s.has_work(WorkerId(1)));
644        assert_eq!(s.pick(WorkerId(1)), Some(1));
645    }

647    // ---- a deterministic one-core kernel -------------------------------
648    //
649    // Fixed-priority preemptive, one core, no time: a woken worker more
650    // urgent than the one running runs at once (nested inside the running
651    // callback, which is where a real kernel would preempt it); any other
652    // wake is pending until the running worker returns. Each callback is a
653    // script of arrivals it raises mid-run.
654
655    struct Kernel {
656        sched: &'static Scheduler,
657        running: Vec<WorkerId>,
658        pending: Vec<WorkerId>,
659        scripts: Vec<Vec<usize>>,
660        trace: Vec<String>,
661    }
662
663    std::thread_local! {
664        static KERNEL: RefCell<Option<Kernel>> = const { RefCell::new(None) };
665    }
666
667    fn with<R>(f: impl FnOnce(&mut Kernel) -> R) -> R {
668        KERNEL.with(|k| f(k.borrow_mut().as_mut().expect("kernel installed")))
669    }
670
671    unsafe extern "C" fn kernel_wake(ctx: *mut c_void) {
672        let w = WorkerId(ctx as usize as u8);
673        let preempts = with(|k| {
674            let current = k.running.last().copied();
675            if current.is_none_or(|c| w < c) {
676                true
677            } else {
678                if !k.pending.contains(&w) && current != Some(w) {
679                    k.pending.push(w);
680                }
681                false
682            }
683        });
684        if preempts {
685            run_worker(w);
686        }
687    }
688
689    fn run_worker(w: WorkerId) {
690        let sched = with(|k| {
691            k.running.push(w);
692            k.sched
693        });
694        sched.set_parked(w, false);
695        sched.run(w, |slot| {
696            let script = with(|k| {
697                k.trace.push(format!("start {slot}"));
698                k.scripts[slot].clone()
699            });
700            for raised in script {
701                sched.arrive(raised);
702            }
703            with(|k| k.trace.push(format!("end {slot}")));
704        });
705        sched.set_parked(w, true);
706        with(|k| k.running.pop());
707        // The kernel's next pick: the most urgent pending worker.
708        loop {
709            let next = with(|k| {
710                k.pending.sort();
711                if k.pending.is_empty() {
712                    None
713                } else {
714                    Some(k.pending.remove(0))
715                }
716            });
717            match next {
718                Some(n) => run_worker(n),
719                None => break,
720            }
721        }
722    }
723
724    /// Band 0 (levels 0-1) holds `H`; band 1 (levels 2-31) holds `A` and
725    /// `B`, `B` the more urgent of the two. Slots: H 0, A 1, B 2.
726    fn two_bands(scripts: Vec<Vec<usize>>) -> &'static Scheduler {
727        let mut s = Scheduler::new();
728        s.add_worker(Band::new(0, 1).unwrap(), Some((kernel_wake, ctx(0))))
729            .unwrap();
730        s.add_worker(Band::new(2, 31).unwrap(), Some((kernel_wake, ctx(1))))
731            .unwrap();
732        s.bind_slot(0, 1).unwrap(); // H
733        s.bind_slot(1, 6).unwrap(); // A
734        s.bind_slot(2, 4).unwrap(); // B
735        let s: &'static Scheduler = Box::leak(Box::new(s));
736        s.set_parked(WorkerId(0), true);
737        s.set_parked(WorkerId(1), true);
738        KERNEL.with(|k| {
739            *k.borrow_mut() = Some(Kernel {
740                sched: s,
741                running: vec![],
742                pending: vec![],
743                scripts,
744                trace: vec![],
745            })
746        });
747        s
748    }
749
750    fn trace() -> Vec<String> {
751        with(|k| k.trace.clone())
752    }
