1//! phase-436 -- one backend wake slot, several executors.
2//!
3//! A backend session has ONE wake-callback slot (`Session::set_wake_callback`).
4//! Before this module, the executor that installed into it was the only one an
5//! arrival could wake. Every board's spawned tiers open over the boot
6//! executor's session (`Executor::open_with_session_*`), and several C
7//! executors can share one `rclc_support_t` session. So those executors' wake
8//! objects were never signalled. With a park primitive installed they slept
9//! out their whole bound before seeing data, and without one they fell back
10//! to driving the transport for the full timeout.
11//!
12//! A [`WakeFanout`] is what the backend's slot holds instead: one callback
13//! whose context is a fixed table of member wake contexts, each of which it
14//! signals. The table is a `static` pool, so the pointer a backend holds can
15//! never dangle, whichever executor leaves first.
16//!
17//! Lock-free, so the callback may run from a backend worker thread or an ISR:
18//!
19//! * the callback counts itself IN before it reads a slot, and OUT after;
20//! * a member leaving clears its slot, then waits (yielding) until no callback
21//!   is in flight. A callback that started after the clear cannot see the
22//!   member; one that started before it is waited for. Both orders are
23//!   `SeqCst`.
24//!
25//! An entry is keyed by the session's address. The owner of an OWNED session
26//! claims an entry while its executor value may still move, so it re-keys the
27//! entry when it hands the session's final address out
28//! (`Executor::session_ptr`), which is what every tier is opened over.
29//!
30//! The cost is that every arrival wakes every member: the callback does not
31//! say which executor the data is for. A member woken for another's data
32//! drains its sessions non-blockingly and parks again.
33
34// The executor's integration exists only with `alloc` + `rmw-cffi`; the
35// table itself is also compiled for the unit tests below.
36#![cfg_attr(not(all(feature = "alloc", feature = "rmw-cffi")), allow(dead_code))]
37
38use core::ffi::c_void;
39use portable_atomic::{AtomicBool, AtomicPtr, AtomicU32, Ordering::SeqCst};
40
41/// Sessions that can be shared at once. Exhausting the pool is not an error:
42/// the executor falls back to installing its own context directly, which is
43/// the behaviour every executor had before this module.
44pub(crate) const MAX_FANOUTS: usize = 4;
45/// Executors per shared session: the boot executor plus its spawned tiers,
46/// or the C executors over one support context. A member past this does not
47/// join and is treated as having no async wake, so it never parks blind.
48pub(crate) const MAX_FANOUT_MEMBERS: usize = 8;
49
50/// What a member's context is signalled with (`nros_rmw_runtime_wake_cb` in a
51/// real build; a test's own function in a unit test).
52pub(crate) type WakeFn = unsafe extern "C" fn(*mut c_void);
53
54pub(crate) struct WakeFanout {
55    /// The session this entry serves; null = free.
56    key: AtomicPtr<c_void>,
57    /// Bumped on claim and on release, so a membership taken from an earlier
58    /// tenancy of this entry can neither leave a later tenant's slot nor read
59    /// it as live.
60    generation: AtomicU32,
61    /// The backend honours the callback the entry is installed as
62    /// (`Session::supports_wake_callback` at install time).
63    live: AtomicBool,
64    inflight: AtomicU32,
65    members: [AtomicPtr<c_void>; MAX_FANOUT_MEMBERS],
66}
67
68impl WakeFanout {
69    const fn new() -> Self {
70        Self {
71            key: AtomicPtr::new(core::ptr::null_mut()),
72            generation: AtomicU32::new(0),
73            live: AtomicBool::new(false),
74            inflight: AtomicU32::new(0),
75            members: [const { AtomicPtr::new(core::ptr::null_mut()) }; MAX_FANOUT_MEMBERS],
76        }
77    }
78
79    /// The pointer to hand a backend as the callback context.
80    pub(crate) fn as_ctx(&'static self) -> *mut c_void {
81        self as *const Self as *mut c_void
82    }
83
84    /// Signal every member with `wake`. What the backend's callback does.
85    pub(crate) fn fire(&self, wake: WakeFn) {
86        self.inflight.fetch_add(1, SeqCst);
87        for slot in &self.members {
88            let ctx = slot.load(SeqCst);
89            if !ctx.is_null() {
90                // SAFETY: a member's context stays valid until its slot is
91                // cleared AND every callback that could have read it has
92                // counted out, which `leave` waits for.
93                unsafe { wake(ctx) };
94            }
95        }
96        self.inflight.fetch_sub(1, SeqCst);
97    }
98
99    /// Join with `ctx`. `None` when every member slot is taken.
100    pub(crate) fn join(&'static self, ctx: *mut c_void) -> Option<Membership> {
101        let generation = self.generation.load(SeqCst);
102        for (idx, slot) in self.members.iter().enumerate() {
103            if slot
104                .compare_exchange(core::ptr::null_mut(), ctx, SeqCst, SeqCst)
105                .is_ok()
106            {
107                return Some(Membership {
108                    fanout: self,
109                    idx,
110                    generation,
111                });
112            }
113        }
114        None
115    }
116
117    /// Members currently joined.
118    pub(crate) fn member_count(&self) -> usize {
119        self.members
120            .iter()
121            .filter(|s| !s.load(SeqCst).is_null())
122            .count()
123    }
124
125    /// Record whether the backend honours the installed callback.
126    pub(crate) fn set_live(&self, live: bool) {
127        self.live.store(live, SeqCst);
128    }
129
130    /// Serve `key` from now on: the owner's session reached its final
131    /// address.
132    pub(crate) fn rekey(&self, key: *mut c_void) {
133        self.key.store(key, SeqCst);
134    }
135
136    /// Return the entry to the pool. The caller has already taken the
137    /// callback off the backend, so no NEW callback can start; this waits out
138    /// any still in flight before the entry can be reused.
139    pub(crate) fn release(&self, yield_now: fn()) {
140        self.live.store(false, SeqCst);
141        self.generation.fetch_add(1, SeqCst);
142        for slot in &self.members {
143            slot.store(core::ptr::null_mut(), SeqCst);
144        }
145        self.quiesce(yield_now);
146        self.key.store(core::ptr::null_mut(), SeqCst);
147    }
148
149    fn quiesce(&self, yield_now: fn()) {
150        while self.inflight.load(SeqCst) != 0 {
151            yield_now();
152        }
153    }
154}

156/// One executor's place in a [`WakeFanout`].
157#[derive(Clone, Copy)]
158pub(crate) struct Membership {
159    fanout: &'static WakeFanout,
160    idx: usize,
161    generation: u32,
162}
163
164impl Membership {
165    pub(crate) fn fanout(&self) -> &'static WakeFanout {
166        self.fanout
167    }
168
169    /// Is the entry still the tenancy this membership joined?
170    pub(crate) fn is_current(&self) -> bool {
171        self.fanout.generation.load(SeqCst) == self.generation
172    }
173
174    /// Does an arrival on the shared session still reach this member? False
175    /// once the entry was released (the session's owner tore its callback
176    /// down) or the backend never honoured it.
177    pub(crate) fn is_live(&self) -> bool {
178        self.is_current() && self.fanout.live.load(SeqCst)
179    }
180
181    /// Leave. After this returns no callback holds this member's context.
182    /// A membership from an earlier tenancy of the entry is a no-op: `release`
183    /// already cleared it, and the slot may now be someone else's.
184    pub(crate) fn leave(self, yield_now: fn()) {
185        if !self.is_current() {
186            return;
187        }
188        self.fanout.members[self.idx].store(core::ptr::null_mut(), SeqCst);
189        self.fanout.quiesce(yield_now);
190    }
191}
192
193static FANOUTS: [WakeFanout; MAX_FANOUTS] = [const { WakeFanout::new() }; MAX_FANOUTS];
194
195/// The entry serving `key`, if any.
196pub(crate) fn find(key: *mut c_void) -> Option<&'static WakeFanout> {
197    if key.is_null() {
198        return None;
199    }
200    FANOUTS.iter().find(|f| f.key.load(SeqCst) == key)
201}
202
203/// Claim a free entry for `key`. `None` when the pool is exhausted.
204pub(crate) fn claim(key: *mut c_void) -> Option<&'static WakeFanout> {
205    if key.is_null() {
206        return None;
207    }
208    for f in &FANOUTS {
209        if f.key
210            .compare_exchange(core::ptr::null_mut(), key, SeqCst, SeqCst)
211            .is_ok()
212        {
213            f.generation.fetch_add(1, SeqCst);
214            f.live.store(false, SeqCst);
215            return Some(f);
216        }
217    }
218    None
219}

226    // The pool is process-global and `cargo test` runs tests in parallel, so
227    // every test holds `POOL`, keys its entries by its own static's address,
228    // and releases what it claims.
229
230    fn spin() {
231        std::thread::yield_now();
232    }
233
234    /// The pool is shared by every test in the process; hold this while
235    /// claiming from it.
236    static POOL: std::sync::Mutex<()> = std::sync::Mutex::new(());
237
238    #[repr(C)]
239    struct Probe {
240        hits: AtomicUsize,
241    }
242
243    unsafe extern "C" fn count(ctx: *mut c_void) {
244        let p = unsafe { &*(ctx as *const Probe) };
245        p.hits.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
246    }
247
248    fn ctx(p: &Probe) -> *mut c_void {
249        p as *const Probe as *mut c_void
250    }
251
252    #[test]
253    fn one_arrival_wakes_every_member() {
254        let _pool = POOL.lock().unwrap();
255        static KEY: u8 = 0;
256        let key = &KEY as *const u8 as *mut c_void;
257        let f = claim(key).expect("a free entry");
258        f.set_live(true);
259        let (a, b, c) = (
260            Probe {
261                hits: AtomicUsize::new(0),
262            },
263            Probe {
264                hits: AtomicUsize::new(0),
265            },
266            Probe {
267                hits: AtomicUsize::new(0),
268            },
269        );
270        let ma = f.join(ctx(&a)).unwrap();
271        let mb = f.join(ctx(&b)).unwrap();
272        let mc = f.join(ctx(&c)).unwrap();
273        assert!(find(key).is_some_and(|g| core::ptr::eq(g, f)));
274        f.fire(count);
275        for p in [&a, &b, &c] {
276            assert_eq!(p.hits.load(std::sync::atomic::Ordering::SeqCst), 1);
277        }
278        assert!(ma.is_live() && mb.is_live() && mc.is_live());
279
280        // A member that left is not woken; the others still are.
281        mb.leave(spin);
282        f.fire(count);
283        assert_eq!(a.hits.load(std::sync::atomic::Ordering::SeqCst), 2);
284        assert_eq!(b.hits.load(std::sync::atomic::Ordering::SeqCst), 1);
285        assert_eq!(c.hits.load(std::sync::atomic::Ordering::SeqCst), 2);
286        assert_eq!(f.member_count(), 2);
287        f.release(spin);
288        assert!(find(key).is_none());
289    }

291    /// The owner tears its callback down while tiers are still joined: they
292    /// read as not live (so the executor stops parking on a wake nothing will
293    /// signal), and their later `leave` cannot touch the entry's next tenant.
294    #[test]
295    fn release_makes_every_membership_dead_and_its_leave_a_no_op() {
296        let _pool = POOL.lock().unwrap();
297        static KEY: u8 = 0;
298        let f = claim(&KEY as *const u8 as *mut c_void).unwrap();
299        f.set_live(true);
300        let p = Probe {
301            hits: AtomicUsize::new(0),
302        };
303        let tier = f.join(ctx(&p)).unwrap();
304        assert!(tier.is_live());
305        f.release(spin);
306        assert!(!tier.is_live(), "a released entry is not a live wake");
307
308        // The next tenant of the same entry.
309        static KEY2: u8 = 0;
310        let g = claim(&KEY2 as *const u8 as *mut c_void).unwrap();
311        assert!(
312            core::ptr::eq(f, g),
313            "the pool hands out the first free entry"
314        );
315        let q = Probe {
316            hits: AtomicUsize::new(0),
317        };
318        let fresh = g.join(ctx(&q)).unwrap();
319        tier.leave(spin);
320        assert_eq!(
321            g.member_count(),
322            1,
323            "a stale membership cleared the new tenant's slot"
324        );
325        fresh.leave(spin);
326        g.release(spin);
327    }

366    /// `leave` must not return while a callback that read the member's context
367    /// is still running: the context is freed right after. A callback is held
368    /// open inside the member's wake; `leave` on another thread has to wait.
369    #[test]
370    fn leave_waits_out_a_callback_in_flight() {
371        let _pool = POOL.lock().unwrap();
372        use std::sync::atomic::{AtomicBool as StdBool, Ordering};
373        static KEY: u8 = 0;
374        static ENTERED: StdBool = StdBool::new(false);
375        static RELEASE: StdBool = StdBool::new(false);
376        unsafe extern "C" fn slow(_ctx: *mut c_void) {
377            ENTERED.store(true, Ordering::SeqCst);
378            while !RELEASE.load(Ordering::SeqCst) {
379                std::thread::yield_now();
380            }
381        }
382        let f = claim(&KEY as *const u8 as *mut c_void).unwrap();
383        f.set_live(true);
384        let p = Box::leak(Box::new(Probe {
385            hits: AtomicUsize::new(0),
386        }));
387        let m = f.join(ctx(p)).unwrap();
388        let firer = std::thread::spawn(move || f.fire(slow));
389        while !ENTERED.load(Ordering::SeqCst) {
390            std::thread::yield_now();
391        }
392        let left = std::sync::Arc::new(StdBool::new(false));
393        let left2 = left.clone();
394        let leaver = std::thread::spawn(move || {
395            m.leave(spin);
396            left2.store(true, Ordering::SeqCst);
397        });
398        std::thread::sleep(std::time::Duration::from_millis(50));
399        assert!(
400            !left.load(Ordering::SeqCst),
401            "leave returned while a callback holding the context was running"
402        );
403        RELEASE.store(true, Ordering::SeqCst);
404        firer.join().unwrap();
405        leaver.join().unwrap();
406        assert!(left.load(Ordering::SeqCst));
407        f.release(spin);
408    }
