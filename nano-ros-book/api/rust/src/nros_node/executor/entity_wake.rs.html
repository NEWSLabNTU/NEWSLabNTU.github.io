1//! An arrival names its ENTITY and its OWNING
2//! executor.
3//!
4//! The session-scoped wake (`set_wake_callback`) carries only the executor's
5//! context, so an arrival is binary: "something happened on this session". Two
6//! things are lost at that ABI. The executor cannot tell which entry is ready,
7//! so it scans them all. And when several executors share one session (the
8//! wake fan-out), every arrival wakes every one of them, at a
9//! measured 7-9 us of CPU per spurious wake.
10//!
11//! Each subscription, service and client entry is registered with a [`Waker`]
12//! from [`waker`], through the handle's `register_waker` (W1 carries it across
13//! the C ABI as upstream's `*_set_on_new_*_callback`). That waker identifies:
14//!
15//! - the owning executor's [`WakeCtx`], an `Arc` the waker holds a strong
16//!   reference to, so a backend that keeps a clone past the entry's release
17//!   cannot dangle;
18//! - the entry's slot, in the low bits of the same pointer: six for an image
19//!   of 64 callbacks or fewer, one more per doubling past that. `WakeCtx` is
20//!   aligned to [`WAKE_CTX_ALIGN`], so those bits are free.
21//!
22//! Waking it sets that slot's bit in [`WakeCtx::ready`] and then does exactly
23//! what the session callback does: flag, probe, signal. A backend that
24//! attributes the arrival this way can skip the session wake for it, so only
25//! the owner wakes. A backend that cannot attribute (NULL slot, or
26//! `UNSUPPORTED`) keeps the session wake, and the fan-out stays the fallback.
27//!
28//! W3 is the consumer of `ready`: `spin_once` takes the word and reads an
29//! attributed slot's readiness from its bit instead of calling its
30//! `has_data` (see `Executor::entity_recheck` for the cases that still ask).
31
32use core::task::{RawWaker, RawWakerVTable, Waker};

34use portable_atomic_util::Arc;

38/// Slots the ready words can name: the image's own callback count, rounded
39/// up to whole machine words (at least one: 64 slots on a 64-bit target, 32
40/// on a 32-bit one). An entry above this keeps the session-scoped wake.
41// Issue 2541 -- this was always one `usize` word; the ready words are now an
42// `AtomicSlotMask` (`slot_mask::ATOMIC_WORDS`) that grows with the image.
43pub(crate) const SLOTS: usize = super::slot_mask::ATOMIC_SLOTS;
44
45/// `WakeCtx`'s alignment: the slot capacity rounded up to a power of two,
46/// and never below the 64 it always had.
47pub(crate) const WAKE_CTX_ALIGN: usize = if SLOTS <= 64 {
48    64
49} else {
50    SLOTS.next_power_of_two()
51};
52
53const SLOT_MASK: usize = WAKE_CTX_ALIGN - 1;
54
55const _: () = assert!(
56    core::mem::align_of::<WakeCtx>() >= WAKE_CTX_ALIGN && SLOT_MASK >= SLOTS - 1,
57    "the slot rides in the low bits of the WakeCtx pointer, so WakeCtx must be aligned past SLOTS"
58);
59
60/// A zero-sized field type that raises `WakeCtx`'s alignment to
61/// [`WAKE_CTX_ALIGN`]: `()` at 64 (its `repr(align(64))` already holds), an
62/// over-aligned marker above. `repr(align)` takes only a literal, hence the
63/// lookup.
64pub(crate) type WakeCtxAlign = <AlignFor<WAKE_CTX_ALIGN> as SelectAlign>::Marker;
65
66pub(crate) struct AlignFor<const N: usize>;
67
68pub(crate) trait SelectAlign {
69    type Marker;
70}
71
72#[repr(align(128))]
73pub(crate) struct Align128;
74#[repr(align(256))]
75pub(crate) struct Align256;
76
77impl SelectAlign for AlignFor<64> {
78    type Marker = ();
79}
80impl SelectAlign for AlignFor<128> {
81    type Marker = Align128;
82}
83impl SelectAlign for AlignFor<256> {
84    type Marker = Align256;
85}

87/// The waker for `slot` of the executor that owns `ctx`, or `None` when the
88/// slot is beyond what the ready word can name.
89pub(crate) fn waker(ctx: &Arc<WakeCtx>, slot: usize) -> Option<Waker> {
90    if slot >= SLOTS {
91        return None;
92    }
93    let base = Arc::into_raw(ctx.clone()) as usize;
94    debug_assert_eq!(base & SLOT_MASK, 0, "WakeCtx is not aligned past SLOTS");
95    let raw = RawWaker::new((base | slot) as *const (), &VTABLE);
96    // SAFETY: the vtable honours the RawWaker contract: `clone` takes a strong
97    // reference, `drop` releases one, and `wake` consumes the one it was given.
98    Some(unsafe { Waker::from_raw(raw) })
99}
100
101fn split(data: *const ()) -> (*const WakeCtx, usize) {
102    let d = data as usize;
103    ((d & !SLOT_MASK) as *const WakeCtx, d & SLOT_MASK)
104}
105
106static VTABLE: RawWakerVTable = RawWakerVTable::new(clone, wake, wake_by_ref, drop);
107
108unsafe fn clone(data: *const ()) -> RawWaker {
109    // SAFETY: `data` came from `waker()`, so its base is an `Arc::into_raw`
110    // pointer whose count this waker holds one of.
111    unsafe { Arc::increment_strong_count(split(data).0) };
112    RawWaker::new(data, &VTABLE)
113}
114
115unsafe fn wake(data: *const ()) {
116    unsafe {
117        wake_by_ref(data);
118        drop(data);
119    }
120}
121
122unsafe fn wake_by_ref(data: *const ()) {
123    let (ctx, slot) = split(data);
124    // SAFETY: this waker holds a strong reference, so the `WakeCtx` is alive.
125    unsafe { &*ctx }.mark_ready(slot);
126}
127
128unsafe fn drop(data: *const ()) {
129    // SAFETY: releases the strong reference this waker held.
130    unsafe { Arc::decrement_strong_count(split(data).0) };
131}
