1//! phase-397 W2 (RFC-0084) — an arrival names its ENTITY and its OWNING
2//! executor.
3//!
4//! The session-scoped wake (`set_wake_callback`) carries only the executor's
5//! context, so an arrival is binary: "something happened on this session". Two
6//! things are lost at that ABI. The executor cannot tell which entry is ready,
7//! so it scans them all. And when several executors share one session (the
8//! wake fan-out, RFC-0106 D6), every arrival wakes every one of them, at a
9//! measured 7-9 us of CPU per spurious wake (phase-436 step 5).
10//!
11//! Each subscription, service and client entry is registered with a [`Waker`]
12//! from [`waker`], through the handle's `register_waker` (W1 carries it across
13//! the C ABI as upstream's `*_set_on_new_*_callback`). That waker identifies:
14//!
15//! - the owning executor's [`WakeCtx`], an `Arc` the waker holds a strong
16//!   reference to, so a backend that keeps a clone past the entry's release
17//!   cannot dangle;
18//! - the entry's slot, in the low bits of the same pointer (six on a 64-bit
19//!   target, five on a 32-bit one). `WakeCtx` is
20//!   `repr(align(64))`, so those bits are free.
21//!
22//! Waking it sets that slot's bit in [`WakeCtx::ready`] and then does exactly
23//! what the session callback does: flag, probe, signal. A backend that
24//! attributes the arrival this way can skip the session wake for it, so only
25//! the owner wakes. A backend that cannot attribute (NULL slot, or
26//! `UNSUPPORTED`) keeps the session wake, and the fan-out stays the fallback.
27//!
28//! W3 is the consumer of `ready`: `spin_once` takes the word and reads an
29//! attributed slot's readiness from its bit instead of calling its
30//! `has_data` (see `Executor::entity_recheck` for the cases that still ask).
31
32use core::task::{RawWaker, RawWakerVTable, Waker};
33
34use portable_atomic_util::Arc;
35
36use super::spin::WakeCtx;
37
38/// Slots the ready word can name: one `usize` (64 on a 64-bit target, 32 on
39/// a 32-bit one, which may have no 64-bit atomics). An entry above this keeps
40/// the session-scoped wake.
41pub(crate) const SLOTS: usize = usize::BITS as usize;
42
43const SLOT_MASK: usize = SLOTS - 1;
44
45const _: () = assert!(
46    core::mem::align_of::<WakeCtx>() >= SLOTS,
47    "the slot rides in the low bits of the WakeCtx pointer, so WakeCtx must be aligned to SLOTS"
48);
49
50/// The waker for `slot` of the executor that owns `ctx`, or `None` when the
51/// slot is beyond what the ready word can name.
52pub(crate) fn waker(ctx: &Arc<WakeCtx>, slot: usize) -> Option<Waker> {
53    if slot >= SLOTS {
54        return None;
55    }
56    let base = Arc::into_raw(ctx.clone()) as usize;
57    debug_assert_eq!(base & SLOT_MASK, 0, "WakeCtx is not aligned to SLOTS");
58    let raw = RawWaker::new((base | slot) as *const (), &VTABLE);
59    // SAFETY: the vtable honours the RawWaker contract: `clone` takes a strong
60    // reference, `drop` releases one, and `wake` consumes the one it was given.
61    Some(unsafe { Waker::from_raw(raw) })
62}
63
64fn split(data: *const ()) -> (*const WakeCtx, usize) {
65    let d = data as usize;
66    ((d & !SLOT_MASK) as *const WakeCtx, d & SLOT_MASK)
67}
68
69static VTABLE: RawWakerVTable = RawWakerVTable::new(clone, wake, wake_by_ref, drop);
70
71unsafe fn clone(data: *const ()) -> RawWaker {
72    // SAFETY: `data` came from `waker()`, so its base is an `Arc::into_raw`
73    // pointer whose count this waker holds one of.
74    unsafe { Arc::increment_strong_count(split(data).0) };
75    RawWaker::new(data, &VTABLE)
76}
77
78unsafe fn wake(data: *const ()) {
79    unsafe {
80        wake_by_ref(data);
81        drop(data);
82    }
83}
84
85unsafe fn wake_by_ref(data: *const ()) {
86    let (ctx, slot) = split(data);
87    // SAFETY: this waker holds a strong reference, so the `WakeCtx` is alive.
88    unsafe { &*ctx }.mark_ready(slot);
89}
90
91unsafe fn drop(data: *const ()) {
92    // SAFETY: releases the strong reference this waker held.
93    unsafe { Arc::decrement_strong_count(split(data).0) };
94}
