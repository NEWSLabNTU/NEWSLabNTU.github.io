1//! One bit per callback slot, over as many `u64` words as the image needs.
2//!
3//! The executor names a set of callback slots (ready, attributed, deferred,
4//! in a band, ...) as a [`SlotMask`]: `crate::config::SLOT_WORDS` words, sized
5//! from the image, so the most callbacks one executor holds is
6//! `crate::config::CALLBACK_SLOT_CAPACITY`. An image of 64 callbacks or fewer
7//! has one word, the layout the masks had as a bare `u64`.
8//!
9//! [`AtomicSlotMask`] is the cross-thread twin, written by a backend's arrival
10//! path (and an interrupt's): one `usize` atomic per machine word, so each
11//! update stays a single-word, lock-free read-modify-write and a target with
12//! no 64-bit atomics keeps the `portable_atomic` fallback it had.
13
14// Issue 2541 -- before this the masks were one `u64` and the table could not
15// pass 64 callbacks.
16

21/// A set of callback slots, one bit per slot, `SLOT_WORDS` words of 64.
22///
23/// An index at or above the capacity is outside every mask: [`Self::bit`] of
24/// it is empty, [`Self::test`] of it is `false`, and setting it does nothing.
25#[derive(Clone, Copy, PartialEq, Eq, Debug)]
26pub(crate) struct SlotMask(pub(crate) [u64; SLOT_WORDS]);
27
28impl Default for SlotMask {
29    #[inline]
30    fn default() -> Self {
31        Self::EMPTY
32    }
33}
34
35#[allow(dead_code)] // each build uses the subset its features reach
36impl SlotMask {
37    /// No slot.
38    pub(crate) const EMPTY: Self = Self([0; SLOT_WORDS]);
39    /// Every slot up to the capacity (which is a whole number of words).
40    pub(crate) const ALL: Self = Self([u64::MAX; SLOT_WORDS]);
41    /// Slots one mask can name.
42    pub(crate) const CAPACITY: usize = CALLBACK_SLOT_CAPACITY;
43
44    /// The mask holding `slot` alone; empty when `slot` is past the capacity.
45    #[inline]
46    pub(crate) const fn bit(slot: usize) -> Self {
47        let mut m = Self::EMPTY;
48        if slot < CALLBACK_SLOT_CAPACITY {
49            m.0[slot / 64] = 1u64 << (slot % 64);
50        }
51        m
52    }
53
54    /// The mask whose first word is `word` (slots 0..64) and every other
55    /// word empty.
56    #[inline]
57    pub(crate) const fn from_word0(word: u64) -> Self {
58        let mut m = Self::EMPTY;
59        m.0[0] = word;
60        m
61    }
62
63    /// Word `k` (slots `64 * k .. 64 * k + 64`); 0 past the last word.
64    #[inline]
65    pub(crate) const fn word(self, k: usize) -> u64 {
66        if k < SLOT_WORDS { self.0[k] } else { 0 }
67    }
68
69    /// `self` with `slot` added.
70    #[inline]
71    pub(crate) const fn with(self, slot: usize) -> Self {
72        self.union(Self::bit(slot))
73    }
74
75    /// Add `slot`.
76    #[inline]
77    pub(crate) fn set(&mut self, slot: usize) {
78        if slot < CALLBACK_SLOT_CAPACITY {
79            self.0[slot / 64] |= 1u64 << (slot % 64);
80        }
81    }
82
83    /// Remove `slot`.
84    #[inline]
85    pub(crate) fn clear(&mut self, slot: usize) {
86        if slot < CALLBACK_SLOT_CAPACITY {
87            self.0[slot / 64] &= !(1u64 << (slot % 64));
88        }
89    }
90
91    /// Whether `slot` is in the mask.
92    #[inline]
93    pub(crate) const fn test(self, slot: usize) -> bool {
94        slot < CALLBACK_SLOT_CAPACITY && self.0[slot / 64] & (1u64 << (slot % 64)) != 0
95    }
96
97    /// No slot is in the mask.
98    #[inline]
99    pub(crate) const fn is_empty(self) -> bool {
100        let mut k = 0;
101        while k < SLOT_WORDS {
102            if self.0[k] != 0 {
103                return false;
104            }
105            k += 1;
106        }
107        true
108    }
109
110    /// Slots in either.
111    #[inline]
112    pub(crate) const fn union(self, other: Self) -> Self {
113        let mut m = self;
114        let mut k = 0;
115        while k < SLOT_WORDS {
116            m.0[k] |= other.0[k];
117            k += 1;
118        }
119        m
120    }
121
122    /// Slots in both.
123    #[inline]
124    pub(crate) const fn intersection(self, other: Self) -> Self {
125        let mut m = self;
126        let mut k = 0;
127        while k < SLOT_WORDS {
128            m.0[k] &= other.0[k];
129            k += 1;
130        }
131        m
132    }
133
134    /// Slots in `self` and not in `other`.
135    #[inline]
136    pub(crate) const fn and_not(self, other: Self) -> Self {
137        let mut m = self;
138        let mut k = 0;
139        while k < SLOT_WORDS {
140            m.0[k] &= !other.0[k];
141            k += 1;
142        }
143        m
144    }
145
146    /// Every slot not in `self`.
147    #[inline]
148    pub(crate) const fn complement(self) -> Self {
149        Self::ALL.and_not(self)
150    }
151
152    /// Some slot is in both.
153    #[inline]
154    pub(crate) const fn intersects(self, other: Self) -> bool {
155        !self.intersection(other).is_empty()
156    }
157
158    /// Every slot of `other` is in `self`.
159    #[inline]
160    pub(crate) const fn contains_all(self, other: Self) -> bool {
161        other.and_not(self).is_empty()
162    }
163
164    /// Slots in the mask.
165    #[inline]
166    pub(crate) const fn count(self) -> u32 {
167        let mut n = 0;
168        let mut k = 0;
169        while k < SLOT_WORDS {
170            n += self.0[k].count_ones();
171            k += 1;
172        }
173        n
174    }
175
176    /// The lowest slot in the mask.
177    #[inline]
178    pub(crate) const fn first_set(self) -> Option<usize> {
179        let mut k = 0;
180        while k < SLOT_WORDS {
181            if self.0[k] != 0 {
182                return Some(k * 64 + self.0[k].trailing_zeros() as usize);
183            }
184            k += 1;
185        }
186        None
187    }
188
189    /// The slots in the mask, lowest first.
190    #[inline]
191    pub(crate) fn iter_ones(self) -> IterOnes {
192        IterOnes { rest: self }
193    }
194}
195
196/// [`SlotMask::iter_ones`].
197#[allow(dead_code)]
198pub(crate) struct IterOnes {
199    rest: SlotMask,
200}
201
202impl Iterator for IterOnes {
203    type Item = usize;
204
205    #[inline]
206    fn next(&mut self) -> Option<usize> {
207        let mut k = 0;
208        while k < SLOT_WORDS {
209            let w = self.rest.0[k];
210            if w != 0 {
211                self.rest.0[k] = w & (w - 1);
212                return Some(k * 64 + w.trailing_zeros() as usize);
213            }
214            k += 1;
215        }
216        None
217    }
218}

258/// Bits in one atomic word: a machine word, so a target whose widest native
259/// atomic is 32 bits uses 32-bit words.
260const ATOMIC_WORD_BITS: usize = usize::BITS as usize;
261
262// Issue 2541 -- the atomic words cover the image's OWN callback count, not
263// the slot masks' capacity, rounded up to a machine word: an image of up to
264// `usize::BITS` callbacks keeps exactly the one word it always had (32 slots
265// on a 32-bit target, 64 on a 64-bit one), so its wake path is unchanged; a
266// bigger image gets one more word per `usize::BITS` callbacks, up to the
267// capacity. nros-node's own unit tests cover the whole capacity.
268/// Atomic words in an [`AtomicSlotMask`].
269#[allow(dead_code)]
270pub(crate) const ATOMIC_WORDS: usize = {
271    let cap_words = CALLBACK_SLOT_CAPACITY.div_ceil(ATOMIC_WORD_BITS);
272    let need = if cfg!(test) {
273        CALLBACK_SLOT_CAPACITY
274    } else if crate::config::MAX_EXECUTOR_CBS < CALLBACK_SLOT_CAPACITY {
275        crate::config::MAX_EXECUTOR_CBS
276    } else {
277        CALLBACK_SLOT_CAPACITY
278    };
279    let words = need.div_ceil(ATOMIC_WORD_BITS);
280    let words = if words == 0 { 1 } else { words };
281    if words < cap_words { words } else { cap_words }
282};
283
284/// One atomic word and one mask word: every [`AtomicSlotMask`] operation then
285/// takes the single-word path the masks had before they grew words, so such
286/// an image compiles to the code it always did.
287const ONE_WORD: bool = ATOMIC_WORDS == 1 && SLOT_WORDS == 1;
288
289/// Slots an [`AtomicSlotMask`] can name: the cross-thread wake words'
290/// width. A slot past it keeps the session-scoped wake, as a slot past one
291/// `usize` always did.
292#[allow(dead_code)]
293pub(crate) const ATOMIC_SLOTS: usize = ATOMIC_WORDS * ATOMIC_WORD_BITS;
294
295const _: () = assert!(
296    64 % ATOMIC_WORD_BITS == 0,
297    "a mask word splits into whole atomic words"
298);
299
300/// A [`SlotMask`] other threads (and interrupts) write: one
301/// `portable_atomic::AtomicUsize` per machine word. Every operation on it is
302/// a sequence of single-word atomic operations, `SeqCst` as the one word
303/// was; a multi-word read is therefore not one snapshot, which no reader
304/// needs (each slot's bit is independent).
305#[allow(dead_code)]
306pub(crate) struct AtomicSlotMask([portable_atomic::AtomicUsize; ATOMIC_WORDS]);
307
308impl Default for AtomicSlotMask {
309    fn default() -> Self {
310        Self::new()
311    }
312}

320#[allow(dead_code)] // each build uses the subset its features reach
321impl AtomicSlotMask {
322    pub(crate) const fn new() -> Self {
323        Self([const { portable_atomic::AtomicUsize::new(0) }; ATOMIC_WORDS])
324    }
325
326    /// The atomic word holding `slot`, and the slot's bit in it.
327    #[inline]
328    const fn locate(slot: usize) -> (usize, usize) {
329        (slot / ATOMIC_WORD_BITS, 1usize << (slot % ATOMIC_WORD_BITS))
330    }
331
332    /// The part of `mask` atomic word `j` covers.
333    #[inline]
334    const fn part(mask: SlotMask, j: usize) -> usize {
335        let bit = j * ATOMIC_WORD_BITS;
336        (mask.0[bit / 64] >> (bit % 64)) as usize
337    }
338
339    /// Fold atomic word `j`'s value `v` into `mask`.
340    #[inline]
341    fn fold(mask: &mut SlotMask, j: usize, v: usize) {
342        let bit = j * ATOMIC_WORD_BITS;
343        mask.0[bit / 64] |= (v as u64) << (bit % 64);
344    }
345
346    /// Set `slot`'s bit: one `fetch_or`. A slot past the capacity sets
347    /// nothing.
348    #[inline]
349    pub(crate) fn set(&self, slot: usize) {
350        if slot < ATOMIC_SLOTS {
351            let (j, b) = Self::locate(slot);
352            self.0[j].fetch_or(b, portable_atomic::Ordering::SeqCst);
353        }
354    }
355
356    /// Whether `slot`'s bit is set.
357    #[inline]
358    pub(crate) fn test(&self, slot: usize) -> bool {
359        if slot >= ATOMIC_SLOTS {
360            return false;
361        }
362        let (j, b) = Self::locate(slot);
363        self.0[j].load(portable_atomic::Ordering::SeqCst) & b != 0
364    }
365
366    /// Clear `slot`'s bit only if the word still reads `seen` (one
367    /// `compare_exchange` on the slot's word). Returns whether it did.
368    #[inline]
369    pub(crate) fn clear_if_word_unchanged(&self, slot: usize, seen: &SlotMask) -> bool {
370        if slot >= ATOMIC_SLOTS {
371            return false;
372        }
373        let (j, b) = Self::locate(slot);
374        let was = Self::part(*seen, j);
375        self.0[j]
376            .compare_exchange(
377                was,
378                was & !b,
379                portable_atomic::Ordering::SeqCst,
380                portable_atomic::Ordering::SeqCst,
381            )
382            .is_ok()
383    }
384
385    /// Every bit, word by word.
386    #[inline]
387    pub(crate) fn load(&self) -> SlotMask {
388        if ONE_WORD {
389            return SlotMask::from_word0(self.0[0].load(portable_atomic::Ordering::SeqCst) as u64);
390        }
391        let mut m = SlotMask::EMPTY;
392        for (j, w) in self.0.iter().enumerate() {
393            Self::fold(&mut m, j, w.load(portable_atomic::Ordering::SeqCst));
394        }
395        m
396    }
397
398    /// Replace every word.
399    #[inline]
400    pub(crate) fn store(&self, mask: SlotMask) {
401        for (j, w) in self.0.iter().enumerate() {
402            w.store(Self::part(mask, j), portable_atomic::Ordering::SeqCst);
403        }
404    }
405
406    /// Take every bit: one `swap(0)` per word.
407    #[inline]
408    pub(crate) fn take(&self) -> SlotMask {
409        if ONE_WORD {
410            return SlotMask::from_word0(
411                self.0[0].swap(0, portable_atomic::Ordering::SeqCst) as u64
412            );
413        }
414        let mut m = SlotMask::EMPTY;
415        for (j, w) in self.0.iter().enumerate() {
416            Self::fold(&mut m, j, w.swap(0, portable_atomic::Ordering::SeqCst));
417        }
418        m
419    }
420
421    /// Take the bits of `mask` and leave the rest: one `fetch_and` per word
422    /// `mask` touches (a `swap(0)` where it covers the whole word).
423    #[inline]
424    pub(crate) fn take_masked(&self, mask: SlotMask) -> SlotMask {
425        if ONE_WORD {
426            // The one-word code as it was before the masks grew words.
427            let part = mask.word(0) as usize;
428            let v = self.0[0].fetch_and(!part, portable_atomic::Ordering::SeqCst) & part;
429            return SlotMask::from_word0(v as u64);
430        }
431        let mut m = SlotMask::EMPTY;
432        for (j, w) in self.0.iter().enumerate() {
433            let part = Self::part(mask, j);
434            let v = match part {
435                0 => continue,
436                usize::MAX => w.swap(0, portable_atomic::Ordering::SeqCst),
437                _ => w.fetch_and(!part, portable_atomic::Ordering::SeqCst) & part,
438            };
439            Self::fold(&mut m, j, v);
440        }
441        m
442    }
443
444    /// Whether any bit of `mask` is set.
445    #[inline]
446    pub(crate) fn intersects(&self, mask: SlotMask) -> bool {
447        if ONE_WORD {
448            return self.0[0].load(portable_atomic::Ordering::SeqCst) as u64 & mask.word(0) != 0;
449        }
450        self.0.iter().enumerate().any(|(j, w)| {
451            let part = Self::part(mask, j);
452            part != 0 && w.load(portable_atomic::Ordering::SeqCst) & part != 0
453        })
454    }
455}

458mod tests {
459    use super::*;
460    use std::vec::Vec;

462    #[test]
463    fn the_unit_tests_run_more_than_one_word() {
464        const { assert!(SLOT_WORDS >= 2, "cfg(test) floors the masks at two words") };
465        assert_eq!(SlotMask::CAPACITY, 64 * SLOT_WORDS);
466    }
467
468    #[test]
469    fn bits_on_each_side_of_a_word_boundary_are_distinct() {
470        for i in [0usize, 1, 63, 64, 65, 127] {
471            let m = SlotMask::bit(i);
472            assert!(m.test(i), "slot {i}");
473            assert_eq!(m.count(), 1);
474            assert_eq!(m.first_set(), Some(i));
475            for j in [0usize, 1, 63, 64, 65, 127] {
476                assert_eq!(m.test(j), i == j, "bit {i} read at {j}");
477            }
478        }
479        assert_ne!(SlotMask::bit(0), SlotMask::bit(64), "no aliasing mod 64");
480        assert_ne!(SlotMask::bit(63), SlotMask::bit(127));
481    }
482
483    #[test]
484    fn an_index_past_the_capacity_is_in_no_mask() {
485        let cap = SlotMask::CAPACITY;
486        assert!(SlotMask::bit(cap).is_empty());
487        assert!(!SlotMask::ALL.test(cap));
488        let mut m = SlotMask::EMPTY;
489        m.set(cap);
490        assert!(m.is_empty());
491        m.set(cap - 1);
492        assert_eq!(m.iter_ones().collect::<Vec<_>>(), [cap - 1]);
493        m.clear(cap);
494        assert_eq!(m.count(), 1);
495    }
496
497    #[test]
498    fn set_algebra_spans_the_words() {
499        let a = SlotMask::bit(3) | SlotMask::bit(64) | SlotMask::bit(127);
500        let b = SlotMask::bit(64) | SlotMask::bit(100);
501        assert_eq!((a & b), SlotMask::bit(64));
502        assert_eq!((a | b).count(), 4);
503        assert_eq!(a.and_not(b), SlotMask::bit(3) | SlotMask::bit(127));
504        assert!(a.intersects(b));
505        assert!(!a.intersects(SlotMask::bit(100) | SlotMask::bit(4)));
506        assert!(a.contains_all(SlotMask::bit(127) | SlotMask::bit(3)));
507        assert!(!a.contains_all(b));
508        assert_eq!((!a).count(), SlotMask::CAPACITY as u32 - 3);
509        assert!(!(!a).test(64));
510        assert!((!a).test(63));
511        assert_eq!(a.iter_ones().collect::<Vec<_>>(), [3, 64, 127]);
512        let mut c = a;
513        c.clear(64);
514        assert_eq!(c.iter_ones().collect::<Vec<_>>(), [3, 127]);
515        assert_eq!(SlotMask::EMPTY.first_set(), None);
516        assert_eq!(SlotMask::ALL.count(), SlotMask::CAPACITY as u32);
517        assert_eq!(
518            SlotMask::from_word0(0b101).iter_ones().collect::<Vec<_>>(),
519            [0, 2]
520        );
521        assert_eq!(SlotMask::bit(127).word(1), 1u64 << 63);
522    }
523
524    #[test]
525    fn the_atomic_mask_takes_and_leaves_bits_across_words() {
526        let a = AtomicSlotMask::new();
527        for i in [0usize, 31, 32, 63, 64, 127] {
528            a.set(i);
529            assert!(a.test(i), "slot {i}");
530        }
531        a.set(SlotMask::CAPACITY);
532        let all = [0usize, 31, 32, 63, 64, 127]
533            .into_iter()
534            .fold(SlotMask::EMPTY, |m, i| m.with(i));
535        assert_eq!(a.load(), all);
536        let band = SlotMask::bit(32) | SlotMask::bit(127) | SlotMask::bit(5);
537        assert!(a.intersects(band));
538        assert_eq!(a.take_masked(band), SlotMask::bit(32) | SlotMask::bit(127));
539        assert_eq!(a.load(), all.and_not(band));
540        assert_eq!(a.take_masked(SlotMask::ALL), all.and_not(band));
541        assert!(a.load().is_empty());
542
543        a.store(SlotMask::bit(64) | SlotMask::bit(1));
544        let seen = a.load();
545        a.set(65);
546        assert!(!a.clear_if_word_unchanged(64, &seen), "word 64 moved");
547        let seen = a.load();
548        assert!(a.clear_if_word_unchanged(64, &seen));
549        assert_eq!(a.take(), SlotMask::bit(65) | SlotMask::bit(1));
550    }
