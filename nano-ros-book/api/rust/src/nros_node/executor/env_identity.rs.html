3//! What `$NROS_NODE_NAME` and `$NROS_NODE_NAMESPACE` do to a node.
4//!
5//! Both variables are read by the hosted edge (`nros::env::try_resolve_hosted`,
6//! which every Linux road reaches: Rust, C's `rclc_support_init`, C++'s
7//! `nros_cpp_init`). Until this module they reached only the executor's own
8//! default name, so a node named in code or by the launch file kept its name
9//! and the variable was read and dropped, silently. A read whose result is
10//! discarded is not allowed: a value is honoured and a mismatch is warned
11//! once, or it is ignored with a notice that says where to state it.
12//!
13//! The rules:
14//!
15//! - `$NROS_NODE_NAMESPACE` applies to EVERY node of the image. A namespace
16//!   does not collide, so the node count does not matter. When a node stated a
17//!   different non-root namespace, one warning per process says so.
18//! - `$NROS_NODE_NAME` renames the node only when the image's sizing
19//!   descriptor STATES that it creates exactly one
20//!   ([`crate::config::STATED_NODE_COUNT`] `== Some(1)`), warning once when the
21//!   stated name differs. With two or more nodes one name cannot name them all
22//!   without a collision, and with an unstated count nothing here can tell, so
23//!   in both cases the variable is ignored with a once-per-process notice
24//!   naming where a node's name is authored.
25//!
26//! The values live in process-global slots because the environment is
27//! process-wide and one road (C's `rclc_node_init_default`) names its node
28//! before any executor exists. They are installed once, by the edge that read
29//! them; nothing on an embedded target installs anything, so there every node
30//! keeps its stated identity, as before.
31
32use core::cell::UnsafeCell;
33use portable_atomic::{AtomicBool, AtomicU8, AtomicUsize, Ordering};
34
35/// A node name or namespace is held in 64 bytes everywhere in the executor
36/// (`NodeRecord`), so a longer value could never be applied.
37const CAP: usize = 64;
38
39const EMPTY: u8 = 0;
40const WRITING: u8 = 1;
41const READY: u8 = 2;
42
43/// A once-install string slot: written by the one edge that reads the
44/// environment, read by every node creation after it. `no_std`, no lock.
45struct Slot {
46    state: AtomicU8,
47    len: AtomicUsize,
48    buf: UnsafeCell<[u8; CAP]>,
49}
50
51// SAFETY: `buf` is written only by the thread that won the EMPTY -> WRITING
52// exchange, and read only after a READY acquire load, which orders the write
53// before every read. It is never written again once READY.
54unsafe impl Sync for Slot {}
55
56impl Slot {
57    const fn new() -> Self {
58        Self {
59            state: AtomicU8::new(EMPTY),
60            len: AtomicUsize::new(0),
61            buf: UnsafeCell::new([0; CAP]),
62        }
63    }
64
65    /// First writer wins. `false` when `v` does not fit, so the caller can
66    /// say the value was not applied rather than drop it silently.
67    fn install(&self, v: &str) -> bool {
68        if v.len() > CAP {
69            return false;
70        }
71        if self
72            .state
73            .compare_exchange(EMPTY, WRITING, Ordering::AcqRel, Ordering::Acquire)
74            .is_err()
75        {
76            return true;
77        }
78        // SAFETY: this thread won the exchange, so it is the only writer and
79        // no reader looks at `buf` until READY is published below.
80        unsafe { (&mut *self.buf.get())[..v.len()].copy_from_slice(v.as_bytes()) };
81        self.len.store(v.len(), Ordering::Relaxed);
82        self.state.store(READY, Ordering::Release);
83        true
84    }
85
86    /// Test hook only (`set_for_tests`): forget the value.
87    fn reset(&self) {
88        self.state.store(EMPTY, Ordering::Release);
89    }
90
91    fn get(&self) -> Option<&str> {
92        if self.state.load(Ordering::Acquire) != READY {
93            return None;
94        }
95        let len = self.len.load(Ordering::Relaxed);
96        // SAFETY: READY was published after the write (see `install`), and the
97        // bytes are a copy of a `&str`, so they are valid UTF-8.
98        let bytes = unsafe { &(&*self.buf.get())[..len] };
99        core::str::from_utf8(bytes).ok()
100    }
101}

103static ENV_NODE_NAME: Slot = Slot::new();
104static ENV_NODE_NAMESPACE: Slot = Slot::new();
105
106static WARNED_RENAME: AtomicBool = AtomicBool::new(false);
107static NOTED_NAME_IGNORED: AtomicBool = AtomicBool::new(false);
108static WARNED_NAMESPACE: AtomicBool = AtomicBool::new(false);
109
110/// Test hook: `NO_OVERRIDE` reads the build's [`crate::config::STATED_NODE_COUNT`];
111/// `OVERRIDE_UNSTATED` answers `None`; any other value is that count.
112static COUNT_OVERRIDE: AtomicUsize = AtomicUsize::new(NO_OVERRIDE);
113const NO_OVERRIDE: usize = usize::MAX;
114const OVERRIDE_UNSTATED: usize = usize::MAX - 1;
115
116/// The image's stated node count, as the rule reads it.
117fn stated_count() -> Option<usize> {
118    match COUNT_OVERRIDE.load(Ordering::Acquire) {
119        NO_OVERRIDE => crate::config::STATED_NODE_COUNT,
120        OVERRIDE_UNSTATED => None,
121        n => Some(n),
122    }
123}

125/// Test hook for the route tests (Rust here, C in `nros-c`): install `name` /
126/// `namespace` as if the environment had them, and answer `count` as the
127/// image's stated node count, replacing whatever a previous call installed and
128/// re-arming the once-per-process messages. Never called outside tests: the
129/// count of a real image is a build fact, not something to set.
130#[doc(hidden)]
131pub fn set_for_tests(name: Option<&str>, namespace: Option<&str>, count: Option<usize>) {
132    ENV_NODE_NAME.reset();
133    ENV_NODE_NAMESPACE.reset();
134    install(name, namespace);
135    COUNT_OVERRIDE.store(count.unwrap_or(OVERRIDE_UNSTATED), Ordering::Release);
136    for f in [&WARNED_RENAME, &NOTED_NAME_IGNORED, &WARNED_NAMESPACE] {
137        f.store(false, Ordering::Release);
138    }
139}
140
141/// Test hook: back to "the environment said nothing", the build's own count.
142#[doc(hidden)]
143pub fn clear_for_tests() {
144    ENV_NODE_NAME.reset();
145    ENV_NODE_NAMESPACE.reset();
146    COUNT_OVERRIDE.store(NO_OVERRIDE, Ordering::Release);
147}

149/// Record the environment's node name and namespace for this process.
150///
151/// Called by the hosted edge with what it read (`None` = unset or empty).
152/// Only the first call installs a value; the environment does not change
153/// under a running process, so a later executor resolves the same values.
154pub fn install(node_name: Option<&str>, namespace: Option<&str>) {
155    for (var, slot, value) in [
156        ("NROS_NODE_NAME", &ENV_NODE_NAME, node_name),
157        ("NROS_NODE_NAMESPACE", &ENV_NODE_NAMESPACE, namespace),
158    ] {
159        if let Some(v) = value
160            && !slot.install(v)
161        {
162            nros_log::log_warn!(
163                nros_log::get_logger("nros"),
164                "{}={} is ignored: it is longer than {} bytes, the most a node's {} holds",
165                var,
166                v,
167                CAP,
168                if var == "NROS_NODE_NAME" {
169                    "name"
170                } else {
171                    "namespace"
172                }
173            );
174        }
175    }
176}

178/// Whether this process's `$NROS_NODE_NAME` renames its node: set, and the

183}

192/// What the environment did to `$NROS_NODE_NAME`, for the notice and the tests.
193#[derive(Debug, Clone, Copy, PartialEq, Eq)]
194pub enum NameRule {
195    /// The variable is unset: the stated name stands.
196    Unset,
197    /// The image states one node: the variable names it.
198    Renamed,
199    /// The image states this many nodes (two or more, or zero): one name
200    /// cannot name them all, so the variable is ignored.
201    IgnoredCount(usize),
202    /// The image does not state its node count: ignored.
203    IgnoredUnstated,
204}
205
206/// One node's identity after the environment rung, and what decided it.
207#[derive(Debug, Clone, Copy, PartialEq, Eq)]
208pub struct Resolved<'a> {
209    /// The name the node is created with.
210    pub name: &'a str,
211    /// The namespace the node is created with.
212    pub namespace: &'a str,
213    /// What `$NROS_NODE_NAME` did.
214    pub name_rule: NameRule,
215    /// `Some(stated)` when `$NROS_NODE_NAMESPACE` replaced a node's own
216    /// non-root namespace that differs from it.
217    pub namespace_overrode: Option<&'a str>,
218}
219
220/// The pure rule: no globals, no logging.
221#[must_use]
222pub fn resolve<'a>(
223    env_name: Option<&'a str>,
224    env_namespace: Option<&'a str>,
225    stated_count: Option<usize>,
226    stated_name: &'a str,
227    stated_namespace: &'a str,
228) -> Resolved<'a> {
229    let (name, name_rule) = match (env_name, stated_count) {
230        (None, _) => (stated_name, NameRule::Unset),
231        (Some(env), Some(1)) => (env, NameRule::Renamed),
232        (Some(_), Some(n)) => (stated_name, NameRule::IgnoredCount(n)),
233        (Some(_), None) => (stated_name, NameRule::IgnoredUnstated),
234    };
235    let (namespace, namespace_overrode) = match env_namespace {
236        None => (stated_namespace, None),
237        Some(env) => {
238            let root = stated_namespace.is_empty() || stated_namespace == "/";
239            let overrode = (!root && stated_namespace != env).then_some(stated_namespace);
240            (env, overrode)
241        }
242    };
243    Resolved {
244        name,
245        namespace,
246        name_rule,
247        namespace_overrode,
248    }
249}

251/// Apply this process's environment to a node about to be created with
252/// `stated_name` / `stated_namespace`, warning or noting once per process.
253///
254/// Every road that names a node calls this exactly where the name is fixed:
255/// `NodeBuilder::build` (Rust, C++, C's `rclc_executor_node_init`) and C's
256/// legacy `rclc_node_init_ex`.
257#[must_use]
258pub fn apply<'a>(stated_name: &'a str, stated_namespace: &'a str) -> Resolved<'a> {
259    let r = resolve(
260        ENV_NODE_NAME.get(),
261        ENV_NODE_NAMESPACE.get(),
262        stated_count(),
263        stated_name,
264        stated_namespace,
265    );
266    let logger = || nros_log::get_logger("nros");
267    match r.name_rule {
268        NameRule::Renamed
269            if r.name != stated_name && !WARNED_RENAME.swap(true, Ordering::AcqRel) =>
270        {
271            nros_log::log_warn!(
272                logger(),
273                "NROS_NODE_NAME={} renames node `{}`, the image's only node",
274                r.name,
275                stated_name
276            );
277        }
278        NameRule::IgnoredCount(n) if !NOTED_NAME_IGNORED.swap(true, Ordering::AcqRel) => {
279            nros_log::log_warn!(
280                logger(),
281                "NROS_NODE_NAME={} is ignored: this image creates {} nodes, and one name \
282                 cannot name them all. Name each node in system.toml, or with a launch remap",
283                ENV_NODE_NAME.get().unwrap_or(""),
284                n
285            );
286        }
287        NameRule::IgnoredUnstated if !NOTED_NAME_IGNORED.swap(true, Ordering::AcqRel) => {
288            nros_log::log_warn!(
289                logger(),
290                "NROS_NODE_NAME={} is ignored: this image does not state how many nodes it \
291                 creates ([image] node_count, which `nros build` writes from the contract), \
292                 so a rename could collide. Name the node in system.toml, or with a launch remap",
293                ENV_NODE_NAME.get().unwrap_or("")
294            );
295        }
296        _ => {}
297    }
298    if let Some(stated) = r.namespace_overrode
299        && !WARNED_NAMESPACE.swap(true, Ordering::AcqRel)
300    {
301        nros_log::log_warn!(
302            logger(),
303            "NROS_NODE_NAMESPACE={} overrides node `{}`'s namespace `{}`",
304            r.namespace,
305            r.name,
306            stated
307        );
308    }
309    r
310}

318        let r = resolve(None, None, Some(1), "talker", "/ns");

366    #[test]
367    fn a_slot_installs_once_and_refuses_what_cannot_fit() {
368        let s = Slot::new();
369        assert_eq!(s.get(), None);
370        assert!(s.install("first"));
371        assert!(s.install("second"), "a later install is not an error");
372        assert_eq!(s.get(), Some("first"));

375        assert!(!Slot::new().install(long));

379/// The Rust route (issue 2739 acceptance): a node created through the
380/// executor, `NodeBuilder::build`, carries the environment's identity by the
381/// rule. Gated like `node.rs`'s builder tests: `MockSession` is the backend.
382#[cfg(all(test, feature = "alloc", not(feature = "rmw-cffi")))]
383mod route_tests {
384    extern crate std;
385    use super::{clear_for_tests, set_for_tests};
386    use crate::{executor::Executor, mock::MockSession};
387    use std::sync::{Mutex, MutexGuard};
388
389    /// The slots are process-wide: these tests take turns, and put the
390    /// environment back for every other test in the binary.
391    static LOCK: Mutex<()> = Mutex::new(());
392
393    struct Env(#[allow(dead_code)] MutexGuard<'static, ()>);
394    impl Env {
395        fn set(name: Option<&str>, ns: Option<&str>, count: Option<usize>) -> Self {
396            let g = LOCK.lock().unwrap_or_else(|e| e.into_inner());
397            set_for_tests(name, ns, count);
398            Env(g)
399        }
400    }
401    impl Drop for Env {
402        fn drop(&mut self) {
403            clear_for_tests();
404        }
405    }
406
407    fn identity(
408        exec: &Executor,
409        id: crate::executor::NodeId,
410    ) -> (std::string::String, std::string::String) {
411        let r = exec.node(id).expect("node");
412        (r.name.as_str().into(), r.namespace.as_str().into())
413    }
414
415    #[test]
416    fn one_stated_node_takes_the_env_name() {
417        let _e = Env::set(Some("w0node"), None, Some(1));
418        let mut exec: Executor = Executor::from_session(MockSession::new());
419        let id = exec.node_builder("talker").build().expect("node");
420        assert_eq!(identity(&exec, id).0, "w0node");
421    }
422
423    #[test]
424    fn two_stated_nodes_keep_their_names() {
425        let _e = Env::set(Some("w0node"), None, Some(2));
426        let mut exec: Executor = Executor::from_session(MockSession::new());
427        let a = exec.node_builder("talker").build().expect("node");
428        let b = exec.node_builder("listener").build().expect("node");
429        assert_eq!(identity(&exec, a).0, "talker");
430        assert_eq!(identity(&exec, b).0, "listener");
431    }
432
433    #[test]
434    fn an_unstated_count_keeps_the_name() {

436        let mut exec: Executor = Executor::from_session(MockSession::new());
437        let id = exec.node_builder("talker").build().expect("node");
438        assert_eq!(identity(&exec, id).0, "talker");
