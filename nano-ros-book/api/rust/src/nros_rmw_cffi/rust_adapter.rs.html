1//! Phase 115.L.0 — Generic Rust-trait → C-vtable adapter.
2//!
3//! `RustBackendAdapter<R>` converts any `R: nros_rmw::Rmw` whose
4//! associated types implement the matching `Session` / `Publisher` /
5//! `Subscription` / `ServiceTrait` / `ClientTrait` traits
6//! into a `static NrosRmwVtable`. Each per-backend cffi crate then
7//! collapses to ~10 LOC:
8//!
9//! ```ignore
10//! #[unsafe(no_mangle)]
11//! pub extern "C" fn nros_rmw_zenoh_register() -> nros_rmw_cffi::NrosRmwRet {
12//!     nros_rmw_cffi::RustBackendAdapter::<nros_rmw_zenoh::ZenohRmw>::register()
13//! }
14//! ```
15//!
16//! # Storage discipline
17//!
18//! - Session: `Box::into_raw(Box::new(session))` is stashed in
19//!   `NrosRmwSession::backend_data`. `close` reclaims via
20//!   `Box::from_raw` (drops the box, runs `Drop`, frees the alloc).
21//! - Publisher / Subscription / Service / Client: same
22//!   pattern with their respective handle types.
23//!
24//! # 'static
25//!
26//! Every handle must be `'static` (the `Box` outlives the call). We
27//! deliberately do **not** require `Send`: the C runtime hands the
28//! `backend_data` pointer back to the same caller that minted it,
29//! and the executor's single-thread-per-session invariant matches
30//! the Rust trait surface. Zenoh-pico's `ZenohSession`, for
31//! instance, holds an unmovable `*const Context` which the
32//! upstream library does not mark `Send`.
33//!
34//! # Bounds
35//!
36//! All error types must be `TransportError` (or `Into<TransportError>`).
37//! This matches every in-tree backend today.
38

58mod static_subscriber_storage {
59    use core::{cell::UnsafeCell, mem, ptr};
60
61    use portable_atomic::{AtomicBool, Ordering};
62
63    // Issue 0269 — the pool size is build-time configurable via
64    // `NROS_RMW_SUBSCRIBER_SLOTS` (build.rs, default 8). The old
65    // hardcoded 4 silently capped every embedded session at four
66    // subscriptions: the fifth `create_subscription` hit the exhausted
67    // pool, returned BAD_ALLOC, and the executor surfaced it as an
68    // opaque `SubscriberCreationFailed` (the autoware_sentinel comp-all
69    // wall — every std target Boxes instead and never sees a limit).
70    const SLOT_COUNT: usize = crate::parse_env_usize(
71        env!("NROS_RMW_SUBSCRIBER_SLOTS"),
72        "NROS_RMW_SUBSCRIBER_SLOTS must be a decimal integer",
73    );
74    const SLOT_SIZE: usize = 1024;
75    const SLOT_ALIGN: usize = 16;
76
77    #[repr(align(16))]
78    struct Slot {
79        bytes: UnsafeCell<[u8; SLOT_SIZE]>,
80    }
81
82    // Phase 192.5 — `#[repr(align(16))]` can't take the `SLOT_ALIGN` const, so
83    // assert they stay in lockstep (the insert() guard compares against SLOT_ALIGN).
84    const _: () = assert!(mem::align_of::<Slot>() == SLOT_ALIGN);
85
86    unsafe impl Sync for Slot {}
87
88    impl Slot {
89        const fn new() -> Self {
90            Self {
91                bytes: UnsafeCell::new([0; SLOT_SIZE]),
92            }
93        }
94    }
95
96    // issue 0739 — declare the arithmetic so the pool inventory can price it.
97    // 0271 measured this pool at 8,192 bytes on an image that had never heard of
98    // the knob; `SLOT_SIZE` is the literal above, so the figure is derivable.
99    // nros-pool: SLOTS = NROS_RMW_SUBSCRIBER_SLOTS * 1024
100    static USED: [AtomicBool; SLOT_COUNT] = [const { AtomicBool::new(false) }; SLOT_COUNT];
101    static SLOTS: [Slot; SLOT_COUNT] = [const { Slot::new() }; SLOT_COUNT];
102
103    pub unsafe fn insert<T>(value: T) -> Option<*mut T> {
104        if mem::size_of::<T>() > SLOT_SIZE || mem::align_of::<T>() > SLOT_ALIGN {
105            return None;
106        }
107
108        for index in 0..SLOT_COUNT {
109            if USED[index]
110                .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
111                .is_err()
112            {
113                continue;
114            }
115
116            let ptr = SLOTS[index].bytes.get().cast::<T>();
117            unsafe { ptr.write(value) };
118            return Some(ptr);
119        }
120
121        None
122    }
123
124    pub unsafe fn take<T>(ptr: *mut T) -> bool {
125        if ptr.is_null() {
126            return false;
127        }
128
129        for index in 0..SLOT_COUNT {
130            let slot_ptr = SLOTS[index].bytes.get().cast::<T>();
131            if ptr != slot_ptr {
132                continue;
133            }
134
135            unsafe { ptr::drop_in_place(ptr) };
136            USED[index].store(false, Ordering::Release);
137            return true;
138        }
139
140        false
141    }
142}

145// Trait alias bundle
146// ============================================================================
147//
148// `RustBackend` ties together every constraint the adapter needs. The
149// alternative — long `where` clauses on every fn — is unreadable. This
150// trait is sealed: backends don't impl it directly; the blanket impl
151// below picks it up automatically for any `R: Rmw` whose associated
152// types line up.
153
154/// Bundle of trait bounds an `Rmw` backend must satisfy to be exposed
155/// through [`RustBackendAdapter`]. Implemented automatically for any
156/// `R: Rmw` whose handle types use `TransportError` and are
157/// `'static`.
158pub trait RustBackend: Sized {
159    type Session: Session<
160            Error = TransportError,
161            PublisherHandle = Self::Publisher,
162            SubscriptionHandle = Self::Subscription,
163            ServiceHandle = Self::Service,
164            ClientHandle = Self::Client,
165        > + 'static;
166    type Publisher: Publisher<Error = TransportError> + 'static;
167    type Subscription: Subscription<Error = TransportError> + 'static;
168    type Service: ServiceTrait<Error = TransportError> + 'static;
169    type Client: ClientTrait<Error = TransportError> + 'static;
170
171    /// Construct a fresh factory instance. Called inside the `open`
172    /// trampoline. Equivalent to `R::default()` for backends that
173    /// `derive(Default)`; the indirection keeps the door open for
174    /// future per-backend registration knobs.
175    fn factory() -> Self;
176
177    /// Move the factory into a session per the `Rmw::open` contract.
178    fn open(self, config: &RmwConfig) -> Result<Self::Session, TransportError>;
179}
180
181impl<R> RustBackend for R
182where
183    R: Rmw<Error = TransportError> + Default + Sized,
184    R::Session: Session<Error = TransportError> + 'static,
185    <R::Session as Session>::PublisherHandle: Publisher<Error = TransportError> + 'static,
186    <R::Session as Session>::SubscriptionHandle: Subscription<Error = TransportError> + 'static,
187    <R::Session as Session>::ServiceHandle: ServiceTrait<Error = TransportError> + 'static,
188    <R::Session as Session>::ClientHandle: ClientTrait<Error = TransportError> + 'static,
189{
190    type Session = R::Session;
191    type Publisher = <R::Session as Session>::PublisherHandle;
192    type Subscription = <R::Session as Session>::SubscriptionHandle;
193    type Service = <R::Session as Session>::ServiceHandle;
194    type Client = <R::Session as Session>::ClientHandle;
195
196    fn factory() -> Self {
197        R::default()
198    }
199
200    fn open(self, config: &RmwConfig) -> Result<Self::Session, TransportError> {
201        Rmw::open(self, config)
202    }
203}

209/// Read a null-terminated C string into a Rust `&str`. Returns the
210/// empty string if `ptr` is null or contains invalid UTF-8.
211///
212/// # Safety
213///
214/// `ptr`, if non-null, must point to a valid null-terminated byte
215/// sequence that outlives the returned borrow.
216unsafe fn cstr_to_str<'a>(ptr: *const core::ffi::c_char) -> &'a str {
217    if ptr.is_null() {
218        return "";
219    }
220    let ptr = ptr.cast::<u8>();
221    let mut len = 0usize;
222    // Bound the scan so a missing terminator can't read off the end of
223    // a small caller buffer. 1 KiB matches the typical NAME_BUF_LEN
224    // (256) plus headroom for type-name / hash strings.
225    while len < 4096 && unsafe { *ptr.add(len) } != 0 {
226        len += 1;
227    }
228    let slice = unsafe { core::slice::from_raw_parts(ptr, len) };
229    core::str::from_utf8(slice).unwrap_or("")
230}

232/// Convert the cffi QoS view back into a `nros_rmw::QosSettings`. The
233/// adapter trampolines call this when forwarding `create_publisher` /
234/// `create_subscription` into the Rust trait.
235fn qos_from_cffi(q: &NrosRmwQos) -> QosSettings {
236    use nros_rmw::{
237        QosDurabilityPolicy, QosHistoryPolicy, QosLivelinessPolicy, QosReliabilityPolicy,
238    };
239    QosSettings {
240        reliability: if q.reliability == 0 {
241            QosReliabilityPolicy::BestEffort
242        } else {
243            QosReliabilityPolicy::Reliable
244        },
245        durability: if q.durability == 0 {
246            QosDurabilityPolicy::Volatile
247        } else {
248            QosDurabilityPolicy::TransientLocal
249        },
250        history: if q.history == 0 {
251            QosHistoryPolicy::KeepLast
252        } else {
253            QosHistoryPolicy::KeepAll
254        },
255        depth: q.depth as u32,
256        // phase-301 (issue 0240): the express hint left the QoS struct; the
257        // create trampolines thread it via `TopicInfo` from the options param.
258        tx_express: false,
259        deadline_ms: q.deadline_ms,
260        lifespan_ms: q.lifespan_ms,
261        liveliness_kind: match q.liveliness_kind {
262            1 => QosLivelinessPolicy::Automatic,
263            2 => QosLivelinessPolicy::ManualByTopic,
264            3 => QosLivelinessPolicy::ManualByNode,
265            _ => QosLivelinessPolicy::None,
266        },
267        liveliness_lease_ms: q.liveliness_lease_ms,
268        avoid_ros_namespace_conventions: q.avoid_ros_namespace_conventions != 0,
269    }
270}

351    }

361        unsafe { crate::nros_rmw_cffi_register_named(name, &Self::VTABLE) }
362    }

369unsafe extern "C" fn create_session_trampoline<R: RustBackend>(
370    locator: *const core::ffi::c_char,
371    mode: u8,
372    domain_id: u32,
373    node_name: *const core::ffi::c_char,
374    out: *mut NrosRmwSession,
375) -> NrosRmwRet {
376    if out.is_null() {
377        return NROS_RMW_RET_INVALID_ARGUMENT;
378    }
379    let cfg = RmwConfig {
380        locator: unsafe { cstr_to_str(locator) },
381        mode: if mode == 0 {
382            SessionMode::Client
383        } else {
384            SessionMode::Peer
385        },
386        domain_id,
387        node_name: unsafe { cstr_to_str(node_name) },
388        namespace: "",
389        properties: &[],
390    };
391    let factory = R::factory();
392    match factory.open(&cfg) {
393        Ok(session) => {
394            let boxed = Box::into_raw(Box::new(session));
395            unsafe {
396                (*out).backend_data = boxed as *mut c_void;
397            }
398            NROS_RMW_RET_OK
399        }
400        Err(e) => ret_from_error(&e),
401    }
402}
403
404unsafe extern "C" fn destroy_session_trampoline<R: RustBackend>(
405    session: *mut NrosRmwSession,
406) -> NrosRmwRet {
407    let Some(boxed) = (unsafe { take_box::<R::Session>(session_data_mut(session)) }) else {
408        return NROS_RMW_RET_INVALID_ARGUMENT;
409    };
410    let mut s = boxed;
411    let ret = match Session::close(&mut *s) {
412        Ok(()) => NROS_RMW_RET_OK,
413        Err(e) => ret_from_error(&e),
414    };
415    drop(s);
416    ret
417}

419unsafe extern "C" fn drive_io_trampoline<R: RustBackend>(
420    session: *mut NrosRmwSession,
421    timeout_ms: i32,
422) -> NrosRmwRet {
423    let Some(s) = (unsafe { session_mut::<R::Session>(session) }) else {

425    };
426    match Session::drive_io(s, timeout_ms) {
427        Ok(()) => NROS_RMW_RET_OK,
428        Err(e) => ret_from_error(&e),
429    }

436unsafe extern "C" fn create_publisher_trampoline<R: RustBackend>(
437    session: *mut NrosRmwSession,
438    topic_name: *const core::ffi::c_char,
439    type_name: *const core::ffi::c_char,
440    type_hash: *const core::ffi::c_char,
441    domain_id: u32,
442    qos: *const NrosRmwQos,
443    options: *const nros_rmw_publisher_options_t,
444    out: *mut NrosRmwPublisher,
445) -> NrosRmwRet {
446    if out.is_null() || qos.is_null() {
447        return NROS_RMW_RET_INVALID_ARGUMENT;
448    }
449    let Some(s) = (unsafe { session_mut::<R::Session>(session) }) else {
450        return NROS_RMW_RET_INVALID_ARGUMENT;
451    };
452    let node_name = unsafe { session_node_name(session) };
453    let namespace = unsafe { session_namespace(session) };
454    let topic = TopicInfo {
455        name: unsafe { cstr_to_str(topic_name) },
456        type_name: unsafe { cstr_to_str(type_name) },
457        type_hash: unsafe { cstr_to_str(type_hash) },
458        domain_id,
459        node_name,
460        namespace,
461        // Publisher side — the receive-buffer hint is subscription-only.
462        rx_buffer_hint: 0,
463        // phase-301 (issue 0240) — express hint from the NULLable options
464        // struct (NULL = default, not express).
465        tx_express: unsafe { options.as_ref() }.is_some_and(|o| o.tx_express != 0),
466    };
467    let qos_settings = qos_from_cffi(unsafe { &*qos });
468    match Session::create_publisher(s, &topic, qos_settings) {
469        Ok(pub_handle) => {
470            let boxed = Box::into_raw(Box::new(pub_handle));
471            unsafe {
472                (*out).backend_data = boxed as *mut c_void;
473            }
474            NROS_RMW_RET_OK
475        }
476        Err(e) => ret_from_error(&e),
477    }
478}

486unsafe extern "C" fn publish_raw_trampoline<R: RustBackend>(
487    publisher: *mut NrosRmwPublisher,
488    data: *const u8,
489    len: usize,
490) -> NrosRmwRet {
491    let Some(p) = (unsafe { publisher_ref::<R::Publisher>(publisher) }) else {
492        return NROS_RMW_RET_INVALID_ARGUMENT;
493    };
494    if data.is_null() && len != 0 {
495        return NROS_RMW_RET_INVALID_ARGUMENT;

497    let slice = unsafe { core::slice::from_raw_parts(data, len) };
498    match Publisher::publish_raw(p, slice) {
499        Ok(()) => NROS_RMW_RET_OK,
500        Err(e) => ret_from_error(&e),
501    }
502}

508unsafe extern "C" fn create_subscription_trampoline<R: RustBackend>(
509    session: *mut NrosRmwSession,
510    topic_name: *const core::ffi::c_char,
511    type_name: *const core::ffi::c_char,
512    type_hash: *const core::ffi::c_char,
513    domain_id: u32,
514    qos: *const NrosRmwQos,
515    options: *const nros_rmw_subscription_options_t,
516    out: *mut NrosRmwSubscription,
517) -> NrosRmwRet {
518    if out.is_null() || qos.is_null() {
519        return NROS_RMW_RET_INVALID_ARGUMENT;
520    }
521    let Some(s) = (unsafe { session_mut::<R::Session>(session) }) else {
522        return NROS_RMW_RET_INVALID_ARGUMENT;
523    };
524    let node_name = unsafe { session_node_name(session) };
525    let namespace = unsafe { session_namespace(session) };
526    let topic = TopicInfo {
527        name: unsafe { cstr_to_str(topic_name) },
528        type_name: unsafe { cstr_to_str(type_name) },
529        type_hash: unsafe { cstr_to_str(type_hash) },
530        domain_id,
531        node_name,
532        namespace,
533        // Phase 231 (RFC-0038) / phase-301 (issue 0240) — receive-buffer size
534        // hint from the NULLable options struct (NULL = unset), so the backend
535        // can size-class its receive storage.
536        rx_buffer_hint: unsafe { options.as_ref() }.map_or(0, |o| o.rx_buffer_hint as usize),
537        // Subscription side — express is a publisher-only hint.
538        tx_express: false,
539    };
540    let qos_settings = qos_from_cffi(unsafe { &*qos });
541    match Session::create_subscription(s, &topic, qos_settings) {
542        Ok(sub_handle) => {
543            #[cfg(all(target_os = "none", not(feature = "std")))]
544            {
545                let Some(ptr) = (unsafe { static_subscriber_storage::insert(sub_handle) }) else {
546                    return crate::NROS_RMW_RET_BAD_ALLOC;
547                };
548                unsafe {
549                    (*out).backend_data = ptr as *mut c_void;
550                }
551                NROS_RMW_RET_OK
552            }
553            #[cfg(not(all(target_os = "none", not(feature = "std"))))]
554            {
555                let boxed = Box::into_raw(Box::new(sub_handle));
556                unsafe {
557                    (*out).backend_data = boxed as *mut c_void;
558                }
559                NROS_RMW_RET_OK
560            }
561        }
562        Err(e) => ret_from_error(&e),
563    }
564}

566unsafe extern "C" fn destroy_subscription_trampoline<R: RustBackend>(
567    subscriber: *mut NrosRmwSubscription,
568) {

583    subscriber: *mut NrosRmwSubscription,

728unsafe extern "C" fn try_recv_request_trampoline<R: RustBackend>(
729    server: *mut NrosRmwService,
730    buf: *mut u8,
731    buf_len: usize,
732    seq_out: *mut i64,
733) -> i32 {
734    let Some(s) = (unsafe { service_mut::<R::Service>(server) }) else {
735        return NROS_RMW_RET_INVALID_ARGUMENT;
736    };
737    if (buf.is_null() && buf_len != 0) || seq_out.is_null() {
738        return NROS_RMW_RET_INVALID_ARGUMENT;
739    }
740    let slice = unsafe { core::slice::from_raw_parts_mut(buf, buf_len) };
741    let buf_start = slice.as_ptr() as usize;
742    match ServiceTrait::try_recv_request(s, slice) {
743        Ok(Some(req)) => {
744            // The handle's `data` slice borrows from `buf`. Use its
745            // offset within the caller's buffer to compute the payload
746            // length; some backends prepend an envelope header so the
747            // payload doesn't start at offset 0.
748            let offset = (req.data.as_ptr() as usize).saturating_sub(buf_start);
749            let len = req.data.len();
750            unsafe {
751                *seq_out = req.sequence_number;
752            }
753            // Move the payload to the start of the buffer so the C
754            // caller sees a `(buf, len)` pair starting at offset 0,
755            // matching the cyclonedds backend's contract.
756            if offset != 0 {
757                let total = offset + len;
758                // SAFETY: `slice[..total]` is initialised by the
759                // backend; copy_within respects overlapping ranges.
760                slice.copy_within(offset..total, 0);
761            }
762            len as i32
763        }
764        Ok(None) => NROS_RMW_RET_NO_DATA,
765        Err(e) => ret_from_error(&e),
766    }
767}

889// ============================================================================
890// Event-callback bridge (Phase 115.L.0.events)
891// ============================================================================
892//
893// Events / liveliness / deadline are wired through the vtable (see the
894// `register_subscription_event` / `register_publisher_event` /
895// `assert_publisher_liveliness` / `next_deadline_ms` trampolines registered
896// above) — the stale "TODO: wire through" header was removed (Phase 192.9).
897//
898// `NrosRmwEventCallback` (cffi shape) and `nros_rmw::EventCallback`
899// (trait shape) have *layout-identical* arguments:
900//
901//   * NrosRmwEventKind  ↔ EventKind     — both `#[repr(u8)]`, same variant ids.
902//   * *const NrosRmwEventPayload ↔ *const c_void  — fn-ptr-width pointer; the
903//     trait callback dereferences as `*const LivelinessChangedStatus` /
904//     `*const CountStatus`, the cffi callback dereferences as the matching
905//     union member. The field-by-field layout matches today (see the
906//     `_event_payload_layout_match` const-asserts below).
907//   * *mut c_void       ↔ *mut c_void   — identical.
908//
909// Therefore the cffi callback can be transmuted into a trait callback
910// pointer; the receiving Rust trait code calls it via the trait
911// signature, but the bytes on the wire are the cffi struct.
912
913const _: () = {
914    use core::mem::{align_of, size_of};
915    assert!(
916        size_of::<crate::NrosRmwLivelinessChangedStatus>()
917            == size_of::<nros_rmw::LivelinessChangedStatus>()
918    );
919    assert!(
920        align_of::<crate::NrosRmwLivelinessChangedStatus>()
921            == align_of::<nros_rmw::LivelinessChangedStatus>()
922    );
923    assert!(size_of::<crate::NrosRmwCountStatus>() == size_of::<nros_rmw::CountStatus>());
924    assert!(align_of::<crate::NrosRmwCountStatus>() == align_of::<nros_rmw::CountStatus>());
925    // EventKind tags must round-trip 0..=4 between the generated C
926    // discriminants and the `#[repr(u8)]` trait enum.
927    use crate::nros_rmw_event_kind_t as ck;
928    assert!(
929        ck::NROS_RMW_EVENT_LIVELINESS_CHANGED == nros_rmw::EventKind::LivelinessChanged as ck::Type
930    );
931    assert!(
932        ck::NROS_RMW_EVENT_REQUESTED_DEADLINE_MISSED
933            == nros_rmw::EventKind::RequestedDeadlineMissed as ck::Type
934    );
935    assert!(ck::NROS_RMW_EVENT_MESSAGE_LOST == nros_rmw::EventKind::MessageLost as ck::Type);
936    assert!(ck::NROS_RMW_EVENT_LIVELINESS_LOST == nros_rmw::EventKind::LivelinessLost as ck::Type);
937    assert!(
938        ck::NROS_RMW_EVENT_OFFERED_DEADLINE_MISSED
939            == nros_rmw::EventKind::OfferedDeadlineMissed as ck::Type
940    );
941};

943unsafe extern "C" fn register_subscription_event_trampoline<R: RustBackend>(
944    subscriber: *mut NrosRmwSubscription,
945    kind: NrosRmwEventKind,
946    deadline_ms: u32,
947    cb: NrosRmwEventCallback,
948    user_context: *mut c_void,
949) -> NrosRmwRet {
950    let Some(s) = (unsafe { subscription_mut::<R::Subscription>(subscriber) }) else {
951        return NROS_RMW_RET_INVALID_ARGUMENT;
952    };
953    // Explicit conversion, not a transmute: `NrosRmwEventKind` is
954    // C-uint-sized (#238) while trait `EventKind` is `#[repr(u8)]` —
955    // reinterpreting bytes would be UB.
956    let trait_kind: nros_rmw::EventKind = event_kind_from_c(kind);
957    // The generated callback type is nullable; a NULL callback can't be
958    // forwarded into the trait's non-null callback surface.
959    let Some(cb) = cb else {
960        return NROS_RMW_RET_INVALID_ARGUMENT;
961    };
962    // SAFETY: see module-level note. `cb`'s parameter types
963    // `(NrosRmwEventKind, *const NrosRmwEventPayload, *mut c_void)` are
964    // ABI-compatible with the trait's `(EventKind, *const c_void, *mut c_void)`.
965    let trait_cb: nros_rmw::EventCallback = unsafe { core::mem::transmute(cb) };
966    let res = unsafe {
967        Subscription::register_event_callback(s, trait_kind, deadline_ms, trait_cb, user_context)
968    };
969    match res {
970        Ok(()) => NROS_RMW_RET_OK,
971        Err(e) => {
972            // Backend's `Unsupported` mapping for events is its
973            // `serialization_error()` by trait-doc default; the cffi
974            // contract for "this event kind unsupported" is its own
975            // ret code, so map any error here to UNSUPPORTED rather
976            // than e.g. SERIALIZATION_ERROR — the caller's mental
977            // model is "vtable said no events," not "marshalling
978            // failed."
979            let _ = e;
980            NROS_RMW_RET_UNSUPPORTED
981        }
982    }
983}

1105        unsafe extern "C" fn(
1106            out_buf: *mut u8,
1107            cap: usize,
1108            out_written: *mut usize,
1109            user_ctx: *mut c_void,
1110        ),
1111    >,
1112    user_ctx: *mut c_void,
1113) -> NrosRmwRet {
1114    // Phase 124.E.1 — delegate to the Rust backend's
1115    // `Publisher::publish_streamed` impl. Default trait body fires
1116    // the staging-buffer fallback (124.E.2); concrete backends opt
1117    // in by overriding for true streamed publish into the network
1118    // buffer.
1119    let Some(p) = (unsafe { publisher_ref::<R::Publisher>(publisher) }) else {
1120        return NROS_RMW_RET_INVALID_ARGUMENT;
1121    };
1122    // Generated ABI marks the callbacks nullable; the trait surface is not.
1123    let (Some(size_cb), Some(chunk_cb)) = (size_cb, chunk_cb) else {
1124        return NROS_RMW_RET_INVALID_ARGUMENT;
1125    };
1126    // SAFETY: this trampoline is entered from the C ABI with the same
1127    // callback/user_ctx lifetime contract required by `Publisher::publish_streamed`.
1128    match unsafe { Publisher::publish_streamed(p, size_cb, chunk_cb, user_ctx) } {
1129        Ok(()) => NROS_RMW_RET_OK,
1130        Err(e) => ret_from_error(&e),
1131    }
1132}

1134unsafe extern "C" fn try_recv_sequence_trampoline<R: RustBackend>(
1135    subscriber: *mut NrosRmwSubscription,
1136    buf: *mut u8,
1137    per_msg_cap: usize,
1138    max_msgs: usize,
1139    out_lens: *mut usize,
1140) -> i32 {
1141    // Phase 124.D.1 — delegate to the Rust backend's
1142    // `Subscription::try_recv_sequence` impl. Default trait body
1143    // loop-drives `try_recv_raw`; concrete backends opt in by
1144    // overriding for a native batch take.
1145    let Some(s) = (unsafe { subscription_mut::<R::Subscription>(subscriber) }) else {
1146        return NROS_RMW_RET_INVALID_ARGUMENT;
1147    };
1148    if buf.is_null() || out_lens.is_null() || per_msg_cap == 0 {
1149        return NROS_RMW_RET_INVALID_ARGUMENT;
1150    }
1151    // SAFETY: caller pinky-promised a contiguous block of
1152    // `max_msgs * per_msg_cap` bytes at `buf` and at least
1153    // `max_msgs` `usize` slots at `out_lens`. The buffer is
1154    // exclusively borrowed for the duration of this call.
1155    let buf_slice =
1156        unsafe { core::slice::from_raw_parts_mut(buf, max_msgs.saturating_mul(per_msg_cap)) };
1157    let lens_slice = unsafe { core::slice::from_raw_parts_mut(out_lens, max_msgs) };
1158    match Subscription::try_recv_sequence(s, buf_slice, per_msg_cap, max_msgs, lens_slice) {
1159        Ok(count) => count as i32,
1160        Err(e) => ret_from_error(&e),
1161    }
1162}

1169// Validity contract:
1170//
1171//   - The C-side wrappers (`CffiSession`, `CffiPublisher`, …) hand
1172//     each trampoline an entity-struct pointer whose `backend_data`
1173//     was last written by *this* adapter's create-trampoline. That
1174//     write is `Box::into_raw(Box::new(handle))` of `R::Session` /

1177//     per the executor's single-thread invariant.
1178//   - `take_box` reclaims ownership; subsequent calls see a null
1179//     `backend_data` and return `INVALID_ARGUMENT`.

1310#[inline]
1311unsafe fn take_box<T>(slot: &mut *mut c_void) -> Option<Box<T>> {
1312    if slot.is_null() {
1313        return None;
1314    }
1315    let ptr = *slot as *mut T;
1316    *slot = core::ptr::null_mut();
1317    if ptr.is_null() {
1318        None
1319    } else {
1320        // SAFETY: this pointer was minted by `Box::into_raw` in the
1321        // corresponding create-trampoline; we are taking ownership
1322        // back and the slot has been cleared so no second take is
1323        // possible.
1324        Some(unsafe { Box::from_raw(ptr) })
1325    }
