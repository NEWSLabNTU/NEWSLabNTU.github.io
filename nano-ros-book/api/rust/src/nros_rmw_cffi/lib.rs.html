1//! C function table adapter for nros RMW backends.
2//!
3//! This crate provides a vtable-based bridge so that backends written in C,
4//! C++, Zig, Ada, or any language with a C-compatible ABI can implement the
5//! nros `Session` / `Publisher` / `Subscription` / service traits without
6//! writing Rust code.
7//!
8//! # Usage (C backend implementor)
9//!
10//! 1. Include `<nros/rmw_vtable.h>`
11//! 2. Implement all function pointers in `nros_rmw_vtable_t`
12//! 3. Call `nros_rmw_cffi_register(&my_vtable)` before creating sessions
13//!
14//! # Usage (Rust consumer)
15//!
16//! Enable the `rmw-cffi` feature on `nros` and use `Executor<CffiSession>`.

94/// Map a `TransportError` to the corresponding `rmw_ret_t` code.
95///
96/// By-reference because `TransportError` carries a `String` on its
97/// dynamic-diagnostic variant and is not `Copy`. The string itself is
98/// dropped at the boundary — embedded RMW callers cannot afford a
99/// thread-local error buffer.
100pub fn ret_from_error(err: &TransportError) -> NrosRmwRet {
101    match err {
102        TransportError::Timeout => NROS_RMW_RET_TIMEOUT,
103        TransportError::WouldBlock => NROS_RMW_RET_WOULD_BLOCK,
104        TransportError::TooLarge => NROS_RMW_RET_MESSAGE_TOO_LARGE,
105        TransportError::BufferTooSmall => NROS_RMW_RET_BUFFER_TOO_SMALL,
106        TransportError::MessageTooLarge => NROS_RMW_RET_MESSAGE_TOO_LARGE,
107        TransportError::InvalidArgument => NROS_RMW_RET_INVALID_ARGUMENT,
108        // Issue 0468 — InvalidConfig used to borrow INVALID_ARGUMENT, so a
109        // capacity the BUILD cannot honour arrived looking like a caller
110        // passing something wrong. It has its own code now.
111        TransportError::InvalidConfig => NROS_RMW_RET_INVALID_CONFIG,
112        TransportError::Unsupported => NROS_RMW_RET_UNSUPPORTED,
113        TransportError::BadAlloc => NROS_RMW_RET_BAD_ALLOC,
114        TransportError::IncompatibleQos => NROS_RMW_RET_INCOMPATIBLE_QOS,
115        TransportError::TopicNameInvalid => NROS_RMW_RET_TOPIC_NAME_INVALID,
116        TransportError::NodeNameNonExistent => NROS_RMW_RET_NODE_NAME_NON_EXISTENT,
117        TransportError::LoanNotSupported => NROS_RMW_RET_LOAN_NOT_SUPPORTED,
118        TransportError::NoData => NROS_RMW_RET_NO_DATA,
119        TransportError::IncompatibleAbi => NROS_RMW_RET_INCOMPATIBLE_ABI,
120        // Phase 155.B.3 — distinguish wire-level connection failure
121        // from generic backend error so the FreeRTOS / RV64 C+C++
122        // `init -> -X` logs identify the actual class. zenoh-pico's
123        // `ZpicoError::Session` (zpico_open returned -3) and
124        // `ZpicoError::Generic` (zpico_init returned -1) both flow
125        // through `ZpicoError → ConnectionFailed`; the cmake-built
126        // FreeRTOS C/C++ tests will now surface NOT_FOUND (the
127        // user-side mapping in `nros_support_init`) instead of the
128        // generic NROS_RET_ERROR catch-all.
129        TransportError::ConnectionFailed | TransportError::Disconnected => {
130            NROS_RMW_RET_CONNECTION_FAILED
131        }
132        // Everything else collapses to NROS_RMW_RET_ERROR. Backends
133        // that want fine-grained reporting should adopt the named
134        // variants above (Phase 102.2 sweep).
135        _ => NROS_RMW_RET_ERROR,
136    }
137}

139/// Map a `rmw_ret_t` returned by a C-side vtable function back to
140/// a `TransportError` for the Rust caller. Inverse of `ret_from_error`
141/// — used when `nros-rmw-cffi`'s `CffiSession` etc. receive a code
142/// from the registered C backend.
143///
144/// `NROS_RMW_RET_OK` is mapped to `TransportError::Backend("ok")` as a
145/// programming-error sentinel; callers should branch on the success
146/// path before calling this. Unknown negative codes collapse to the
147/// generic `TransportError::Backend("unknown rmw_ret_t")` so a future
148/// constant added to the C header degrades gracefully on the Rust side.
149pub fn error_from_ret(ret: NrosRmwRet) -> TransportError {
150    match ret {
151        NROS_RMW_RET_OK => {
152            TransportError::Backend("ok (logic error: positive ret_t at error site)")
153        }
154        NROS_RMW_RET_ERROR => TransportError::Backend("rmw_ret error"),
155        NROS_RMW_RET_TIMEOUT => TransportError::Timeout,
156        NROS_RMW_RET_BAD_ALLOC => TransportError::BadAlloc,
157        NROS_RMW_RET_INVALID_ARGUMENT => TransportError::InvalidArgument,
158        NROS_RMW_RET_INVALID_CONFIG => TransportError::InvalidConfig,
159        NROS_RMW_RET_UNSUPPORTED => TransportError::Unsupported,
160        NROS_RMW_RET_INCOMPATIBLE_QOS => TransportError::IncompatibleQos,
161        NROS_RMW_RET_TOPIC_NAME_INVALID => TransportError::TopicNameInvalid,
162        NROS_RMW_RET_NODE_NAME_NON_EXISTENT => TransportError::NodeNameNonExistent,
163        NROS_RMW_RET_LOAN_NOT_SUPPORTED => TransportError::LoanNotSupported,
164        NROS_RMW_RET_NO_DATA => TransportError::NoData,
165        NROS_RMW_RET_WOULD_BLOCK => TransportError::WouldBlock,
166        NROS_RMW_RET_BUFFER_TOO_SMALL => TransportError::BufferTooSmall,
167        NROS_RMW_RET_MESSAGE_TOO_LARGE => TransportError::MessageTooLarge,
168        NROS_RMW_RET_INCOMPATIBLE_ABI => TransportError::IncompatibleAbi,
169        // Phase 155.B.3 — inverse of `ret_from_error`'s
170        // `ConnectionFailed | Disconnected → CONNECTION_FAILED`
171        // mapping. Decodes the new vtable-level code back to the
172        // `TransportError::ConnectionFailed` variant; downstream
173        // `transport_error_to_ret` in nros-c surfaces it as
174        // `NROS_RET_NOT_FOUND` (-4) to the user.
175        NROS_RMW_RET_CONNECTION_FAILED => TransportError::ConnectionFailed,
176        _ => TransportError::Backend("unknown rmw_ret_t"),
177    }
178}

392// Phase-301 (issue 0241) — the QoS lowering is FALLIBLE: a depth the
393// C ABI's u16 cannot represent is a create-time error, never a silent
394// saturate. The duration fields are u32 ms on both sides (0 = unset,
395// `NROS_RMW_DURATION_INFINITE_MS` = explicit infinite) and pass
396// through unchanged; finer-grained callers lower via
397// `nros_rmw::duration_to_qos_ms` (sub-ms CEILs to 1 ms, past-u32
398// errors).
399impl TryFrom<QoSProfile> for NrosRmwQos {
400    type Error = TransportError;
401
402    fn try_from(qos: QoSProfile) -> Result<Self, TransportError> {
403        if qos.depth > u16::MAX as u32 {
404            return Err(TransportError::InvalidArgument);
405        }
406        Ok(Self {
407            // issue 0829 — the sentinel LOWERS; it is not resolved here.
408            // `NrosRmwQos` is what a C backend receives, and resolving the
409            // absence is that backend's job (cyclone answers depth 1, xrce
410            // leaves it for the Agent). Folding it to a concrete value on this
411            // side would hand every C backend the same answer and make the
412            // per-backend resolution unreachable.
413            reliability: match qos.reliability {
414                QoSReliabilityPolicy::SystemDefault => {
415                    generated::NROS_RMW_RELIABILITY_SYSTEM_DEFAULT as u8
416                }
417                QoSReliabilityPolicy::BestEffort => {
418                    generated::NROS_RMW_RELIABILITY_BEST_EFFORT as u8
419                }
420                QoSReliabilityPolicy::Reliable => generated::NROS_RMW_RELIABILITY_RELIABLE as u8,
421            },
422            durability: match qos.durability {
423                QoSDurabilityPolicy::SystemDefault => {
424                    generated::NROS_RMW_DURABILITY_SYSTEM_DEFAULT as u8
425                }
426                QoSDurabilityPolicy::Volatile => generated::NROS_RMW_DURABILITY_VOLATILE as u8,
427                QoSDurabilityPolicy::TransientLocal => {
428                    generated::NROS_RMW_DURABILITY_TRANSIENT_LOCAL as u8
429                }
430            },
431            history: match qos.history {
432                QoSHistoryPolicy::SystemDefault => generated::NROS_RMW_HISTORY_SYSTEM_DEFAULT as u8,
433                QoSHistoryPolicy::KeepLast => generated::NROS_RMW_HISTORY_KEEP_LAST as u8,
434                QoSHistoryPolicy::KeepAll => generated::NROS_RMW_HISTORY_KEEP_ALL as u8,
435            },
436            liveliness_kind: qos.liveliness_kind as u8,
437            depth: qos.depth as u16,
438            _reserved0: 0,
439            deadline_ms: qos.deadline_ms,
440            lifespan_ms: qos.lifespan_ms,
441            liveliness_lease_ms: qos.liveliness_lease_ms,
442            avoid_ros_namespace_conventions: qos.avoid_ros_namespace_conventions as u8,
443            _reserved1: [0; 3],
444        })
445    }
446}

504// ============================================================================
505//
506// Phase 104.B.2 — named registry replaces the singleton vtable.
507// Backends register under a stable identifier (`"zenoh"`, `"dds"`,
508// `"xrce"`, future `"uorb"`, `"cyclonedds"`); consumers look up
509// vtables by name via `nros_rmw_cffi_lookup`. Multiple backends can
510// coexist in the same process (bridge nodes).
511//
512// Capacity comes from the `NROS_RMW_MAX_BACKENDS` build-time env
513// var (default 8). See `build.rs`.
514//
515// Implementation: a fixed-size `[BackendSlot; MAX_BACKENDS]`
516// guarded by an atomic length counter. No alloc; `no_std`
517// compatible. Slot scan is O(N) for lookup but N is tiny (8 by
518// default). Each slot owns its name buffer; `name_ptr` returned
519// to consumers points into the slot and stays valid for the
520// program's lifetime.

522/// Compile-time max number of concurrently registered backends.
523/// Set via `NROS_RMW_MAX_BACKENDS` env var at build time
524/// (`build.rs`). Default 8.
525pub const MAX_BACKENDS: usize = parse_max_backends(env!("NROS_RMW_MAX_BACKENDS"));

527const fn parse_max_backends(s: &str) -> usize {
528    parse_env_usize(s, "NROS_RMW_MAX_BACKENDS must be a decimal integer")
529}

531/// Const decimal parser for build.rs-emitted envs (`MAX_BACKENDS`,
532/// `NROS_RMW_SUBSCRIBER_SLOTS`).
533pub(crate) const fn parse_env_usize(s: &str, msg: &str) -> usize {
534    let bytes = s.as_bytes();
535    let mut i = 0usize;
536    let mut acc: usize = 0;
537    while i < bytes.len() {
538        let d = bytes[i];
539        if !d.is_ascii_digit() {
540            let _ = msg;
541            panic!("nros-rmw-cffi: build-time env must be a decimal integer");
542        }
543        acc = acc * 10 + (d - b'0') as usize;
544        i += 1;
545    }
546    acc
547}

549/// Maximum length of a backend name. Names are short ASCII
550/// identifiers (`"zenoh"`, `"cyclonedds"`); 32 bytes is generous.
551const BACKEND_NAME_MAX: usize = 32;

553#[repr(C)]
554struct BackendSlot {
555    /// Null-terminated UTF-8 backend name. Zero-initialized when
556    /// unused (`name[0] == 0`).
557    name: [u8; BACKEND_NAME_MAX],
558    vtable: *const NrosRmwVtable,
559}
560
561impl BackendSlot {
562    const fn empty() -> Self {
563        Self {
564            name: [0u8; BACKEND_NAME_MAX],
565            vtable: core::ptr::null(),
566        }
567    }
568
569    #[inline]
570    fn is_empty(&self) -> bool {
571        self.name[0] == 0
572    }
573
574    #[inline]
575    fn name_matches(&self, candidate: &[u8]) -> bool {
576        if self.is_empty() {
577            return false;
578        }
579        // Compare up to the first NUL or candidate length.
580        let mut i = 0usize;
581        while i < self.name.len() && i < candidate.len() {
582            if self.name[i] == 0 {
583                return false; // slot name shorter than candidate
584            }
585            if self.name[i] != candidate[i] {
586                return false;
587            }
588            i += 1;
589        }
590        // candidate fully consumed; slot must be NUL at i (same length)
591        i == candidate.len() && (i == self.name.len() || self.name[i] == 0)
592    }
593}

595// SAFETY: `BackendSlot::vtable` is a `*const` pointer used in a
596// `'static` context; once written it's never freed and the registry
597// is guarded by an atomic length counter for publication. Marker
598// trait implementations are required so the static array is
599// `Sync` across threads.
600unsafe impl Sync for BackendSlot {}

602/// Fixed-size registry. `slots[0..len]` are live; `slots[len..]`
603/// are zero-initialized. `len` is the publication fence.
604///
605/// `slots` lives in an `UnsafeCell` because we mutate through
606/// `&'static REGISTRY`. Safety invariants:
607/// * Slot writes happen only inside `nros_rmw_cffi_register_named`,
608///   which is documented "call before `Executor::open`" — backend
609///   ctors fire pre-main, manual calls precede session creation.
610/// * Slot reads via `nros_rmw_cffi_lookup` and `get_vtable` happen
611///   after `Executor::open`, well after registration completes.
612/// * The atomic `len` provides the release-acquire fence so any
613///   reader that sees `len = N` also sees the populated slot
614///   contents for indices `< N`.
615#[doc(hidden)]
616pub struct Registry {
617    slots: core::cell::UnsafeCell<[BackendSlot; MAX_BACKENDS]>,
618    len: portable_atomic::AtomicUsize,
619}
620
621impl Registry {
622    #[doc(hidden)]
623    pub const fn new() -> Self {
624        let slots = {
625            #[allow(clippy::declare_interior_mutable_const)]
626            const E: BackendSlot = BackendSlot::empty();
627            [E; MAX_BACKENDS]
628        };
629        Self {
630            slots: core::cell::UnsafeCell::new(slots),
631            len: portable_atomic::AtomicUsize::new(0),
632        }
633    }
634
635    /// Borrow slot `i` immutably. Caller must guarantee
636    /// `i < self.len.load(Acquire)`.
637    #[inline]
638    unsafe fn slot(&self, i: usize) -> &BackendSlot {
639        // SAFETY: registry protocol guarantees slot stability once
640        // published via the atomic len fence.
641        unsafe { &(*self.slots.get())[i] }
642    }
643
644    /// Borrow slot `i` mutably. Caller must guarantee exclusive
645    /// access — either pre-publication (idx > current `len`) or
646    /// during an idempotent overwrite of an already-registered name.
647    #[inline]
648    #[allow(clippy::mut_from_ref)]
649    unsafe fn slot_mut(&self, i: usize) -> &mut BackendSlot {
650        // SAFETY: see Registry doc — writer-side discipline.
651        unsafe { &mut (*self.slots.get())[i] }
652    }
653}

726// SAFETY: each slot is published by `key` and `valid` atomics. Writers store
727// `info` before setting `valid = true` with Release ordering; readers take
728// `valid` with AcqRel before copying the `MessageInfo`.
729unsafe impl Sync for MessageInfoSlot {}
730
731// issue 0739 — deliberately NOT annotated with a `// nros-pool:` formula.
732// `MessageInfoSlot`'s width depends on cfg (`alloc` + `safety-e2e` add three
733// more fields), so any constant here would be right for one build and wrong for
734// the rest. Issue 0271 measured 3,584 bytes at 64 slots in ITS configuration;
735// stating that as the cost would be the fabrication the inventory exists to
736// avoid. The knob still appears in the table with its default — the table says
737// "no byte figure", which is true, rather than implying it is free.
738static MESSAGE_INFO_TABLE: [MessageInfoSlot; MESSAGE_INFO_SLOTS] = {
739    #[allow(clippy::declare_interior_mutable_const)]
740    const E: MessageInfoSlot = MessageInfoSlot::empty();
741    [E; MESSAGE_INFO_SLOTS]
742};

744fn lookup_message_info_slot(key: usize) -> Option<&'static MessageInfoSlot> {
745    if key == 0 {
746        return None;
747    }
748    MESSAGE_INFO_TABLE
749        .iter()
750        .find(|slot| slot.key.load(Ordering::Acquire) == key)
751}
752
753#[cfg(feature = "alloc")]
754fn get_or_insert_message_info_slot(key: usize) -> Option<&'static MessageInfoSlot> {
755    if key == 0 {
756        return None;
757    }
758    for slot in &MESSAGE_INFO_TABLE {
759        let current = slot.key.load(Ordering::Acquire);
760        if current == key {
761            return Some(slot);
762        }
763        if current == 0
764            && slot
765                .key
766                .compare_exchange(0, key, Ordering::AcqRel, Ordering::Acquire)
767                .is_ok()
768        {
769            return Some(slot);
770        }
771    }
772    None
773}

775#[cfg(feature = "alloc")]
776pub(crate) fn store_cffi_message_info(key: usize, info: Option<MessageInfo>) {
777    let Some(slot) = get_or_insert_message_info_slot(key) else {
778        return;
779    };
780    match info {
781        Some(info) => {
782            // SAFETY: this slot is keyed to one subscriber backend handle. The
783            // executor owns each subscriber mutably while receiving, so writes
784            // for the same key are serialized.
785            unsafe {
786                *slot.info.get() = info;
787            }
788            slot.valid.store(true, Ordering::Release);
789        }
790        None => slot.valid.store(false, Ordering::Release),
791    }
792}

794fn take_cffi_message_info(key: usize) -> Option<MessageInfo> {
795    let slot = lookup_message_info_slot(key)?;
796    if !slot.valid.swap(false, Ordering::AcqRel) {
797        return None;
798    }
799    // SAFETY: `valid.swap(false)` gives this reader exclusive consumption of the
800    // last stored `MessageInfo` for this key.
801    Some(unsafe { *slot.info.get() })
802}

941/// Register a backend under a stable name. Multiple backends can
942/// coexist; consumers select via [`nros_rmw_cffi_lookup`] or the
943/// higher-level `Executor::node_builder(...).rmw(...)` path.
944///
945/// Names must be UTF-8, NUL-terminated, ≤ 31 bytes (excluding NUL).
946/// Reserved names today: `"zenoh"`, `"dds"`, `"xrce"`,
947/// `"cyclonedds"`, future `"uorb"`. The string `"default"` is the
948/// implicit name used by the legacy single-arg
949/// [`nros_rmw_cffi_register`] shim.
950///
951/// Returns:
952/// * `NROS_RMW_RET_OK` on success.
953/// * `NROS_RMW_RET_INVALID_ARGUMENT` if `name` / `vtable` is
954///   NULL, the name is empty, or exceeds 31 bytes.
955/// * `NROS_RMW_RET_ERROR` if the registry is full
956///   (`MAX_BACKENDS` reached without a matching entry).
957///
958/// Duplicate registration of the same name overwrites the
959/// previous vtable (idempotent for ctor-fires-twice cases).
960///
961/// # Safety
962///
963/// * `name` must be a valid NUL-terminated UTF-8 string.
964/// * `vtable` must remain valid for the program's lifetime.
965#[unsafe(no_mangle)]
966pub unsafe extern "C" fn nros_rmw_cffi_register_named(
967    name: *const core::ffi::c_char,
968    vtable: *const NrosRmwVtable,
969) -> NrosRmwRet {
970    if name.is_null() || vtable.is_null() {
971        return NROS_RMW_RET_INVALID_ARGUMENT;
972    }
973
974    // Issue 0332 — reject an incomplete vtable at registration rather than
975    // panicking mid-spin. SAFETY: `vtable` is non-null (checked) and the caller
976    // guarantees it is valid for the program's lifetime (see `# Safety`).
977    if let Some(missing) = first_missing_vtable_slot(unsafe { &*vtable }) {
978        let _ = missing; // named for debuggers; INVALID_ARGUMENT is the ABI signal
979        return NROS_RMW_RET_INVALID_ARGUMENT;
980    }
981
982    let name_u8 = name.cast::<u8>();
983
984    // Length-check the input. We scan up to BACKEND_NAME_MAX + 1
985    // bytes; anything longer is rejected.
986    let mut len = 0usize;
987    while len < BACKEND_NAME_MAX {
988        let b = unsafe { *name_u8.add(len) };
989        if b == 0 {
990            break;
991        }
992        len += 1;
993    }
994    if len == 0 {
995        return NROS_RMW_RET_INVALID_ARGUMENT;
996    }
997    // Must have found a NUL within BACKEND_NAME_MAX.
998    if unsafe { *name_u8.add(len) } != 0 {
999        return NROS_RMW_RET_INVALID_ARGUMENT;
1000    }
1001
1002    let name_bytes = unsafe { core::slice::from_raw_parts(name_u8, len) };
1003
1004    // First pass: look for existing entry with same name → overwrite.
1005    let current_len = registry().len.load(Ordering::Acquire);
1006    for i in 0..current_len {
1007        // SAFETY: i < current_len, indices in bounds.
1008        let slot = unsafe { registry().slot(i) };
1009        if slot.name_matches(name_bytes) {
1010            // SAFETY: writer-side idempotent overwrite. The slot is
1011            // already published; concurrent readers will see either
1012            // the old or new vtable consistently, both valid.
1013            unsafe {
1014                let slot_mut = registry().slot_mut(i);
1015                slot_mut.vtable = vtable;
1016            }
1017            core::sync::atomic::fence(Ordering::Release);
1018            return NROS_RMW_RET_OK;
1019        }
1020    }
1021
1022    // No existing entry; append. Reserve a slot via atomic increment.
1023    let idx = registry().len.fetch_add(1, Ordering::AcqRel);
1024    if idx >= MAX_BACKENDS {
1025        // Roll back the increment so subsequent registers don't see a
1026        // stale `len > MAX_BACKENDS`. (Race window negligible — once
1027        // we hit capacity, no further append succeeds.)
1028        registry().len.store(MAX_BACKENDS, Ordering::Release);
1029        return NROS_RMW_RET_ERROR;
1030    }
1031
1032    // SAFETY: idx < MAX_BACKENDS, mutating an as-yet-unpublished slot.
1033    unsafe {
1034        let slot = registry().slot_mut(idx);
1035        slot.name[..len].copy_from_slice(name_bytes);
1036        slot.name[len] = 0;
1037        slot.vtable = vtable;
1038    }
1039    // Release-fence so concurrent lookups see both the name and the
1040    // vtable consistently with the updated `len`.
1041    core::sync::atomic::fence(Ordering::Release);
1042    NROS_RMW_RET_OK
1043}

1047///
1048/// # Safety
1049///
1050/// * `name` must be a valid NUL-terminated UTF-8 string.
1051#[unsafe(no_mangle)]
1052pub unsafe extern "C" fn nros_rmw_cffi_lookup(
1053    name: *const core::ffi::c_char,
1054) -> *const NrosRmwVtable {
1055    if name.is_null() {
1056        return core::ptr::null();
1057    }
1058    let name_u8 = name.cast::<u8>();
1059    let mut len = 0usize;
1060    while len < BACKEND_NAME_MAX {
1061        if unsafe { *name_u8.add(len) } == 0 {
1062            break;
1063        }
1064        len += 1;
1065    }
1066    if len == 0 || len == BACKEND_NAME_MAX {
1067        return core::ptr::null();
1068    }
1069    let name_bytes = unsafe { core::slice::from_raw_parts(name_u8, len) };
1070
1071    let current_len = registry().len.load(Ordering::Acquire);
1072    for i in 0..current_len {
1073        // SAFETY: i < current_len, indices in bounds; publication
1074        // fence via the atomic-len Acquire load.
1075        let slot = unsafe { registry().slot(i) };
1076        if slot.name_matches(name_bytes) {
1077            return slot.vtable;
1078        }
1079    }
1080    core::ptr::null()
1081}

1138/// Phase 128.A.3 — outcome of `resolve_backend`.
1139pub enum BackendResolution {
1140    /// Exactly one matching backend; use its vtable.
1141    Single(&'static NrosRmwVtable),
1142    /// No backend linked into the binary. Maps to
1143    /// [`NROS_RMW_RET_NO_BACKEND`].
1144    NoBackend,
1145    /// More than one backend linked and no selector given. Maps to
1146    /// [`NROS_RMW_RET_AMBIGUOUS_BACKEND`].
1147    Ambiguous,
1148    /// Selector did not match any registered backend. Maps to
1149    /// [`NROS_RMW_RET_UNKNOWN_BACKEND`].
1150    Unknown,
1151}
1152
1153/// Phase 128.A.3 — selection policy for the single-backend
1154/// `Executor::open` / `nros::init` path.
1155///
1156/// Algorithm:
1157///
1158/// 1. If `selector` is `Some(name)` (typically from `$NROS_RMW`),
1159///    look it up in the registry. Hit → [`BackendResolution::Single`];
1160///    miss → [`BackendResolution::Unknown`].
1161/// 2. Otherwise, if exactly one backend is registered, return it.
1162/// 3. Otherwise, if zero, [`BackendResolution::NoBackend`]; if more
1163///    than one, [`BackendResolution::Ambiguous`].
1164///
1165/// Callers convert the resolution to a [`NrosRmwRet`] via
1166/// [`backend_resolution_to_ret`].
1167///
1168/// Bridge consumers (`Executor::open_multi`) bypass this function and
1169/// call `nros_rmw_cffi_lookup` per spec instead.
1170pub fn resolve_backend(selector: Option<&[u8]>) -> BackendResolution {
1171    let n = registry().len.load(Ordering::Acquire);
1172    if let Some(name) = selector {
1173        let mut i = 0usize;
1174        while i < n {
1175            // SAFETY: i < n, registry len-Acquire fence orders the read.
1176            let slot = unsafe { registry().slot(i) };
1177            if slot.name_matches(name) {
1178                if slot.vtable.is_null() {
1179                    return BackendResolution::Unknown;
1180                }
1181                return BackendResolution::Single(unsafe { &*slot.vtable });
1182            }
1183            i += 1;
1184        }
1185        return BackendResolution::Unknown;
1186    }
1187    match n {
1188        0 => BackendResolution::NoBackend,
1189        1 => default_vtable()
1190            .map(BackendResolution::Single)
1191            .unwrap_or(BackendResolution::NoBackend),
1192        _ => BackendResolution::Ambiguous,
1193    }
1194}

1196/// Phase 128.A.3 — map a [`BackendResolution`] to its canonical
1197/// [`NrosRmwRet`]. [`BackendResolution::Single`] is *not* an error and
1198/// returns [`NROS_RMW_RET_OK`]; callers needing the vtable should
1199/// pattern-match on the resolution itself.
1200pub fn backend_resolution_to_ret(res: &BackendResolution) -> NrosRmwRet {
1201    match res {
1202        BackendResolution::Single(_) => NROS_RMW_RET_OK,
1203        BackendResolution::NoBackend => NROS_RMW_RET_NO_BACKEND,
1204        BackendResolution::Ambiguous => NROS_RMW_RET_AMBIGUOUS_BACKEND,
1205        BackendResolution::Unknown => NROS_RMW_RET_UNKNOWN_BACKEND,
1206    }
1207}

1235/// Phase 115.A.2 — C entry point for installing a custom transport.
1236///
1237/// Mirrors the Rust-side `nros_rmw::set_custom_transport(Some(...))`
1238/// (or `None` when `ops == NULL`) but returns the canonical
1239/// `rmw_ret_t` codes so non-Rust consumers don't have to
1240/// reach into nros-c's higher-level error enum.
1241///
1242/// The struct's contents are copied internally; the caller may
1243/// stack-allocate. Pass `NULL` to clear the slot.
1244///
1245/// # Safety
1246///
1247/// `ops` must either be `NULL` or point at a valid
1248/// `nros_transport_ops_t` whose four fn pointers stay live for the
1249/// lifetime of the registration (i.e. until a subsequent
1250/// `nros_rmw_cffi_set_custom_transport(NULL)` or a replacement
1251/// install).
1252#[unsafe(no_mangle)]
1253pub unsafe extern "C" fn nros_rmw_cffi_set_custom_transport(
1254    ops: *const generated::nros_transport_ops_t,
1255) -> NrosRmwRet {
1256    if ops.is_null() {
1257        // Clear: ignore any error (None is always accepted).
1258        let _ = unsafe { nros_rmw::set_custom_transport(None) };
1259        return NROS_RMW_RET_OK;
1260    }
1261    // SAFETY: caller guarantees `ops` is valid for one read.
1262    let src = unsafe { &*ops };
1263
1264    // issue 0331 — the generated type's fn-pointer slots are `Option<fn>`
1265    // (C nullability); `NrosTransportOps`' are plain `fn`. The two are
1266    // layout-identical via the null-pointer optimization, so a NULL slot
1267    // transmutes into a `fn` that is UB the moment the runtime calls it.
1268    // Taking the hand-written Rust type at this boundary made that
1269    // unrepresentable-looking but not unreachable — a C caller could always
1270    // pass NULL. Reject it here, before the copy.
1271    if src.open.is_none() || src.close.is_none() || src.write.is_none() || src.read.is_none() {
1272        return NROS_RMW_RET_INVALID_ARGUMENT;
1273    }
1274
1275    // SAFETY: layout equivalence of the two representations is asserted
1276    // above, and every fn slot is non-NULL per the check just made, so this
1277    // reinterpretation can neither silently mismatch nor produce a null `fn`.
1278    let copy: nros_rmw::NrosTransportOps = unsafe { core::mem::transmute_copy(src) };
1279    match unsafe { nros_rmw::set_custom_transport(Some(copy)) } {
1280        Ok(()) => NROS_RMW_RET_OK,
1281        Err(e) => ret_from_error(&e),
1282    }
1283}

1297/// Write a Rust `&str` as a null-terminated byte sequence into a fixed buffer.
1298/// Returns a pointer to the buffer start (as C `char*`, matching the
1299/// generated ABI's string parameters).
1300fn to_c_str<const N: usize>(s: &str, buf: &mut [u8; N]) -> *const core::ffi::c_char {
1301    let len = s.len().min(N - 1);
1302    buf[..len].copy_from_slice(&s.as_bytes()[..len]);
1303    buf[len] = 0;
1304    buf.as_ptr().cast()
1305}

1307/// Inverse of [`to_c_str`] — read a null-terminated byte buffer back
1308/// as a `&str`, stopping at the first NUL byte. Used by the
1309/// `topic_name()` / `type_name()` / `node_name()` accessors on the
1310/// `Cffi*` types so callers can introspect without round-tripping
1311/// through the vtable. Phase 102.5.
1312fn cstr_buf_to_str<const N: usize>(buf: &[u8; N]) -> &str {
1313    let len = buf.iter().position(|&b| b == 0).unwrap_or(N);
1314    // The buffers are written via `to_c_str` from a `&str`, so the
1315    // bytes between [..len] are guaranteed valid UTF-8. `from_utf8`
1316    // handles the (impossible) corruption case by returning empty.
1317    core::str::from_utf8(&buf[..len]).unwrap_or("")
1318}

1321// Node table (phase-376 W5/B1)
1322// ============================================================================
1323//
1324// A SIDE table, not a field on `CffiSession`, for the same reason
1325// `MESSAGE_INFO_TABLE` above is one: the session's size is an ABI surface. It
1326// is what every C and C++ `_opaque` buffer is sized from, through a build-time
1327// probe that has to agree across every coordinate, and the guards that catch a
1328// disagreement are compile-time asserts in `nros-c` (issue 0472). Per-node
1329// bookkeeping is runtime state, not part of the session's shape — putting four
1330// name buffers inside the session grew `_opaque` by ~544 bytes in every C
1331// consumer and tripped those guards, which is the machinery working correctly
1332// and saying "this does not belong here".

1334/// Distinct `(session, name, namespace)` triples the shim tracks. Raise with
1335/// `NROS_RMW_MAX_NODES`; the default mirrors the executor's `MAX_NODES`.
1336pub const MAX_NODES: usize = parse_env_usize(
1337    env!("NROS_RMW_MAX_NODES"),
1338    "NROS_RMW_MAX_NODES must be a decimal integer",
1339);

1341struct NodeSlot {
1342    /// The owning session's `backend_data`, or 0 when free.
1343    session_key: portable_atomic::AtomicUsize,
1344    name: UnsafeCell<[u8; NODE_NAME_BUF_LEN]>,
1345    namespace_: UnsafeCell<[u8; NODE_NAME_BUF_LEN]>,
1346    backend_data: UnsafeCell<*mut c_void>,
1347}
1348
1349impl NodeSlot {
1350    const fn empty() -> Self {
1351        Self {
1352            session_key: portable_atomic::AtomicUsize::new(0),
1353            name: UnsafeCell::new([0u8; NODE_NAME_BUF_LEN]),
1354            namespace_: UnsafeCell::new([0u8; NODE_NAME_BUF_LEN]),
1355            backend_data: UnsafeCell::new(core::ptr::null_mut()),
1356        }
1357    }
1358}

1360// SAFETY: a slot is claimed by a CAS on `session_key` from 0, and only the
1361// claiming thread writes its cells before any other reader can match it (a
1362// reader matches on `session_key` AND the names, which are written first).
1363// Entity creation is a setup-time operation on the executor thread.
1364unsafe impl Sync for NodeSlot {}
1365
1366static NODE_TABLE: [NodeSlot; MAX_NODES] = {
1367    #[allow(clippy::declare_interior_mutable_const)]
1368    const E: NodeSlot = NodeSlot::empty();
1369    [E; MAX_NODES]
1370};

1372/// Release every node this session claimed. Called from `destroy_session`: a
1373/// static table that is never reclaimed would let a long-running image that
1374/// opens and closes sessions exhaust it.
1375/// Hand this session's node slots back, telling the backend first.
1376///
1377/// Issue 0800 — `create_node` was dispatched and `destroy_node` never was, so a
1378/// backend that allocated state in the create leaked it on every session close:
1379/// the shim forgot the node and the backend was never told it was gone. Nothing
1380/// caught it because `destroy_node` had no producer AND no consumer, which is
1381/// the state that reads as "optional slot" and is indistinguishable from
1382/// "nobody wired the other half".
1383///
1384/// Order matters: the nodes go before `destroy_session`, because a backend's
1385/// node state hangs off its session state.
1386fn release_session_nodes(
1387    session_key: usize,
1388    vtable: &'static NrosRmwVtable,
1389    session_view: &mut NrosRmwSession,
1390) {
1391    if session_key == 0 {
1392        return;
1393    }
1394    for slot in NODE_TABLE.iter() {
1395        if slot.session_key.load(Ordering::Acquire) != session_key {
1396            continue;
1397        }
1398        // SAFETY: written once by the claiming thread before the slot became
1399        // findable, and this session owns it.
1400        let backend_data = unsafe { *slot.backend_data.get() };
1401        if let Some(destroy) = vtable.destroy_node
1402            && !backend_data.is_null()
1403        {
1404            let mut view = node_view_of(slot, session_view);
1405            // SAFETY: `view` describes a node this session created through
1406            // `create_node`, and the session is still open.
1407            let ret = unsafe { destroy(&mut view) };
1408            if ret != NROS_RMW_RET_OK {
1409                nros_log::nros_error!(
1410                    nros_log::get_logger("nros_rmw_cffi"),
1411                    "destroy_node failed with {}; the backend may still hold node state",
1412                    ret
1413                );
1414            }
1415        }
1416        // SAFETY: same slot ownership as above; cleared before the slot is
1417        // published as free so a re-claim cannot see a stale pointer.
1418        unsafe { *slot.backend_data.get() = core::ptr::null_mut() };
1419        slot.session_key.store(0, Ordering::Release);
1420    }
1421}

1423fn slot_str(cell: &UnsafeCell<[u8; NODE_NAME_BUF_LEN]>) -> &str {
1424    // SAFETY: cells are written once by the thread that claimed the slot,
1425    // before `session_key` makes it findable.
1426    cstr_buf_to_str(unsafe { &*cell.get() })
1427}

1429fn find_node_slot(key: usize, name: &str, namespace: &str) -> Option<&'static NodeSlot> {
1430    NODE_TABLE.iter().find(|slot| {
1431        slot.session_key.load(Ordering::Acquire) == key
1432            && slot_str(&slot.name) == name
1433            && slot_str(&slot.namespace_) == namespace
1434    })
1435}

1437fn claim_node_slot(key: usize, name: &str, namespace: &str) -> Option<&'static NodeSlot> {
1438    for slot in NODE_TABLE.iter() {
1439        if slot
1440            .session_key
1441            .compare_exchange(0, key, Ordering::AcqRel, Ordering::Acquire)
1442            .is_ok()
1443        {
1444            // SAFETY: the CAS above makes this thread the slot's owner, and a
1445            // reader cannot match it until the names are in place.
1446            unsafe {
1447                let _ = to_c_str(name, &mut *slot.name.get());
1448                let _ = to_c_str(namespace, &mut *slot.namespace_.get());
1449                *slot.backend_data.get() = core::ptr::null_mut();
1450            }
1451            return Some(slot);
1452        }
1453    }
1454    None
1455}

1537    node_namespace: *const core::ffi::c_char,
1538    enclave: *const core::ffi::c_char,

1574unsafe extern "C" fn endpoint_info_visit_trampoline(
1575    ctx: *mut core::ffi::c_void,
1576    info: *const generated::rmw_topic_endpoint_info_t,
1577) -> bool {
1578    if ctx.is_null() || info.is_null() {
1579        return true; // skip this entry, keep enumerating
1580    }
1581    let info = unsafe { &*info };
1582    let cb = unsafe { &mut *(ctx as *mut &mut dyn FnMut(&GraphEndpointInfo<'_>) -> bool) };
1583
1584    // Each borrowed string is skipped rather than lossily converted, for the
1585    // same reason as the names-and-types side.
1586    let cstr = |p: *const core::ffi::c_char| -> Option<&str> {
1587        if p.is_null() {
1588            None
1589        } else {
1590            unsafe { core::ffi::CStr::from_ptr(p) }.to_str().ok()
1591        }
1592    };
1593    let (Some(node_name), Some(node_namespace), Some(topic_type)) = (
1594        cstr(info.node_name),
1595        cstr(info.node_namespace),
1596        cstr(info.topic_type),
1597    ) else {
1598        return true;
1599    };
1600
1601    let view = GraphEndpointInfo {
1602        node_name,
1603        node_namespace,
1604        topic_type,
1605        is_publisher: info.endpoint_type == generated::rmw_endpoint_type_t::RMW_ENDPOINT_PUBLISHER,
1606        endpoint_gid: info.endpoint_gid.data,
1607    };
1608    cb(&view)
1609}

1611/// phase-381 — the C names-and-types visitor, back into a Rust closure call.
1612///
1613/// # Safety
1614/// Called only by a backend slot this crate handed `ctx` to.
1615unsafe extern "C" fn names_and_types_visit_trampoline(
1616    ctx: *mut core::ffi::c_void,
1617    name: *const core::ffi::c_char,
1618    types: *const *const core::ffi::c_char,
1619    types_count: usize,
1620) -> bool {
1621    if ctx.is_null() || name.is_null() {
1622        return true; // skip this entry, keep enumerating
1623    }
1624    let cb = unsafe { &mut *(ctx as *mut &mut dyn FnMut(&str, &[&str]) -> bool) };
1625    let Ok(name) = (unsafe { core::ffi::CStr::from_ptr(name) }).to_str() else {
1626        return true;
1627    };
1628    // Bounded on the stack: the C side owns the array for this call only, and a
1629    // type that is not valid UTF-8 is SKIPPED rather than lossily converted — a
1630    // mangled type name is a different, plausible type.
1631    const TYPES_MAX: usize = 8;
1632    let mut buf: [&str; TYPES_MAX] = [""; TYPES_MAX];
1633    let mut n = 0usize;
1634    if !types.is_null() {
1635        for i in 0..types_count.min(TYPES_MAX) {
1636            let p = unsafe { *types.add(i) };
1637            if p.is_null() {
1638                continue;
1639            }
1640            if let Ok(t) = (unsafe { core::ffi::CStr::from_ptr(p) }).to_str() {
1641                buf[n] = t;
1642                n += 1;
1643            }
1644        }
1645    }
1646    cb(name, &buf[..n])
1647}

1649/// Shared body for `count_publishers` / `count_subscribers`.
1650fn count_on_topic(
1651    view: &NrosRmwSession,
1652    f: unsafe extern "C" fn(
1653        *const NrosRmwSession,
1654        *const core::ffi::c_char,
1655        *mut usize,
1656    ) -> NrosRmwRet,
1657    topic_name: &str,
1658) -> Result<usize, TransportError> {
1659    // NUL-terminate on the stack; the C side borrows for the call only.
1660    const NAME_MAX: usize = 256;
1661    if topic_name.len() >= NAME_MAX {
1662        return Err(TransportError::InvalidArgument);
1663    }
1664    let mut buf = [0u8; NAME_MAX];
1665    buf[..topic_name.len()].copy_from_slice(topic_name.as_bytes());

1668        f(
1669            view as *const NrosRmwSession,

1672        )
1673    };
1674    if rc == NROS_RMW_RET_OK {
1675        Ok(out)
1676    } else {
1677        Err(error_from_ret(rc))

1689    /// backend speaks, as a NUL-terminated C string with `'static` lifetime,
1690    /// or NULL when the backend does not declare one.

1708            // installed; it takes no arguments and returns static storage.
1709            Some(f) => unsafe { f() },
1710            None => core::ptr::null(),
1711        }
1712    }

1735    /// Phase 376 W5/B1 — find or create the node an entity belongs to, and
1736    /// return a view a `create_*` slot can take.
1737    ///
1738    /// This replaces `entity_view`, which fabricated a per-call
1739    /// `NrosRmwSession` whose `node_name` carried the entity's owning node.
1740    /// That worked and cost no ABI change, which is why phase-268 chose it —
1741    /// but it meant a backend learned about a node by reading a STRING off a
1742    /// session it was also using for transport state, and had to re-derive the
1743    /// set of nodes itself. zenoh does precisely that, linear-scanning declared
1744    /// liveliness tokens to answer "have I seen this node before".
1745    ///
1746    /// Now the runtime owns that question: `create_node` fires once per
1747    /// distinct `(name, namespace)`, which is only true because
1748    /// `Executor::create_node` deduplicates too (W5/B1.a).
1749    ///
1750    /// The returned `rmw_node_t` borrows `session_view` and the table's name
1751    /// cells, so the caller keeps `session_view` alive across the slot call —
1752    /// the same discipline `entity_view` documented, moved one level up.
1753    fn node_slot(
1754        &mut self,
1755        node_name: Option<&str>,
1756        namespace: &str,
1757        session_view: &mut NrosRmwSession,
1758    ) -> Result<NrosRmwNode, TransportError> {
1759        let key = self.backend_data as usize;
1760        if key == 0 {
1761            return Err(TransportError::InvalidArgument);
1762        }
1763
1764        // No node identity on the entity (direct-RMW / single-node path): fall
1765        // back to the session's own open-time name, which is what phase-268's
1766        // fallback did and keeps a one-node image working unchanged.
1767        let mut scratch = [0u8; NODE_NAME_BUF_LEN];
1768        let name: &str = match node_name {
1769            Some(n) if !n.is_empty() => n,
1770            _ => {
1771                let owned = self.node_name();
1772                let len = owned.len().min(NODE_NAME_BUF_LEN - 1);
1773                scratch[..len].copy_from_slice(&owned.as_bytes()[..len]);
1774                cstr_buf_to_str(&scratch)
1775            }
1776        };
1777
1778        // Truncating would MERGE two distinct nodes into one record and then
1779        // declare the wrong identity on the graph, so it is refused. The
1780        // executor bounds both at 64, so a name that arrived through it fits.
1781        if name.len() >= NODE_NAME_BUF_LEN || namespace.len() >= NODE_NAME_BUF_LEN {
1782            return Err(TransportError::InvalidArgument);
1783        }
1784
1785        if let Some(slot) = find_node_slot(key, name, namespace) {
1786            return Ok(node_view_of(slot, session_view));
1787        }
1788
1789        let slot = claim_node_slot(key, name, namespace).ok_or(TransportError::ConnectionFailed)?;
1790
1791        // A backend with no `create_node` keeps working: the node is then a
1792        // pure identity carrier with a NULL `backend_data`, which is what every
1793        // backend that has not implemented the slot already sees.
1794        if let Some(create) = self.vtable.create_node {
1795            let mut view = node_view_of(slot, session_view);
1796            let ret = unsafe {
1797                create(
1798                    session_view as *mut NrosRmwSession,
1799                    view.name,
1800                    view.namespace_,
1801                    &mut view,
1802                )
1803            };
1804            if ret != NROS_RMW_RET_OK {
1805                slot.session_key.store(0, Ordering::Release);
1806                return Err(error_from_ret(ret));
1807            }
1808            // SAFETY: this thread claimed the slot above and nothing else
1809            // writes it.
1810            unsafe { *slot.backend_data.get() = view.backend_data };
1811        }
1812        Ok(node_view_of(slot, session_view))
1813    }

1880    /// The property-carrying half of [`open_with_properties`], kept in its own
1881    /// never-inlined frame so the marshalling buffers (up to
1882    /// `RMW_SESSION_MAX_PROPERTIES` × 2 × `SESSION_PROPERTY_BUF_LEN`) are only
1883    /// on the stack of a call that actually has properties to carry. An MCU
1884    /// image that configures nothing pays nothing.
1885    #[inline(never)]
1886    fn open_marshalling_properties(
1887        vtable: &'static NrosRmwVtable,
1888        locator: &str,
1889        mode: u8,
1890        domain_id: u32,
1891        node_name: &str,
1892        properties: &[(&str, &str)],
1893    ) -> Result<Self, TransportError> {
1894        if properties.len() > MAX_SESSION_PROPERTIES {
1895            return Err(TransportError::InvalidArgument);
1896        }
1897        let mut key_bufs = [[0u8; SESSION_PROPERTY_BUF_LEN]; MAX_SESSION_PROPERTIES];
1898        let mut val_bufs = [[0u8; SESSION_PROPERTY_BUF_LEN]; MAX_SESSION_PROPERTIES];
1899        let mut entries = [rmw_session_property_t {
1900            key: core::ptr::null(),
1901            value: core::ptr::null(),
1902        }; MAX_SESSION_PROPERTIES];
1903        for (i, (key, value)) in properties.iter().enumerate() {
1904            // Empty or over-long is REFUSED. `to_c_str` truncates, which for a
1905            // name is a wrong-but-visible entity and for a TLS certificate
1906            // path is a session that fails somewhere unrelated.
1907            if key.is_empty()
1908                || value.is_empty()
1909                || key.len() >= SESSION_PROPERTY_BUF_LEN
1910                || value.len() >= SESSION_PROPERTY_BUF_LEN
1911            {
1912                return Err(TransportError::InvalidArgument);
1913            }
1914            key_bufs[i][..key.len()].copy_from_slice(key.as_bytes());
1915            val_bufs[i][..value.len()].copy_from_slice(value.as_bytes());
1916            entries[i] = rmw_session_property_t {
1917                key: key_bufs[i].as_ptr().cast(),
1918                value: val_bufs[i].as_ptr().cast(),
1919            };
1920        }
1921        let options = NrosRmwSessionOptions {
1922            localhost_only: 0,
1923            _reserved: [0u8; 7],
1924            enclave: core::ptr::null(),
1925            properties: entries.as_ptr(),
1926            property_count: properties.len(),
1927        };
1928        Self::open_with_vtable(vtable, locator, mode, domain_id, node_name, &options)
1929    }

2069/// Report a QoS DOWNGRADE — the granted profile differing from the requested
2070/// one — once per entity, at creation.
2071///
2072/// Issue 0823 gave the runtime the ability to READ the granted QoS. This is the
2073/// half that makes it a diagnostic. Silence from a QoS mismatch is
2074/// indistinguishable from a topic-name typo, a domain split (issue 0801) and a
2075/// discovery failure (issue 0803); all three were run to ground this month and
2076/// each cost hours precisely because nothing separated them. A line naming the
2077/// field that changed separates this one in a sentence.
2078///
2079/// Only a DIFFERENCE is reported. Equality is the common case and printing it
2080/// would bury the signal — the same reason `nros_platform_task_init` only warns
2081/// when the kernel disagreed with the priority it was asked for.
2082///
2083/// Best-effort by construction: a backend with no read-back slot reports
2084/// nothing, which is correct rather than missing (zenoh-pico's QoS is
2085/// per-message flags with no negotiation to read).
2086#[cfg(feature = "alloc")]
2087fn report_qos_downgrade(kind: &str, name: &str, requested: &NrosRmwQos, granted: &NrosRmwQos) {
2088    // The generated constants are `i32` (bindgen) while the struct field is
2089    // `u8`; compare in the constants' type rather than casting them down.
2090    fn reliability(v: u8) -> &'static str {
2091        match v as i32 {
2092            NROS_RMW_RELIABILITY_RELIABLE => "RELIABLE",
2093            NROS_RMW_RELIABILITY_BEST_EFFORT => "BEST_EFFORT",
2094            _ => "?",
2095        }
2096    }
2097    fn durability(v: u8) -> &'static str {
2098        match v as i32 {
2099            NROS_RMW_DURABILITY_TRANSIENT_LOCAL => "TRANSIENT_LOCAL",
2100            NROS_RMW_DURABILITY_VOLATILE => "VOLATILE",
2101            _ => "?",
2102        }
2103    }
2104    if requested.reliability != granted.reliability {
2105        nros_log::nros_warn!(
2106            nros_log::get_logger("nros_rmw_cffi"),
2107            "{kind} `{name}`: asked for reliability {} and the backend granted {}. A RELIABLE \
2108             reader does not match a BEST_EFFORT writer, so this is the usual reason a pair \
2109             goes silent.",
2110            reliability(requested.reliability),
2111            reliability(granted.reliability)
2112        );
2113    }
2114    if requested.durability != granted.durability {
2115        nros_log::nros_warn!(
2116            nros_log::get_logger("nros_rmw_cffi"),
2117            "{kind} `{name}`: asked for durability {} and the backend granted {}.",
2118            durability(requested.durability),
2119            durability(granted.durability)

3181// ---------------------------------------------------------------------------
3182// Issue 0812 — the loan path across the C / C++ boundary, without a heap.
3183//
3184// `SlotLending` hands back a `CffiSlot`, which is a Rust value with a `Drop`
3185// impl. An FFI entry point has to keep that value alive between `_loan` and
3186// `_commit` / `_discard` and has nowhere to put it, so `nros-c` / `nros-cpp`
3187// each boxed one per loan: a `malloc` on the path whose whole purpose is to
3188// remove copies and, worse, a hard `alloc` dependency on a surface that a
3189// heap-free image is supposed to be able to use.
3190//
3191// None of that state has to cross the boundary. Commit and discard are handed
3192// the publisher back as a parameter; whether the loan took the staging
3193// fallback is a property of that publisher's vtable, not of the individual
3194// loan; and the backend already minted a stable token of its own. So the FFI
3195// layers pass the BACKEND's token straight through and store nothing at all.

3285    }

3799    fn take_sequence(
3800        &mut self,
3801        buf: &mut [u8],

3803        max_msgs: usize,
3804        out_lens: &mut [usize],
3805    ) -> Result<usize, TransportError> {
3806        // Phase 124.D.2 — runtime fallback. If the backend exposes
3807        // `take_sequence` natively, call it in one hop; otherwise
3808        // delegate to the trait's default body which loop-drives
3809        // `take_serialized`. Either way the caller sees the same shape:
3810        // contiguous slot block + per-slot length array + count
3811        // return.
3812        if let Some(f) = self.vtable.take_sequence {
3813            if per_msg_cap == 0 || max_msgs == 0 {
3814                return Ok(0);
3815            }
3816            let limit = max_msgs.min(out_lens.len());
3817            if buf.len() < limit.saturating_mul(per_msg_cap) {
3818                return Err(TransportError::BufferTooSmall);
3819            }
3820            let view = self.make_view();
3821            // Phase 376 W3.b/W3.d step A — the count arrives in `taken`.
3822            let mut taken = 0usize;
3823            let rc = unsafe {
3824                f(
3825                    &view,
3826                    buf.as_mut_ptr(),
3827                    per_msg_cap,
3828                    limit,
3829                    out_lens.as_mut_ptr(),
3830                    &mut taken,
3831                )
3832            };
3833            if rc != NROS_RMW_RET_OK {
3834                return Err(error_from_ret(rc));
3835            }
3836            return Ok(taken);
3837        }
3838        // Phase 124.D.2 — `take_serialized` loop fallback. Inlined
3839        // here (rather than dispatching back through the trait
3840        // default body) so the recursion is structurally
3841        // impossible — `Subscription::take_sequence` on
3842        // `CffiSubscription` is THIS function, and forwarding to
3843        // the default body would deadlock the override.
3844        if per_msg_cap == 0 || max_msgs == 0 {
3845            return Ok(0);
3846        }
3847        // Issue 0971 — a status parked by an earlier drain is delivered before
3848        // any new take, and taking nothing this call is the point: the caller
3849        // has to be able to act on it before it asks for more.
3850        if let Some(status) = self.pending_status.take() {
3851            return Err(status);
3852        }
3853        let limit = max_msgs.min(out_lens.len());
3854        if buf.len() < limit.saturating_mul(per_msg_cap) {
3855            return Err(TransportError::BufferTooSmall);
3856        }
3857        let mut count = 0;
3858        for i in 0..limit {
3859            let slot = &mut buf[i * per_msg_cap..(i + 1) * per_msg_cap];
3860            match self.take_serialized(slot) {
3861                Ok(Some(len)) => {
3862                    out_lens[i] = len;
3863                    count += 1;
3864                }
3865                Ok(None) => break,
3866                // Issue 0971 — this arm used to be `?`, which discarded `count`
3867                // and reported the error, so a caller could not tell how many
3868                // of the slots it can see are valid. The messages already
3869                // written are real and are reported; the status is parked for
3870                // the next call. With nothing delivered yet there is no count
3871                // worth protecting, so it goes out immediately — which is also
3872                // what makes the single-message case identical to `take_serialized`.
3873                Err(e) => {
3874                    if count == 0 {
3875                        return Err(e);
3876                    }
3877                    self.pending_status = Some(e);
3878                    break;
3879                }
3880            }
3881        }
3882        Ok(count)
3883    }

4335/// loads one implementation.
4336///
4337/// Returns `Unsupported` when no registered backend implements the slot, the
4338/// first error any backend reported otherwise, and `Ok` when at least one
4339/// accepted it.
4340pub fn set_backend_log_severity(severity: nros_log::Severity) -> Result<(), TransportError> {
4341    const MAX: usize = 8;
4342    let mut names: [*const core::ffi::c_char; MAX] = [core::ptr::null(); MAX];
4343    // SAFETY: `names` is `MAX` entries and the callee writes at most that many.
4344    let n = unsafe { nros_rmw_cffi_registered_names(names.as_mut_ptr(), MAX) };
4345
4346    let wire = rmw_severity_of(severity);
4347    let mut applied = false;
4348    let mut first_err = None;
4349    for name in names.iter().take(n.min(MAX)) {
4350        // SAFETY: the registry hands back NUL-terminated static names.
4351        let vt = unsafe { nros_rmw_cffi_lookup(*name) };
4352        if vt.is_null() {
4353            continue;
4354        }
4355        // SAFETY: a non-null lookup yields a vtable valid for the image's life.
4356        let Some(f) = (unsafe { (*vt).set_log_severity }) else {
4357            continue;
4358        };
4359        // SAFETY: the slot takes a plain enum by value.
4360        let rc = unsafe { f(wire) };
4361        if rc == NROS_RMW_RET_OK {
4362            applied = true;
4363        } else if first_err.is_none() {
4364            first_err = Some(error_from_ret(rc));
4365        }
4366    }
4367
4368    match (applied, first_err) {
4369        (true, _) => Ok(()),
4370        (false, Some(e)) => Err(e),
4371        (false, None) => Err(TransportError::Unsupported),
4372    }
4373}

4377// ============================================================================
4378//
4379// Declared in `nros/rmw_entity.h`, which explains at length why they are plain
4380// exported functions rather than vtable slots. The one-line version: a vtable
4381// slot is the mechanism for letting backends DIFFER, and these two must not.
4382// They compute over types the ABI defines, they take no entity, and they are
4383// wanted at create time before any backend has registered.
4384//
4385// Defined here for the same reason `nros_rmw_cffi_register_named` is: it keeps
4386// `nros-rmw-abi` a header-only INTERFACE target, with no compiled TU and no new
4387// link edge, and it keeps the implementation in ONE place — which is the whole
4388// point.

4390/// Reason strings, one per clash bit. SELECTED, never formatted: upstream's
4391/// implementations `snprintf` their reason, which would pull the printf engine
4392/// into images that deliberately excluded it.
4393const CLASH_REASONS: &[(u32, &str)] = &[
4394    (
4395        generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_RELIABILITY,
4396        "reliability: publisher is best-effort, subscription requires reliable; ",
4397    ),
4398    (
4399        generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_DURABILITY,
4400        "durability: publisher is volatile, subscription requires transient-local; ",
4401    ),
4402    (
4403        generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_DEADLINE,
4404        "deadline: publisher's period is longer than the subscription requires; ",
4405    ),
4406    (
4407        generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_LIVELINESS_KIND,
4408        "liveliness: publisher's kind is weaker than the subscription requires; ",
4409    ),
4410    (
4411        generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_LIVELINESS_LEASE,
4412        "liveliness lease: publisher's lease is longer than the subscription requires; ",
4413    ),
4414];
4415
4416/// `0` means "unset/no check", and `NROS_RMW_DURATION_INFINITE_MS` means
4417/// explicit infinity — so neither can be compared as a plain number. Map both to
4418/// "infinitely lax" for an OFFERED duration and "infinitely tolerant" for a
4419/// REQUESTED one, which is the same thing: `u64::MAX`.
4420fn duration_or_infinite(ms: u32) -> u64 {
4421    if ms == 0 || u64::from(ms) == generated::NROS_RMW_DURATION_INFINITE_MS as u64 {
4422        u64::MAX
4423    } else {
4424        u64::from(ms)
4425    }
4426}

4428/// How strict a liveliness kind is. A publisher must assert at least as
4429/// strongly as the subscription asks.
4430fn liveliness_strength(kind: u8) -> u8 {
4431    match u32::from(kind) {
4432        generated::rmw_liveliness_kind_t::NROS_RMW_LIVELINESS_MANUAL_BY_TOPIC => 3,
4433        generated::rmw_liveliness_kind_t::NROS_RMW_LIVELINESS_MANUAL_BY_NODE => 2,
4434        generated::rmw_liveliness_kind_t::NROS_RMW_LIVELINESS_AUTOMATIC => 1,
4435        _ => 0,
4436    }
4437}

4439/// The DDS request-offered rules, in one place.
4440/// Does this profile leave any policy UNDETERMINED?
4441///
4442/// Phase 376 W5/B2. `*_UNKNOWN` means "the backend could not read this back",
4443/// which is an ABSENCE, not a value: comparing it as if it were one produces a
4444/// confident wrong verdict. Upstream's answer to that is
4445/// `RMW_QOS_COMPATIBILITY_WARNING` — "these look compatible, but I could not
4446/// check everything" — which was unreachable here until there was a sentinel
4447/// that could trigger it.
4448fn qos_has_unknown(q: &NrosRmwQos) -> bool {
4449    i32::from(q.reliability) == generated::NROS_RMW_RELIABILITY_UNKNOWN
4450        || i32::from(q.durability) == generated::NROS_RMW_DURABILITY_UNKNOWN
4451        || i32::from(q.history) == generated::NROS_RMW_HISTORY_UNKNOWN
4452        || u32::from(q.liveliness_kind)
4453            == generated::rmw_liveliness_kind_t::NROS_RMW_LIVELINESS_UNKNOWN
4454}
4455
4456fn qos_clash_mask(offered: &NrosRmwQos, requested: &NrosRmwQos) -> u32 {
4457    let mut mask = 0u32;
4458    if requested.reliability == generated::NROS_RMW_RELIABILITY_RELIABLE as u8
4459        && offered.reliability == generated::NROS_RMW_RELIABILITY_BEST_EFFORT as u8
4460    {
4461        mask |= generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_RELIABILITY;
4462    }
4463    if requested.durability == generated::NROS_RMW_DURABILITY_TRANSIENT_LOCAL as u8
4464        && offered.durability == generated::NROS_RMW_DURABILITY_VOLATILE as u8
4465    {
4466        mask |= generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_DURABILITY;
4467    }
4468    // A publisher promising samples no more often than every N ms cannot
4469    // satisfy a subscription that demands one at least every M ms when N > M.
4470    if duration_or_infinite(offered.deadline_ms) > duration_or_infinite(requested.deadline_ms) {
4471        mask |= generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_DEADLINE;
4472    }
4473    if liveliness_strength(offered.liveliness_kind) < liveliness_strength(requested.liveliness_kind)
4474    {
4475        mask |= generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_LIVELINESS_KIND;
4476    }
4477    if duration_or_infinite(offered.liveliness_lease_ms)
4478        > duration_or_infinite(requested.liveliness_lease_ms)
4479    {
4480        mask |= generated::nros_rmw_qos_clash_t::NROS_RMW_QOS_CLASH_LIVELINESS_LEASE;
4481    }
4482    mask
4483}

4485/// See `nros/rmw_entity.h`.
4486///
4487/// # Safety
4488/// `compatibility` and `clash_mask` must be valid for writes when non-NULL.
4489#[unsafe(no_mangle)]
4490pub unsafe extern "C" fn nros_rmw_qos_incompatibility_mask(
4491    offered: NrosRmwQos,
4492    requested: NrosRmwQos,
4493    compatibility: *mut generated::rmw_qos_compatibility_type_t::Type,
4494    clash_mask: *mut u32,
4495) -> NrosRmwRet {
4496    if compatibility.is_null() || clash_mask.is_null() {
4497        return NROS_RMW_RET_INVALID_ARGUMENT;
4498    }
4499    let mask = qos_clash_mask(&offered, &requested);
4500    let undetermined = qos_has_unknown(&offered) || qos_has_unknown(&requested);
4501    // SAFETY: both checked non-null above.
4502    unsafe {
4503        *clash_mask = mask;
4504        // A definite clash outranks an unknown: policies we COULD compare are
4505        // already incompatible, and reporting that as a warning would soften a
4506        // verdict the caller can act on.
4507        *compatibility = if mask != 0 {
4508            generated::rmw_qos_compatibility_type_t::RMW_QOS_COMPATIBILITY_ERROR
4509        } else if undetermined {
4510            generated::rmw_qos_compatibility_type_t::RMW_QOS_COMPATIBILITY_WARNING
4511        } else {
4512            generated::rmw_qos_compatibility_type_t::RMW_QOS_COMPATIBILITY_OK
4513        };
4514    }
4515    NROS_RMW_RET_OK
4516}

4518/// See `nros/rmw_entity.h`.
4519///
4520/// # Safety
4521/// `compatibility` must be valid for writes; `reason` must be valid for
4522/// `reason_size` bytes when non-NULL.
4523#[unsafe(no_mangle)]
4524pub unsafe extern "C" fn rmw_qos_profile_check_compatible(
4525    publisher_profile: NrosRmwQos,
4526    subscription_profile: NrosRmwQos,
4527    compatibility: *mut generated::rmw_qos_compatibility_type_t::Type,
4528    reason: *mut core::ffi::c_char,
4529    reason_size: usize,
4530) -> NrosRmwRet {
4531    let mut mask = 0u32;
4532    // SAFETY: forwarding the caller's `compatibility` pointer, which the callee
4533    // null-checks; `mask` is a local.
4534    let rc = unsafe {
4535        nros_rmw_qos_incompatibility_mask(
4536            publisher_profile,
4537            subscription_profile,
4538            compatibility,
4539            &mut mask,
4540        )
4541    };
4542    if rc != NROS_RMW_RET_OK {
4543        return rc;
4544    }
4545    if reason.is_null() || reason_size == 0 {
4546        // The create-time path: the verdict is the whole answer.
4547        return NROS_RMW_RET_OK;
4548    }
4549    // Bounded copy, always NUL-terminated. Truncation is NOT an error: a
4550    // `BUFFER_TOO_SMALL` here would cost the caller the verdict, which is the
4551    // half that matters.
4552    let mut written = 0usize;
4553    for (bit, text) in CLASH_REASONS {
4554        if mask & bit == 0 {
4555            continue;
4556        }
4557        for byte in text.as_bytes() {
4558            if written + 1 >= reason_size {
4559                break;
4560            }
4561            // SAFETY: `written + 1 < reason_size`, so this and the NUL below are
4562            // both inside the caller's buffer.
4563            unsafe { *reason.add(written) = *byte as core::ffi::c_char };
4564            written += 1;
4565        }
4566    }
4567    // SAFETY: `written < reason_size` by the bound above.
4568    unsafe { *reason.add(written) = 0 };
4569    NROS_RMW_RET_OK
4570}
